   269→/* ─────────── VERIFICATIONS TAB ─────────── */
   270→function VerificationsTab() {
   271→  const [verifications, setVerifications] = useState<any[]>([]);
   272→  const [loading, setLoading] = useState(true);
   273→  const [rejecting, setRejecting] = useState<string | null>(null);
   274→  const [approving, setApproving] = useState<string | null>(null);
   275→  const [reasons, setReasons] = useState<Record<string, string>>({});
   276→
   277→  const fetchVerifications = useCallback(async () => {
   278→    setLoading(true);
   279→    try {
   280→      const res = await api.admin.verifications.list();
   281→      setVerifications(res.verifications ?? []);
   282→    } catch {
   283→      toast.error('Failed to load verifications');
   284→    } finally {
   285→      setLoading(false);
   286→    }
   287→  }, []);
   288→
   289→  useEffect(() => { fetchVerifications(); }, [fetchVerifications]);
   290→
   291→  const handleApprove = async (id: string) => {
   292→    setApproving(id);
   293→    try {
   294→      await api.admin.verifications.update(id, { status: 'APPROVED' });
   295→      toast.success('Verification approved successfully');
   296→      fetchVerifications();
   297→    } catch {
   298→      toast.error('Failed to approve verification');
   299→    } finally {
   300→      setApproving(null);
   301→    }
   302→  };
   303→
   304→  const handleReject = async (id: string) => {
   305→    const reason = reasons[id]?.trim();
   306→    if (!reason) { toast.error('Please provide a rejection reason'); return; }
   307→    setRejecting(id);
   308→    try {
   309→      await api.admin.verifications.update(id, { status: 'REJECTED', rejectionReason: reason });
   310→      toast.success('Verification rejected');
   311→      setReasons((prev) => { const n = { ...prev }; delete n[id]; return n; });
   312→      fetchVerifications();
   313→    } catch {
   314→      toast.error('Failed to reject verification');
   315→    } finally {
   316→      setRejecting(null);
   317→    }
   318→  };
   319→
   320→  if (loading) {
   321→    return <div className="space-y-3">{Array.from({ length: 3 }).map((_, i) => <Skeleton key={i} className="h-24 rounded-2xl" />)}</div>;
   322→  }
   323→
   324→  if (verifications.length === 0) {
   325→    return <Card className="rounded-2xl border-0 shadow-sm"><CardContent className="py-16 text-center"><ShieldCheck className="w-12 h-12 mx-auto text-[#14532d]/30 mb-3" /><p className="text-muted-foreground">No pending verifications.</p></CardContent></Card>;
   326→  }
   327→
   328→  return (
   329→    <div className="space-y-3">
   330→      {verifications.map((v) => (
   331→        <motion.div key={v.id} variants={fadeVariants} initial="hidden" animate="visible">
   332→          <Card className="rounded-2xl card-hover border-0 shadow-sm">
   333→            <CardContent className="p-4 space-y-3">
   334→              <div className="flex flex-col sm:flex-row sm:items-center justify-between gap-2">
   335→                <div className="flex items-center gap-2 flex-wrap">
   336→                  <span className="font-semibold text-sm">{v.caregiver?.user?.name ?? 'Caregiver'}</span>
   337→                  <Badge className={docTypeBadge[v.docType] ?? 'bg-gray-100 text-gray-700'}>{v.docType ?? 'DOC'}</Badge>
   338→                </div>
   339→                <span className="text-xs text-muted-foreground flex items-center gap-1">
   340→                  <Clock className="w-3 h-3" />{v.createdAt ? new Date(v.createdAt).toLocaleDateString('en-IN', { day: 'numeric', month: 'short', year: 'numeric' }) : '—'}
   341→                </span>
   342→              </div>
   343→              {v.docNumber && <p className="text-xs text-muted-foreground font-mono bg-muted/50 rounded-lg px-3 py-1.5 w-fit">{v.docNumber}</p>}
   344→              <div className="flex items-center gap-2 pt-1">
   345→                <Button
   346→                  size="sm"
   347→                  className="bg-[#14532d] hover:bg-[#14532d]/90 text-white rounded-full"
   348→                  disabled={approving === v.id}
   349→                  onClick={() => handleApprove(v.id)}
   350→                >
   351→                  {approving === v.id ? <Loader2 className="w-4 h-4 animate-spin" /> : <><CheckCircle2 className="w-4 h-4 mr-1" />Approve</>}
   352→                </Button>
   353→                <Button
   354→                  size="sm"
   355→                  variant="outline"
   356→                  className="text-red-600 border-red-200 hover:bg-red-50 rounded-full"
   357→                  disabled={rejecting === v.id}
   358→                  onClick={() => {
   359→                    if (!reasons[v.id]?.trim()) {
   360→                      setReasons((prev) => ({ ...prev, [v.id]: prev[v.id] ?? '' }));
   361→                    } else {
   362→                      handleReject(v.id);
   363→                    }
   364→                  }}
   365→                >
   366→                  {rejecting === v.id ? <Loader2 className="w-4 h-4 animate-spin" /> : <><XCircle className="w-4 h-4 mr-1" />Reject</>}
   367→                </Button>
   368→              </div>
   369→              {reasons[v.id] !== undefined && (
   370→                <motion.div initial={{ height: 0, opacity: 0 }} animate={{ height: 'auto', opacity: 1 }} className="space-y-2">
   371→                  <Textarea
   372→                    placeholder="Enter rejection reason..."
   373→                    value={reasons[v.id]}
   374→                    onChange={(e) => setReasons((prev) => ({ ...prev, [v.id]: e.target.value }))}
   375→                    className="text-sm rounded-xl min-h-[60px] resize-none"
   376→                  />
   377→                  <Button size="sm" variant="destructive" className="rounded-full" disabled={rejecting === v.id} onClick={() => handleReject(v.id)}>
   378→                    {rejecting === v.id ? <Loader2 className="w-4 h-4 animate-spin" /> : <Ban className="w-4 h-4 mr-1" />}Confirm Rejection
   379→                  </Button>
   380→                </motion.div>
   381→              )}
   382→            </CardContent>
   383→          </Card>
   384→        </motion.div>
   385→      ))}
   386→    </div>
   387→  );
   388→}

   419→      } catch {
   420→        toast.error('Failed to load bookings');
   421→      } finally {
   422→        setLoading(false);

   449→                      {b.date && <span className="flex items-center gap-1"><Clock className="w-3 h-3" />{new Date(b.date).toLocaleDateString('en-IN', { day: 'numeric', month: 'short' })}</span>}
   450→                      {b.amount != null && <span className="flex items-center gap-1"><IndianRupee className="w-3 h-3" />{Number(b.amount).toLocaleString('en-IN')}</span>}
   451→                    </div>
   452→                  </div>
   453→                  <Badge className={statusColors[b.status] ?? 'bg-gray-100 text-gray-700'}>{b.status?.replace('_', ' ')}</Badge>
   454→                </CardContent>

   464→/* ─────────── COMPLAINTS TAB ─────────── */
   465→function ComplaintsTab() {
   466→  const [complaints, setComplaints] = useState<any[]>([]);
   467→  const [loading, setLoading] = useState(true);
   468→  const [dashboardStats, setDashboardStats] = useState<any>(null);
   469→
   470→  useEffect(() => {
   471→    (async () => {
   472→      try {
   473→        const [cRes, dRes] = await Promise.allSettled([
   474→          api.complaints.list({}),
   475→          api.admin.dashboard(),
   476→        ]);
   477→        if (cRes.status === 'fulfilled') setComplaints(cRes.value.complaints ?? []);
   478→        else {
   479→          const res = await api.admin.users({ limit: 10 });
   480→          const all: any[] = [];
   481→          await Promise.all((res.users ?? []).map(async (u: any) => {
   482→            try {
   483→              const cr = await api.complaints.list({ familyId: u.id });
   484→              all.push(...(cr.complaints ?? []));
   485→            } catch { /* skip */ }
   486→          }));
   487→          setComplaints(all);
   488→        }
   489→        if (dRes.status === 'fulfilled') setDashboardStats(dRes.value);
   490→      } catch {
   491→        toast.error('Failed to load complaints');
   492→      } finally {
   493→        setLoading(false);
   494→      }
   495→    })();
   496→  }, []);
   497→
   498→  const totalComplaints = dashboardStats?.complaintsByStatus
   499→    ? Object.values(dashboardStats.complaintsByStatus).reduce((a: number, b: any) => a + Number(b), 0)
   500→    : complaints.length;
   501→
   502→  if (loading) {
   503→    return <div className="space-y-3">{Array.from({ length: 3 }).map((_, i) => <Skeleton key={i} className="h-24 rounded-2xl" />)}</div>;
   504→  }
   505→
   506→  return (
   507→    <div className="space-y-4">
   508→      <div className="flex items-center gap-3">
   509→        <div className="w-10 h-10 rounded-xl bg-[#14532d]/10 flex items-center justify-center">
   510→          <MessageSquare className="w-5 h-5 text-[#14532d]" />
   511→        </div>
   512→        <div>
   513→          <p className="font-semibold text-sm">Complaints Management</p>
   514→          <p className="text-xs text-muted-foreground">{totalComplaints} total complaint{totalComplaints !== 1 ? 's' : ''} on the platform</p>
   515→        </div>
   516→      </div>
   517→
   518→      {complaints.length === 0 ? (
   519→        <Card className="rounded-2xl border-0 shadow-sm"><CardContent className="py-16 text-center"><AlertTriangle className="w-12 h-12 mx-auto text-[#14532d]/30 mb-3" /><p className="text-muted-foreground">No complaints found.</p></CardContent></Card>
   520→      ) : (
   521→        <div className="space-y-3 max-h-[600px] overflow-y-auto">
   522→          {complaints.map((c) => (
   523→            <motion.div key={c.id} variants={fadeVariants} initial="hidden" animate="visible">
   524→              <Card className="rounded-2xl card-hover border-0 shadow-sm">
   525→                <CardContent className="p-4 space-y-2">
   526→                  <div className="flex items-center justify-between flex-wrap gap-2">
   527→                    <div className="flex items-center gap-2">
   528→                      <span className="font-semibold text-sm">{c.subject ?? 'Complaint'}</span>
   529→                      <Badge className={statusColors[c.status] ?? 'bg-gray-100 text-gray-700'}>{c.status}</Badge>
   530→                    </div>
   531→                    <span className="text-xs text-muted-foreground">{c.createdAt ? new Date(c.createdAt).toLocaleDateString('en-IN', { day: 'numeric', month: 'short', year: 'numeric' }) : ''}</span>
   532→                  </div>
   533→                  {c.description && <p className="text-sm text-muted-foreground line-clamp-2">{c.description}</p>}
   534→                  {c.priority && <Badge variant="outline" className="text-xs rounded-full">{c.priority} priority</Badge>}
   535→                </CardContent>
   536→              </Card>
   537→            </motion.div>
   538→          ))}
   539→        </div>
   540→      )}
   541→    </div>
   542→  );
   543→}
