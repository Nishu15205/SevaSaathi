   145→function OverviewTab() {
   146→  const user = useAuthStore((s) => s.user);
   147→  const [stats, setStats] = useState({ patients: 0, activeBookings: 0, reports: 0, pendingReviews: 0 });
   148→  const [recentBookings, setRecentBookings] = useState<any[]>([]);
   149→  const [loading, setLoading] = useState(true);
   150→  const [error, setError] = useState('');
   151→
   152→  const fetchData = useCallback(async () => {
   153→    if (!user?.id) return;
   154→    setLoading(true);
   155→    setError('');
   156→    try {
   157→      const [patientsRes, bookingsRes] = await Promise.all([
   158→        api.patients.list(user.id),
   159→        api.bookings.list({ familyId: user.id }),
   160→      ]);
   161→
   162→      const patients = patientsRes.patients || [];
   163→      const bookings = bookingsRes.bookings || [];
   164→
   165→      const activeBookings = bookings.filter((b: any) => ['PENDING', 'CONFIRMED', 'IN_PROGRESS'].includes(b.status));
   166→      const totalReports = bookings.reduce((acc: number, b: any) => acc + (b.careReports?.length || 0), 0);
   167→      const completedWithoutReview = bookings.filter(
   168→        (b: any) => b.status === 'COMPLETED' && (!b.reviews || b.reviews.length === 0)
   169→      ).length;
   170→
   171→      setStats({
   172→        patients: patients.length,
   173→        activeBookings: activeBookings.length,
   174→        reports: totalReports,
   175→        pendingReviews: completedWithoutReview,
   176→      });
   177→      setRecentBookings(bookings.slice(0, 5));
   178→    } catch (err: any) {
   179→      setError(err.message || 'Failed to load dashboard data');
   180→    } finally {
   181→      setLoading(false);
   182→    }
   183→  }, [user?.id]);
   184→
   185→  useEffect(() => {
   186→    fetchData();
   187→  }, [fetchData]);
   188→
   189→  if (loading) {
   190→    return (
   191→      <div className="space-y-6">
   192→        <LoadingGrid count={4} />
   193→        <div>
   194→          <Skeleton className="h-6 w-40 mb-3" />
   195→          <LoadingCards count={3} />
   196→        </div>
   197→      </div>
   198→    );
   199→  }
   200→
   201→  if (error) {
   202→    return <ErrorState message={error} onRetry={fetchData} />;
   203→  }
   204→
   205→  const statCards = [
   206→    { label: 'Total Patients', value: stats.patients, icon: <Users className="h-5 w-5" />, color: 'bg-forest-50 text-forest-700' },
   207→    { label: 'Active Bookings', value: stats.activeBookings, icon: <CalendarCheck className="h-5 w-5" />, color: 'bg-blue-50 text-blue-700' },
   208→    { label: 'Care Reports', value: stats.reports, icon: <FileText className="h-5 w-5" />, color: 'bg-amber-50 text-amber-700' },
   209→    { label: 'Pending Reviews', value: stats.pendingReviews, icon: <Star className="h-5 w-5" />, color: 'bg-lime-50 text-lime-700' },
   210→  ];
   211→
   212→  return (
   213→    <div className="space-y-6">
   214→      {/* Welcome */}
   215→      <div>
   216→        <h2 className="text-2xl font-bold text-gray-900">
   217→          Welcome back, {user?.name?.split(' ')[0]}!
   218→        </h2>
   219→        <p className="text-sm text-gray-400 mt-1">Here's what's happening with your care requests.</p>
   220→      </div>
   221→
   222→      {/* Stat Cards */}
   223→      <div className="grid grid-cols-1 sm:grid-cols-2 lg:grid-cols-4 gap-4">
   224→        {statCards.map((stat, i) => (
   225→          <motion.div
   226→            key={stat.label}
   227→            initial={{ opacity: 0, y: 12 }}
   228→            animate={{ opacity: 1, y: 0 }}
   229→            transition={{ delay: i * 0.05 }}
   230→          >
   231→            <Card className="rounded-2xl border-gray-100 hover:shadow-md transition-shadow">
   232→              <CardContent className="p-5">
   233→                <div className="flex items-center justify-between">
   234→                  <div>
   235→                    <p className="text-sm text-gray-500 font-medium">{stat.label}</p>
   236→                    <p className="text-3xl font-bold text-gray-900 mt-1">{stat.value}</p>
   237→                  </div>
   238→                  <div className={`w-11 h-11 rounded-2xl flex items-center justify-center ${stat.color}`}>
   239→                    {stat.icon}
   240→                  </div>
   241→                </div>
   242→              </CardContent>
   243→            </Card>
   244→          </motion.div>
   245→        ))}
   246→      </div>
   247→
   248→      {/* Recent Bookings */}
   249→      <div>
   250→        <h3 className="text-base font-semibold text-gray-800 mb-3">Recent Bookings</h3>
   251→        {recentBookings.length === 0 ? (
   252→          <EmptyState
   253→            icon={<CalendarCheck className="h-7 w-7" />}

   260→              <Card key={booking.id} className="rounded-2xl border-gray-100">
   261→                <CardContent className="p-4">

   263→                    <div className="flex items-center gap-3 min-w-0">

   265→                        <UserRound className="h-5 w-5 text-forest-700" />

   298→function PatientsTab() {
   299→  const user = useAuthStore((s) => s.user);
   300→  const [patients, setPatients] = useState<any[]>([]);
   301→  const [loading, setLoading] = useState(true);
   302→  const [error, setError] = useState('');
   303→  const [modalOpen, setModalOpen] = useState(false);
   304→  const [submitting, setSubmitting] = useState(false);
   305→
   306→  // Form state
   307→  const [form, setForm] = useState({
   308→    name: '', age: '', gender: '', relationship: '', address: '', city: '',
   309→    pincode: '', mobilityStatus: '', medicalHistory: '', careRequirements: '',
   310→    emergencyName: '', emergencyPhone: '',
   311→  });
   312→
   313→  const fetchPatients = useCallback(async () => {
   314→    if (!user?.id) return;
   315→    setLoading(true);
   316→    setError('');
   317→    try {
   318→      const res = await api.patients.list(user.id);
   319→      setPatients(res.patients || []);
   320→    } catch (err: any) {
   321→      setError(err.message || 'Failed to load patients');
   322→    } finally {
   323→      setLoading(false);
   324→    }
   325→  }, [user?.id]);
   326→
   327→  useEffect(() => {
   328→    fetchPatients();
   329→  }, [fetchPatients]);
   330→
   331→  const updateForm = (field: string, value: string) => {
   332→    setForm((prev) => ({ ...prev, [field]: value }));
   333→  };
   334→
   335→  const handleSubmit = async (e: React.FormEvent) => {
   336→    e.preventDefault();
   337→    if (!user?.id) return;
   338→    setSubmitting(true);

   340→      await api.patients.create({
   341→        familyId: user.id,

   357→        name: '', age: '', gender: '', relationship: '', address: '', city: '',
   358→        pincode: '', mobilityStatus: '', medicalHistory: '', careRequirements: '',
   359→        emergencyName: '', emergencyPhone: '',
   360→      });
   361→      fetchPatients();
   362→    } catch (err: any) {
   363→      toast.error(err.message || 'Failed to create patient');
