   206→   206→function parseJsonSafe(jsonStr: string | null | undefined): string[] {
   207→   207→  if (!jsonStr) return [];
   208→   208→  try {
   209→   209→    const parsed = JSON.parse(jsonStr);
   210→   210→    return Array.isArray(parsed) ? parsed : [];
   211→   211→  } catch {
   212→   212→    return jsonStr.split(',').map((s) => s.trim()).filter(Boolean);
   213→   213→  }
   214→   214→}
   215→   215→
   216→   216→function formatDate(dateStr: string): string {
   217→   217→  try {
   218→   218→    return new Date(dateStr).toLocaleDateString('en-IN', {
   219→   219→      day: 'numeric',
   220→   220→      month: 'short',
   221→   221→      year: 'numeric',
   222→   222→    });
   223→   223→  } catch {
   224→   224→    return dateStr;
   225→   225→  }
   226→   226→}
   227→   227→
   228→   228→function getTodayDate(): string {
   229→   229→  const now = new Date();
   230→   230→  const year = now.getFullYear();
   231→   231→  const month = String(now.getMonth() + 1).padStart(2, '0');
   232→   232→  const day = String(now.getDate()).padStart(2, '0');
   233→   233→  return `${year}-${month}-${day}`;
   234→   234→}

   790→   790→  const handleFileSelect = async (e: React.ChangeEvent<HTMLInputElement>) => {

   796→   796→    reader.onload = (ev) => {

   813→   813→        toast.success('Aadhar card verified successfully!');

   819→   819→      setAadharResult({ verified: false, error: err.message });

   821→   821→      setVerifying(false);
   822→   822→    }
   823→   823→  };
   824→   824→
   825→   825→  return (
   826→   826→    <div className="mt-6">
