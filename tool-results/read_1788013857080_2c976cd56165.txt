   185→      setLoading(false);

   225→        {statCards.map((stat, i) => (

   227→            key={stat.label}

   229→            animate={{ opacity: 1, y: 0 }}
   230→            transition={{ delay: i * 0.05 }}
   231→          >
   232→            <Card className="rounded-2xl border-gray-100 hover:shadow-md transition-shadow">

   244→            </Card>

   246→        ))}
   247→      </div>
   248→
   249→      {/* Onboarding Guide for new users */}
   250→      {isNewUser && (
   251→        <Card className="rounded-2xl border-2 border-forest-200 bg-gradient-to-br from-forest-50 to-lime-50/30 overflow-hidden">
   252→          <CardContent className="p-6">
   253→            <div className="flex items-center gap-3 mb-4">

   339→function PatientsTab() {
   340→  const user = useAuthStore((s) => s.user);
   341→  const [patients, setPatients] = useState<any[]>([]);
   342→  const [loading, setLoading] = useState(false);
   343→  const [error, setError] = useState('');
   344→  const [modalOpen, setModalOpen] = useState(false);
   345→  const [submitting, setSubmitting] = useState(false);
   346→
   347→  // Form state
   348→  const [form, setForm] = useState({
   349→    name: '', age: '', gender: '', relationship: '', address: '', city: '',
   350→    pincode: '', mobilityStatus: '', medicalHistory: '', careRequirements: '',
   351→    emergencyName: '', emergencyPhone: '',
   352→  });
   353→
   354→  const fetchPatients = useCallback(async () => {
   355→    if (!user?.id) return;
   356→    setLoading(true);
   357→    setError('');
   358→    try {
   359→      const res = await api.patients.list(user.id);
   360→      setPatients(res.patients || []);
   361→    } catch (err: any) {
   362→      setError(err.message || 'Failed to load patients');
   363→    } finally {
   364→      setLoading(false);
   365→    }
   366→  }, [user?.id]);
   367→
   368→  useEffect(() => {
   369→    fetchPatients();
   370→  }, [fetchPatients]);
   371→
   372→  const updateForm = (field: string, value: string) => {
   373→    setForm((prev) => ({ ...prev, [field]: value }));
   374→  };
   375→
   376→  const handleSubmit = async (e: React.FormEvent) => {
   377→    e.preventDefault();
   378→    if (!user?.id) return;
   379→    setSubmitting(true);
   380→    try {
   381→      await api.patients.create({
   382→        familyId: user.id,
   383→        name: form.name,
   384→        age: parseInt(form.age, 10) || 0,
   385→        gender: form.gender,
   386→        relationship: form.relationship,
   387→        address: form.address,
   388→        city: form.city,
   389→        pincode: form.pincode || undefined,
   390→        mobilityStatus: form.mobilityStatus || 'mobile',
   391→        medicalHistory: form.medicalHistory,
   392→        careRequirements: form.careRequirements,
   393→        emergencyContact: JSON.stringify({ name: form.emergencyName, phone: form.emergencyPhone }),
   394→      });
   395→      toast.success('Member profile created successfully!');
   396→      setModalOpen(false);
   397→      setForm({
   398→        name: '', age: '', gender: '', relationship: '', address: '', city: '',
   399→        pincode: '', mobilityStatus: '', medicalHistory: '', careRequirements: '',
   400→        emergencyName: '', emergencyPhone: '',
   401→      });
   402→      fetchPatients();
   403→    } catch (err: any) {
   404→      toast.error(err.message || 'Failed to create member');
   405→    } finally {
   406→      setSubmitting(false);
   407→    }
   408→  };
   409→
   410→  if (error) {
   411→    return <ErrorState message={error} onRetry={fetchPatients} />;
   412→  }
   413→
   414→  const mobilityLabel: Record<string, string> = {
   415→    mobile: 'Mobile', bedridden: 'Bedridden', limited: 'Limited Mobility', wheelchair: 'Wheelchair',
   416→  };
   417→
   418→  return (
   419→    <div className="space-y-6">
   420→      <div className="flex items-center justify-between">
   421→        <div>
   422→          <h2 className="text-xl font-bold text-gray-900">Member Profiles</h2>
   423→          <p className="text-sm text-gray-400">Manage care recipients in your family.</p>
   424→        </div>
   425→        <Dialog open={modalOpen} onOpenChange={setModalOpen}>
   426→          <DialogTrigger asChild>
   427→            <Button className="btn-black text-sm gap-2">
   428→              <Plus className="h-4 w-4" /> Add Member
   429→            </Button>
   430→          </DialogTrigger>
   431→          <DialogContent className="max-w-lg max-h-[90vh] overflow-y-auto rounded-2xl">
   432→            <DialogHeader>
   433→              <DialogTitle>Add New Member</DialogTitle>
   434→            </DialogHeader>
   435→            <form onSubmit={handleSubmit} className="space-y-4 mt-2">
   436→              <div className="grid grid-cols-2 gap-3">
   437→                <div className="col-span-2">
   438→                  <Label className="text-xs font-medium text-gray-600">Full Name *</Label>
   439→                  <Input
   440→                    required
   441→                    value={form.name}
   442→                    onChange={(e) => updateForm('name', e.target.value)}
   443→                    placeholder="Member name"
   444→                    className="mt-1 rounded-xl"
   445→                  />
   446→                </div>
   447→                <div>
   448→                  <Label className="text-xs font-medium text-gray-600">Age *</Label>
   449→                  <Input
   450→                    required
   451→                    type="number"
   452→                    min={0}
   453→                    max={150}
   454→                    value={form.age}
   455→                    onChange={(e) => updateForm('age', e.target.value)}
   456→                    placeholder="Age"
   457→                    className="mt-1 rounded-xl"
   458→                  />
   459→                </div>
   460→                <div>
   461→                  <Label className="text-xs font-medium text-gray-600">Gender *</Label>
   462→                  <Select value={form.gender} onValueChange={(v) => updateForm('gender', v)}>
   463→                    <SelectTrigger className="mt-1 rounded-xl">
   464→                      <SelectValue placeholder="Select gender" />
   465→                    </SelectTrigger>
   466→                    <SelectContent>
   467→                      <SelectItem value="Male">Male</SelectItem>
