   210→   210→function parseJsonSafe(jsonStr: string | null | undefined): string[] {
   211→   211→  if (!jsonStr) return [];
   212→   212→  try {
   213→   213→    const parsed = JSON.parse(jsonStr);
   214→   214→    return Array.isArray(parsed) ? parsed : [];
   215→   215→  } catch {
   216→   216→    return jsonStr.split(',').map((s) => s.trim()).filter(Boolean);
   217→   217→  }
   218→   218→}

   220→   220→function formatDate(dateStr: string): string {
   221→   221→  try {
   222→   222→    return new Date(dateStr).toLocaleDateString('en-IN', {
   223→   223→      day: 'numeric',
   224→   224→      month: 'short',
   225→   225→      year: 'numeric',
   226→   226→    });
   227→   227→  } catch {
   228→   228→    return dateStr;
   229→   229→  }
   230→   230→}

   232→   232→function getTodayDate(): string {
   233→   233→  const now = new Date();
   234→   234→  const year = now.getFullYear();
   235→   235→  const month = String(now.getMonth() + 1).padStart(2, '0');
   236→   236→  const day = String(now.getDate()).padStart(2, '0');
   237→   237→  return `${year}-${month}-${day}`;
   238→   238→}

   262→   262→      const res = await api.auth.sendPhoneOtp(user.phone);

   267→   267→      toast.error(err.message || 'Failed to send OTP');

   269→   269→      setPhoneSending(false);
   270→   270→    }

   272→   272→

   301→   301→      }));

   303→   303→    } catch (err: any) {
   304→   304→      setError(err.message || 'Failed to load dashboard data');
   305→   305→    } finally {
   306→   306→      setLoading(false);
   307→   307→    }
   308→   308→  }, [profile?.id]);

   310→   310→  useEffect(() => {
   311→   311→    fetchData();
   312→   312→  }, [fetchData]);

   314→   314→  // No profile yet — show setup guidance immediately, no loading flash

   325→   325→              <ClipboardList className="h-7 w-7 text-white" />

   982→   982→        <p className="text-xs text-gray-500 mb-4">Enter your 12-digit Aadhaar number to get verified. Verified caregivers get more bookings and trust from families.</p>
