   124→/* ================================================================== */
   125→/*  MAIN COMPONENT                                                     */
   126→/* ================================================================== */
   127→export default function LoginModal({ isOpen, onClose, defaultTab }: LoginModalProps) {
   128→  /* ---- state ---- */
   129→  const [tab, setTab] = useState<Tab>(defaultTab === "register" ? "register" : "login");
   130→  const [showPassword, setShowPassword] = useState(false);
   131→  const [loading, setLoading] = useState(false);
   132→  const [googleLoading, setGoogleLoading] = useState(false);
   133→  const [error, setError] = useState("");
   134→  const [resetOpen, setResetOpen] = useState(false);
   135→
   136→  // Login form
   137→  const [loginEmail, setLoginEmail] = useState("");
   138→  const [loginPassword, setLoginPassword] = useState("");
   139→
   140→  // Register form
   141→  const [regName, setRegName] = useState("");
   142→  const [regEmail, setRegEmail] = useState("");
   143→  const [regPhone, setRegPhone] = useState("");
   144→  const [regPassword, setRegPassword] = useState("");
   145→  const [regRole, setRegRole] = useState<UserRole>("FAMILY");
   146→
   147→  // Reset form (simplified — no OTP)
   148→  const [resetEmail, setResetEmail] = useState("");
   149→  const [resetNewPassword, setResetNewPassword] = useState("");
   150→  const [resetLoading, setResetLoading] = useState(false);
   151→  const [resetError, setResetError] = useState("");
   152→  const [resetDone, setResetDone] = useState(false);
   153→
   154→  const { setAuth } = useAuthStore();
   155→
   156→  /* ---- reset form fields when switching tabs ---- */
   157→  useEffect(() => {
   158→    setError("");
   159→    setLoginPassword("");
   160→  }, [tab]);
   161→
   162→  /* ---- reset entire state when modal closes ---- */
   163→  const handleClose = useCallback(() => {
   164→    setTab(defaultTab === "register" ? "register" : "login");
   165→    setError("");
   166→    setShowPassword(false);
   167→    setLoading(false);
   168→    setGoogleLoading(false);
   169→    setLoginEmail("");
   170→    setLoginPassword("");
   171→    setRegName("");
   172→    setRegEmail("");
   173→    setRegPhone("");
   174→    setRegPassword("");
   175→    setRegRole("FAMILY");
   176→    setResetOpen(false);
   177→    setResetEmail("");
   178→    setResetNewPassword("");
   179→    setResetLoading(false);
   180→    setResetError("");
   181→    setResetDone(false);
   182→    onClose();
   183→  }, [onClose, defaultTab]);
   184→
   185→  /* ================================================================ */
   186→  /*  HANDLERS                                                         */
   187→  /* ================================================================ */
   188→
   189→  /* ---- Google sign-in ---- */
   190→  const handleGoogleSignIn = async () => {
   191→    setGoogleLoading(true);
   192→    setError("");
   193→    try {
   194→      await signIn("google", { redirect: false });
   195→      toast.success("Welcome to SevaSaathi!");
   196→      handleClose();
   197→    } catch {
   198→      setError("Google sign-in failed. Please try again.");
   199→    } finally {
   200→      setGoogleLoading(false);
   201→    }
   202→  };
   203→
   204→  /* ---- Email / password login ---- */
   205→  const handleLogin = async (e: React.FormEvent) => {
   206→    e.preventDefault();
   207→    if (!loginEmail.trim() || !loginPassword.trim()) {
   208→      setError("Please enter your email and password.");
   209→      return;
   210→    }
   211→    setLoading(true);
   212→    setError("");
   213→    try {
   214→      const result = await signIn("credentials", {
   215→        email: loginEmail.trim(),
   216→        password: loginPassword,
   217→        redirect: false,
   218→      });
   219→
   220→      if (result?.error) {
   221→        setError("Invalid email or password. Please try again.");
   222→        return;
   223→      }
   224→
   225→      // AuthSync in AuthProvider will automatically sync session → Zustand store
   226→      toast.success("Welcome back! 🎉");
   227→      handleClose();
   228→    } catch {
   229→      setError("Something went wrong. Please try again.");
   230→    } finally {
   231→      setLoading(false);
   232→    }
   233→  };
   234→
   235→  /* ---- Registration ---- */
   236→  const handleRegister = async (e: React.FormEvent) => {
   237→    e.preventDefault();
   238→    if (!regName.trim() || !regEmail.trim() || !regPhone.trim() || !regPassword.trim()) {
   239→      setError("Please fill in all fields.");
   240→      return;
   241→    }
   242→    if (regPassword.length < 6) {
   243→      setError("Password must be at least 6 characters.");
   244→      return;
   245→    }
   246→    setLoading(true);
   247→    setError("");
   248→    try {
   249→      // 1. Register
   250→      const regRes = await api.auth.register({
   251→        name: regName.trim(),
   252→        email: regEmail.trim(),
   253→        phone: regPhone.trim(),
   254→        password: regPassword,
   255→        role: regRole,
   256→      });
   257→
   258→      // 2. Auto-login with credentials
   259→      const loginResult = await signIn("credentials", {
   260→        email: regEmail.trim(),
   261→        password: regPassword,
   262→        redirect: false,
   263→      });
   264→
   265→      if (loginResult?.error) {
   266→        // Registration succeeded but auto-login failed — that's ok, just notify
   267→        toast.success("Account created! Please log in.");
   268→        setTab("login");
   269→        setLoginEmail(regEmail.trim());
   270→        return;
   271→      }
   272→
   273→      toast.success("Welcome to SevaSaathi! 🎉");
   274→      handleClose();
   275→    } catch (err: any) {
   276→      setError(err?.message || "Registration failed. Please try again.");
   277→    } finally {
   278→      setLoading(false);
   279→    }
   280→  };
   281→
   282→  /* ---- Reset password (simplified) ---- */
   283→  const handleReset = async (e: React.FormEvent) => {
   284→    e.preventDefault();
   285→    if (!resetEmail.trim() || !resetNewPassword.trim()) {
   286→      setResetError("Please enter your email and new password.");
   287→      return;
   288→    }
   289→    if (resetNewPassword.length < 6) {
   290→      setResetError("Password must be at least 6 characters.");
   291→      return;
   292→    }
   293→    setResetLoading(true);
   294→    setResetError("");
   295→    try {
   296→      const res = await fetch("/api/auth/reset-password", {
   297→        method: "POST",
   298→        headers: { "Content-Type": "application/json" },
   299→        body: JSON.stringify({ email: resetEmail.trim(), newPassword: resetNewPassword }),
   300→      });
   301→      const data = await res.json();
   302→      if (!res.ok) {
   303→        setResetError(data.error || "Failed to reset password.");
   304→        return;
   305→      }
   306→      setResetDone(true);
   307→      toast.success("Password reset successfully!");
   308→    } catch {
   309→      setResetError("Something went wrong. Please try again.");
   310→    } finally {
   311→      setResetLoading(false);
   312→    }
   313→  };
   314→
   315→  /* ================================================================ */
   316→  /*  RENDER                                                           */
   317→  /* ================================================================ */
   318→  return (
   319→    <Dialog open={isOpen} onOpenChange={(open) => !open && handleClose()}>
   320→      <DialogContent className="sm:max-w-md p-0 overflow-hidden border-forest-200/40">
   321→        {/* Custom X button to override default position */}
   322→        <button
   323→          onClick={handleClose}
   324→          className="absolute top-3 right-3 z-50 rounded-full p-1.5 text-gray-400 hover:text-gray-600 hover:bg-gray-100 transition-colors"
   325→          aria-label="Close"
   326→        >
   327→          <X className="w-5 h-5" />
   328→        </button>
   329→
   330→        <div className="px-6 pt-6 pb-2">
   331→          <DialogHeader className="space-y-1 mb-1">
   332→            <div className="flex items-center gap-2 justify-center">
   333→              <div className="w-9 h-9 rounded-xl bg-forest-900 flex items-center justify-center">
   334→                <Heart className="w-5 h-5 text-lime-400" />
   335→              </div>
   336→              <DialogTitle className="text-xl font-bold text-forest-900">
   337→                SevaSaathi
   338→              </DialogTitle>
   339→            </div>
   340→            <DialogDescription className="text-sm text-gray-500 mt-1">
   341→              {tab === "login"
   342→                ? "Sign in to your account"
   343→                : "Create your account to get started"}
   344→            </DialogDescription>
   345→          </DialogHeader>
   346→        </div>
   347→
   348→        {/* ---- Error banner ---- */}
   349→        <AnimatePresence>
   350→          {error && (
   351→            <motion.div
   352→              initial={{ height: 0, opacity: 0 }}
   353→              animate={{ height: "auto", opacity: 1 }}
   354→              exit={{ height: 0, opacity: 0 }}
   355→              className="overflow-hidden"
   356→            >
   357→              <div className="mx-6 mt-1 px-3 py-2 rounded-lg bg-red-50 border border-red-200 text-red-700 text-sm flex items-center gap-2">
   358→                <X className="w-4 h-4 flex-shrink-0" />
   359→                {error}
   360→              </div>
   361→            </motion.div>
   362→          )}
   363→        </AnimatePresence>
   364→
   365→        {/* ---- Tab switcher ---- */}
   366→        <div className="px-6 pt-3">
   367→          <div className="flex bg-gray-100 rounded-lg p-1">
   368→            <button
   369→              onClick={() => setTab("login")}
   370→              className={`flex-1 py-2 px-4 rounded-md text-sm font-medium transition-all duration-200 ${
   371→                tab === "login"
   372→                  ? "bg-white text-forest-900 shadow-sm"
   373→                  : "text-gray-500 hover:text-gray-700"
   374→              }`}
   375→            >
   376→              Sign In
   377→            </button>
   378→            <button
   379→              onClick={() => setTab("register")}
   380→              className={`flex-1 py-2 px-4 rounded-md text-sm font-medium transition-all duration-200 ${
   381→                tab === "register"
   382→                  ? "bg-white text-forest-900 shadow-sm"
   383→                  : "text-gray-500 hover:text-gray-700"
   384→              }`}
   385→            >
   386→              Create Account
   387→            </button>
   388→          </div>
   389→        </div>
   390→
   391→        {/* ---- Animated content ---- */}
   392→        <div className="px-6 py-4 overflow-hidden">
   393→          <AnimatePresence mode="wait" custom={tab}>
   394→            {/* ========== LOGIN TAB ========== */}
   395→            {tab === "login" && (
   396→              <motion.div
   397→                key="login"
   398→                custom={tab}
   399→                variants={contentVariants}
   400→                initial="enter"
   401→                animate="center"
   402→                exit="exit"
   403→                className="space-y-4"
   404→              >
   405→                {/* Google button */}
   406→                <button
   407→                  type="button"
   408→                  onClick={handleGoogleSignIn}
   409→                  disabled={googleLoading}
   410→                  className="w-full flex items-center justify-center gap-3 h-11 px-4 rounded-lg border border-gray-300 bg-white hover:bg-gray-50 active:bg-gray-100 transition-all duration-150 text-sm font-medium text-gray-700 disabled:opacity-60 disabled:cursor-not-allowed shadow-sm"
   411→                >
   412→                  {googleLoading ? (
   413→                    <Loader2 className="w-5 h-5 animate-spin text-gray-500" />
   414→                  ) : (
   415→                    <GoogleIcon />
   416→                  )}
   417→                  Continue with Google
   418→                </button>
   419→
   420→                {/* Divider */}
   421→                <div className="flex items-center gap-3">
   422→                  <div className="flex-1 h-px bg-gray-200" />
   423→                  <span className="text-xs text-gray-400 font-medium uppercase tracking-wider">
   424→                    or continue with email
   425→                  </span>
   426→                  <div className="flex-1 h-px bg-gray-200" />
   427→                </div>
   428→
   429→                {/* Email / password form */}
   430→                <form onSubmit={handleLogin} className="space-y-3">
   431→                  <div className="space-y-1.5">
   432→                    <Label htmlFor="login-email" className="text-sm font-medium text-gray-700">
   433→                      Email
   434→                    </Label>
   435→                    <div className="relative">
   436→                      <Mail className="absolute left-3 top-1/2 -translate-y-1/2 w-4 h-4 text-gray-400" />
   437→                      <Input
   438→                        id="login-email"
   439→                        type="email"
   440→                        placeholder="you@example.com"
   441→                        value={loginEmail}
   442→                        onChange={(e) => setLoginEmail(e.target.value)}
   443→                        className="pl-10 h-11 border-gray-300 focus:border-forest-500 focus:ring-forest-500"
   444→                        autoComplete="email"
   445→                        disabled={loading}
   446→                      />
   447→                    </div>
   448→                  </div>
   449→
   450→                  <div className="space-y-1.5">
   451→                    <div className="flex items-center justify-between">
   452→                      <Label htmlFor="login-password" className="text-sm font-medium text-gray-700">
   453→                        Password
   454→                      </Label>
   455→                      <button
   456→                        type="button"
   457→                        onClick={() => setResetOpen(true)}
   458→                        className="text-xs font-medium text-forest-700 hover:text-forest-900 transition-colors"
   459→                      >
   460→                        Forgot password?
   461→                      </button>
   462→                    </div>
   463→                    <div className="relative">
   464→                      <Lock className="absolute left-3 top-1/2 -translate-y-1/2 w-4 h-4 text-gray-400" />
   465→                      <Input
   466→                        id="login-password"
   467→                        type={showPassword ? "text" : "password"}
   468→                        placeholder="Enter your password"
   469→                        value={loginPassword}
   470→                        onChange={(e) => setLoginPassword(e.target.value)}
   471→                        className="pl-10 pr-10 h-11 border-gray-300 focus:border-forest-500 focus:ring-forest-500"
   472→                        autoComplete="current-password"
   473→                        disabled={loading}
   474→                      />
   475→                      <button
   476→                        type="button"
   477→                        onClick={() => setShowPassword(!showPassword)}
   478→                        className="absolute right-3 top-1/2 -translate-y-1/2 text-gray-400 hover:text-gray-600 transition-colors"
   479→                        tabIndex={-1}
   480→                      >
   481→                        {showPassword ? <EyeOff className="w-4 h-4" /> : <Eye className="w-4 h-4" />}
   482→                      </button>
   483→                    </div>
   484→                  </div>
   485→
   486→                  <Button
   487→                    type="submit"
   488→                    disabled={loading}
   489→                    className="w-full h-11 bg-forest-900 hover:bg-forest-800 text-white font-semibold rounded-lg transition-all duration-150 shadow-sm hover:shadow-md"
   490→                  >
   491→                    {loading ? (
   492→                      <>
   493→                        <Loader2 className="w-4 h-4 mr-2 animate-spin" />
   494→                        Signing in...
   495→                      </>
   496→                    ) : (
   497→                      <>
   498→                        Sign In
   499→                        <ArrowRight className="w-4 h-4 ml-2" />
   500→                      </>
   501→                    )}
   502→                  </Button>
   503→                </form>
   504→
   505→                {/* Switch to register */}
   506→                <p className="text-center text-sm text-gray-500 pt-1">
   507→                  Don&apos;t have an account?{" "}
   508→                  <button
   509→                    onClick={() => setTab("register")}
   510→                    className="font-semibold text-forest-700 hover:text-forest-900 transition-colors"
   511→                  >
   512→                    Create one
   513→                  </button>
   514→                </p>
   515→              </motion.div>
   516→            )}
   517→
   518→            {/* ========== REGISTER TAB ========== */}
   519→            {tab === "register" && (
   520→              <motion.div
   521→                key="register"
   522→                custom={tab}
   523→                variants={contentVariants}
   524→                initial="enter"
   525→                animate="center"
   526→                exit="exit"
   527→                className="space-y-4"
   528→              >
   529→                {/* Google button */}
   530→                <button
   531→                  type="button"
   532→                  onClick={handleGoogleSignIn}
   533→                  disabled={googleLoading}
   534→                  className="w-full flex items-center justify-center gap-3 h-11 px-4 rounded-lg border border-gray-300 bg-white hover:bg-gray-50 active:bg-gray-100 transition-all duration-150 text-sm font-medium text-gray-700 disabled:opacity-60 disabled:cursor-not-allowed shadow-sm"
   535→                >
   536→                  {googleLoading ? (
   537→                    <Loader2 className="w-5 h-5 animate-spin text-gray-500" />
   538→                  ) : (
   539→                    <GoogleIcon />
   540→                  )}
   541→                  Sign up with Google
   542→                </button>
   543→
   544→                {/* Divider */}
   545→                <div className="flex items-center gap-3">
   546→                  <div className="flex-1 h-px bg-gray-200" />
   547→                  <span className="text-xs text-gray-400 font-medium uppercase tracking-wider">
   548→                    or register with email
   549→                  </span>
   550→                  <div className="flex-1 h-px bg-gray-200" />
   551→                </div>
   552→
   553→                {/* Registration form */}
   554→                <form onSubmit={handleRegister} className="space-y-3">
   555→                  {/* Name */}
   556→                  <div className="space-y-1.5">
   557→                    <Label htmlFor="reg-name" className="text-sm font-medium text-gray-700">
   558→                      Full Name
   559→                    </Label>
   560→                    <div className="relative">
   561→                      <User className="absolute left-3 top-1/2 -translate-y-1/2 w-4 h-4 text-gray-400" />
   562→                      <Input
   563→                        id="reg-name"
   564→                        type="text"
   565→                        placeholder="Your full name"
   566→                        value={regName}
   567→                        onChange={(e) => setRegName(e.target.value)}
   568→                        className="pl-10 h-11 border-gray-300 focus:border-forest-500 focus:ring-forest-500"
   569→                        autoComplete="name"
   570→                        disabled={loading}
   571→                      />
   572→                    </div>
   573→                  </div>
   574→
   575→                  {/* Email */}
   576→                  <div className="space-y-1.5">
   577→                    <Label htmlFor="reg-email" className="text-sm font-medium text-gray-700">
   578→                      Email
   579→                    </Label>
   580→                    <div className="relative">
   581→                      <Mail className="absolute left-3 top-1/2 -translate-y-1/2 w-4 h-4 text-gray-400" />
   582→                      <Input
   583→                        id="reg-email"
   584→                        type="email"
   585→                        placeholder="you@example.com"
   586→                        value={regEmail}
   587→                        onChange={(e) => setRegEmail(e.target.value)}
   588→                        className="pl-10 h-11 border-gray-300 focus:border-forest-500 focus:ring-forest-500"
   589→                        autoComplete="email"
   590→                        disabled={loading}
   591→                      />
   592→                    </div>
   593→                  </div>
   594→
