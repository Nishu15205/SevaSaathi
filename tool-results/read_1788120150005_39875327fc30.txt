   215→function parseJsonSafe(input: string | null | undefined | any): string[] {
   216→  if (!input) return [];
   217→  if (Array.isArray(input)) return input;
   218→  try {
   219→    const parsed = JSON.parse(input);
   220→    return Array.isArray(parsed) ? parsed : [];
   221→  } catch {
   222→    if (typeof input === 'string') {
   223→      return input.split(',').map((s) => s.trim()).filter(Boolean);
   224→    }
   225→    return [];
   226→  }
   227→}

   229→function formatDate(dateStr: string): string {
   230→  try {
   231→    return new Date(dateStr).toLocaleDateString('en-IN', {
   232→      day: 'numeric',
   233→      month: 'short',
   234→      year: 'numeric',
   235→    });
   236→  } catch {
   237→    return dateStr;
   238→  }
   239→}
   240→
   241→function getTodayDate(): string {
   242→  const now = new Date();
   243→  const year = now.getFullYear();
   244→  const month = String(now.getMonth() + 1).padStart(2, '0');
   245→  const day = String(now.getDate()).padStart(2, '0');
   246→  return `${year}-${month}-${day}`;
   247→}

   253→function OverviewTab({ user }: { user: User }) {
   254→  const profile = user.caregiverProfile;
   255→  const [bookings, setBookings] = useState<Booking[]>([]);
   256→  // Start with loading=false so new caregivers (no profile) never see a skeleton flash
   257→  const [loading, setLoading] = useState(false);
   258→  const [error, setError] = useState('');
   259→
   260→  // Phone verification — MSG91 OTP
   261→  const [phoneVerifying, setPhoneVerifying] = useState(false);
   262→  const [showOtpDialog, setShowOtpDialog] = useState(false);
   263→  const [otpSent, setOtpSent] = useState(false);
   264→  const [otpError, setOtpError] = useState<string | null>(null);
   265→  const [otpValue, setOtpValue] = useState('');
   266→  const [devOtp, setDevOtp] = useState<string | null>(null);
   267→  const [otpVia, setOtpVia] = useState<'sms' | 'dev' | 'msg91'>('sms');
   268→
   269→  const maskPhone = (p: string) => {
   270→    const d = p.replace(/\D/g, '');
   271→    if (d.length < 4) return p;
   272→    return '*'.repeat(d.length - 4) + d.slice(-4);
   273→  };
   274→
   275→  const handleSendOtp = async () => {
   276→    setPhoneVerifying(true);
   277→    setShowOtpDialog(true);
   278→    setOtpSent(false);
   279→    setOtpError(null);
   280→    setOtpValue('');
   281→    setDevOtp(null);
   282→
   283→    try {
   284→      const res = await fetch('/api/auth/send-phone-otp', {
   285→        method: 'POST',
   286→        headers: { 'Content-Type': 'application/json' },
   287→        body: JSON.stringify({ phone: user.phone, userId: user.id }),
   288→      });
   289→      const data = await res.json();
   290→
   291→      if (!res.ok) {
   292→        setOtpError(data.error || 'Failed to send OTP');
   293→        return;
   294→      }
   295→
   296→      setOtpSent(true);
   297→      setOtpVia(data.via || 'sms');
   298→      if (data.via === 'dev' && data.devOtp) {
   299→        setDevOtp(data.devOtp);
   300→      }
   301→      toast.success(data.via === 'dev'
   302→        ? 'OTP generated (dev mode)'
   303→        : 'OTP sent to your phone via SMS!'
   304→      );
   305→    } catch (e: any) {
   306→      console.error('Phone OTP send error:', e);
   307→      setOtpError(e?.message || 'Something went wrong. Please try again.');
   308→    } finally {
   309→      setPhoneVerifying(false);
   310→    }
   311→  };
   312→
   313→  const handleVerifyOtp = async () => {
   314→    if (!otpValue || otpValue.length < (otpVia === 'msg91' ? 4 : 6)) {
   315→      toast.error('Enter the 6-digit OTP');
   316→      return;
   317→    }
   318→    try {
   319→      const res = await fetch('/api/auth/verify-phone-otp', {
   320→        method: 'POST',
   321→        headers: { 'Content-Type': 'application/json' },
   322→        body: JSON.stringify({
   323→          phone: user.phone,
   324→          otp: otpValue,
   325→        }),
   326→      });
   327→      const data = await res.json();
   328→      if (!res.ok) {
   329→        toast.error(data.error || 'Verification failed');
   330→        return;
   331→      }
   332→      toast.success('Phone number verified successfully!');
   333→      useAuthStore.getState().setAuth({ ...user, phoneVerified: true } as any);
   334→      setShowOtpDialog(false);
   335→      setOtpValue('');
   336→      setOtpSent(false);
   337→      setDevOtp(null);
   338→    } catch (err: any) {
   339→      toast.error(err?.message || 'Verification failed');
   340→    }
   341→  };
   342→
   343→  const fetchData = useCallback(async () => {
   344→    if (!profile?.id) { setLoading(false); return; }
   345→    setLoading(true);
   346→    setError('');
   347→    try {
   348→      const res = await api.bookings.list({ caregiverId: profile.id });
   349→      const all = (res.bookings || []).map((b: any) => ({
   350→        ...b,
   351→        date: b.startDate,
   352→        patientName: b.patient?.name,
   353→        familyName: b.family?.name,
   354→        careType: b.careRequirements ? (typeof b.careRequirements === 'string' ? JSON.parse(b.careRequirements) : b.careRequirements)?.needs?.join(', ') : undefined,
   355→      }));
   356→      setBookings(all);
   357→    } catch (err: any) {
   358→      setError(err.message || 'Failed to load dashboard data');
   359→    } finally {
   360→      setLoading(false);
   361→    }
   362→  }, [profile?.id]);
   363→
   364→  useEffect(() => {
   365→    fetchData();
   366→  }, [fetchData]);
   367→
   368→  // No profile yet — show setup guidance immediately, no loading flash
   369→  if (!profile) {
   370→    return (
   371→      <div className="space-y-6">
   372→        <div>
   373→          <h2 className="text-2xl font-bold text-gray-900">Welcome, {user.name.split(' ')[0]}!</h2>
   374→          <p className="text-sm text-gray-400 mt-1">Complete your profile to start receiving care requests.</p>
   375→        </div>
   376→        <Card className="rounded-2xl border-dashed border-2 border-forest-200 bg-forest-50/50">
   377→          <CardContent className="p-8 text-center">
   378→            <div className="w-16 h-16 rounded-2xl green-gradient-bg flex items-center justify-center mx-auto mb-4">
   379→              <ClipboardList className="h-7 w-7 text-white" />
   380→            </div>
   381→            <h3 className="text-lg font-bold text-gray-800 mb-2">Set Up Your Caregiver Profile</h3>
   382→            <p className="text-sm text-gray-500 max-w-md mx-auto">Go to the <strong>My Profile</strong> tab to create your caregiver profile. Families will be able to find and book you once your profile is set up.</p>
   383→          </CardContent>
   384→        </Card>
   385→      </div>
   386→    );
   387→  }
   388→
   389→  if (error) {
   390→    return <ErrorState message={error} onRetry={fetchData} />;
   391→  }
   392→
   393→  const upcomingBookings = bookings
   394→    .filter((b) => ['CONFIRMED', 'IN_PROGRESS'].includes(b.status))
   395→    .sort((a, b) => new Date(a.date).getTime() - new Date(b.date).getTime());
   396→
   397→  const totalBookings = bookings.length;
   398→  const completedJobs = profile?.completedJobs ?? bookings.filter((b) => b.status === 'COMPLETED').length;
   399→  const overallRating = profile?.overallRating ?? 0;
   400→  const hourlyRate = profile?.hourlyRate ?? 0;
   401→
   402→  const statCards = [
   403→    { label: 'Total Bookings', value: totalBookings, icon: <CalendarCheck className="h-5 w-5" />, color: 'bg-forest-50 text-forest-700' },
   404→    { label: 'Completed Jobs', value: completedJobs, icon: <CheckCircle2 className="h-5 w-5" />, color: 'bg-lime-50 text-lime-700' },
   405→    { label: 'Overall Rating', value: overallRating.toFixed(1), icon: <Star className="h-5 w-5" />, color: 'bg-amber-50 text-amber-700', suffix: '/5' },
   406→    { label: 'Hourly Rate', value: `₹${hourlyRate}`, icon: <IndianRupee className="h-5 w-5" />, color: 'bg-purple-50 text-purple-700', isString: true },
   407→  ];
   408→
   409→  const skills = parseJsonSafe(profile?.skills);
   410→  const languages = parseJsonSafe(profile?.languages);
   411→
   412→  return (
   413→    <div className="space-y-6">
   414→      {/* Phone Verification Banner */}
   415→      {!user.phoneVerified && user.phone && (
   416→        <Card className="rounded-2xl border-2 border-amber-200 bg-gradient-to-r from-amber-50 to-orange-50">
   417→          <CardContent className="p-4 flex items-center justify-between gap-4">
   418→            <div className="flex items-center gap-3 min-w-0">
   419→              <div className="w-10 h-10 rounded-xl bg-gradient-to-br from-amber-400 to-orange-500 flex items-center justify-center shrink-0 shadow-sm">
   420→                <Phone className="h-5 w-5 text-white" />
   421→              </div>
   422→              <div className="min-w-0">
   423→                <p className="text-sm font-semibold text-amber-900">Verify Your Phone Number</p>
   424→                <p className="text-xs text-amber-700/70">Required to receive booking requests — {maskPhone(user.phone)}</p>
   425→              </div>
   426→            </div>
   427→            <Button onClick={handleSendOtp} disabled={phoneVerifying} size="sm" className="bg-gradient-to-r from-forest-700 to-forest-900 hover:from-forest-800 hover:to-forest-950 text-white rounded-xl shrink-0 shadow-sm">
   428→              {phoneVerifying && <Loader2 className="h-4 w-4 animate-spin mr-1.5" />}
   429→              {phoneVerifying ? 'Sending...' : 'Send OTP'}
   430→            </Button>
   431→          </CardContent>
   432→        </Card>
   433→      )}
   434→
   435→      {/* Phone OTP Dialog */}
   436→      <Dialog open={showOtpDialog} onOpenChange={(open) => { if (!open) { setShowOtpDialog(false); setOtpValue(''); setOtpError(null); setOtpSent(false); setDevOtp(null); } }}>
   437→        <DialogContent className="sm:max-w-lg">
   438→          <DialogHeader>
   439→            <DialogTitle>Verify Phone Number</DialogTitle>
   440→            <DialogDescription>Enter the OTP sent via SMS to {maskPhone(user.phone)}</DialogDescription>
   441→          </DialogHeader>
   442→          <div className="space-y-4 pt-2">
   443→            {/* Error display */}
   444→            {otpError && (
   445→              <div className="flex items-start gap-2 bg-red-50 border border-red-200 rounded-xl p-3">
   446→                <AlertTriangle className="h-4 w-4 text-red-500 shrink-0 mt-0.5" />
   447→                <p className="text-xs text-red-700">{otpError}</p>
   448→              </div>
   449→            )}
   450→
   451→            {/* OTP sent success */}
   452→            {otpSent && !otpError && (
   453→              <div className="flex items-center gap-2 bg-green-50 border border-green-200 rounded-xl p-3">
   454→                <div className="w-6 h-6 rounded-full bg-green-500 flex items-center justify-center shrink-0">
   455→                  <span className="text-white text-xs font-bold">✓</span>
   456→                </div>
   457→                <p className="text-xs text-green-700 font-medium">OTP sent to {maskPhone(user.phone)} via SMS</p>
   458→              </div>
   459→            )}
   460→
   461→            {/* Dev mode OTP display */}
   462→            {devOtp && (
   463→              <div className="flex items-center gap-2 bg-violet-50 border border-violet-200 rounded-xl p-3">
   464→                <MessageSquare className="h-4 w-4 text-violet-500 shrink-0" />
   465→                <p className="text-xs text-violet-700 font-medium">
   466→                  Dev OTP: <span className="font-mono font-bold tracking-widest">{devOtp}</span>
   467→                </p>
   468→              </div>
   469→            )}
   470→
   471→            {/* OTP input */}
   472→            {otpSent && !otpError && (
   473→              <>
   474→                <div>
   475→                  <Label className="text-sm">OTP</Label>
   476→                  <Input
   477→                    value={otpValue}
   478→                    onChange={(e) => setOtpValue(e.target.value.replace(/\D/g, '').slice(0, otpVia === 'msg91' ? 4 : 6))}
   479→                    placeholder={otpVia === 'msg91' ? '4-digit OTP' : '6-digit OTP'}
   480→                    className="mt-1 text-center text-lg tracking-[0.5em] font-mono rounded-xl"
   481→                    maxLength={otpVia === 'msg91' ? 4 : 6}
   482→                    autoFocus
   483→                  />
   484→                </div>
   485→                <Button
   486→                  onClick={handleVerifyOtp}
   487→                  disabled={otpValue.length < (otpVia === 'msg91' ? 4 : 6)}
   488→                  className="w-full bg-gradient-to-r from-forest-700 to-forest-900 hover:from-forest-800 hover:to-forest-950 text-white rounded-xl shadow-sm"
   489→                >
   490→                  Verify OTP
   491→                </Button>
   492→                <button
   493→                  type="button"
   494→                  onClick={async () => { setOtpValue(''); setOtpSent(false); setDevOtp(null); setOtpError(null); setShowOtpDialog(false); await handleSendOtp(); }}
   495→                  disabled={phoneVerifying}
   496→                  className="w-full text-xs text-forest-700 hover:text-forest-900 font-medium text-center disabled:opacity-50"
   497→                >
   498→                  {phoneVerifying ? 'Resending...' : 'Resend OTP'}
   499→                </button>
   500→              </>
   501→            )}
   502→
   503→            {/* Still loading */}
   504→            {phoneVerifying && !otpError && !otpSent && (
   505→              <div className="flex items-center gap-2 bg-amber-50 border border-amber-200 rounded-xl p-3">
   506→                <Loader2 className="h-4 w-4 text-amber-500 animate-spin" />
   507→                <p className="text-xs text-amber-700">Sending OTP via SMS...</p>
   508→              </div>
   509→            )}
   510→          </div>
   511→        </DialogContent>
   512→      </Dialog>
   513→
   514→      {/* Welcome */}
   515→      <div>
   516→        <h2 className="text-2xl font-bold text-gray-900">
   517→          Welcome back, {user.name.split(' ')[0]}!
   518→        </h2>
   519→        <p className="text-sm text-gray-400 mt-1">Here&apos;s your caregiving dashboard at a glance.</p>
   520→      </div>
   521→
   522→      {/* Verification Progress Banner */}

   919→function WithdrawalForm({ caregiverId, userId, availableBalance }: { caregiverId: string; userId: string; availableBalance: number }) {
   920→  const [open, setOpen] = useState(false);
   921→  const [method, setMethod] = useState<'upi' | 'bank_transfer'>('upi');
   922→  const [amount, setAmount] = useState('');
   923→  const [upiId, setUpiId] = useState('');
   924→  const [bankName, setBankName] = useState('');
   925→  const [accountNumber, setAccountNumber] = useState('');
   926→  const [ifscCode, setIfscCode] = useState('');
   927→  const [accountHolder, setAccountHolder] = useState('');
   928→  const [submitting, setSubmitting] = useState(false);
   929→
   930→  const resetForm = () => {
   931→    setAmount(''); setUpiId(''); setBankName(''); setAccountNumber(''); setIfscCode(''); setAccountHolder(''); setMethod('upi');
   932→  };
   933→
   934→  const handleSubmit = async (e: React.FormEvent) => {
   935→    e.preventDefault();
   936→    const amountPaise = Math.round(parseFloat(amount) * 100);
   937→    if (!amountPaise || amountPaise <= 0) { toast.error('Enter a valid amount'); return; }
   938→    if (amountPaise > availableBalance) { toast.error(`Insufficient balance. Available: ₹${(availableBalance / 100).toLocaleString('en-IN')}`); return; }
   939→    if (method === 'upi' && !upiId) { toast.error('Enter UPI ID'); return; }
   940→    if (method === 'bank_transfer' && (!bankName || !accountNumber || !ifscCode || !accountHolder)) { toast.error('Fill all bank details'); return; }
   941→
   942→    setSubmitting(true);
   943→    try {
   944→      await api.withdrawals.create({
   945→        userId, caregiverId, amount: amountPaise, method,
   946→        upiId: method === 'upi' ? upiId : undefined,
   947→        bankName: method === 'bank_transfer' ? bankName : undefined,
   948→        accountNumber: method === 'bank_transfer' ? accountNumber : undefined,
   949→        ifscCode: method === 'bank_transfer' ? ifscCode : undefined,
   950→        accountHolder: method === 'bank_transfer' ? accountHolder : undefined,
   951→      });
   952→      toast.success('Withdrawal request submitted! Admin will process it soon.');
   953→      setOpen(false);
   954→      resetForm();
   955→    } catch (err: any) {
   956→      toast.error(err.message || 'Failed to submit withdrawal');
