    90→function useDebounce(value: string, delay: number) {
    91→  const [debounced, setDebounced] = useState(value);
    92→  useEffect(() => {
    93→    const timer = setTimeout(() => setDebounced(value), delay);
    94→    return () => clearTimeout(timer);
    95→  }, [value, delay]);
    96→  return debounced;
    97→}

   235→/* ─────────── USERS TAB ─────────── */
   236→function UsersTab({ onViewUser }: { onViewUser: (userId: string, userName: string) => void }) {
   237→  const [role, setRole] = useState<string>('');
   238→  const [search, setSearch] = useState('');
   239→  const [users, setUsers] = useState<any[]>([]);
   240→  const [loading, setLoading] = useState(true);
   241→  const [page, setPage] = useState(1);
   242→  const [totalPages, setTotalPages] = useState(1);
   243→  const [deleting, setDeleting] = useState<string | null>(null);
   244→
   245→  const debouncedSearch = useDebounce(search, 300);
   246→
   247→  useEffect(() => {
   248→    setPage(1);
   249→  }, [debouncedSearch, role]);
   250→
   251→  const fetchUsers = useCallback(async () => {
   252→    setLoading(true);
   253→    try {
   254→      const res = await api.admin.users({ role: role || undefined, page, limit: 20, search: debouncedSearch || undefined });
   255→      setUsers(res.users ?? []);
   256→      setTotalPages(res.pagination?.totalPages ?? 1);
   257→    } catch {
   258→      toast.error('Failed to load users');
   259→    } finally {
   260→      setLoading(false);
   261→    }
   262→  }, [role, page, debouncedSearch]);
   263→
   264→  useEffect(() => { fetchUsers(); }, [fetchUsers]);
   265→
   266→  const handleDelete = async (userId: string, userName: string) => {
   267→    if (!window.confirm(`Are you sure you want to delete user "${userName}"? This action cannot be undone.`)) return;
   268→    setDeleting(userId);
   269→    try {
   270→      await api.admin.deleteUser(userId);
   271→      toast.success(`User "${userName}" deleted successfully`);
   272→      fetchUsers();
   273→    } catch {
   274→      toast.error('Failed to delete user');
   275→    } finally {
   276→      setDeleting(null);
   277→    }
   278→  };
   279→
   280→  const roleFilters = [
   281→    { label: 'All', value: '' },
   282→    { label: 'Family', value: 'FAMILY' },
   283→    { label: 'Caregiver', value: 'CAREGIVER' },
   284→  ];
   285→
   286→  return (
   287→    <div className="space-y-4">
   288→      <div className="flex flex-col sm:flex-row gap-3 sm:items-center sm:justify-between">
   289→        <SearchBar value={search} onChange={setSearch} placeholder="Search users by name, email, phone..." />
   290→        <div className="flex gap-2">
   291→          {roleFilters.map((f) => (
   292→            <Button
   293→              key={f.value}
   294→              variant={role === f.value ? 'default' : 'outline'}
   295→              size="sm"
   296→              className={role === f.value ? 'bg-[#14532d] hover:bg-[#14532d]/90 text-white rounded-full' : 'rounded-full'}
   297→              onClick={() => { setRole(f.value); setPage(1); }}
   298→            >
   299→              {f.label}
   300→            </Button>
   301→          ))}
   302→        </div>
   303→      </div>
   304→
   305→      {loading ? (
   306→        <div className="space-y-3">{Array.from({ length: 5 }).map((_, i) => <Skeleton key={i} className="h-20 rounded-2xl" />)}</div>
   307→      ) : users.length === 0 ? (
   308→        <Card className="rounded-2xl border-0 shadow-sm"><CardContent className="py-12 text-center text-muted-foreground">No users found.</CardContent></Card>
   309→      ) : (
   310→        <div className="space-y-3">
   311→          {users.map((u) => (
   312→            <motion.div key={u.id} variants={fadeVariants} initial="hidden" animate="visible">
   313→              <Card className="rounded-2xl card-hover border-0 shadow-sm">
   314→                <CardContent className="p-4 flex flex-col sm:flex-row sm:items-center gap-3">
   315→                  <div className="flex-1 min-w-0">
   316→                    <div className="flex items-center gap-2 flex-wrap">
   317→                      <UserNameButton id={u.id} name={u.name} onClick={onViewUser} />
   318→                      <Badge className={statusColors[u.role] ?? 'bg-gray-100 text-gray-700'}>{u.role}</Badge>
   319→                      {u.role === 'CAREGIVER' && u.isVerified && <BadgeCheck className="w-4 h-4 text-[#14532d]" />}
   320→                    </div>
   321→                    <div className="flex items-center gap-4 mt-1 text-xs text-muted-foreground flex-wrap">
   322→                      <span>{u.email}</span>
   323→                      <span>{u.phone}</span>
   324→                      <span>Joined {u.createdAt ? new Date(u.createdAt).toLocaleDateString('en-IN', { day: 'numeric', month: 'short', year: 'numeric' }) : '—'}</span>
   325→                    </div>
   326→                    {u.role === 'CAREGIVER' && (
   327→                      <div className="flex items-center gap-3 mt-1 text-xs text-muted-foreground">
   328→                        {u.city && <span className="flex items-center gap-1"><MapPin className="w-3 h-3" />{u.city}</span>}
   329→                        {u.rating != null && <span className="flex items-center gap-1"><Star className="w-3 h-3 text-amber-500" />{Number(u.rating).toFixed(1)}</span>}
   330→                      </div>
   331→                    )}
   332→                    {u.role === 'FAMILY' && u.patientCount != null && (
   333→                      <div className="mt-1 text-xs text-muted-foreground"><Users className="w-3 h-3 inline mr-1" />{u.patientCount} patient{u.patientCount !== 1 ? 's' : ''}</div>
   334→                    )}
   335→                  </div>
   336→                  <Button
   337→                    size="sm"
   338→                    variant="outline"
   339→                    className="text-red-500 hover:text-red-600 hover:bg-red-50 border-red-200 shrink-0"
   340→                    disabled={deleting === u.id}
   341→                    onClick={() => handleDelete(u.id, u.name)}
   342→                  >
   343→                    {deleting === u.id ? <Loader2 className="w-4 h-4 animate-spin" /> : <Trash2 className="w-4 h-4" />}
   344→                  </Button>
   345→                </CardContent>
   346→              </Card>
   347→            </motion.div>
   348→          ))}
   349→        </div>
   350→      )}
   351→
   352→      {totalPages > 1 && (
   353→        <div className="flex justify-center gap-2 pt-2">
   354→          <Button variant="outline" size="sm" className="rounded-full" disabled={page <= 1} onClick={() => setPage(page - 1)}>Previous</Button>
   355→          <span className="flex items-center text-sm text-muted-foreground px-3">Page {page} of {totalPages}</span>
   356→          <Button variant="outline" size="sm" className="rounded-full" disabled={page >= totalPages} onClick={() => setPage(page + 1)}>Next</Button>
   357→        </div>
   358→      )}
   359→    </div>
   360→  );
   361→}

   363→/* ─────────── VERIFICATIONS TAB ─────────── */
   364→function VerificationsTab({ onViewUser }: { onViewUser: (userId: string, userName: string) => void }) {
   365→  const [search, setSearch] = useState('');
   366→  const [verifications, setVerifications] = useState<any[]>([]);
   367→  const [loading, setLoading] = useState(true);
   368→  const [rejecting, setRejecting] = useState<string | null>(null);
   369→  const [approving, setApproving] = useState<string | null>(null);
   370→  const [reasons, setReasons] = useState<Record<string, string>>({});
   371→
   372→  const debouncedSearch = useDebounce(search, 300);
   373→
   374→  const fetchVerifications = useCallback(async () => {
   375→    setLoading(true);
   376→    try {
   377→      const res = await api.admin.verifications.list({ search: debouncedSearch || undefined });
   378→      setVerifications(res.verifications ?? []);
   379→    } catch {
   380→      toast.error('Failed to load verifications');
   381→    } finally {
   382→      setLoading(false);
   383→    }
   384→  }, [debouncedSearch]);
   385→
   386→  useEffect(() => { fetchVerifications(); }, [fetchVerifications]);
   387→
   388→  const handleApprove = async (id: string) => {
   389→    setApproving(id);
   390→    try {
   391→      await api.admin.verifications.update(id, { status: 'APPROVED' });
   392→      toast.success('Verification approved successfully');
   393→      fetchVerifications();
   394→    } catch {
   395→      toast.error('Failed to approve verification');
   396→    } finally {
   397→      setApproving(null);
   398→    }
   399→  };
   400→
   401→  const handleReject = async (id: string) => {
   402→    const reason = reasons[id]?.trim();
   403→    if (!reason) { toast.error('Please provide a rejection reason'); return; }
   404→    setRejecting(id);
   405→    try {
   406→      await api.admin.verifications.update(id, { status: 'REJECTED', rejectionReason: reason });
   407→      toast.success('Verification rejected');
   408→      setReasons((prev) => { const n = { ...prev }; delete n[id]; return n; });
   409→      fetchVerifications();
   410→    } catch {
   411→      toast.error('Failed to reject verification');
   412→    } finally {
   413→      setRejecting(null);
   414→    }
   415→  };
   416→
   417→  if (loading) {
   418→    return <div className="space-y-3">{Array.from({ length: 3 }).map((_, i) => <Skeleton key={i} className="h-24 rounded-2xl" />)}</div>;
   419→  }
   420→
   421→  if (verifications.length === 0) {
   422→    return (
   423→      <div className="space-y-4">
   424→        <SearchBar value={search} onChange={setSearch} placeholder="Search verifications..." />
   425→        <Card className="rounded-2xl border-0 shadow-sm"><CardContent className="py-16 text-center"><ShieldCheck className="w-12 h-12 mx-auto text-[#14532d]/30 mb-3" /><p className="text-muted-foreground">No pending verifications.</p></CardContent></Card>
   426→      </div>
   427→    );
   428→  }
   429→
   430→  return (
   431→    <div className="space-y-4">
   432→      <SearchBar value={search} onChange={setSearch} placeholder="Search verifications..." />
   433→      {verifications.map((v) => (
   434→        <motion.div key={v.id} variants={fadeVariants} initial="hidden" animate="visible">
   435→          <Card className="rounded-2xl card-hover border-0 shadow-sm">
   436→            <CardContent className="p-4 space-y-3">
   437→              <div className="flex flex-col sm:flex-row sm:items-center justify-between gap-2">
   438→                <div className="flex items-center gap-2 flex-wrap">
   439→                  {v.caregiver?.user?.id ? (
   440→                    <UserNameButton id={v.caregiver.user.id} name={v.caregiver?.user?.name ?? 'Caregiver'} onClick={onViewUser} />
   441→                  ) : (
   442→                    <span className="font-semibold text-sm">{v.caregiver?.user?.name ?? 'Caregiver'}</span>
   443→                  )}
   444→                  <Badge className={docTypeBadge[v.docType] ?? 'bg-gray-100 text-gray-700'}>{v.docType ?? 'DOC'}</Badge>
   445→                </div>
   446→                <span className="text-xs text-muted-foreground flex items-center gap-1">
   447→                  <Clock className="w-3 h-3" />{v.createdAt ? new Date(v.createdAt).toLocaleDateString('en-IN', { day: 'numeric', month: 'short', year: 'numeric' }) : '—'}
   448→                </span>
   449→              </div>
   450→              {v.docNumber && <p className="text-xs text-muted-foreground font-mono bg-muted/50 rounded-lg px-3 py-1.5 w-fit">{v.docNumber}</p>}
   451→              <div className="flex items-center gap-2 pt-1">
   452→                <Button
   453→                  size="sm"
