    82→    82→/* ================================================================== */
    83→    83→export default function LoginModal({ isOpen, onClose, defaultTab }: LoginModalProps) {
    84→    84→  /* ---- state ---- */
    85→    85→  const [tab, setTab] = useState<Tab>(defaultTab === "register" ? "register" : "login");
    86→    86→  const [showPassword, setShowPassword] = useState(false);
    87→    87→  const [loading, setLoading] = useState(false);
    88→    88→  const [error, setError] = useState("");
    89→    89→  const [resetOpen, setResetOpen] = useState(false);
    90→    90→
    91→    91→  // Login form
    92→    92→  const [loginEmail, setLoginEmail] = useState("");
    93→    93→  const [loginPassword, setLoginPassword] = useState("");
    94→    94→
    95→    95→  // Register form
    96→    96→  const [regName, setRegName] = useState("");
    97→    97→  const [regEmail, setRegEmail] = useState("");
    98→    98→  const [regPhone, setRegPhone] = useState("");
    99→    99→  const [regPassword, setRegPassword] = useState("");
   100→   100→  const [regRole, setRegRole] = useState<UserRole>("FAMILY");
   101→   101→
   102→   102→  // Reset form
   103→   103→  const [resetEmail, setResetEmail] = useState("");
   104→   104→  const [resetNewPassword, setResetNewPassword] = useState("");
   105→   105→  const [resetLoading, setResetLoading] = useState(false);
   106→   106→  const [resetError, setResetError] = useState("");
   107→   107→  const [resetDone, setResetDone] = useState(false);
   108→   108→
   109→   109→  const { setAuth } = useAuthStore();
   110→   110→  const { toast } = useToast();
   111→   111→
   112→   112→  // Google popup form
   113→   113→  const [googleOpen, setGoogleOpen] = useState(false);
   114→   114→  const [googleEmail, setGoogleEmail] = useState("");
   115→   115→  const [googleName, setGoogleName] = useState("");
   116→   116→  const [googleError, setGoogleError] = useState("");
   117→   117→  // googleSubmitting removed - redirect-based flow doesn't need it
   118→   118→
   119→   119→  /* ---- reset form fields when switching tabs ---- */
   120→   120→  useEffect(() => { setError(""); setLoginPassword(""); }, [tab]);
   121→   121→
   122→   122→  /* ---- reset entire state when modal closes ---- */
   123→   123→  const handleClose = useCallback(() => {
   124→   124→    setTab(defaultTab === "register" ? "register" : "login");
   125→   125→    setError(""); setShowPassword(false); setLoading(false);
   126→   126→    setLoginEmail(""); setLoginPassword("");
   127→   127→    setRegName(""); setRegEmail(""); setRegPhone(""); setRegPassword(""); setRegRole("FAMILY");
   128→   128→    setResetOpen(false); setResetEmail(""); setResetNewPassword("");
   129→   129→    setResetLoading(false); setResetError(""); setResetDone(false);
   130→   130→    setGoogleOpen(false); setGoogleEmail(""); setGoogleName(""); setGoogleError("");
   131→   131→    onClose();
   132→   132→  }, [onClose, defaultTab]);
   133→   133→
   134→   134→  /* ================================================================ */
   135→   135→  /*  HANDLERS                                                         */
   136→   136→/* ================================================================ */
   137→   137→
   138→   138→  const handleGoogleSignIn = async () => {
   139→   139→    setGoogleError("");
   140→   140→    try {
   141→   141→      const checkRes = await fetch("/api/auth/google-configured");
   142→   142→      const { configured } = await checkRes.json();
   143→   143→      if (configured) {
   144→   144→        const origin = window.location.origin;
   145→   145→        const role = tab === "register" ? regRole : "";
   146→   146→        window.location.href = `/api/auth/google-go?origin=${encodeURIComponent(origin)}&role=${encodeURIComponent(role)}`;
   147→   147→        return;
   148→   148→      }
   149→   149→    } catch { /* fall through */ }
   150→   150→    setGoogleOpen(true);
   151→   151→  };
   152→   152→
   153→   153→  const handleGoogleSubmit = (e: React.FormEvent) => {
   154→   154→    e.preventDefault();
   155→   155→    if (!googleEmail.trim() || !googleName.trim()) { setGoogleError("Please enter your name and email."); return; }
   156→   156→    if (!googleEmail.trim().includes("@")) { setGoogleError("Please enter a valid email address."); return; }
   157→   157→    setGoogleError("");
   158→   158→    // Use redirect-based flow so cookies survive the proxy chain
   159→   159→    const origin = window.location.origin;
   160→   160→    const role = tab === "register" ? regRole : "";
   161→   161→    const params = new URLSearchParams({
   162→   162→      email: googleEmail.trim(),
   163→   163→      name: googleName.trim(),
   164→   164→      origin,
   165→   165→    });
   166→   166→    if (role) params.set("role", role);
   167→   167→    window.location.href = `/api/auth/google-simulate?${params.toString()}`;
   168→   168→  };
   169→   169→
   170→   170→  const handleLogin = async (e: React.FormEvent) => {
   171→   171→    e.preventDefault();
   172→   172→    if (!loginEmail.trim() || !loginPassword.trim()) { setError("Please enter your email and password."); return; }
   173→   173→    setLoading(true); setError("");
   174→   174→    try {
   175→   175→      const result = await signIn("credentials", { email: loginEmail.trim(), password: loginPassword, redirect: false });
   176→   176→      if (result?.error) { setError("Invalid email or password. Please try again."); return; }
   177→   177→      try {
   178→   178→        const sessionRes = await fetch("/api/auth/session");
   179→   179→        const sessionData = await sessionRes.json();
   180→   180→        if (sessionData?.user) {
   181→   181→          const u = sessionData.user as any;
   182→   182→          setAuth({ id: u.id || "", email: u.email || "", name: u.name || "", phone: "", role: u.role || "FAMILY", avatarUrl: u.image || null, subscription: "NONE", patientProfiles: [], caregiverProfile: null });
   183→   183→        }
   184→   184→      } catch { /* AuthSync will pick it up */ }
   185→   185→      toast({ title: "Welcome back! \ud83c\udf89", description: "You have successfully logged in." });
   186→   186→      handleClose();
   187→   187→    } catch { setError("Something went wrong. Please try again."); } finally { setLoading(false); }
   188→   188→  };
   189→   189→
   190→   190→  const handleRegister = async (e: React.FormEvent) => {
   191→   191→    e.preventDefault();
   192→   192→    if (!regName.trim() || !regEmail.trim() || !regPhone.trim() || !regPassword.trim()) { setError("Please fill in all fields."); return; }
   193→   193→    if (regPassword.length < 6) { setError("Password must be at least 6 characters."); return; }
   194→   194→    setLoading(true); setError("");
   195→   195→    try {
   196→   196→      await api.auth.register({ name: regName.trim(), email: regEmail.trim(), phone: regPhone.trim(), password: regPassword, role: regRole });
   197→   197→      const loginResult = await signIn("credentials", { email: regEmail.trim(), password: regPassword, redirect: false });
   198→   198→      if (loginResult?.error) {
   199→   199→        toast({ title: "Account created!", description: "Please log in with your credentials." });
   200→   200→        setTab("login"); setLoginEmail(regEmail.trim()); return;
   201→   201→      }
   202→   202→      try {
   203→   203→        const sessionRes = await fetch("/api/auth/session");
   204→   204→        const sessionData = await sessionRes.json();
   205→   205→        if (sessionData?.user) {
   206→   206→          const u = sessionData.user as any;
   207→   207→          setAuth({ id: u.id || "", email: u.email || "", name: u.name || "", phone: regPhone.trim(), role: u.role || regRole, avatarUrl: u.image || null, subscription: "NONE", patientProfiles: [], caregiverProfile: null });
   208→   208→        }
   209→   209→      } catch { /* AuthSync will pick it up */ }
   210→   210→      toast({ title: "Welcome to SevaSaathi! \ud83c\udf89", description: "Your account has been created." });
   211→   211→      handleClose();
   212→   212→    } catch (err: any) { setError(err?.message || "Registration failed. Please try again."); } finally { setLoading(false); }
   213→   213→  };
   214→   214→
   215→   215→  const handleReset = async (e: React.FormEvent) => {
   216→   216→    e.preventDefault();
   217→   217→    if (!resetEmail.trim() || !resetNewPassword.trim()) { setResetError("Please enter your email and new password."); return; }
   218→   218→    if (resetNewPassword.length < 6) { setResetError("Password must be at least 6 characters."); return; }
   219→   219→    setResetLoading(true); setResetError("");
   220→   220→    try {
   221→   221→      const res = await fetch("/api/auth/reset-password", { method: "POST", headers: { "Content-Type": "application/json" }, body: JSON.stringify({ email: resetEmail.trim(), newPassword: resetNewPassword }) });
   222→   222→      const data = await res.json();
   223→   223→      if (!res.ok) { setResetError(data.error || "Failed to reset password."); return; }
   224→   224→      setResetDone(true);
   225→   225→      toast({ title: "Password reset successfully!", description: "You can now log in with your new password." });
   226→   226→    } catch { setResetError("Something went wrong. Please try again."); } finally { setResetLoading(false); }
   227→   227→  };
   228→   228→
   229→   229→  /* ================================================================ */
   230→   230→  /*  RENDER                                                           */
   231→   231→/* ================================================================ */
   232→   232→  return (
   233→   233→    <Dialog open={isOpen} onOpenChange={(open) => !open && handleClose()}>
   234→   234→      <DialogContent showCloseButton={false} className="sm:max-w-md p-0 overflow-hidden border-forest-200/40 max-h-[92vh] flex flex-col">
   235→   235→        <button onClick={handleClose} className="absolute top-3 right-3 z-50 rounded-full p-1.5 text-gray-400 hover:text-gray-600 hover:bg-gray-100 transition-colors" aria-label="Close">
   236→   236→          <X className="w-5 h-5" />
   237→   237→        </button>
   238→   238→
   239→   239→        <div className="px-6 pt-6 pb-2">
   240→   240→          <DialogHeader className="space-y-1 mb-1">
   241→   241→            <div className="flex items-center gap-2 justify-center">
   242→   242→              <div className="w-9 h-9 rounded-xl bg-forest-900 flex items-center justify-center">
   243→   243→                <Heart className="w-5 h-5 text-lime-400" />
   244→   244→              </div>
   245→   245→              <DialogTitle className="text-xl font-bold text-forest-900">SevaSaathi</DialogTitle>
   246→   246→            </div>
   247→   247→            <DialogDescription className="text-sm text-gray-500 mt-1">
   248→   248→              {tab === "login" ? "Sign in to your account" : "Create your account to get started"}
   249→   249→            </DialogDescription>
   250→   250→          </DialogHeader>
   251→   251→        </div>
   252→   252→
   253→   253→        <AnimatePresence>
   254→   254→          {error && (
   255→   255→            <motion.div initial={{ height: 0, opacity: 0 }} animate={{ height: "auto", opacity: 1 }} exit={{ height: 0, opacity: 0 }} className="overflow-hidden">
   256→   256→              <div className="mx-6 mt-1 px-3 py-2 rounded-lg bg-red-50 border border-red-200 text-red-700 text-sm flex items-center gap-2">
   257→   257→                <X className="w-4 h-4 flex-shrink-0" />{error}
   258→   258→              </div>
   259→   259→            </motion.div>
   260→   260→          )}
   261→   261→        </AnimatePresence>
   262→   262→
   263→   263→        <div className="px-6 pt-3">
   264→   264→          <div className="flex bg-gray-100 rounded-lg p-1">
   265→   265→            <button onClick={() => setTab("login")} className={`flex-1 py-2 px-4 rounded-md text-sm font-medium transition-all ${tab === "login" ? "bg-white text-forest-900 shadow-sm" : "text-gray-500 hover:text-gray-700"}`}>Sign In</button>
   266→   266→            <button onClick={() => setTab("register")} className={`flex-1 py-2 px-4 rounded-md text-sm font-medium transition-all ${tab === "register" ? "bg-white text-forest-900 shadow-sm" : "text-gray-500 hover:text-gray-700"}`}>Create Account</button>
   267→   267→          </div>
   268→   268→        </div>
   269→   269→
   270→   270→        <div className="px-6 py-4 overflow-y-auto flex-1 min-h-0">
   271→   271→          <AnimatePresence mode="wait" custom={tab}>
   272→   272→            {tab === "login" && (
   273→   273→              <motion.div key="login" custom={tab} variants={contentVariants} initial="enter" animate="center" exit="exit" className="space-y-4">
   274→   274→                <button type="button" onClick={handleGoogleSignIn} className="w-full flex items-center justify-center gap-3 h-11 px-4 rounded-lg border border-gray-300 bg-white hover:bg-gray-50 transition-all text-sm font-medium text-gray-700 shadow-sm">
   275→   275→                  <GoogleIcon /> Continue with Google
   276→   276→                </button>
   277→   277→                <div className="flex items-center gap-3">
   278→   278→                  <div className="flex-1 h-px bg-gray-200" />
   279→   279→                  <span className="text-xs text-gray-400 font-medium uppercase tracking-wider">or continue with email</span>
   280→   280→                  <div className="flex-1 h-px bg-gray-200" />
   281→   281→                </div>
   282→   282→                <form onSubmit={handleLogin} className="space-y-3">
   283→   283→                  <div className="space-y-1.5">
   284→   284→                    <Label htmlFor="login-email" className="text-sm font-medium text-gray-700">Email</Label>
   285→   285→                    <div className="relative">
   286→   286→                      <Mail className="absolute left-3 top-1/2 -translate-y-1/2 w-4 h-4 text-gray-400" />
   287→   287→                      <Input id="login-email" type="email" placeholder="you@example.com" value={loginEmail} onChange={(e) => setLoginEmail(e.target.value)} className="pl-10 h-11 border-gray-300" autoComplete="email" disabled={loading} />
   288→   288→                    </div>
   289→   289→                  </div>
   290→   290→                  <div className="space-y-1.5">
   291→   291→                    <div className="flex items-center justify-between">
   292→   292→                      <Label htmlFor="login-password" className="text-sm font-medium text-gray-700">Password</Label>
   293→   293→                      <button type="button" onClick={() => setResetOpen(true)} className="text-xs font-medium text-forest-700 hover:text-forest-900">Forgot password?</button>
   294→   294→                    </div>
   295→   295→                    <div className="relative">
   296→   296→                      <Lock className="absolute left-3 top-1/2 -translate-y-1/2 w-4 h-4 text-gray-400" />
   297→   297→                      <Input id="login-password" type={showPassword ? "text" : "password"} placeholder="Enter your password" value={loginPassword} onChange={(e) => setLoginPassword(e.target.value)} className="pl-10 pr-10 h-11 border-gray-300" autoComplete="current-password" disabled={loading} />
   298→   298→                      <button type="button" onClick={() => setShowPassword(!showPassword)} className="absolute right-3 top-1/2 -translate-y-1/2 text-gray-400 hover:text-gray-600" tabIndex={-1}>
   299→   299→                        {showPassword ? <EyeOff className="w-4 h-4" /> : <Eye className="w-4 h-4" />}
   300→   300→                      </button>
   301→   301→                    </div>
   302→   302→                  </div>
   303→   303→                  <Button type="submit" disabled={loading} className="w-full h-11 bg-forest-900 hover:bg-forest-800 text-white font-semibold rounded-lg shadow-sm hover:shadow-md">
   304→   304→                    {loading ? <><Loader2 className="w-4 h-4 mr-2 animate-spin" />Signing in...</> : <>Sign In <ArrowRight className="w-4 h-4 ml-2" /></>}
   305→   305→                  </Button>
   306→   306→                </form>
   307→   307→                <p className="text-center text-sm text-gray-500 pt-1">Don&apos;t have an account? <button onClick={() => setTab("register")} className="font-semibold text-forest-700 hover:text-forest-900">Create one</button></p>
   308→   308→              </motion.div>
   309→   309→            )}
   310→   310→
   311→   311→            {tab === "register" && (
   312→   312→              <motion.div key="register" custom={tab} variants={contentVariants} initial="enter" animate="center" exit="exit" className="space-y-4">
   313→   313→                {/* Role selector */}
   314→   314→                <div className="space-y-2">
   315→   315→                  <Label className="text-sm font-semibold text-gray-800">How would you like to use SevaSaathi?</Label>
   316→   316→                  <div className="grid grid-cols-1 gap-2">
   317→   317→                    <button type="button" onClick={() => setRegRole("FAMILY")} className={`flex items-start gap-3 p-3 rounded-xl border-2 text-left transition-all ${regRole === "FAMILY" ? "border-forest-500 bg-forest-50" : "border-gray-200 bg-white hover:border-gray-300"}`}>
   318→   318→                      <div className={`w-9 h-9 rounded-lg flex items-center justify-center shrink-0 mt-0.5 ${regRole === "FAMILY" ? "bg-forest-100" : "bg-gray-100"}`}>
   319→   319→                        <Heart className={`w-4.5 h-4.5 ${regRole === "FAMILY" ? "text-forest-700" : "text-gray-400"}`} />
   320→   320→                      </div>
   321→   321→                      <div className="min-w-0">
   322→   322→                        <p className={`text-sm font-semibold ${regRole === "FAMILY" ? "text-forest-900" : "text-gray-700"}`}>I need care for my family</p>
   323→   323→                        <p className={`text-xs mt-0.5 ${regRole === "FAMILY" ? "text-forest-600/80" : "text-gray-400"}`}>Find trusted caregivers for parents, elders, or family members at home</p>
   324→   324→                      </div>
   325→   325→                    </button>
   326→   326→                    <button type="button" onClick={() => setRegRole("CAREGIVER")} className={`flex items-start gap-3 p-3 rounded-xl border-2 text-left transition-all ${regRole === "CAREGIVER" ? "border-forest-500 bg-forest-50" : "border-gray-200 bg-white hover:border-gray-300"}`}>
   327→   327→                      <div className={`w-9 h-9 rounded-lg flex items-center justify-center shrink-0 mt-0.5 ${regRole === "CAREGIVER" ? "bg-forest-100" : "bg-gray-100"}`}>
   328→   328→                        <ShieldCheck className={`w-4.5 h-4.5 ${regRole === "CAREGIVER" ? "text-forest-700" : "text-gray-400"}`} />
   329→   329→                      </div>
   330→   330→                      <div className="min-w-0">
   331→   331→                        <p className={`text-sm font-semibold ${regRole === "CAREGIVER" ? "text-forest-900" : "text-gray-700"}`}>I want to provide care services</p>
   332→   332→                        <p className={`text-xs mt-0.5 ${regRole === "CAREGIVER" ? "text-forest-600/80" : "text-gray-400"}`}>Join as a caregiver, get bookings, and earn money</p>
   333→   333→                      </div>
   334→   334→                    </button>
   335→   335→                  </div>
   336→   336→                </div>
   337→   337→
