   166→   166→      const totalReports = bookings.reduce((acc: number, b: any) => acc + (b.careReports?.length || 0), 0);
   167→   167→      const completedWithoutReview = bookings.filter(
   168→   168→        (b: any) => b.status === 'COMPLETED' && (!b.reviews || b.reviews.length === 0)
   169→   169→      ).length;

   171→   171→      setStats({
   172→   172→        patients: patients.length,
   173→   173→        activeBookings: activeBookings.length,
   174→   174→        reports: totalReports,
   175→   175→        pendingReviews: completedWithoutReview,
   176→   176→      });
   177→   177→      setRecentBookings(bookings.slice(0, 5));
   178→   178→    } catch (err: any) {
   179→   179→      setError(err.message || 'Failed to load dashboard data');
   180→   180→    } finally {
   181→   181→      setLoading(false);
   182→   182→    }
   183→   183→  }, [user?.id]);

   186→   186→    fetchData();
   187→   187→  }, [fetchData]);

   207→   207→    { label: 'Active Bookings', value: stats.activeBookings, icon: <CalendarCheck className="h-5 w-5" />, color: 'bg-blue-50 text-blue-700' },
   208→   208→    { label: 'Care Reports', value: stats.reports, icon: <FileText className="h-5 w-5" />, color: 'bg-amber-50 text-amber-700' },
   209→   209→    { label: 'Pending Reviews', value: stats.pendingReviews, icon: <Star className="h-5 w-5" />, color: 'bg-lime-50 text-lime-700' },
   210→   210→  ];

   212→   212→  return (
   213→   213→    <div className="space-y-6">
   214→   214→      {/* Welcome */}
   215→   215→      <div>
   216→   216→        <h2 className="text-2xl font-bold text-gray-900">
   217→   217→          Welcome back, {user?.name?.split(' ')[0]}!
   218→   218→        </h2>

   220→   220→      </div>
   221→   221→
   222→   222→      {/* Stat Cards */}
   223→   223→      <div className="grid grid-cols-1 sm:grid-cols-2 lg:grid-cols-4 gap-4">
   224→   224→        {statCards.map((stat, i) => (
   225→   225→          <motion.div

   227→   227→            initial={{ opacity: 0, y: 12 }}

   232→   232→              <CardContent className="p-5">
   233→   233→                <div className="flex items-center justify-between">
   234→   234→                  <div>

   309→   309→function PatientsTab() {
   310→   310→  const user = useAuthStore((s) => s.user);
   311→   311→  const [patients, setPatients] = useState<any[]>([]);
   312→   312→  const [loading, setLoading] = useState(true);
   313→   313→  const [error, setError] = useState('');
   314→   314→  const [modalOpen, setModalOpen] = useState(false);
   315→   315→  const [submitting, setSubmitting] = useState(false);
   316→   316→
   317→   317→  // Form state
   318→   318→  const [form, setForm] = useState({
   319→   319→    name: '', age: '', gender: '', relationship: '', address: '', city: '',
   320→   320→    pincode: '', mobilityStatus: '', medicalHistory: '', careRequirements: '',
   321→   321→    emergencyName: '', emergencyPhone: '',
   322→   322→  });
   323→   323→
   324→   324→  const fetchPatients = useCallback(async () => {
   325→   325→    if (!user?.id) return;
   326→   326→    setLoading(true);
   327→   327→    setError('');
   328→   328→    try {
   329→   329→      const res = await api.patients.list(user.id);
   330→   330→      setPatients(res.patients || []);
   331→   331→    } catch (err: any) {
   332→   332→      setError(err.message || 'Failed to load patients');
   333→   333→    } finally {
   334→   334→      setLoading(false);
   335→   335→    }
   336→   336→  }, [user?.id]);
   337→   337→
   338→   338→  useEffect(() => {
   339→   339→    fetchPatients();
   340→   340→  }, [fetchPatients]);
   341→   341→
   342→   342→  const updateForm = (field: string, value: string) => {
   343→   343→    setForm((prev) => ({ ...prev, [field]: value }));

   345→   345→
   346→   346→  const handleSubmit = async (e: React.FormEvent) => {
   347→   347→    e.preventDefault();

   349→   349→    setSubmitting(true);
   350→   350→    try {
   351→   351→      await api.patients.create({
   352→   352→        familyId: user.id,
   353→   353→        name: form.name,
   354→   354→        age: parseInt(form.age, 10) || 0,
   355→   355→        gender: form.gender,
   356→   356→        relationship: form.relationship,
   357→   357→        address: form.address,
   358→   358→        city: form.city,
   359→   359→        pincode: form.pincode || undefined,
   360→   360→        mobilityStatus: form.mobilityStatus || 'mobile',
   361→   361→        medicalHistory: form.medicalHistory,
   362→   362→        careRequirements: form.careRequirements,
   363→   363→        emergencyContact: JSON.stringify({ name: form.emergencyName, phone: form.emergencyPhone }),
   364→   364→      });
   365→   365→      toast.success('Patient profile created successfully!');
   366→   366→      setModalOpen(false);
   367→   367→      setForm({
   368→   368→        name: '', age: '', gender: '', relationship: '', address: '', city: '',
   369→   369→        pincode: '', mobilityStatus: '', medicalHistory: '', careRequirements: '',
   370→   370→        emergencyName: '', emergencyPhone: '',
   371→   371→      });
   372→   372→      fetchPatients();
   373→   373→    } catch (err: any) {
   374→   374→      toast.error(err.message || 'Failed to create patient');
   375→   375→    } finally {
   376→   376→      setSubmitting(false);
   377→   377→    }
   378→   378→  };
   379→   379→
   380→   380→  if (loading) {
   381→   381→    return <LoadingCards count={3} />;
   382→   382→  }
   383→   383→
   384→   384→  if (error) {
   385→   385→    return <ErrorState message={error} onRetry={fetchPatients} />;
   386→   386→  }
   387→   387→
   388→   388→  const mobilityLabel: Record<string, string> = {
   389→   389→    mobile: 'Mobile', bedridden: 'Bedridden', limited: 'Limited Mobility', wheelchair: 'Wheelchair',
   390→   390→  };
   391→   391→
   392→   392→  return (
   393→   393→    <div className="space-y-6">
   394→   394→      <div className="flex items-center justify-between">
   395→   395→        <div>
   396→   396→          <h2 className="text-xl font-bold text-gray-900">Patient Profiles</h2>
   397→   397→          <p className="text-sm text-gray-400">Manage care recipients in your family.</p>
   398→   398→        </div>
   399→   399→        <Dialog open={modalOpen} onOpenChange={setModalOpen}>
   400→   400→          <DialogTrigger asChild>
   401→   401→            <Button className="btn-black text-sm gap-2">
   402→   402→              <Plus className="h-4 w-4" /> Add Patient
   403→   403→            </Button>
   404→   404→          </DialogTrigger>
   405→   405→          <DialogContent className="max-w-lg max-h-[90vh] overflow-y-auto rounded-2xl">
   406→   406→            <DialogHeader>
   407→   407→              <DialogTitle>Add New Patient</DialogTitle>
   408→   408→            </DialogHeader>
   409→   409→            <form onSubmit={handleSubmit} className="space-y-4 mt-2">
   410→   410→              <div className="grid grid-cols-2 gap-3">
   411→   411→                <div className="col-span-2">
   412→   412→                  <Label className="text-xs font-medium text-gray-600">Full Name *</Label>
   413→   413→                  <Input
   414→   414→                    required
   415→   415→                    value={form.name}
   416→   416→                    onChange={(e) => updateForm('name', e.target.value)}
   417→   417→                    placeholder="Patient name"
   418→   418→                    className="mt-1 rounded-xl"
   419→   419→                  />
   420→   420→                </div>
   421→   421→                <div>

   430→   430→                    placeholder="Age"
   431→   431→                    className="mt-1 rounded-xl"
   432→   432→                  />
   433→   433→                </div>
   434→   434→                <div>
