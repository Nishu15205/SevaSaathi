   286→   286→/* ─────────── VERIFICATIONS TAB ─────────── */
   287→   287→function VerificationsTab() {
   288→   288→  const [verifications, setVerifications] = useState<any[]>([]);
   289→   289→  const [loading, setLoading] = useState(true);
   290→   290→  const [rejecting, setRejecting] = useState<string | null>(null);
   291→   291→  const [approving, setApproving] = useState<string | null>(null);
   292→   292→  const [reasons, setReasons] = useState<Record<string, string>>({});
   293→   293→
   294→   294→  const fetchVerifications = useCallback(async () => {
   295→   295→    setLoading(true);
   296→   296→    try {
   297→   297→      const res = await api.admin.verifications.list();
   298→   298→      setVerifications(res.verifications ?? []);
   299→   299→    } catch {
   300→   300→      toast.error('Failed to load verifications');
   301→   301→    } finally {
   302→   302→      setLoading(false);
   303→   303→    }
   304→   304→  }, []);
   305→   305→
   306→   306→  useEffect(() => { fetchVerifications(); }, [fetchVerifications]);
   307→   307→
   308→   308→  const handleApprove = async (id: string) => {
   309→   309→    setApproving(id);
   310→   310→    try {
   311→   311→      await api.admin.verifications.update(id, { status: 'APPROVED' });
   312→   312→      toast.success('Verification approved successfully');
   313→   313→      fetchVerifications();
   314→   314→    } catch {
   315→   315→      toast.error('Failed to approve verification');
   316→   316→    } finally {
   317→   317→      setApproving(null);
   318→   318→    }
   319→   319→  };
   320→   320→
   321→   321→  const handleReject = async (id: string) => {
   322→   322→    const reason = reasons[id]?.trim();
   323→   323→    if (!reason) { toast.error('Please provide a rejection reason'); return; }
   324→   324→    setRejecting(id);
   325→   325→    try {
   326→   326→      await api.admin.verifications.update(id, { status: 'REJECTED', rejectionReason: reason });
   327→   327→      toast.success('Verification rejected');
   328→   328→      setReasons((prev) => { const n = { ...prev }; delete n[id]; return n; });
   329→   329→      fetchVerifications();
   330→   330→    } catch {
   331→   331→      toast.error('Failed to reject verification');
   332→   332→    } finally {
   333→   333→      setRejecting(null);
   334→   334→    }
   335→   335→  };
   336→   336→
   337→   337→  if (loading) {
   338→   338→    return <div className="space-y-3">{Array.from({ length: 3 }).map((_, i) => <Skeleton key={i} className="h-24 rounded-2xl" />)}</div>;
   339→   339→  }
   340→   340→
   341→   341→  if (verifications.length === 0) {
   342→   342→    return <Card className="rounded-2xl border-0 shadow-sm"><CardContent className="py-16 text-center"><ShieldCheck className="w-12 h-12 mx-auto text-[#14532d]/30 mb-3" /><p className="text-muted-foreground">No pending verifications.</p></CardContent></Card>;
   343→   343→  }
   344→   344→
   345→   345→  return (
   346→   346→    <div className="space-y-3">
   347→   347→      {verifications.map((v) => (
   348→   348→        <motion.div key={v.id} variants={fadeVariants} initial="hidden" animate="visible">
   349→   349→          <Card className="rounded-2xl card-hover border-0 shadow-sm">
   350→   350→            <CardContent className="p-4 space-y-3">
   351→   351→              <div className="flex flex-col sm:flex-row sm:items-center justify-between gap-2">
   352→   352→                <div className="flex items-center gap-2 flex-wrap">
   353→   353→                  <span className="font-semibold text-sm">{v.caregiver?.user?.name ?? 'Caregiver'}</span>
   354→   354→                  <Badge className={docTypeBadge[v.docType] ?? 'bg-gray-100 text-gray-700'}>{v.docType ?? 'DOC'}</Badge>
   355→   355→                </div>
   356→   356→                <span className="text-xs text-muted-foreground flex items-center gap-1">
   357→   357→                  <Clock className="w-3 h-3" />{v.createdAt ? new Date(v.createdAt).toLocaleDateString('en-IN', { day: 'numeric', month: 'short', year: 'numeric' }) : '—'}
   358→   358→                </span>
   359→   359→              </div>
   360→   360→              {v.docNumber && <p className="text-xs text-muted-foreground font-mono bg-muted/50 rounded-lg px-3 py-1.5 w-fit">{v.docNumber}</p>}
   361→   361→              <div className="flex items-center gap-2 pt-1">
   362→   362→                <Button
   363→   363→                  size="sm"
   364→   364→                  className="bg-[#14532d] hover:bg-[#14532d]/90 text-white rounded-full"
   365→   365→                  disabled={approving === v.id}
   366→   366→                  onClick={() => handleApprove(v.id)}
   367→   367→                >
   368→   368→                  {approving === v.id ? <Loader2 className="w-4 h-4 animate-spin" /> : <><CheckCircle2 className="w-4 h-4 mr-1" />Approve</>}
   369→   369→                </Button>
   370→   370→                <Button
   371→   371→                  size="sm"
   372→   372→                  variant="outline"
   373→   373→                  className="text-red-600 border-red-200 hover:bg-red-50 rounded-full"
   374→   374→                  disabled={rejecting === v.id}
   375→   375→                  onClick={() => {
   376→   376→                    if (!reasons[v.id]?.trim()) {
   377→   377→                      setReasons((prev) => ({ ...prev, [v.id]: prev[v.id] ?? '' }));
   378→   378→                    } else {
   379→   379→                      handleReject(v.id);
   380→   380→                    }
   381→   381→                  }}
   382→   382→                >
   383→   383→                  {rejecting === v.id ? <Loader2 className="w-4 h-4 animate-spin" /> : <><XCircle className="w-4 h-4 mr-1" />Reject</>}
   384→   384→                </Button>
