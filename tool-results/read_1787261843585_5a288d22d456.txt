   479→function CreateProfileForm({ userId, onCreated }: { userId: string; onCreated: (profile: any) => void }) {
   480→  const [saving, setSaving] = useState(false);

   490→  const handleSubmit = async (e: React.FormEvent) => {
   491→    e.preventDefault();
   492→    if (!form.city || !form.yearsExperience || !form.hourlyRate) {
   493→      toast.error('Please fill in City, Experience, and Hourly Rate.');
   494→      return;
   495→    }
   496→    if (selectedSkills.length === 0) {
   497→      toast.error('Please select at least one skill.');
   498→      return;
   499→    }
   500→    setSaving(true);
   501→    try {
   502→      const res = await api.caregivers.create({
   503→        userId,
   504→        city: form.city,
   505→        yearsExperience: parseInt(form.yearsExperience),
   506→        hourlyRate: parseInt(form.hourlyRate),
   507→        skills: JSON.stringify(selectedSkills),

   509→        languages: JSON.stringify(form.languages.split(',').map((s: string) => s.trim()).filter(Boolean)),
   510→        bio: form.bio,
   511→      });
   512→      toast.success('Profile created successfully!');
   513→      onCreated(res.caregiver);
   514→    } catch (err: any) {
   515→      toast.error(err.message || 'Failed to create profile');

   517→      setSaving(false);
   518→    }

   532→                <Label className="text-xs font-medium text-gray-600">City *</Label>
