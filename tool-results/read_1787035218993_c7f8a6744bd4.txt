   143→   143→  const [regPhone, setRegPhone] = useState("");

   145→   145→  const [regRole, setRegRole] = useState<UserRole>("FAMILY");
   146→   146→
   147→   147→  // Reset form (simplified — no OTP)
   148→   148→  const [resetEmail, setResetEmail] = useState("");
   149→   149→  const [resetNewPassword, setResetNewPassword] = useState("");
   150→   150→  const [resetLoading, setResetLoading] = useState(false);
   151→   151→  const [resetError, setResetError] = useState("");
   152→   152→  const [resetDone, setResetDone] = useState(false);
   153→   153→
   154→   154→  const { setAuth } = useAuthStore();
   155→   155→  const { toast } = useToast();
   156→   156→
   157→   157→  // Google popup form
   158→   158→  const [googleOpen, setGoogleOpen] = useState(false);
   159→   159→  const [googleEmail, setGoogleEmail] = useState("");
   160→   160→  const [googleName, setGoogleName] = useState("");
   161→   161→  const [googleError, setGoogleError] = useState("");
   162→   162→  const [googleSubmitting, setGoogleSubmitting] = useState(false);
   163→   163→
   164→   164→  /* ---- reset form fields when switching tabs ---- */

   185→   185→    setResetEmail("");
   186→   186→    setResetNewPassword("");

   212→   212→      setGoogleError("Please enter your name and email.");
   213→   213→      return;
   214→   214→    }
   215→   215→    if (!googleEmail.trim().includes("@")) {
   216→   216→      setGoogleError("Please enter a valid email address.");
   217→   217→      return;
   218→   218→    }
   219→   219→    setGoogleSubmitting(true);
   220→   220→    setGoogleError("");

   222→   222→      const res = await fetch("/api/auth/google-simulate", {
   223→   223→        method: "POST",
   224→   224→        headers: { "Content-Type": "application/json" },
   225→   225→        body: JSON.stringify({ email: googleEmail.trim(), name: googleName.trim() }),
   226→   226→      });

   228→   228→      if (!res.ok) {

   231→   231→      }

   234→   234→      toast({
   235→   235→        title: "Welcome" + (data.user.isNewUser ? " to SevaSaathi!" : " back!") + " \ud83c\udf89",

   237→   237→      });
   238→   238→      // Use router.push to trigger a full navigation which re-fetches the session
   239→   239→      window.location.href = window.location.origin;
   240→   240→      return;
   241→   241→    } catch {
   242→   242→      setGoogleError("Something went wrong. Please try again.");
   243→   243→    } finally {
   244→   244→      setGoogleSubmitting(false);
   245→   245→    }
   246→   246→  };
   247→   247→
   248→   248→  /* ---- Email / password login ---- */
   249→   249→  const handleLogin = async (e: React.FormEvent) => {
   250→   250→    e.preventDefault();
   251→   251→    if (!loginEmail.trim() || !loginPassword.trim()) {
   252→   252→      setError("Please enter your email and password.");
   253→   253→      return;
   254→   254→    }
   255→   255→    setLoading(true);
   256→   256→    setError("");

   258→   258→      const result = await signIn("credentials", {
   259→   259→        email: loginEmail.trim(),
   260→   260→        password: loginPassword,
   261→   261→        redirect: false,
   262→   262→      });
   263→   263→
   264→   264→      if (result?.error) {
   265→   265→        setError("Invalid email or password. Please try again.");
   266→   266→        return;
   267→   267→      }
   268→   268→
   269→   269→      // AuthSync in AuthProvider will automatically sync session → Zustand store
   270→   270→      toast({
   271→   271→        title: "Welcome back! 🎉",
   272→   272→        description: "You have successfully logged in.",
   273→   273→      });
   274→   274→      handleClose();
   275→   275→    } catch {
   276→   276→      setError("Something went wrong. Please try again.");
   277→   277→    } finally {
   278→   278→      setLoading(false);
   279→   279→    }
   280→   280→  };
   281→   281→
   282→   282→  /* ---- Registration ---- */
   283→   283→  const handleRegister = async (e: React.FormEvent) => {
   284→   284→    e.preventDefault();

   286→   286→      setError("Please fill in all fields.");
   287→   287→      return;
   288→   288→    }
   289→   289→    if (regPassword.length < 6) {
   290→   290→      setError("Password must be at least 6 characters.");
   291→   291→      return;
   292→   292→    }
   293→   293→    setLoading(true);
   294→   294→    setError("");

   296→   296→      // 1. Register
   297→   297→      const regRes = await api.auth.register({
   298→   298→        name: regName.trim(),
   299→   299→        email: regEmail.trim(),
   300→   300→        phone: regPhone.trim(),
   301→   301→        password: regPassword,
   302→   302→        role: regRole,
   303→   303→      });
   304→   304→
   305→   305→      // 2. Auto-login with credentials
   306→   306→      const loginResult = await signIn("credentials", {
   307→   307→        email: regEmail.trim(),
   308→   308→        password: regPassword,
   309→   309→        redirect: false,
   310→   310→      });
   311→   311→
   312→   312→      if (loginResult?.error) {
   313→   313→        // Registration succeeded but auto-login failed — that's ok, just notify
   314→   314→        toast({ title: "Account created!", description: "Please log in with your credentials." });
   315→   315→        setTab("login");
   316→   316→        setLoginEmail(regEmail.trim());
   317→   317→        return;
   318→   318→      }
   319→   319→
   320→   320→      toast({ title: "Welcome to SevaSaathi! 🎉", description: "Your account has been created." });
   321→   321→      handleClose();
   322→   322→    } catch (err: any) {
   323→   323→      setError(err?.message || "Registration failed. Please try again.");
   324→   324→    } finally {
   325→   325→      setLoading(false);
   326→   326→    }
   327→   327→  };
   328→   328→

   352→   352→      }

   375→   375→        </button>

   391→   391→            </DialogDescription>
   392→   392→          </DialogHeader>
   393→   393→        </div>
   394→   394→
   395→   395→        {/* ---- Error banner ---- */}

   406→   406→                {error}
   407→   407→              </div>

   409→   409→          )}
   410→   410→        </AnimatePresence>
   411→   411→
   412→   412→        {/* ---- Tab switcher ---- */}
   413→   413→        <div className="px-6 pt-3">

   415→   415→            <button

   417→   417→              className={`flex-1 py-2 px-4 rounded-md text-sm font-medium transition-all duration-200 ${
   418→   418→                tab === "login"

   420→   420→                  : "text-gray-500 hover:text-gray-700"

   422→   422→            >
