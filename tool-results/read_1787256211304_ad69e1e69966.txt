   205→function parseJsonSafe(jsonStr: string | null | undefined): string[] {
   206→  if (!jsonStr) return [];
   207→  try {
   208→    const parsed = JSON.parse(jsonStr);
   209→    return Array.isArray(parsed) ? parsed : [];
   210→  } catch {
   211→    return jsonStr.split(',').map((s) => s.trim()).filter(Boolean);
   212→  }
   213→}

   215→function formatDate(dateStr: string): string {
   216→  try {
   217→    return new Date(dateStr).toLocaleDateString('en-IN', {
   218→      day: 'numeric',
   219→      month: 'short',
   220→      year: 'numeric',
   221→    });
   222→  } catch {
   223→    return dateStr;
   224→  }
   225→}
   226→
   227→function getTodayDate(): string {
   228→  const now = new Date();
   229→  const year = now.getFullYear();
   230→  const month = String(now.getMonth() + 1).padStart(2, '0');
   231→  const day = String(now.getDate()).padStart(2, '0');
   232→  return `${year}-${month}-${day}`;
   233→}

   496→  const handleSubmit = async (e: React.FormEvent) => {

   503→      toast.error('Please select at least one skill.');

   506→    setSaving(true);

   515→        languages: JSON.stringify(form.languages.split(',').map((s: string) => s.trim()).filter(Boolean)),
   516→        bio: form.bio,
   517→      });
   518→      toast.success('Profile created successfully!');
   519→      onCreated(res.caregiver);
   520→    } catch (err: any) {
   521→      toast.error(err.message || 'Failed to create profile');
   522→    } finally {
   523→      setSaving(false);
   524→    }
   525→  };
   526→
   527→  return (
   528→    <div className="space-y-6 max-w-2xl">
   529→      <div>
   530→        <h2 className="text-2xl font-bold text-gray-900">Create Your Profile</h2>

   788→function AadharVerificationSection({ caregiverId }: { caregiverId: string }) {
   789→  const [aadharResult, setAadharResult] = useState<any>(null);
   790→  const [verifying, setVerifying] = useState(false);
   791→  const [imagePreview, setImagePreview] = useState<string | null>(null);
   792→
   793→  const handleFileSelect = async (e: React.ChangeEvent<HTMLInputElement>) => {
   794→    const file = e.target.files?.[0];
   795→    if (!file) return;
   796→
   797→    // Show preview
   798→    const reader = new FileReader();
   799→    reader.onload = (ev) => {
   800→      setImagePreview(ev.target?.result as string);
   801→      setAadharResult(null);
   802→    };
   803→    reader.readAsDataURL(file);
   804→  };
   805→
   806→  const handleVerify = async () => {
   807→    if (!imagePreview) {
   808→      toast.error('Please upload an Aadhar card image first');
   809→      return;
   810→    }
   811→    setVerifying(true);
   812→    try {
   813→      const result = await api.verifyAadhar(imagePreview);
   814→      setAadharResult(result);
   815→      if (result.verified) {
   816→        toast.success('Aadhar card verified successfully!');
   817→      } else {
   818→        toast.error(result.error || 'Aadhar verification failed');
   819→      }
   820→    } catch (err: any) {
   821→      toast.error(err.message || 'Verification failed');
   822→      setAadharResult({ verified: false, error: err.message });
   823→    } finally {
   824→      setVerifying(false);
   825→    }
   826→  };
   827→
   828→  return (
   829→    <div className="mt-6">
   830→      <div className="flex items-center gap-2 mb-3">
   831→        <ShieldCheck className="h-4 w-4 text-forest-700" />
   832→        <p className="text-sm font-semibold text-gray-700">Aadhar Card Verification</p>
   833→      </div>
   834→      <div className="bg-gray-50 rounded-2xl p-5">
   835→        <p className="text-xs text-gray-500 mb-4">Upload your Aadhar card for AI-powered instant verification. Your data is processed securely and not stored.</p>
   836→        
   837→        <div className="space-y-4">
   838→          {/* Upload Area */}
   839→          {!imagePreview ? (
   840→            <label className="flex flex-col items-center justify-center border-2 border-dashed border-gray-300 rounded-xl p-8 cursor-pointer hover:border-forest-400 hover:bg-forest-50/50 transition-colors">
   841→              <Camera className="h-10 w-10 text-gray-400 mb-3" />
   842→              <p className="text-sm font-medium text-gray-600">Upload Aadhar Card</p>
   843→              <p className="text-xs text-gray-400 mt-1">JPG, PNG - Clear photo of your Aadhar card</p>
   844→              <input
   845→                type="file"
   846→                accept="image/*"
   847→                onChange={handleFileSelect}
   848→                className="hidden"
   849→              />
   850→            </label>
   851→          ) : (
   852→            <div className="space-y-3">
   853→              <div className="relative rounded-xl overflow-hidden border border-gray-200">
   854→                <img src={imagePreview} alt="Aadhar preview" className="w-full max-h-48 object-contain bg-white" />
   855→                <button
   856→                  onClick={() => { setImagePreview(null); setAadharResult(null); }}
   857→                  className="absolute top-2 right-2 w-7 h-7 rounded-full bg-black/60 text-white flex items-center justify-center hover:bg-black/80"
   858→                >
   859→                  <XCircle className="h-4 w-4" />
   860→                </button>
   861→              </div>
   862→              <div className="flex gap-2">
   863→                <Button
   864→                  onClick={handleVerify}
   865→                  disabled={verifying}
   866→                  className="btn-green text-sm gap-2"
   867→                >
   868→                  {verifying ? <Loader2 className="h-4 w-4 animate-spin" /> : <ShieldCheck className="h-4 w-4" />}
   869→                  {verifying ? 'Verifying...' : 'Verify Aadhar'}
   870→                </Button>
   871→                <Button variant="outline" size="sm" onClick={() => { setImagePreview(null); setAadharResult(null); }} className="rounded-full">
   872→                  Change Image
   873→                </Button>
   874→              </div>
   875→            </div>
   876→          )}
   877→
   878→          {/* Result */}
   879→          {aadharResult && (
   880→            <motion.div
   881→              initial={{ opacity: 0, y: 8 }}
   882→              animate={{ opacity: 1, y: 0 }}
   883→              className={`rounded-xl p-4 ${
   884→                aadharResult.verified
   885→                  ? 'bg-green-50 border border-green-200'
   886→                  : 'bg-red-50 border border-red-200'
   887→              }`}
   888→            >
   889→              <div className="flex items-center gap-2 mb-2">
   890→                {aadharResult.verified ? (
   891→                  <><CheckCircle2 className="h-5 w-5 text-green-600" /><p className="text-sm font-semibold text-green-800">Verification Successful</p></>
   892→                ) : (
   893→                  <><XCircle className="h-5 w-5 text-red-600" /><p className="text-sm font-semibold text-red-800">Verification Failed</p></>
   894→                )}
   895→              </div>
   896→              {aadharResult.verified ? (
   897→                <div className="grid grid-cols-1 sm:grid-cols-2 gap-2 mt-3">
   898→                  {aadharResult.name && (
   899→                    <div><p className="text-[10px] text-green-600 font-medium">Name</p><p className="text-sm font-semibold text-green-900">{aadharResult.name}</p></div>
   900→                  )}
   901→                  {aadharResult.aadharNumber && (
   902→                    <div><p className="text-[10px] text-green-600 font-medium">Aadhar Number</p><p className="text-sm font-semibold text-green-900 font-mono">{aadharResult.aadharNumber.replace(/(\d{4})/g, '$1 ').trim()}</p></div>

   919→        </div>

   929→function BookingsTab({ user }: { user: User }) {
   930→  const profile = user.caregiverProfile;
   931→  const [bookings, setBookings] = useState<Booking[]>([]);
   932→  const [loading, setLoading] = useState(true);
   933→  const [error, setError] = useState('');
   934→  const [completingId, setCompletingId] = useState<string | null>(null);
   935→  const [statusFilter, setStatusFilter] = useState('');
   936→
   937→  const fetchBookings = useCallback(async () => {
   938→    if (!profile?.id) return;
   939→    setLoading(true);
   940→    setError('');
   941→    try {
   942→      const params: any = { caregiverId: profile.id };
   943→      if (statusFilter) params.status = statusFilter;
   944→      const res = await api.bookings.list(params);
   945→      const mapped = (res.bookings || []).map((b: any) => ({
   946→        ...b,
   947→        date: b.startDate,
   948→        patientName: b.patient?.name,
   949→        familyName: b.family?.name,
   950→        careType: b.careRequirements ? (typeof b.careRequirements === 'string' ? JSON.parse(b.careRequirements) : b.careRequirements)?.needs?.join(', ') : undefined,
   951→      }));
   952→      setBookings(mapped);
   953→    } catch (err: any) {
   954→      setError(err.message || 'Failed to load bookings');
   955→    } finally {
   956→      setLoading(false);
   957→    }
   958→  }, [profile?.id, statusFilter]);
   959→
   960→  useEffect(() => {
   961→    fetchBookings();
   962→  }, [fetchBookings]);
   963→
   964→  const handleStatusChange = async (bookingId: string, status: string, reason?: string) => {
   965→    setCompletingId(bookingId);
   966→    try {
   967→      await api.bookings.updateStatus(bookingId, { status, cancellationReason: reason });
   968→      const labels: Record<string, string> = {
   969→        CONFIRMED: 'Booking accepted!',
   970→        IN_PROGRESS: 'Care started!',
   971→        COMPLETED: 'Booking completed!',
   972→        CANCELLED: 'Booking declined.',
   973→      };
   974→      toast.success(labels[status] || `Booking updated`);
   975→      fetchBookings();
   976→    } catch (err: any) {
   977→      toast.error(err.message || 'Failed to update booking');
   978→    } finally {
   979→      setCompletingId(null);
   980→    }
   981→  };
   982→
   983→  if (loading) {
   984→    return (
   985→      <div className="space-y-6">
   986→        <Skeleton className="h-8 w-48" />
   987→        <LoadingCards count={4} />
   988→      </div>
   989→    );
   990→  }
   991→
   992→  if (error) {
   993→    return <ErrorState message={error} onRetry={fetchBookings} />;
   994→  }
   995→
   996→  if (bookings.length === 0) {
   997→    return (
   998→      <div className="space-y-6">
   999→        <div>
  1000→          <h2 className="text-2xl font-bold text-gray-900">My Bookings</h2>
  1001→          <p className="text-sm text-gray-400 mt-1">Manage all your care bookings.</p>
  1002→        </div>
  1003→        <EmptyState
  1004→          icon={<CalendarCheck className="h-7 w-7" />}
  1005→          title="No bookings yet"
  1006→          description="When families book your services, their requests will appear here."
  1007→        />
  1008→      </div>
  1009→    );
