   210→function parseJsonSafe(jsonStr: string | null | undefined): string[] {
   211→  if (!jsonStr) return [];
   212→  try {
   213→    const parsed = JSON.parse(jsonStr);
   214→    return Array.isArray(parsed) ? parsed : [];
   215→  } catch {
   216→    return jsonStr.split(',').map((s) => s.trim()).filter(Boolean);
   217→  }
   218→}
   219→
   220→function formatDate(dateStr: string): string {
   221→  try {
   222→    return new Date(dateStr).toLocaleDateString('en-IN', {
   223→      day: 'numeric',
   224→      month: 'short',
   225→      year: 'numeric',
   226→    });
   227→  } catch {
   228→    return dateStr;
   229→  }
   230→}
   231→
   232→function getTodayDate(): string {
   233→  const now = new Date();
   234→  const year = now.getFullYear();
   235→  const month = String(now.getMonth() + 1).padStart(2, '0');
   236→  const day = String(now.getDate()).padStart(2, '0');
   237→  return `${year}-${month}-${day}`;
   238→}

   247→  // Start with loading=false so new caregivers (no profile) never see a skeleton flash

   250→

   252→  const [phoneDialogOpen, setPhoneDialogOpen] = useState(false);
   253→  const [phoneOtpSent, setPhoneOtpSent] = useState(false);
   254→  const [phoneOtp, setPhoneOtp] = useState('');
   255→  const [phoneSending, setPhoneSending] = useState(false);
   256→  const [phoneVerifying, setPhoneVerifying] = useState(false);
   257→  const [phoneDevOtp, setPhoneDevOtp] = useState('');
   258→
   259→  const sendPhoneOtp = async () => {
   260→    setPhoneSending(true);
   261→    try {
   262→      const res = await api.auth.sendPhoneOtp(user.phone);
   263→      setPhoneDevOtp(res.devOtp || '');
   264→      setPhoneOtpSent(true);
   265→      toast.success('OTP sent to your phone!');
   266→    } catch (err: any) {
   267→      toast.error(err.message || 'Failed to send OTP');
   268→    } finally {
   269→      setPhoneSending(false);
   270→    }
   271→  };
   272→
   273→  const verifyPhoneOtp = async () => {
   274→    if (phoneOtp.length !== 6) { toast.error('Enter 6-digit OTP'); return; }
   275→    setPhoneVerifying(true);
   276→    try {
   277→      await api.auth.verifyPhoneOtp(user.phone, phoneOtp);
   278→      toast.success('Phone verified successfully!');
   279→      setPhoneDialogOpen(false);
   280→      // Update local state
   281→      useAuthStore.getState().setAuth({ ...user, phoneVerified: true } as any);
   282→    } catch (err: any) {
   283→      toast.error(err.message || 'Verification failed');
   284→    } finally {
   285→      setPhoneVerifying(false);
   286→    }
   287→  };
   288→
   289→  const fetchData = useCallback(async () => {
   290→    if (!profile?.id) { setLoading(false); return; }
   291→    setLoading(true);
   292→    setError('');
   293→    try {
   294→      const res = await api.bookings.list({ caregiverId: profile.id });
   295→      const all = (res.bookings || []).map((b: any) => ({
   296→        ...b,
   297→        date: b.startDate,
   298→        patientName: b.patient?.name,
   299→        familyName: b.family?.name,
   300→        careType: b.careRequirements ? (typeof b.careRequirements === 'string' ? JSON.parse(b.careRequirements) : b.careRequirements)?.needs?.join(', ') : undefined,
   301→      }));
   302→      setBookings(all);
   303→    } catch (err: any) {
   304→      setError(err.message || 'Failed to load dashboard data');
   305→    } finally {
   306→      setLoading(false);
   307→    }
   308→  }, [profile?.id]);
   309→
   310→  useEffect(() => {
   311→    fetchData();
   312→  }, [fetchData]);
   313→
   314→  // No profile yet — show setup guidance immediately, no loading flash
   315→  if (!profile) {
   316→    return (
   317→      <div className="space-y-6">
   318→        <div>
   319→          <h2 className="text-2xl font-bold text-gray-900">Welcome, {user.name.split(' ')[0]}!</h2>
   320→          <p className="text-sm text-gray-400 mt-1">Complete your profile to start receiving care requests.</p>
   321→        </div>
   322→        <Card className="rounded-2xl border-dashed border-2 border-forest-200 bg-forest-50/50">
   323→          <CardContent className="p-8 text-center">
   324→            <div className="w-16 h-16 rounded-2xl green-gradient-bg flex items-center justify-center mx-auto mb-4">
   325→              <ClipboardList className="h-7 w-7 text-white" />
   326→            </div>
   327→            <h3 className="text-lg font-bold text-gray-800 mb-2">Set Up Your Caregiver Profile</h3>
   328→            <p className="text-sm text-gray-500 max-w-md mx-auto">Go to the <strong>My Profile</strong> tab to create your caregiver profile. Families will be able to find and book you once your profile is set up.</p>
   329→          </CardContent>
   330→        </Card>
   331→      </div>
   332→    );
   333→  }
   334→
   335→  if (error) {
   336→    return <ErrorState message={error} onRetry={fetchData} />;
   337→  }
   338→
   339→  const upcomingBookings = bookings
   340→    .filter((b) => ['CONFIRMED', 'IN_PROGRESS'].includes(b.status))
   341→    .sort((a, b) => new Date(a.date).getTime() - new Date(b.date).getTime());
   342→
   343→  const totalBookings = bookings.length;
   344→  const completedJobs = profile?.completedJobs ?? bookings.filter((b) => b.status === 'COMPLETED').length;
   345→  const overallRating = profile?.overallRating ?? 0;
   346→  const hourlyRate = profile?.hourlyRate ?? 0;
   347→
   348→  const statCards = [
   349→    { label: 'Total Bookings', value: totalBookings, icon: <CalendarCheck className="h-5 w-5" />, color: 'bg-forest-50 text-forest-700' },
   350→    { label: 'Completed Jobs', value: completedJobs, icon: <CheckCircle2 className="h-5 w-5" />, color: 'bg-lime-50 text-lime-700' },
   351→    { label: 'Overall Rating', value: overallRating.toFixed(1), icon: <Star className="h-5 w-5" />, color: 'bg-amber-50 text-amber-700', suffix: '/5' },
   352→    { label: 'Hourly Rate', value: `₹${hourlyRate}`, icon: <IndianRupee className="h-5 w-5" />, color: 'bg-purple-50 text-purple-700', isString: true },
   353→  ];
   354→
   355→  const skills = parseJsonSafe(profile?.skills);
   356→  const languages = parseJsonSafe(profile?.languages);
   357→
   358→  return (
   359→    <div className="space-y-6">
   360→      {/* Phone Verification Banner */}
   361→      {!user.phoneVerified && user.phone && (
   362→        <PhoneVerificationBanner user={user} />
   363→      )}
   364→
   365→      {/* Welcome */}
   366→      <div>
   367→        <h2 className="text-2xl font-bold text-gray-900">
   368→          Welcome back, {user.name.split(' ')[0]}!
   369→        </h2>
   370→        <p className="text-sm text-gray-400 mt-1">Here&apos;s your caregiving dashboard at a glance.</p>
   371→      </div>
   372→
   373→      {/* Phone Verification Banner */}
   374→      {!user.phoneVerified && (
   375→        <Card className="rounded-2xl border-2 border-amber-200 bg-amber-50">
   376→          <CardContent className="p-4 flex items-center justify-between gap-4">
   377→            <div className="flex items-center gap-3 min-w-0">
   378→              <div className="w-10 h-10 rounded-xl bg-amber-100 flex items-center justify-center shrink-0">
   379→                <Phone className="h-5 w-5 text-amber-600" />
   380→              </div>
   381→              <div className="min-w-0">
   382→                <p className="text-sm font-semibold text-amber-900">Verify Your Phone Number</p>

   384→              </div>
   385→            </div>
   386→            <Button onClick={() => setPhoneDialogOpen(true)} size="sm" className="bg-amber-600 hover:bg-amber-700 text-white rounded-xl shrink-0">
   387→              Verify Now
   388→            </Button>
   389→          </CardContent>
   390→        </Card>
   391→      )}
   392→
   393→      {/* Phone OTP Dialog */}
   394→      <Dialog open={phoneDialogOpen} onOpenChange={setPhoneDialogOpen}>
   395→        <DialogContent className="max-w-sm rounded-2xl">
   396→          <DialogHeader>
   397→            <DialogTitle>Verify Phone Number</DialogTitle>
   398→            <DialogDescription>We&apos;ll send a 6-digit OTP to {user.phone}</DialogDescription>
   399→          </DialogHeader>
   400→          <div className="space-y-4 mt-3">
   401→            {!phoneOtpSent ? (
   402→              <Button onClick={sendPhoneOtp} disabled={phoneSending} className="w-full rounded-xl">
   403→                {phoneSending && <Loader2 className="h-4 w-4 animate-spin mr-2" />}
   404→                Send OTP
   405→              </Button>
   406→            ) : (
   407→              <>
   408→                <div className="text-center">
   409→                  <p className="text-sm text-gray-600">Enter the 6-digit OTP</p>

   452→      {/* Stat Cards */}

   454→        {statCards.map((stat, i) => (

   456→            key={stat.label}
   457→            initial={{ opacity: 0, y: 12 }}

   460→          >
   461→            <Card className="rounded-2xl border-gray-100 hover:shadow-md transition-shadow">
   462→              <CardContent className="p-5">
   463→                <div className="flex items-center justify-between">
   464→                  <div>
   465→                    <p className="text-sm text-gray-500 font-medium">{stat.label}</p>
   466→                    <div className="flex items-baseline gap-1 mt-1">
   467→                      <p className="text-3xl font-bold text-gray-900">
   468→                        {stat.isString ? stat.value : stat.value}
   469→                      </p>
   470→                      {stat.suffix && <span className="text-sm text-gray-400">{stat.suffix}</span>}
   471→                    </div>
   472→                  </div>

   483→      {/* Upcoming Bookings */}
   484→      <div>
   485→        <h3 className="text-base font-semibold text-gray-800 mb-3">Upcoming & Active Bookings</h3>

   488→            icon={<CalendarCheck className="h-7 w-7" />}
   489→            title="No upcoming bookings"
   490→            description="New bookings from families will appear here once confirmed."
   491→          />
   492→        ) : (
   493→          <div className="space-y-3">
   494→            {upcomingBookings.map((booking) => (
   495→              <motion.div
   496→                key={booking.id}
   497→                initial={{ opacity: 0, x: -8 }}
   498→                animate={{ opacity: 1, x: 0 }}
   499→                transition={{ duration: 0.2 }}
   500→              >
   501→                <Card className="rounded-2xl border-gray-100">
   502→                  <CardContent className="p-4">
   503→                    <div className="flex items-center justify-between gap-4">
   504→                      <div className="flex items-center gap-3 min-w-0">
   505→                        <div className="w-10 h-10 rounded-xl bg-forest-50 flex items-center justify-center shrink-0">
   506→                          <UserRound className="h-5 w-5 text-forest-700" />
   507→                        </div>
   508→                        <div className="min-w-0">

   925→/* AADHAR VERIFICATION SECTION                                    */
   926→/* ============================================================ */
   927→
   928→function AadharVerificationSection({ caregiverId }: { caregiverId: string }) {
   929→  const [aadharNumber, setAadharNumber] = useState('');
   930→  const [verifying, setVerifying] = useState(false);
   931→  const [verified, setVerified] = useState(false);
   932→  const [verifiedNumber, setVerifiedNumber] = useState('');
   933→  const [error, setError] = useState('');
   934→
   935→  const formatAadhar = (value: string) => {
   936→    const digits = value.replace(/\D/g, '').slice(0, 12);
   937→    if (digits.length <= 4) return digits;
   938→    if (digits.length <= 8) return `${digits.slice(0, 4)} ${digits.slice(4)}`;
   939→    return `${digits.slice(0, 4)} ${digits.slice(4, 8)} ${digits.slice(8)}`;
   940→  };
   941→
   942→  const handleAadharChange = (e: React.ChangeEvent<HTMLInputElement>) => {
   943→    const formatted = formatAadhar(e.target.value);
   944→    setAadharNumber(formatted);
   945→    setError('');
   946→  };
   947→
   948→  const handleVerify = async () => {
   949→    const digits = aadharNumber.replace(/\s/g, '');
   950→    if (digits.length !== 12) {
   951→      setError('Enter complete 12-digit Aadhaar number');
   952→      return;
   953→    }
   954→    setVerifying(true);
   955→    setError('');
   956→    try {
   957→      const result = await api.verifyAadhar(digits, caregiverId);
   958→      if (result.verified) {
   959→        setVerified(true);
   960→        setVerifiedNumber(result.aadharNumber || aadharNumber);
   961→        toast.success('Aadhaar verified successfully! You are now a verified caregiver.');
   962→        // Refresh auth to get updated isVerified status
   963→        const me = await api.auth.me(useAuthStore.getState().user?.id || '');
   964→        if (me.user) useAuthStore.getState().setAuth(me.user);
   965→      } else {
   966→        setError(result.error || 'Verification failed');
   967→      }
   968→    } catch (err: any) {
   969→      setError(err.message || 'Verification failed');
   970→    } finally {
   971→      setVerifying(false);
   972→    }
   973→  };
   974→
   975→  if (verified) {
   976→    return (
   977→      <div className="mt-6">
   978→        <div className="flex items-center gap-2 mb-3">
   979→          <ShieldCheck className="h-4 w-4 text-green-600" />
   980→          <p className="text-sm font-semibold text-green-800">Aadhaar Verified</p>
   981→        </div>
   982→        <div className="bg-green-50 border border-green-200 rounded-2xl p-5">
   983→          <div className="flex items-center gap-3">
   984→            <div className="w-12 h-12 rounded-xl bg-green-100 flex items-center justify-center shrink-0">
   985→              <CheckCircle2 className="h-6 w-6 text-green-600" />
   986→            </div>
   987→            <div>
   988→              <p className="text-sm font-semibold text-green-900">Verification Complete</p>
   989→              <p className="text-sm font-mono text-green-700 mt-0.5">{verifiedNumber}</p>
   990→            </div>
   991→          </div>
   992→        </div>
   993→      </div>
   994→    );
   995→  }
   996→
