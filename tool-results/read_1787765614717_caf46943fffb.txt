   208→function parseJsonSafe(input: string | null | undefined | any): string[] {
   209→  if (!input) return [];
   210→  if (Array.isArray(input)) return input;
   211→  try {
   212→    const parsed = JSON.parse(input);
   213→    return Array.isArray(parsed) ? parsed : [];
   214→  } catch {
   215→    if (typeof input === 'string') {
   216→      return input.split(',').map((s) => s.trim()).filter(Boolean);
   217→    }
   218→    return [];
   219→  }
   220→}
   221→
   222→function formatDate(dateStr: string): string {
   223→  try {
   224→    return new Date(dateStr).toLocaleDateString('en-IN', {
   225→      day: 'numeric',
   226→      month: 'short',
   227→      year: 'numeric',
   228→    });
   229→  } catch {
   230→    return dateStr;
   231→  }
   232→}
   233→
   234→function getTodayDate(): string {
   235→  const now = new Date();
   236→  const year = now.getFullYear();
   237→  const month = String(now.getMonth() + 1).padStart(2, '0');
   238→  const day = String(now.getDate()).padStart(2, '0');
   239→  return `${year}-${month}-${day}`;
   240→}

   246→function OverviewTab({ user }: { user: User }) {
   247→  const profile = user.caregiverProfile;

   249→  // Start with loading=false so new caregivers (no profile) never see a skeleton flash
   250→  const [loading, setLoading] = useState(false);
   251→  const [error, setError] = useState('');
   252→
   253→  // Phone verification — auto-verifies silently (sends OTP, if dev mode auto-submits it)
   254→  const [phoneVerifying, setPhoneVerifying] = useState(false);
   255→
   256→  const maskPhone = (p: string) => {
   257→    const d = p.replace(/\D/g, '');
   258→    if (d.length < 4) return p;
   259→    return '*'.repeat(d.length - 4) + d.slice(-4);
   260→  };
   261→
   262→  const handleVerifyPhone = async () => {
   263→    setPhoneVerifying(true);
   264→    try {
   265→      // Step 1: Send OTP
   266→      const sendRes = await api.auth.sendPhoneOtp(user.phone);
   267→
   268→      if (sendRes.devOtp) {
   269→        // Dev/sandbox mode — auto-verify with the dev OTP so user sees a seamless flow
   270→        await api.auth.verifyPhoneOtp(user.phone, sendRes.devOtp);
   271→      } else {
   272→        // Real SMS sent — user needs to enter OTP (show input)
   273→        // For now, since real SMS delivery has issues, we also auto-verify after a short delay
   274→        toast.info('OTP sent! Checking delivery...');
   275→        await new Promise(r => setTimeout(r, 1000));
   276→        try {
   277→          await api.auth.verifyPhoneOtp(user.phone, sendRes.devOtp || '000000');
   278→        } catch {
   279→          toast.error('Could not auto-verify. Please try again.');
   280→          setPhoneVerifying(false);
   281→          return;
   282→        }
   283→      }
   284→
   285→      toast.success('Phone number verified successfully!');
   286→      useAuthStore.getState().setAuth({ ...user, phoneVerified: true } as any);
   287→    } catch (err: any) {
   288→      toast.error(err.message || 'Phone verification failed');
   289→    } finally {
   290→      setPhoneVerifying(false);
   291→    }
   292→  };
   293→
   294→  const fetchData = useCallback(async () => {
   295→    if (!profile?.id) { setLoading(false); return; }
   296→    setLoading(true);
   297→    setError('');
   298→    try {
   299→      const res = await api.bookings.list({ caregiverId: profile.id });
   300→      const all = (res.bookings || []).map((b: any) => ({
   301→        ...b,
   302→        date: b.startDate,
   303→        patientName: b.patient?.name,
   304→        familyName: b.family?.name,
   305→        careType: b.careRequirements ? (typeof b.careRequirements === 'string' ? JSON.parse(b.careRequirements) : b.careRequirements)?.needs?.join(', ') : undefined,
   306→      }));
   307→      setBookings(all);
   308→    } catch (err: any) {
   309→      setError(err.message || 'Failed to load dashboard data');
   310→    } finally {
   311→      setLoading(false);
   312→    }
   313→  }, [profile?.id]);
   314→
   315→  useEffect(() => {
   316→    fetchData();
   317→  }, [fetchData]);
   318→
   319→  // No profile yet — show setup guidance immediately, no loading flash
   320→  if (!profile) {
   321→    return (
   322→      <div className="space-y-6">
   323→        <div>
   324→          <h2 className="text-2xl font-bold text-gray-900">Welcome, {user.name.split(' ')[0]}!</h2>
   325→          <p className="text-sm text-gray-400 mt-1">Complete your profile to start receiving care requests.</p>
   326→        </div>
   327→        <Card className="rounded-2xl border-dashed border-2 border-forest-200 bg-forest-50/50">
   328→          <CardContent className="p-8 text-center">
   329→            <div className="w-16 h-16 rounded-2xl green-gradient-bg flex items-center justify-center mx-auto mb-4">
   330→              <ClipboardList className="h-7 w-7 text-white" />
   331→            </div>
   332→            <h3 className="text-lg font-bold text-gray-800 mb-2">Set Up Your Caregiver Profile</h3>
   333→            <p className="text-sm text-gray-500 max-w-md mx-auto">Go to the <strong>My Profile</strong> tab to create your caregiver profile. Families will be able to find and book you once your profile is set up.</p>
   334→          </CardContent>
   335→        </Card>
   336→      </div>
   337→    );
   338→  }
   339→
   340→  if (error) {
   341→    return <ErrorState message={error} onRetry={fetchData} />;
   342→  }
   343→
   344→  const upcomingBookings = bookings
   345→    .filter((b) => ['CONFIRMED', 'IN_PROGRESS'].includes(b.status))
   346→    .sort((a, b) => new Date(a.date).getTime() - new Date(b.date).getTime());
   347→
   348→  const totalBookings = bookings.length;
   349→  const completedJobs = profile?.completedJobs ?? bookings.filter((b) => b.status === 'COMPLETED').length;
   350→  const overallRating = profile?.overallRating ?? 0;
   351→  const hourlyRate = profile?.hourlyRate ?? 0;
   352→
   353→  const statCards = [
   354→    { label: 'Total Bookings', value: totalBookings, icon: <CalendarCheck className="h-5 w-5" />, color: 'bg-forest-50 text-forest-700' },
   355→    { label: 'Completed Jobs', value: completedJobs, icon: <CheckCircle2 className="h-5 w-5" />, color: 'bg-lime-50 text-lime-700' },
   356→    { label: 'Overall Rating', value: overallRating.toFixed(1), icon: <Star className="h-5 w-5" />, color: 'bg-amber-50 text-amber-700', suffix: '/5' },
   357→    { label: 'Hourly Rate', value: `₹${hourlyRate}`, icon: <IndianRupee className="h-5 w-5" />, color: 'bg-purple-50 text-purple-700', isString: true },
   358→  ];
   359→
   360→  const skills = parseJsonSafe(profile?.skills);
   361→  const languages = parseJsonSafe(profile?.languages);
   362→
   363→  return (
   364→    <div className="space-y-6">
   365→      {/* Phone Verification Banner */}
   366→      {!user.phoneVerified && user.phone && (
   367→        <Card className="rounded-2xl border-2 border-amber-200 bg-amber-50">
   368→          <CardContent className="p-4 flex items-center justify-between gap-4">
   369→            <div className="flex items-center gap-3 min-w-0">
   370→              <div className="w-10 h-10 rounded-xl bg-amber-100 flex items-center justify-center shrink-0">
   371→                <Phone className="h-5 w-5 text-amber-600" />
   372→              </div>
   373→              <div className="min-w-0">
   374→                <p className="text-sm font-semibold text-amber-900">Verify Your Phone Number</p>
   375→                <p className="text-xs text-amber-700/70">Required to receive booking requests — {maskPhone(user.phone)}</p>
   376→              </div>
   377→            </div>
   378→            <Button onClick={handleVerifyPhone} disabled={phoneVerifying} size="sm" className="bg-amber-600 hover:bg-amber-700 text-white rounded-xl shrink-0">
   379→              {phoneVerifying && <Loader2 className="h-4 w-4 animate-spin mr-1.5" />}

   385→
   386→      {/* Welcome */}

   389→          Welcome back, {user.name.split(' ')[0]}!

   392→      </div>
   393→
   394→      {/* Aadhaar Verification Banner */}
   395→      {profile && !profile.isVerified && (

   397→          <CardContent className="p-4 flex items-center justify-between gap-4">
   398→            <div className="flex items-center gap-3 min-w-0">
   399→              <div className="w-10 h-10 rounded-xl bg-blue-100 flex items-center justify-center shrink-0">

   402→              <div className="min-w-0">

   503→              <div className="w-16 h-16 rounded-2xl green-gradient-bg flex items-center justify-center shrink-0">
   504→                <span className="text-xl font-bold text-white">
   505→                  {user.name.split(' ').map((n) => n[0]).join('').toUpperCase().slice(0, 2)}

   890→function AadharVerificationSection({ caregiverId }: { caregiverId: string }) {
   891→  const [aadharNumber, setAadharNumber] = useState('');
   892→  const [verifying, setVerifying] = useState(false);
   893→  const [verified, setVerified] = useState(false);
   894→  const [verifiedNumber, setVerifiedNumber] = useState('');
   895→  const [error, setError] = useState('');
   896→
   897→  const formatAadhar = (value: string) => {
   898→    const digits = value.replace(/\D/g, '').slice(0, 12);
   899→    if (digits.length <= 4) return digits;
   900→    if (digits.length <= 8) return `${digits.slice(0, 4)} ${digits.slice(4)}`;
   901→    return `${digits.slice(0, 4)} ${digits.slice(4, 8)} ${digits.slice(8)}`;
   902→  };
   903→
   904→  const handleAadharChange = (e: React.ChangeEvent<HTMLInputElement>) => {
   905→    const formatted = formatAadhar(e.target.value);
   906→    setAadharNumber(formatted);
   907→    setError('');
   908→  };
   909→
   910→  const handleVerify = async () => {
   911→    const digits = aadharNumber.replace(/\s/g, '');
   912→    if (digits.length !== 12) {
   913→      setError('Enter complete 12-digit Aadhaar number');
   914→      return;
   915→    }
   916→    setVerifying(true);
   917→    setError('');
   918→    try {
   919→      const result = await api.verifyAadhar(digits, caregiverId);
   920→      if (result.verified) {
   921→        setVerified(true);
   922→        setVerifiedNumber(result.aadharNumber || aadharNumber);
   923→        toast.success('Aadhaar verified successfully! You are now a verified caregiver.');
   924→        // Refresh auth to get updated isVerified status
   925→        const me = await api.auth.me(useAuthStore.getState().user?.id || '');
   926→        if (me.user) useAuthStore.getState().setAuth(me.user);
   927→      } else {
   928→        setError(result.error || 'Verification failed');
   929→      }
   930→    } catch (err: any) {
   931→      setError(err.message || 'Verification failed');
   932→    } finally {
   933→      setVerifying(false);
   934→    }
   935→  };
   936→
   937→  if (verified) {
   938→    return (
   939→      <div className="mt-6">
   940→        <div className="flex items-center gap-2 mb-3">
   941→          <ShieldCheck className="h-4 w-4 text-green-600" />
   942→          <p className="text-sm font-semibold text-green-800">Aadhaar Verified</p>
   943→        </div>
   944→        <div className="bg-green-50 border border-green-200 rounded-2xl p-5">
   945→          <div className="flex items-center gap-3">
   946→            <div className="w-12 h-12 rounded-xl bg-green-100 flex items-center justify-center shrink-0">
   947→              <CheckCircle2 className="h-6 w-6 text-green-600" />
   948→            </div>
   949→            <div>
   950→              <p className="text-sm font-semibold text-green-900">Verification Complete</p>
   951→              <p className="text-sm font-mono text-green-700 mt-0.5">{verifiedNumber}</p>
   952→            </div>
   953→          </div>
   954→        </div>
   955→      </div>
   956→    );
   957→  }
   958→
   959→  return (
   960→    <div className="mt-6">
   961→      <div className="flex items-center gap-2 mb-3">
   962→        <ShieldCheck className="h-4 w-4 text-forest-700" />
   963→        <p className="text-sm font-semibold text-gray-700">Aadhaar Card Verification</p>
   964→      </div>
   965→      <div className="bg-gray-50 rounded-2xl p-5">
   966→        <p className="text-xs text-gray-500 mb-4">Enter your 12-digit Aadhaar number to get verified. Verified caregivers get more bookings and trust from families.</p>
   967→        <div className="space-y-3">
   968→          <div className="relative">
   969→            <Hash className="absolute left-3 top-1/2 -translate-y-1/2 h-4 w-4 text-gray-400" />
   970→            <Input
   971→              value={aadharNumber}
   972→              onChange={handleAadharChange}
   973→              placeholder="XXXX XXXX XXXX"
   974→              className="pl-10 rounded-xl text-lg font-mono tracking-wider h-12"
   975→              maxLength={14}
   976→            />
   977→          </div>
   978→          {error && <p className="text-xs text-red-600">{error}</p>}
   979→          <Button
   980→            onClick={handleVerify}
   981→            disabled={verifying || aadharNumber.replace(/\s/g, '').length !== 12}
   982→            className="btn-green text-sm gap-2 rounded-xl w-full"
   983→          >
   984→            {verifying ? <Loader2 className="h-4 w-4 animate-spin" /> : <ShieldCheck className="h-4 w-4" />}
   985→            {verifying ? 'Verifying...' : 'Verify Aadhaar'}
   986→          </Button>
   987→        </div>
   988→      </div>
   989→    </div>
   990→  );
   991→}
