   128→   128→  /* ---- state ---- */
   129→   129→  const [tab, setTab] = useState<Tab>(defaultTab === "register" ? "register" : "login");
   130→   130→  const [showPassword, setShowPassword] = useState(false);
   131→   131→  const [loading, setLoading] = useState(false);
   132→   132→  const [error, setError] = useState("");
   133→   133→  const [resetOpen, setResetOpen] = useState(false);
   134→   134→
   135→   135→  // Login form
   136→   136→  const [loginEmail, setLoginEmail] = useState("");
   137→   137→  const [loginPassword, setLoginPassword] = useState("");
   138→   138→
   139→   139→  // Register form
   140→   140→  const [regName, setRegName] = useState("");
   141→   141→  const [regEmail, setRegEmail] = useState("");
   142→   142→  const [regPhone, setRegPhone] = useState("");
   143→   143→  const [regPassword, setRegPassword] = useState("");
   144→   144→  const [regRole, setRegRole] = useState<UserRole>("FAMILY");
   145→   145→
   146→   146→  // Reset form (simplified — no OTP)
   147→   147→  const [resetEmail, setResetEmail] = useState("");
   148→   148→  const [resetNewPassword, setResetNewPassword] = useState("");
   149→   149→  const [resetLoading, setResetLoading] = useState(false);
   150→   150→  const [resetError, setResetError] = useState("");
   151→   151→  const [resetDone, setResetDone] = useState(false);
   152→   152→
   153→   153→  const { setAuth } = useAuthStore();
   154→   154→  const { toast } = useToast();
   155→   155→
   156→   156→  // Google popup form
   157→   157→  const [googleOpen, setGoogleOpen] = useState(false);
   158→   158→  const [googleEmail, setGoogleEmail] = useState("");
   159→   159→  const [googleName, setGoogleName] = useState("");
   160→   160→  const [googleError, setGoogleError] = useState("");
   161→   161→  const [googleSubmitting, setGoogleSubmitting] = useState(false);
   162→   162→
   163→   163→  /* ---- reset form fields when switching tabs ---- */
   164→   164→  useEffect(() => {
   165→   165→    setError("");
   166→   166→    setLoginPassword("");
   167→   167→  }, [tab]);
   168→   168→
   169→   169→  /* ---- reset entire state when modal closes ---- */
   170→   170→  const handleClose = useCallback(() => {
   171→   171→    setTab(defaultTab === "register" ? "register" : "login");
   172→   172→    setError("");
   173→   173→    setShowPassword(false);
   174→   174→    setLoading(false);
   175→   175→    setLoginEmail("");
   176→   176→    setLoginPassword("");
   177→   177→    setRegName("");
   178→   178→    setRegEmail("");
   179→   179→    setRegPhone("");
   180→   180→    setRegPassword("");
   181→   181→    setRegRole("FAMILY");
   182→   182→    setResetOpen(false);
   183→   183→    setResetEmail("");
   184→   184→    setResetNewPassword("");
   185→   185→    setResetLoading(false);
   186→   186→    setResetError("");
   187→   187→    setResetDone(false);
   188→   188→    setGoogleOpen(false);
   189→   189→    setGoogleEmail("");

   191→   191→    setGoogleError("");

   195→   195→
   196→   196→  /* ================================================================ */
   197→   197→  /*  HANDLERS                                                         */
   198→   198→  /* ================================================================ */
   199→   199→
   200→   200→  /* ---- Google sign-in — redirect flow, fallback to simulated ---- */
   201→   201→  const handleGoogleSignIn = async () => {
   202→   202→    setGoogleError("");
   203→   203→    try {
   204→   204→      // Check if real Google OAuth is configured
   205→   205→      const checkRes = await fetch("/api/auth/google-configured");
   206→   206→      const { configured } = await checkRes.json();
   207→   207→
   208→   208→      if (configured) {
   209→   209→        // Real Google OAuth — full-page redirect to /api/auth/google-go
   210→   210→        const origin = window.location.origin;
   211→   211→        const role = tab === "register" ? regRole : "";
   212→   212→        window.location.href = `/api/auth/google-go?origin=${encodeURIComponent(origin)}&role=${encodeURIComponent(role)}`;
   213→   213→        return;
   214→   214→      }
   215→   215→    } catch {
   216→   216→      // If check fails, fall through to simulated flow
   217→   217→    }
   218→   218→    // Google OAuth not configured — open simulated sign-in
   219→   219→    setGoogleOpen(true);
   220→   220→  };
   221→   221→
   222→   222→  /* ---- Google popup submit ---- */
   223→   223→  const handleGoogleSubmit = async (e: React.FormEvent) => {
   224→   224→    e.preventDefault();
   225→   225→    if (!googleEmail.trim() || !googleName.trim()) {
   226→   226→      setGoogleError("Please enter your name and email.");
   227→   227→      return;
   228→   228→    }
   229→   229→    if (!googleEmail.trim().includes("@")) {
   230→   230→      setGoogleError("Please enter a valid email address.");
   231→   231→      return;
   232→   232→    }
   233→   233→    setGoogleSubmitting(true);
   234→   234→    setGoogleError("");
   235→   235→    try {
   236→   236→      // 1. Call the simulate endpoint (sets the session cookie)
   237→   237→      const res = await fetch("/api/auth/google-simulate", {
   238→   238→        method: "POST",
   239→   239→        headers: { "Content-Type": "application/json" },
   240→   240→        body: JSON.stringify({ email: googleEmail.trim(), name: googleName.trim(), role: tab === "register" ? regRole : undefined }),
   241→   241→      });
   242→   242→      const data = await res.json();
   243→   243→      if (!res.ok) {
   244→   244→        setGoogleError(data.error || "Google sign-in failed.");
   245→   245→        return;
   246→   246→      }
   247→   247→
   248→   248→      // 2. Fetch the NextAuth session (cookie is already set by step 1)
   249→   249→      const sessionRes = await fetch("/api/auth/session");
   250→   250→      const sessionData = await sessionRes.json();
   251→   251→
   252→   252→      if (sessionData?.user) {
   253→   253→        // 3. Directly update the Zustand store — no page reload needed
   254→   254→        setAuth({
   255→   255→          id: sessionData.user.id,
   256→   256→          email: sessionData.user.email,
   257→   257→          name: sessionData.user.name,
   258→   258→          phone: "",
   259→   259→          role: sessionData.user.role,
   260→   260→          avatarUrl: sessionData.user.image || null,
   261→   261→          subscription: "NONE" as any,
   262→   262→          patientProfiles: [],
   263→   263→          caregiverProfile: null,
   264→   264→        });
   265→   265→        setGoogleOpen(false);
   266→   266→        toast({
   267→   267→          title: "Welcome" + (data.user.isNewUser ? " to SevaSaathi!" : " back!") + " \ud83c\udf89",
   268→   268→          description: "Signed in as " + data.user.email,
   269→   269→        });
   270→   270→        handleClose();
   271→   271→      } else {
   272→   272→        // Fallback: reload the page so NextAuth picks up the cookie
   273→   273→        window.location.href = window.location.origin;
   274→   274→      }
   275→   275→    } catch {
   276→   276→      setGoogleError("Something went wrong. Please try again.");
   277→   277→    } finally {
   278→   278→      setGoogleSubmitting(false);
   279→   279→    }
   280→   280→  };
   281→   281→
   282→   282→  /* ---- Email / password login ---- */
   283→   283→  const handleLogin = async (e: React.FormEvent) => {
   284→   284→    e.preventDefault();
   285→   285→    if (!loginEmail.trim() || !loginPassword.trim()) {
   286→   286→      setError("Please enter your email and password.");
   287→   287→      return;
   288→   288→    }
   289→   289→    setLoading(true);
   290→   290→    setError("");
   291→   291→    try {
   292→   292→      const result = await signIn("credentials", {
   293→   293→        email: loginEmail.trim(),
   294→   294→        password: loginPassword,
   295→   295→        redirect: false,
   296→   296→      });
   297→   297→
   298→   298→      if (result?.error) {
   299→   299→        setError("Invalid email or password. Please try again.");
   300→   300→        return;
   301→   301→      }
   302→   302→
   303→   303→      // Fetch session to get user data, then set auth explicitly
   304→   304→      try {
   305→   305→        const sessionRes = await fetch("/api/auth/session");
   306→   306→        const sessionData = await sessionRes.json();
   307→   307→        if (sessionData?.user) {
   308→   308→          const u = sessionData.user as any;
   309→   309→          setAuth({
   310→   310→            id: u.id || "",
   311→   311→            email: u.email || "",
   312→   312→            name: u.name || "",
   313→   313→            phone: "",
   314→   314→            role: u.role || "FAMILY",
   315→   315→            avatarUrl: u.image || null,
   316→   316→            subscription: "NONE",
   317→   317→            patientProfiles: [],
   318→   318→            caregiverProfile: null,
   319→   319→          });
   320→   320→        }
   321→   321→      } catch {
   322→   322→        // AuthSync will pick it up as fallback
   323→   323→      }
   324→   324→
   325→   325→      toast({
   326→   326→        title: "Welcome back! 🎉",
   327→   327→        description: "You have successfully logged in.",
   328→   328→      });
   329→   329→      handleClose();
   330→   330→    } catch {
   331→   331→      setError("Something went wrong. Please try again.");
   332→   332→    } finally {
   333→   333→      setLoading(false);
   334→   334→    }
   335→   335→  };
   336→   336→
   337→   337→  /* ---- Registration ---- */
   338→   338→  const handleRegister = async (e: React.FormEvent) => {
   339→   339→    e.preventDefault();
   340→   340→    if (!regName.trim() || !regEmail.trim() || !regPhone.trim() || !regPassword.trim()) {
   341→   341→      setError("Please fill in all fields.");
   342→   342→      return;
   343→   343→    }
   344→   344→    if (regPassword.length < 6) {
   345→   345→      setError("Password must be at least 6 characters.");
   346→   346→      return;
   347→   347→    }
   348→   348→    setLoading(true);
   349→   349→    setError("");
   350→   350→    try {
   351→   351→      // 1. Register
   352→   352→      const regRes = await api.auth.register({
   353→   353→        name: regName.trim(),
   354→   354→        email: regEmail.trim(),
   355→   355→        phone: regPhone.trim(),
   356→   356→        password: regPassword,
   357→   357→        role: regRole,
   358→   358→      });
   359→   359→
   360→   360→      // 2. Auto-login with credentials
   361→   361→      const loginResult = await signIn("credentials", {
   362→   362→        email: regEmail.trim(),
   363→   363→        password: regPassword,
   364→   364→        redirect: false,
   365→   365→      });
   366→   366→
   367→   367→      if (loginResult?.error) {
   368→   368→        // Registration succeeded but auto-login failed — that's ok, just notify
   369→   369→        toast({ title: "Account created!", description: "Please log in with your credentials." });
   370→   370→        setTab("login");
   371→   371→        setLoginEmail(regEmail.trim());
   372→   372→        return;
   373→   373→      }
   374→   374→
   375→   375→      // Fetch session to get user data, then set auth explicitly
   376→   376→      try {
   377→   377→        const sessionRes = await fetch("/api/auth/session");
   378→   378→        const sessionData = await sessionRes.json();
   379→   379→        if (sessionData?.user) {
   380→   380→          const u = sessionData.user as any;
   381→   381→          setAuth({
   382→   382→            id: u.id || "",
   383→   383→            email: u.email || "",
   384→   384→            name: u.name || "",
   385→   385→            phone: regPhone.trim(),
   386→   386→            role: u.role || regRole,
   387→   387→            avatarUrl: u.image || null,
   388→   388→            subscription: "NONE",
   389→   389→            patientProfiles: [],
   390→   390→            caregiverProfile: null,
   391→   391→          });
   392→   392→        }
   393→   393→      } catch {
   394→   394→        // AuthSync will pick it up as fallback
   395→   395→      }
   396→   396→
   397→   397→      toast({ title: "Welcome to SevaSaathi! 🎉", description: "Your account has been created." });
   398→   398→      handleClose();
   399→   399→    } catch (err: any) {
   400→   400→      setError(err?.message || "Registration failed. Please try again.");
   401→   401→    } finally {
   402→   402→      setLoading(false);
   403→   403→    }
   404→   404→  };
   405→   405→
   406→   406→  /* ---- Reset password (simplified) ---- */
   407→   407→  const handleReset = async (e: React.FormEvent) => {
   408→   408→    e.preventDefault();
   409→   409→    if (!resetEmail.trim() || !resetNewPassword.trim()) {
   410→   410→      setResetError("Please enter your email and new password.");
   411→   411→      return;
   412→   412→    }
   413→   413→    if (resetNewPassword.length < 6) {
   414→   414→      setResetError("Password must be at least 6 characters.");
   415→   415→      return;
   416→   416→    }
   417→   417→    setResetLoading(true);
   418→   418→    setResetError("");
   419→   419→    try {
   420→   420→      const res = await fetch("/api/auth/reset-password", {
   421→   421→        method: "POST",
   422→   422→        headers: { "Content-Type": "application/json" },
   423→   423→        body: JSON.stringify({ email: resetEmail.trim(), newPassword: resetNewPassword }),
   424→   424→      });
   425→   425→      const data = await res.json();
   426→   426→      if (!res.ok) {
   427→   427→        setResetError(data.error || "Failed to reset password.");
   428→   428→        return;
   429→   429→      }
   430→   430→      setResetDone(true);
   431→   431→      toast({ title: "Password reset successfully!", description: "You can now log in with your new password." });
   432→   432→    } catch {
   433→   433→      setResetError("Something went wrong. Please try again.");
   434→   434→    } finally {
   435→   435→      setResetLoading(false);
   436→   436→    }
   437→   437→  };
   438→   438→
   439→   439→  /* ================================================================ */
   440→   440→  /*  RENDER                                                           */
   441→   441→  /* ================================================================ */
   442→   442→  return (
   443→   443→    <Dialog open={isOpen} onOpenChange={(open) => !open && handleClose()}>
   444→   444→      <DialogContent showCloseButton={false} className="sm:max-w-md p-0 overflow-hidden border-forest-200/40">
   445→   445→        {/* Single custom close button */}
   446→   446→        <button
   447→   447→          onClick={handleClose}
   448→   448→          className="absolute top-3 right-3 z-50 rounded-full p-1.5 text-gray-400 hover:text-gray-600 hover:bg-gray-100 transition-colors"
   449→   449→          aria-label="Close"
   450→   450→        >
   451→   451→          <X className="w-5 h-5" />
   452→   452→        </button>
   453→   453→
   454→   454→        <div className="px-6 pt-6 pb-2">
   455→   455→          <DialogHeader className="space-y-1 mb-1">
   456→   456→            <div className="flex items-center gap-2 justify-center">
   457→   457→              <div className="w-9 h-9 rounded-xl bg-forest-900 flex items-center justify-center">
   458→   458→                <Heart className="w-5 h-5 text-lime-400" />
   459→   459→              </div>
   460→   460→              <DialogTitle className="text-xl font-bold text-forest-900">
   461→   461→                SevaSaathi
   462→   462→              </DialogTitle>
   463→   463→            </div>
   464→   464→            <DialogDescription className="text-sm text-gray-500 mt-1">
   465→   465→              {tab === "login"
   466→   466→                ? "Sign in to your account"
   467→   467→                : "Create your account to get started"}
   468→   468→            </DialogDescription>
   469→   469→          </DialogHeader>
   470→   470→        </div>
   471→   471→
   472→   472→        {/* ---- Error banner ---- */}
   473→   473→        <AnimatePresence>
   474→   474→          {error && (
   475→   475→            <motion.div
   476→   476→              initial={{ height: 0, opacity: 0 }}
   477→   477→              animate={{ height: "auto", opacity: 1 }}
   478→   478→              exit={{ height: 0, opacity: 0 }}
   479→   479→              className="overflow-hidden"
   480→   480→            >
   481→   481→              <div className="mx-6 mt-1 px-3 py-2 rounded-lg bg-red-50 border border-red-200 text-red-700 text-sm flex items-center gap-2">
   482→   482→                <X className="w-4 h-4 flex-shrink-0" />
   483→   483→                {error}
   484→   484→              </div>
   485→   485→            </motion.div>
   486→   486→          )}
   487→   487→        </AnimatePresence>
   488→   488→
   489→   489→        {/* ---- Tab switcher ---- */}
   490→   490→        <div className="px-6 pt-3">
   491→   491→          <div className="flex bg-gray-100 rounded-lg p-1">
   492→   492→            <button
