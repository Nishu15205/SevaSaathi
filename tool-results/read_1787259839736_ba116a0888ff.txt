   124→/* ================================================================== */
   125→/*  MAIN COMPONENT                                                     */
   126→/* ================================================================== */
   127→export default function LoginModal({ isOpen, onClose, defaultTab }: LoginModalProps) {
   128→  /* ---- state ---- */
   129→  const [tab, setTab] = useState<Tab>(defaultTab === "register" ? "register" : "login");
   130→  const [showPassword, setShowPassword] = useState(false);
   131→  const [loading, setLoading] = useState(false);
   132→  const [error, setError] = useState("");
   133→  const [resetOpen, setResetOpen] = useState(false);
   134→
   135→  // Login form
   136→  const [loginEmail, setLoginEmail] = useState("");
   137→  const [loginPassword, setLoginPassword] = useState("");
   138→
   139→  // Register form
   140→  const [regName, setRegName] = useState("");
   141→  const [regEmail, setRegEmail] = useState("");
   142→  const [regPhone, setRegPhone] = useState("");
   143→  const [regPassword, setRegPassword] = useState("");
   144→  const [regRole, setRegRole] = useState<UserRole>("FAMILY");
   145→
   146→  // Reset form (simplified — no OTP)
   147→  const [resetEmail, setResetEmail] = useState("");
   148→  const [resetNewPassword, setResetNewPassword] = useState("");
   149→  const [resetLoading, setResetLoading] = useState(false);
   150→  const [resetError, setResetError] = useState("");
   151→  const [resetDone, setResetDone] = useState(false);
   152→
   153→  const { setAuth } = useAuthStore();
   154→  const { toast } = useToast();
   155→
   156→  // Google popup form
   157→  const [googleOpen, setGoogleOpen] = useState(false);
   158→  const [googleEmail, setGoogleEmail] = useState("");
   159→  const [googleName, setGoogleName] = useState("");
   160→  const [googleError, setGoogleError] = useState("");
   161→  const [googleSubmitting, setGoogleSubmitting] = useState(false);
   162→
   163→  /* ---- reset form fields when switching tabs ---- */
   164→  useEffect(() => {
   165→    setError("");
   166→    setLoginPassword("");
   167→  }, [tab]);
   168→
   169→  /* ---- reset entire state when modal closes ---- */
   170→  const handleClose = useCallback(() => {
   171→    setTab(defaultTab === "register" ? "register" : "login");
   172→    setError("");
   173→    setShowPassword(false);
   174→    setLoading(false);
   175→    setLoginEmail("");
   176→    setLoginPassword("");
   177→    setRegName("");
   178→    setRegEmail("");
   179→    setRegPhone("");
   180→    setRegPassword("");
   181→    setRegRole("FAMILY");
   182→    setResetOpen(false);
   183→    setResetEmail("");
   184→    setResetNewPassword("");
   185→    setResetLoading(false);
   186→    setResetError("");
   187→    setResetDone(false);
   188→    setGoogleOpen(false);
   189→    setGoogleEmail("");
   190→    setGoogleName("");
   191→    setGoogleError("");
   192→    setGoogleSubmitting(false);
   193→    onClose();
   194→  }, [onClose, defaultTab]);
   195→
   196→  /* ================================================================ */
   197→  /*  HANDLERS                                                         */
   198→  /* ================================================================ */
   199→
   200→  /* ---- Google sign-in — redirect flow, fallback to simulated ---- */
   201→  const handleGoogleSignIn = async () => {
   202→    setGoogleError("");
   203→    try {
   204→      // Check if real Google OAuth is configured
   205→      const checkRes = await fetch("/api/auth/google-configured");
   206→      const { configured } = await checkRes.json();
   207→
   208→      if (configured) {
   209→        // Real Google OAuth — full-page redirect to /api/auth/google-go
   210→        const origin = window.location.origin;
   211→        const role = tab === "register" ? regRole : "";
   212→        window.location.href = `/api/auth/google-go?origin=${encodeURIComponent(origin)}&role=${encodeURIComponent(role)}`;
   213→        return;
   214→      }
   215→    } catch {
   216→      // If check fails, fall through to simulated flow
   217→    }
   218→    // Google OAuth not configured — open simulated sign-in
   219→    setGoogleOpen(true);
   220→  };
   221→
   222→  /* ---- Google popup submit ---- */
   223→  const handleGoogleSubmit = async (e: React.FormEvent) => {
   224→    e.preventDefault();
   225→    if (!googleEmail.trim() || !googleName.trim()) {
   226→      setGoogleError("Please enter your name and email.");
   227→      return;
   228→    }
   229→    if (!googleEmail.trim().includes("@")) {
   230→      setGoogleError("Please enter a valid email address.");
   231→      return;
   232→    }
   233→    setGoogleSubmitting(true);
   234→    setGoogleError("");
   235→    try {
   236→      // 1. Call the simulate endpoint (sets the session cookie)
   237→      const res = await fetch("/api/auth/google-simulate", {
   238→        method: "POST",
   239→        headers: { "Content-Type": "application/json" },
   240→        body: JSON.stringify({ email: googleEmail.trim(), name: googleName.trim(), role: tab === "register" ? regRole : undefined }),
   241→      });
   242→      const data = await res.json();
   243→      if (!res.ok) {
   244→        setGoogleError(data.error || "Google sign-in failed.");
   245→        return;
   246→      }
   247→
   248→      // 2. Fetch the NextAuth session (cookie is already set by step 1)
   249→      const sessionRes = await fetch("/api/auth/session");
   250→      const sessionData = await sessionRes.json();
   251→
   252→      if (sessionData?.user) {
   253→        // 3. Directly update the Zustand store — no page reload needed
   254→        setAuth({
   255→          id: sessionData.user.id,
   256→          email: sessionData.user.email,
   257→          name: sessionData.user.name,
   258→          phone: "",
   259→          role: sessionData.user.role,
   260→          avatarUrl: sessionData.user.image || null,
   261→          subscription: "NONE" as any,
   262→          patientProfiles: [],
   263→          caregiverProfile: null,
   264→        });
   265→        setGoogleOpen(false);
   266→        toast({
   267→          title: "Welcome" + (data.user.isNewUser ? " to SevaSaathi!" : " back!") + " \ud83c\udf89",
   268→          description: "Signed in as " + data.user.email,
   269→        });
   270→        handleClose();
   271→      } else {
   272→        // Fallback: reload the page so NextAuth picks up the cookie
   273→        window.location.href = window.location.origin;
   274→      }
   275→    } catch {
   276→      setGoogleError("Something went wrong. Please try again.");
   277→    } finally {
   278→      setGoogleSubmitting(false);
   279→    }
   280→  };
   281→
   282→  /* ---- Email / password login ---- */
   283→  const handleLogin = async (e: React.FormEvent) => {
   284→    e.preventDefault();
   285→    if (!loginEmail.trim() || !loginPassword.trim()) {
   286→      setError("Please enter your email and password.");
   287→      return;
   288→    }
   289→    setLoading(true);
   290→    setError("");
   291→    try {
   292→      const result = await signIn("credentials", {
   293→        email: loginEmail.trim(),
   294→        password: loginPassword,
   295→        redirect: false,
   296→      });
   297→
   298→      if (result?.error) {
   299→        setError("Invalid email or password. Please try again.");
   300→        return;
   301→      }
   302→
   303→      // Fetch session to get user data, then set auth explicitly
   304→      try {
   305→        const sessionRes = await fetch("/api/auth/session");
   306→        const sessionData = await sessionRes.json();
   307→        if (sessionData?.user) {
   308→          const u = sessionData.user as any;
   309→          setAuth({
   310→            id: u.id || "",
   311→            email: u.email || "",
   312→            name: u.name || "",
   313→            phone: "",
   314→            role: u.role || "FAMILY",
   315→            avatarUrl: u.image || null,
   316→            subscription: "NONE",
   317→            patientProfiles: [],
   318→            caregiverProfile: null,
   319→          });
   320→        }
   321→      } catch {
   322→        // AuthSync will pick it up as fallback
   323→      }
   324→
   325→      toast({
   326→        title: "Welcome back! 🎉",
   327→        description: "You have successfully logged in.",
   328→      });
   329→      handleClose();
   330→    } catch {
   331→      setError("Something went wrong. Please try again.");
   332→    } finally {
   333→      setLoading(false);
   334→    }
   335→  };
   336→
   337→  /* ---- Registration ---- */
   338→  const handleRegister = async (e: React.FormEvent) => {
   339→    e.preventDefault();
   340→    if (!regName.trim() || !regEmail.trim() || !regPhone.trim() || !regPassword.trim()) {
   341→      setError("Please fill in all fields.");
   342→      return;
   343→    }
   344→    if (regPassword.length < 6) {
   345→      setError("Password must be at least 6 characters.");
   346→      return;
   347→    }
   348→    setLoading(true);
   349→    setError("");
   350→    try {
   351→      // 1. Register
   352→      const regRes = await api.auth.register({
   353→        name: regName.trim(),
   354→        email: regEmail.trim(),
   355→        phone: regPhone.trim(),
   356→        password: regPassword,
   357→        role: regRole,
   358→      });
   359→
   360→      // 2. Auto-login with credentials
   361→      const loginResult = await signIn("credentials", {
   362→        email: regEmail.trim(),
   363→        password: regPassword,
   364→        redirect: false,
   365→      });
   366→
   367→      if (loginResult?.error) {
   368→        // Registration succeeded but auto-login failed — that's ok, just notify
   369→        toast({ title: "Account created!", description: "Please log in with your credentials." });
   370→        setTab("login");
   371→        setLoginEmail(regEmail.trim());
   372→        return;
   373→      }
   374→
   375→      // Fetch session to get user data, then set auth explicitly
   376→      try {
   377→        const sessionRes = await fetch("/api/auth/session");
   378→        const sessionData = await sessionRes.json();
   379→        if (sessionData?.user) {
   380→          const u = sessionData.user as any;
   381→          setAuth({
   382→            id: u.id || "",
   383→            email: u.email || "",
   384→            name: u.name || "",
   385→            phone: regPhone.trim(),
   386→            role: u.role || regRole,
   387→            avatarUrl: u.image || null,
   388→            subscription: "NONE",
   389→            patientProfiles: [],
   390→            caregiverProfile: null,
   391→          });
   392→        }
   393→      } catch {
   394→        // AuthSync will pick it up as fallback
   395→      }
   396→
   397→      toast({ title: "Welcome to SevaSaathi! 🎉", description: "Your account has been created." });
   398→      handleClose();
   399→    } catch (err: any) {
   400→      setError(err?.message || "Registration failed. Please try again.");
   401→    } finally {
   402→      setLoading(false);
   403→    }
   404→  };
   405→
   406→  /* ---- Reset password (simplified) ---- */
   407→  const handleReset = async (e: React.FormEvent) => {
   408→    e.preventDefault();
   409→    if (!resetEmail.trim() || !resetNewPassword.trim()) {
   410→      setResetError("Please enter your email and new password.");
   411→      return;
   412→    }
   413→    if (resetNewPassword.length < 6) {
   414→      setResetError("Password must be at least 6 characters.");
   415→      return;
   416→    }
   417→    setResetLoading(true);
   418→    setResetError("");
   419→    try {
   420→      const res = await fetch("/api/auth/reset-password", {
   421→        method: "POST",
   422→        headers: { "Content-Type": "application/json" },
   423→        body: JSON.stringify({ email: resetEmail.trim(), newPassword: resetNewPassword }),
   424→      });
   425→      const data = await res.json();
   426→      if (!res.ok) {
   427→        setResetError(data.error || "Failed to reset password.");
   428→        return;
   429→      }
   430→      setResetDone(true);
   431→      toast({ title: "Password reset successfully!", description: "You can now log in with your new password." });
   432→    } catch {
   433→      setResetError("Something went wrong. Please try again.");
   434→    } finally {
   435→      setResetLoading(false);
   436→    }
   437→  };
   438→
   439→  /* ================================================================ */
   440→  /*  RENDER                                                           */
   441→  /* ================================================================ */
   442→  return (
   443→    <Dialog open={isOpen} onOpenChange={(open) => !open && handleClose()}>
   444→      <DialogContent showCloseButton={false} className="sm:max-w-md p-0 overflow-hidden border-forest-200/40">
   445→        {/* Single custom close button */}
   446→        <button
   447→          onClick={handleClose}
   448→          className="absolute top-3 right-3 z-50 rounded-full p-1.5 text-gray-400 hover:text-gray-600 hover:bg-gray-100 transition-colors"
   449→          aria-label="Close"
   450→        >
   451→          <X className="w-5 h-5" />
   452→        </button>
   453→
   454→        <div className="px-6 pt-6 pb-2">
   455→          <DialogHeader className="space-y-1 mb-1">
   456→            <div className="flex items-center gap-2 justify-center">
   457→              <div className="w-9 h-9 rounded-xl bg-forest-900 flex items-center justify-center">
   458→                <Heart className="w-5 h-5 text-lime-400" />
   459→              </div>
   460→              <DialogTitle className="text-xl font-bold text-forest-900">
   461→                SevaSaathi
   462→              </DialogTitle>
   463→            </div>
   464→            <DialogDescription className="text-sm text-gray-500 mt-1">
   465→              {tab === "login"
   466→                ? "Sign in to your account"
   467→                : "Create your account to get started"}
   468→            </DialogDescription>
   469→          </DialogHeader>
   470→        </div>
   471→
   472→        {/* ---- Error banner ---- */}
   473→        <AnimatePresence>
   474→          {error && (
   475→            <motion.div
   476→              initial={{ height: 0, opacity: 0 }}
   477→              animate={{ height: "auto", opacity: 1 }}
   478→              exit={{ height: 0, opacity: 0 }}
   479→              className="overflow-hidden"
   480→            >
   481→              <div className="mx-6 mt-1 px-3 py-2 rounded-lg bg-red-50 border border-red-200 text-red-700 text-sm flex items-center gap-2">
   482→                <X className="w-4 h-4 flex-shrink-0" />
   483→                {error}
   484→              </div>
   485→            </motion.div>
   486→          )}
   487→        </AnimatePresence>
   488→
   489→        {/* ---- Tab switcher ---- */}
   490→        <div className="px-6 pt-3">
   491→          <div className="flex bg-gray-100 rounded-lg p-1">
   492→            <button
   493→              onClick={() => setTab("login")}
   494→              className={`flex-1 py-2 px-4 rounded-md text-sm font-medium transition-all duration-200 ${
   495→                tab === "login"
   496→                  ? "bg-white text-forest-900 shadow-sm"
   497→                  : "text-gray-500 hover:text-gray-700"
   498→              }`}
   499→            >
   500→              Sign In
   501→            </button>
   502→            <button
   503→              onClick={() => setTab("register")}
   504→              className={`flex-1 py-2 px-4 rounded-md text-sm font-medium transition-all duration-200 ${
   505→                tab === "register"
   506→                  ? "bg-white text-forest-900 shadow-sm"
   507→                  : "text-gray-500 hover:text-gray-700"
   508→              }`}
   509→            >
   510→              Create Account
   511→            </button>
   512→          </div>
   513→        </div>
   514→
   515→        {/* ---- Animated content ---- */}
   516→        <div className="px-6 py-4 overflow-hidden">
   517→          <AnimatePresence mode="wait" custom={tab}>
   518→            {/* ========== LOGIN TAB ========== */}
   519→            {tab === "login" && (
   520→              <motion.div
   521→                key="login"
   522→                custom={tab}
   523→                variants={contentVariants}
   524→                initial="enter"
   525→                animate="center"
   526→                exit="exit"
   527→                className="space-y-4"
   528→              >
   529→                {/* Google button */}
   530→                <button
   531→                  type="button"
   532→                  onClick={handleGoogleSignIn}
   533→                  className="w-full flex items-center justify-center gap-3 h-11 px-4 rounded-lg border border-gray-300 bg-white hover:bg-gray-50 active:bg-gray-100 transition-all duration-150 text-sm font-medium text-gray-700 shadow-sm"
   534→                >
   535→                  <GoogleIcon />
   536→                  Continue with Google
   537→                </button>
   538→
   539→                {/* Divider */}
   540→                <div className="flex items-center gap-3">
   541→                  <div className="flex-1 h-px bg-gray-200" />
   542→                  <span className="text-xs text-gray-400 font-medium uppercase tracking-wider">
   543→                    or continue with email
   544→                  </span>
   545→                  <div className="flex-1 h-px bg-gray-200" />
   546→                </div>
   547→
   548→                {/* Email / password form */}
   549→                <form onSubmit={handleLogin} className="space-y-3">
   550→                  <div className="space-y-1.5">
   551→                    <Label htmlFor="login-email" className="text-sm font-medium text-gray-700">
   552→                      Email
   553→                    </Label>
   554→                    <div className="relative">
   555→                      <Mail className="absolute left-3 top-1/2 -translate-y-1/2 w-4 h-4 text-gray-400" />
   556→                      <Input
   557→                        id="login-email"
   558→                        type="email"
   559→                        placeholder="you@example.com"
   560→                        value={loginEmail}
   561→                        onChange={(e) => setLoginEmail(e.target.value)}
   562→                        className="pl-10 h-11 border-gray-300 focus:border-forest-500 focus:ring-forest-500"
   563→                        autoComplete="email"
   564→                        disabled={loading}
   565→                      />
   566→                    </div>
   567→                  </div>
   568→
   569→                  <div className="space-y-1.5">
   570→                    <div className="flex items-center justify-between">
   571→                      <Label htmlFor="login-password" className="text-sm font-medium text-gray-700">
   572→                        Password
   573→                      </Label>
   574→                      <button
   575→                        type="button"
   576→                        onClick={() => setResetOpen(true)}
   577→                        className="text-xs font-medium text-forest-700 hover:text-forest-900 transition-colors"
   578→                      >
   579→                        Forgot password?
   580→                      </button>
   581→                    </div>
   582→                    <div className="relative">
   583→                      <Lock className="absolute left-3 top-1/2 -translate-y-1/2 w-4 h-4 text-gray-400" />
   584→                      <Input
   585→                        id="login-password"
   586→                        type={showPassword ? "text" : "password"}
   587→                        placeholder="Enter your password"
   588→                        value={loginPassword}
   589→                        onChange={(e) => setLoginPassword(e.target.value)}
   590→                        className="pl-10 pr-10 h-11 border-gray-300 focus:border-forest-500 focus:ring-forest-500"
   591→                        autoComplete="current-password"
   592→                        disabled={loading}
   593→                      />
   594→                      <button
