   145→function OverviewTab() {
   146→  const user = useAuthStore((s) => s.user);
   147→  const [stats, setStats] = useState({ patients: 0, activeBookings: 0, reports: 0, pendingReviews: 0 });
   148→  const [recentBookings, setRecentBookings] = useState<any[]>([]);
   149→  const [loading, setLoading] = useState(true);
   150→  const [error, setError] = useState('');
   151→
   152→  const fetchData = useCallback(async () => {
   153→    if (!user?.id) return;
   154→    setLoading(true);
   155→    setError('');
   156→    try {
   157→      const [patientsRes, bookingsRes] = await Promise.all([
   158→        api.patients.list(user.id),
   159→        api.bookings.list({ familyId: user.id }),
   160→      ]);
   161→
   162→      const patients = patientsRes.patients || [];
   163→      const bookings = bookingsRes.bookings || [];
   164→
   165→      const activeBookings = bookings.filter((b: any) => ['PENDING', 'CONFIRMED', 'IN_PROGRESS'].includes(b.status));
   166→      const totalReports = bookings.reduce((acc: number, b: any) => acc + (b.careReports?.length || 0), 0);
   167→      const completedWithoutReview = bookings.filter(
   168→        (b: any) => b.status === 'COMPLETED' && (!b.reviews || b.reviews.length === 0)
   169→      ).length;
   170→
   171→      setStats({
   172→        patients: patients.length,
   173→        activeBookings: activeBookings.length,
   174→        reports: totalReports,
   175→        pendingReviews: completedWithoutReview,
   176→      });
   177→      setRecentBookings(bookings.slice(0, 5));
   178→    } catch (err: any) {
   179→      setError(err.message || 'Failed to load dashboard data');
   180→    } finally {
   181→      setLoading(false);
   182→    }
   183→  }, [user?.id]);
   184→
   185→  useEffect(() => {
   186→    fetchData();
   187→  }, [fetchData]);
   188→
   189→  if (loading) {
   190→    return (
   191→      <div className="space-y-6">
   192→        <LoadingGrid count={4} />
   193→        <div>
   194→          <Skeleton className="h-6 w-40 mb-3" />
   195→          <LoadingCards count={3} />
   196→        </div>
   197→      </div>
   198→    );
   199→  }
   200→
   201→  if (error) {
   202→    return <ErrorState message={error} onRetry={fetchData} />;
   203→  }
   204→
   205→  const statCards = [
   206→    { label: 'Total Patients', value: stats.patients, icon: <Users className="h-5 w-5" />, color: 'bg-forest-50 text-forest-700' },
   207→    { label: 'Active Bookings', value: stats.activeBookings, icon: <CalendarCheck className="h-5 w-5" />, color: 'bg-blue-50 text-blue-700' },
   208→    { label: 'Care Reports', value: stats.reports, icon: <FileText className="h-5 w-5" />, color: 'bg-amber-50 text-amber-700' },
   209→    { label: 'Pending Reviews', value: stats.pendingReviews, icon: <Star className="h-5 w-5" />, color: 'bg-lime-50 text-lime-700' },
   210→  ];
   211→
   212→  return (
   213→    <div className="space-y-6">
   214→      {/* Welcome */}
   215→      <div>
   216→        <h2 className="text-2xl font-bold text-gray-900">
   217→          Welcome back, {user?.name?.split(' ')[0]}!
   218→        </h2>
   219→        <p className="text-sm text-gray-400 mt-1">Here's what's happening with your care requests.</p>
   220→      </div>
   221→
   222→      {/* Stat Cards */}
   223→      <div className="grid grid-cols-1 sm:grid-cols-2 lg:grid-cols-4 gap-4">
   224→        {statCards.map((stat, i) => (
   225→          <motion.div
   226→            key={stat.label}
   227→            initial={{ opacity: 0, y: 12 }}
   228→            animate={{ opacity: 1, y: 0 }}
   229→            transition={{ delay: i * 0.05 }}
   230→          >
   231→            <Card className="rounded-2xl border-gray-100 hover:shadow-md transition-shadow">
   232→              <CardContent className="p-5">
   233→                <div className="flex items-center justify-between">
   234→                  <div>
   235→                    <p className="text-sm text-gray-500 font-medium">{stat.label}</p>
   236→                    <p className="text-3xl font-bold text-gray-900 mt-1">{stat.value}</p>
   237→                  </div>
   238→                  <div className={`w-11 h-11 rounded-2xl flex items-center justify-center ${stat.color}`}>
   239→                    {stat.icon}
   240→                  </div>
   241→                </div>
   242→              </CardContent>
   243→            </Card>
   244→          </motion.div>
   245→        ))}
   246→      </div>
   247→
   248→      {/* Recent Bookings */}
   249→      <div>
   250→        <h3 className="text-base font-semibold text-gray-800 mb-3">Recent Bookings</h3>
   251→        {recentBookings.length === 0 ? (
   252→          <EmptyState
   253→            icon={<CalendarCheck className="h-7 w-7" />}
   254→            title="No bookings yet"
   255→            description="Find a caregiver and book your first care session."
   256→          />
   257→        ) : (
   258→          <div className="space-y-3">
   259→            {recentBookings.map((booking: any) => (
   260→              <Card key={booking.id} className="rounded-2xl border-gray-100">
   261→                <CardContent className="p-4">
   262→                  <div className="flex items-center justify-between gap-4">
   263→                    <div className="flex items-center gap-3 min-w-0">
   264→                      <div className="w-10 h-10 rounded-xl bg-forest-50 flex items-center justify-center shrink-0">
   265→                        <UserRound className="h-5 w-5 text-forest-700" />
   266→                      </div>
   267→                      <div className="min-w-0">
   268→                        <p className="text-sm font-semibold text-gray-800 truncate">
   269→                          {booking.patient?.name || 'Patient'}
   270→                        </p>
   271→                        <p className="text-xs text-gray-400">
   272→                          {booking.caregiver?.user?.name || 'Caregiver'} &middot; {booking.shiftType?.replace(/_/g, ' ')}
   273→                        </p>
   274→                      </div>
   275→                    </div>
   276→                    <div className="text-right shrink-0">
   277→                      <Badge variant="outline" className={`text-[11px] rounded-full ${statusColors[booking.status] || ''}`}>
   278→                        {booking.status}
   279→                      </Badge>
   280→                      <p className="text-xs text-gray-400 mt-1">
   281→                        {booking.totalAmount ? `₹${booking.totalAmount}` : '--'}
   282→                      </p>
   283→                    </div>
   284→                  </div>
   285→                </CardContent>
   286→              </Card>
   287→            ))}
   288→          </div>
   289→        )}
   290→      </div>
   291→    </div>
   292→  );
   293→}

   298→function PatientsTab() {
   299→  const user = useAuthStore((s) => s.user);
   300→  const [patients, setPatients] = useState<any[]>([]);
   301→  const [loading, setLoading] = useState(true);
   302→  const [error, setError] = useState('');
   303→  const [modalOpen, setModalOpen] = useState(false);
   304→  const [submitting, setSubmitting] = useState(false);
   305→
   306→  // Form state
   307→  const [form, setForm] = useState({
   308→    name: '', age: '', gender: '', relationship: '', address: '', city: '',
   309→    pincode: '', mobilityStatus: '', medicalHistory: '', careRequirements: '',
   310→    emergencyName: '', emergencyPhone: '',
   311→  });
   312→
   313→  const fetchPatients = useCallback(async () => {
   314→    if (!user?.id) return;
   315→    setLoading(true);
   316→    setError('');
   317→    try {
   318→      const res = await api.patients.list(user.id);
   319→      setPatients(res.patients || []);
   320→    } catch (err: any) {
   321→      setError(err.message || 'Failed to load patients');
   322→    } finally {
   323→      setLoading(false);
   324→    }
   325→  }, [user?.id]);
   326→
   327→  useEffect(() => {
   328→    fetchPatients();
   329→  }, [fetchPatients]);
   330→
   331→  const updateForm = (field: string, value: string) => {
   332→    setForm((prev) => ({ ...prev, [field]: value }));
   333→  };
   334→
   335→  const handleSubmit = async (e: React.FormEvent) => {
   336→    e.preventDefault();
   337→    if (!user?.id) return;
   338→    setSubmitting(true);
   339→    try {
   340→      await api.patients.create({
   341→        familyId: user.id,
   342→        name: form.name,
   343→        age: parseInt(form.age, 10) || 0,
   344→        gender: form.gender,
   345→        relationship: form.relationship,
   346→        address: form.address,
   347→        city: form.city,
   348→        pincode: form.pincode || undefined,
   349→        mobilityStatus: form.mobilityStatus || 'mobile',
   350→        medicalHistory: form.medicalHistory,
   351→        careRequirements: form.careRequirements,
   352→        emergencyContact: JSON.stringify({ name: form.emergencyName, phone: form.emergencyPhone }),
   353→      });
   354→      toast.success('Patient profile created successfully!');
   355→      setModalOpen(false);
   356→      setForm({
   357→        name: '', age: '', gender: '', relationship: '', address: '', city: '',
   358→        pincode: '', mobilityStatus: '', medicalHistory: '', careRequirements: '',
   359→        emergencyName: '', emergencyPhone: '',
   360→      });
   361→      fetchPatients();
   362→    } catch (err: any) {
   363→      toast.error(err.message || 'Failed to create patient');
   364→    } finally {
   365→      setSubmitting(false);
   366→    }
   367→  };
   368→
   369→  if (loading) {
   370→    return <LoadingCards count={3} />;
   371→  }
   372→
   373→  if (error) {
   374→    return <ErrorState message={error} onRetry={fetchPatients} />;
   375→  }
   376→
   377→  const mobilityLabel: Record<string, string> = {
   378→    mobile: 'Mobile', bedridden: 'Bedridden', limited: 'Limited Mobility', wheelchair: 'Wheelchair',
   379→  };
   380→
   381→  return (
   382→    <div className="space-y-6">
   383→      <div className="flex items-center justify-between">
   384→        <div>
   385→          <h2 className="text-xl font-bold text-gray-900">Patient Profiles</h2>
   386→          <p className="text-sm text-gray-400">Manage care recipients in your family.</p>
   387→        </div>
   388→        <Dialog open={modalOpen} onOpenChange={setModalOpen}>
   389→          <DialogTrigger asChild>
   390→            <Button className="btn-black text-sm gap-2">
   391→              <Plus className="h-4 w-4" /> Add Patient
   392→            </Button>
   393→          </DialogTrigger>
   394→          <DialogContent className="max-w-lg max-h-[90vh] overflow-y-auto rounded-2xl">
   395→            <DialogHeader>
   396→              <DialogTitle>Add New Patient</DialogTitle>
   397→            </DialogHeader>
   398→            <form onSubmit={handleSubmit} className="space-y-4 mt-2">
   399→              <div className="grid grid-cols-2 gap-3">
   400→                <div className="col-span-2">
   401→                  <Label className="text-xs font-medium text-gray-600">Full Name *</Label>
   402→                  <Input
   403→                    required
   404→                    value={form.name}
   405→                    onChange={(e) => updateForm('name', e.target.value)}
   406→                    placeholder="Patient name"
   407→                    className="mt-1 rounded-xl"
   408→                  />
   409→                </div>
   410→                <div>
   411→                  <Label className="text-xs font-medium text-gray-600">Age *</Label>
   412→                  <Input
   413→                    required
   414→                    type="number"
   415→                    min={0}
   416→                    max={150}
   417→                    value={form.age}
   418→                    onChange={(e) => updateForm('age', e.target.value)}
   419→                    placeholder="Age"
   420→                    className="mt-1 rounded-xl"
   421→                  />
   422→                </div>
   423→                <div>
   424→                  <Label className="text-xs font-medium text-gray-600">Gender *</Label>

  1532→  const fetchComplaints = useCallback(async () => {
  1533→    if (!user?.id) return;
  1534→    setLoading(true);
  1535→    setError('');
  1536→    try {
  1537→      const res = await api.complaints.list({ familyId: user.id });
  1538→      setComplaints(res.complaints || []);
  1539→      // Also fetch unique caregivers from bookings
  1540→      api.bookings.list({ familyId: user.id }).then(bRes => {
  1541→        const uniqueCaregivers = new Map<string, any>();
  1542→        (bRes.bookings || []).forEach((b: any) => {
  1543→          if (b.caregiver && !uniqueCaregivers.has(b.caregiverId)) {
  1544→            uniqueCaregivers.set(b.caregiverId, b.caregiver);
  1545→          }
  1546→        });
  1547→        setCaregivers(Array.from(uniqueCaregivers.values()));
  1548→      }).catch(() => {});
  1549→    } catch (err: any) {
  1550→      setError(err.message || 'Failed to load complaints');
  1551→    } finally {
  1552→      setLoading(false);
  1553→    }
  1554→  }, [user?.id]);
  1555→
  1556→  useEffect(() => {
  1557→    fetchComplaints();
  1558→  }, [fetchComplaints]);
  1559→
  1560→  const updateForm = (field: string, value: string) => {
  1561→    setForm((prev) => ({ ...prev, [field]: value }));
  1562→  };
  1563→
  1564→  const handleSubmit = async (e: React.FormEvent) => {
  1565→    e.preventDefault();
  1566→    if (!user?.id) return;
  1567→    if (!form.caregiverId) {
  1568→      toast.error('Please select a caregiver');
  1569→      return;
  1570→    }
  1571→    setSubmitting(true);
  1572→    try {
  1573→      await api.complaints.create({
  1574→        familyId: user.id,
  1575→        caregiverId: form.caregiverId,
  1576→        subject: form.subject,
  1577→        description: form.description,
  1578→        priority: form.priority,
  1579→      });
  1580→      toast.success('Complaint filed successfully');
  1581→      setModalOpen(false);
  1582→      setForm({ subject: '', description: '', priority: 'medium', caregiverId: '' });
  1583→      fetchComplaints();
  1584→    } catch (err: any) {
  1585→      toast.error(err.message || 'Failed to file complaint');
  1586→    } finally {
  1587→      setSubmitting(false);
  1588→    }
  1589→  };
  1590→
  1591→  const priorityLabel: Record<string, string> = {
  1592→    low: 'Low', medium: 'Medium', high: 'High', urgent: 'Urgent',
  1593→  };
  1594→
  1595→  if (loading) {
  1596→    return <LoadingCards count={3} />;
  1597→  }
  1598→
  1599→  if (error) {
  1600→    return <ErrorState message={error} onRetry={fetchComplaints} />;
  1601→  }
  1602→
  1603→  return (
  1604→    <div className="space-y-6">
  1605→      <div className="flex items-center justify-between">
  1606→        <div>
  1607→          <h2 className="text-xl font-bold text-gray-900">Complaints</h2>
  1608→          <p className="text-sm text-gray-400">File and track complaints about care services.</p>
  1609→        </div>
  1610→        <Dialog open={modalOpen} onOpenChange={setModalOpen}>
  1611→          <DialogTrigger asChild>
  1612→            <Button className="btn-black text-sm gap-2">
  1613→              <AlertTriangle className="h-4 w-4" /> File Complaint
  1614→            </Button>
  1615→          </DialogTrigger>
  1616→          <DialogContent className="max-w-lg rounded-2xl">
  1617→            <DialogHeader>
  1618→              <DialogTitle>File a Complaint</DialogTitle>
  1619→            </DialogHeader>
  1620→            <form onSubmit={handleSubmit} className="space-y-4 mt-2">
  1621→              <div>
  1622→                <Label className="text-xs font-medium text-gray-600">Caregiver *</Label>
  1623→                <Select value={form.caregiverId} onValueChange={v => updateForm('caregiverId', v)}>
  1624→                  <SelectTrigger className="mt-1 rounded-xl"><SelectValue placeholder="Select caregiver" /></SelectTrigger>
  1625→                  <SelectContent>
  1626→                    {caregivers.map(c => (
  1627→                      <SelectItem key={c.id} value={c.id}>{c.user?.name || 'Unknown Caregiver'}</SelectItem>
  1628→                    ))}
  1629→                  </SelectContent>
  1630→                </Select>
  1631→              </div>
  1632→              <div>
  1633→                <Label className="text-xs font-medium text-gray-600">Subject *</Label>
  1634→                <Input
  1635→                  required
  1636→                  value={form.subject}
  1637→                  onChange={(e) => updateForm('subject', e.target.value)}
  1638→                  placeholder="Brief subject"
  1639→                  className="mt-1 rounded-xl"
  1640→                />
  1641→              </div>
  1642→              <div>
  1643→                <Label className="text-xs font-medium text-gray-600">Description *</Label>
  1644→                <Textarea
  1645→                  required
  1646→                  minLength={10}
  1647→                  value={form.description}
  1648→                  onChange={(e) => updateForm('description', e.target.value)}
  1649→                  placeholder="Describe the issue in detail..."
  1650→                  className="mt-1 rounded-xl min-h-[100px]"
  1651→                />
  1652→              </div>
  1653→              <div>
  1654→                <Label className="text-xs font-medium text-gray-600">Priority</Label>
  1655→                <Select value={form.priority} onValueChange={(v) => updateForm('priority', v)}>
  1656→                  <SelectTrigger className="mt-1 rounded-xl">
  1657→                    <SelectValue />
  1658→                  </SelectTrigger>
  1659→                  <SelectContent>
  1660→                    <SelectItem value="low">Low</SelectItem>
  1661→                    <SelectItem value="medium">Medium</SelectItem>
  1662→                    <SelectItem value="high">High</SelectItem>
  1663→                    <SelectItem value="urgent">Urgent</SelectItem>
  1664→                  </SelectContent>
  1665→                </Select>
  1666→              </div>
  1667→              <div className="flex gap-2 justify-end pt-2">
  1668→                <Button type="button" variant="outline" onClick={() => setModalOpen(false)} className="rounded-xl">
  1669→                  Cancel
  1670→                </Button>
  1671→                <Button type="submit" disabled={submitting} className="btn-black text-sm gap-2">
  1672→                  {submitting ? <Loader2 className="h-4 w-4 animate-spin" /> : <Send className="h-4 w-4" />}
  1673→                  Submit
  1674→                </Button>
  1675→              </div>
  1676→            </form>
  1677→          </DialogContent>
  1678→        </Dialog>
  1679→      </div>
  1680→
  1681→      {complaints.length === 0 ? (
  1682→        <EmptyState
  1683→          icon={<MessageCircle className="h-7 w-7" />}
  1684→          title="No complaints filed"
  1685→          description="If you have any issues with a caregiver or service, file a complaint here."
  1686→        />
  1687→      ) : (
  1688→        <div className="space-y-3">
  1689→          {complaints.map((complaint: any, i: number) => (
  1690→            <motion.div
  1691→              key={complaint.id}
  1692→              initial={{ opacity: 0, y: 6 }}
  1693→              animate={{ opacity: 1, y: 0 }}
  1694→              transition={{ delay: i * 0.03 }}
  1695→            >
  1696→              <Card className="rounded-2xl border-gray-100">
  1697→                <CardContent className="p-4 sm:p-5">
  1698→                  <div className="flex flex-col sm:flex-row sm:items-start justify-between gap-2">
  1699→                    <div className="min-w-0">
  1700→                      <p className="text-sm font-semibold text-gray-800">{complaint.subject}</p>
  1701→                      <p className="text-xs text-gray-400 mt-0.5">
  1702→                        {complaint.caregiver?.user?.name || 'Caregiver'} &middot; {new Date(complaint.createdAt).toLocaleDateString('en-IN', { day: 'numeric', month: 'short', year: 'numeric' })}
  1703→                      </p>
  1704→                      <p className="text-xs text-gray-500 mt-2 line-clamp-2">{complaint.description}</p>
  1705→                    </div>
  1706→                    <div className="flex items-center gap-2 sm:shrink-0">
  1707→                      <Badge className={`text-[10px] rounded-full ${priorityColors[complaint.priority] || ''}`}>
  1708→                        {priorityLabel[complaint.priority] || complaint.priority}
  1709→                      </Badge>
  1710→                      <Badge variant="outline" className={`text-[10px] rounded-full ${complaintStatusColors[complaint.status] || ''}`}>
  1711→                        {complaint.status?.replace(/_/g, ' ')}
  1712→                      </Badge>
  1713→                    </div>
  1714→                  </div>
  1715→                  {complaint.resolution && (
  1716→                    <div className="mt-3 bg-green-50 rounded-xl p-3 text-xs text-green-700">
  1717→                      <p className="font-medium">Resolution:</p>
  1718→                      <p className="mt-0.5">{complaint.resolution}</p>
  1719→                    </div>
  1720→                  )}
  1721→                </CardContent>
  1722→              </Card>
  1723→            </motion.div>
  1724→          ))}
  1725→        </div>
  1726→      )}
  1727→    </div>
  1728→  );
  1729→}
