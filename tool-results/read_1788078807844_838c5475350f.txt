   215→function parseJsonSafe(input: string | null | undefined | any): string[] {
   216→  if (!input) return [];
   217→  if (Array.isArray(input)) return input;
   218→  try {
   219→    const parsed = JSON.parse(input);
   220→    return Array.isArray(parsed) ? parsed : [];
   221→  } catch {
   222→    if (typeof input === 'string') {
   223→      return input.split(',').map((s) => s.trim()).filter(Boolean);
   224→    }
   225→    return [];
   226→  }
   227→}

   229→function formatDate(dateStr: string): string {
   230→  try {
   231→    return new Date(dateStr).toLocaleDateString('en-IN', {
   232→      day: 'numeric',
   233→      month: 'short',
   234→      year: 'numeric',
   235→    });
   236→  } catch {
   237→    return dateStr;
   238→  }
   239→}
   240→
   241→function getTodayDate(): string {
   242→  const now = new Date();
   243→  const year = now.getFullYear();
   244→  const month = String(now.getMonth() + 1).padStart(2, '0');
   245→  const day = String(now.getDate()).padStart(2, '0');
   246→  return `${year}-${month}-${day}`;
   247→}

   253→function OverviewTab({ user }: { user: User }) {
   254→  const profile = user.caregiverProfile;
   255→  const [bookings, setBookings] = useState<Booking[]>([]);
   256→  // Start with loading=false so new caregivers (no profile) never see a skeleton flash
   257→  const [loading, setLoading] = useState(false);
   258→  const [error, setError] = useState('');
   259→
   260→  // Phone verification — real OTP via Firebase
   261→  const [phoneVerifying, setPhoneVerifying] = useState(false);
   262→  const [showOtpDialog, setShowOtpDialog] = useState(false);
   263→  const [otpSent, setOtpSent] = useState(false);
   264→  const [otpValue, setOtpValue] = useState("");
   265→  const firebase = useFirebasePhoneAuth();
   266→
   267→  const maskPhone = (p: string) => {
   268→    const d = p.replace(/\D/g, '');
   269→    if (d.length < 4) return p;
   270→    return '*'.repeat(d.length - 4) + d.slice(-4);
   271→  };
   272→
   273→  const handleSendOtp = async () => {
   274→    setPhoneVerifying(true);
   275→    setShowOtpDialog(true);
   276→    setOtpSent(false);
   277→    setOtpValue('');
   278→
   279→    if (!firebase.isReady) {
   280→      toast.error('OTP service is initializing. If you received an OTP, enter it below.');
   281→      setPhoneVerifying(false);
   282→      return;
   283→    }
   284→
   285→    try {
   286→      const sent = await firebase.sendOtp(user.phone);
   287→      if (sent) {
   288→        setOtpSent(true);
   289→        toast.success('OTP sent to your phone!');
   290→      } else {
   291→        toast.error(firebase.error || 'Could not send OTP automatically. Enter it if received.');
   292→      }
   293→    } catch (err: any) {
   294→      toast.error(err?.message || 'OTP sending failed. Enter it if received.');
   295→    } finally {
   296→      setPhoneVerifying(false);
   297→    }
   298→  };
   299→
   300→  const handleVerifyOtp = async () => {
   301→    if (!otpValue || otpValue.length !== 6) {
   302→      toast.error('Enter the 6-digit OTP sent to your phone');
   303→      return;
   304→    }
   305→    try {
   306→      const result = await firebase.verifyOtp(otpValue);
   307→      if (!result) {
   308→        toast.error(firebase.error || 'Invalid OTP. Please try again.');
   309→        return;
   310→      }
   311→      const res = await fetch('/api/auth/verify-phone-otp', {
   312→        method: 'POST',
   313→        headers: { 'Content-Type': 'application/json' },
   314→        body: JSON.stringify({ phone: user.phone, firebaseToken: result.firebaseToken }),
   315→      });
   316→      const data = await res.json();
   317→      if (!res.ok) {
   318→        toast.error(data.error || 'Verification failed');
   319→        return;
   320→      }
   321→      toast.success('Phone number verified successfully!');
   322→      useAuthStore.getState().setAuth({ ...user, phoneVerified: true } as any);
   323→      setShowOtpDialog(false);
   324→      setOtpValue('');
   325→      firebase.reset();
   326→    } catch (err: any) {
   327→      toast.error(err?.message || 'Verification failed');
   328→    }
   329→  };
   330→
   331→  const fetchData = useCallback(async () => {
   332→    if (!profile?.id) { setLoading(false); return; }
   333→    setLoading(true);
   334→    setError('');
   335→    try {
   336→      const res = await api.bookings.list({ caregiverId: profile.id });
   337→      const all = (res.bookings || []).map((b: any) => ({
   338→        ...b,
   339→        date: b.startDate,
   340→        patientName: b.patient?.name,
   341→        familyName: b.family?.name,
   342→        careType: b.careRequirements ? (typeof b.careRequirements === 'string' ? JSON.parse(b.careRequirements) : b.careRequirements)?.needs?.join(', ') : undefined,
   343→      }));
   344→      setBookings(all);
   345→    } catch (err: any) {
   346→      setError(err.message || 'Failed to load dashboard data');
   347→    } finally {
   348→      setLoading(false);
   349→    }
   350→  }, [profile?.id]);
   351→
   352→  useEffect(() => {
   353→    fetchData();
   354→  }, [fetchData]);
   355→
   356→  // No profile yet — show setup guidance immediately, no loading flash
   357→  if (!profile) {
   358→    return (
   359→      <div className="space-y-6">
   360→        <div>
   361→          <h2 className="text-2xl font-bold text-gray-900">Welcome, {user.name.split(' ')[0]}!</h2>
   362→          <p className="text-sm text-gray-400 mt-1">Complete your profile to start receiving care requests.</p>
   363→        </div>
   364→        <Card className="rounded-2xl border-dashed border-2 border-forest-200 bg-forest-50/50">
   365→          <CardContent className="p-8 text-center">
   366→            <div className="w-16 h-16 rounded-2xl green-gradient-bg flex items-center justify-center mx-auto mb-4">
   367→              <ClipboardList className="h-7 w-7 text-white" />
   368→            </div>
   369→            <h3 className="text-lg font-bold text-gray-800 mb-2">Set Up Your Caregiver Profile</h3>
   370→            <p className="text-sm text-gray-500 max-w-md mx-auto">Go to the <strong>My Profile</strong> tab to create your caregiver profile. Families will be able to find and book you once your profile is set up.</p>
   371→          </CardContent>
   372→        </Card>
   373→      </div>
   374→    );
   375→  }
   376→
   377→  if (error) {
   378→    return <ErrorState message={error} onRetry={fetchData} />;
   379→  }
   380→
   381→  const upcomingBookings = bookings
   382→    .filter((b) => ['CONFIRMED', 'IN_PROGRESS'].includes(b.status))
   383→    .sort((a, b) => new Date(a.date).getTime() - new Date(b.date).getTime());
   384→
   385→  const totalBookings = bookings.length;
   386→  const completedJobs = profile?.completedJobs ?? bookings.filter((b) => b.status === 'COMPLETED').length;
   387→  const overallRating = profile?.overallRating ?? 0;
   388→  const hourlyRate = profile?.hourlyRate ?? 0;
   389→
   390→  const statCards = [
   391→    { label: 'Total Bookings', value: totalBookings, icon: <CalendarCheck className="h-5 w-5" />, color: 'bg-forest-50 text-forest-700' },
   392→    { label: 'Completed Jobs', value: completedJobs, icon: <CheckCircle2 className="h-5 w-5" />, color: 'bg-lime-50 text-lime-700' },
   393→    { label: 'Overall Rating', value: overallRating.toFixed(1), icon: <Star className="h-5 w-5" />, color: 'bg-amber-50 text-amber-700', suffix: '/5' },
   394→    { label: 'Hourly Rate', value: `₹${hourlyRate}`, icon: <IndianRupee className="h-5 w-5" />, color: 'bg-purple-50 text-purple-700', isString: true },
   395→  ];
   396→
   397→  const skills = parseJsonSafe(profile?.skills);
   398→  const languages = parseJsonSafe(profile?.languages);
   399→
   400→  return (
   401→    <div className="space-y-6">
   402→      {/* Phone Verification Banner */}
   403→      {!user.phoneVerified && user.phone && (
   404→        <Card className="rounded-2xl border-2 border-amber-200 bg-gradient-to-r from-amber-50 to-orange-50">
   405→          <CardContent className="p-4 flex items-center justify-between gap-4">
   406→            <div className="flex items-center gap-3 min-w-0">
   407→              <div className="w-10 h-10 rounded-xl bg-gradient-to-br from-amber-400 to-orange-500 flex items-center justify-center shrink-0 shadow-sm">
   408→                <Phone className="h-5 w-5 text-white" />
   409→              </div>
   410→              <div className="min-w-0">
   411→                <p className="text-sm font-semibold text-amber-900">Verify Your Phone Number</p>
   412→                <p className="text-xs text-amber-700/70">Required to receive booking requests — {maskPhone(user.phone)}</p>
   413→              </div>
   414→            </div>
   415→            <Button onClick={handleSendOtp} disabled={phoneVerifying} size="sm" className="bg-gradient-to-r from-forest-700 to-forest-900 hover:from-forest-800 hover:to-forest-950 text-white rounded-xl shrink-0 shadow-sm">
   416→              {phoneVerifying && <Loader2 className="h-4 w-4 animate-spin mr-1.5" />}
   417→              {phoneVerifying ? 'Sending...' : 'Send OTP'}
   418→            </Button>
   419→          </CardContent>
   420→        </Card>
   421→      )}
   422→
   423→      {/* Phone OTP Dialog */}
   424→      <Dialog open={showOtpDialog} onOpenChange={(open) => { if (!open) { setShowOtpDialog(false); setOtpValue(''); firebase.reset(); } }}>
   425→        <DialogContent className="sm:max-w-md">
   426→          <DialogHeader>
   427→            <DialogTitle>Verify Phone Number</DialogTitle>
   428→            <DialogDescription>Enter the OTP sent to {maskPhone(user.phone)}</DialogDescription>
   429→          </DialogHeader>
   430→          <div className="space-y-4 pt-2">
   431→            {otpSent ? (
   432→              <div className="flex items-center gap-2 bg-green-50 border border-green-200 rounded-xl p-3">
   433→                <div className="w-6 h-6 rounded-full bg-green-500 flex items-center justify-center shrink-0">
   434→                  <span className="text-white text-xs font-bold">✓</span>
   435→                </div>
   436→                <p className="text-xs text-green-700 font-medium">OTP sent! Check your phone.</p>
   437→              </div>
   438→            ) : !phoneVerifying ? (
   439→              <div className="flex items-center gap-2 bg-amber-50 border border-amber-200 rounded-xl p-3">
   440→                <div className="w-6 h-6 rounded-full bg-amber-500 flex items-center justify-center shrink-0">
   441→                  <span className="text-white text-xs">!</span>
   442→                </div>
   443→                <p className="text-xs text-amber-700">{firebase.error || 'OTP may take a moment. Enter it if received.'}</p>
   444→              </div>
   445→            ) : null}
   446→            <div>
   447→              <Label className="text-sm">OTP</Label>
   448→              <Input
   449→                value={otpValue}
   450→                onChange={(e) => setOtpValue(e.target.value.replace(/\D/g, '').slice(0, 6))}
   451→                placeholder="6-digit OTP"
   452→                className="mt-1 text-center text-lg tracking-[0.5em] font-mono rounded-xl"
   453→                maxLength={6}
   454→                autoFocus
   455→              />
   456→            </div>
   457→            <Button
   458→              onClick={handleVerifyOtp}
   459→              disabled={firebase.verifying || otpValue.length < 4}
   460→              className="w-full bg-gradient-to-r from-forest-700 to-forest-900 hover:from-forest-800 hover:to-forest-950 text-white rounded-xl shadow-sm"
   461→            >
   462→              {firebase.verifying && <Loader2 className="h-4 w-4 animate-spin mr-2" />}
   463→              Verify OTP
   464→            </Button>
   465→            <button
   466→              type="button"
   467→              onClick={async () => { setOtpValue(''); firebase.reset(); setShowOtpDialog(false); await handleSendOtp(); }}
   468→              disabled={firebase.sendingOtp}
   469→              className="w-full text-xs text-forest-700 hover:text-forest-900 font-medium text-center disabled:opacity-50"
   470→            >
   471→              {firebase.sendingOtp ? 'Resending...' : 'Resend OTP'}
   472→            </button>
   473→          </div>
   474→        </DialogContent>
   475→      </Dialog>
   476→
   477→      {/* Welcome */}
   478→      <div>
   479→        <h2 className="text-2xl font-bold text-gray-900">
   480→          Welcome back, {user.name.split(' ')[0]}!
   481→        </h2>
   482→        <p className="text-sm text-gray-400 mt-1">Here&apos;s your caregiving dashboard at a glance.</p>
   483→      </div>
   484→
   485→      {/* Verification Progress Banner */}
   486→      {profile && (
   487→        profile.isVerified ? (
   488→          <div className="rounded-2xl bg-gradient-to-r from-emerald-600 via-green-600 to-teal-600 p-4 flex items-center gap-3 shadow-sm">
   489→            <div className="w-10 h-10 rounded-xl bg-white/20 backdrop-blur-sm flex items-center justify-center shrink-0">
   490→              <ShieldCheck className="h-5 w-5 text-white" />
   491→            </div>
   492→            <div className="flex-1">
   493→              <div className="flex items-center gap-2">
   494→                <p className="text-sm font-bold text-white">Verified Caregiver</p>
   495→                <Badge className="bg-yellow-400/30 text-yellow-100 border-yellow-400/40 text-xs">40% More Visibility</Badge>
   496→              </div>
   497→              <p className="text-xs text-emerald-100 mt-0.5">Your profile displays the trusted verified badge to families</p>
   498→            </div>
   499→            <div className="w-10 h-10 rounded-full bg-white/20 flex items-center justify-center shrink-0">
   500→              <CheckCircle2 className="h-5 w-5 text-white" />
   501→            </div>
   502→          </div>
   503→        ) : (
   504→          <Card className="rounded-2xl border-2 border-amber-200 bg-amber-50">
   505→            <CardContent className="p-4 flex items-center justify-between gap-4">
   506→              <div className="flex items-center gap-3 min-w-0">
   507→                <div className="w-10 h-10 rounded-xl bg-amber-100 flex items-center justify-center shrink-0">
   508→                  <ShieldCheck className="h-5 w-5 text-amber-600" />
   509→                </div>
   510→                <div className="min-w-0">
   511→                  <p className="text-sm font-semibold text-amber-900">Complete Verification — Get More Bookings</p>
   512→                  <p className="text-xs text-amber-700/70">Verified caregivers get 40% more visibility and a trusted badge</p>

   882→function WithdrawalForm({ caregiverId, userId, availableBalance }: { caregiverId: string; userId: string; availableBalance: number }) {
   883→  const [open, setOpen] = useState(false);
   884→  const [method, setMethod] = useState<'upi' | 'bank_transfer'>('upi');
   885→  const [amount, setAmount] = useState('');
   886→  const [upiId, setUpiId] = useState('');
   887→  const [bankName, setBankName] = useState('');
   888→  const [accountNumber, setAccountNumber] = useState('');
   889→  const [ifscCode, setIfscCode] = useState('');
   890→  const [accountHolder, setAccountHolder] = useState('');
   891→  const [submitting, setSubmitting] = useState(false);
   892→
   893→  const resetForm = () => {
   894→    setAmount(''); setUpiId(''); setBankName(''); setAccountNumber(''); setIfscCode(''); setAccountHolder(''); setMethod('upi');
   895→  };
   896→
   897→  const handleSubmit = async (e: React.FormEvent) => {
   898→    e.preventDefault();
   899→    const amountPaise = Math.round(parseFloat(amount) * 100);
   900→    if (!amountPaise || amountPaise <= 0) { toast.error('Enter a valid amount'); return; }
   901→    if (amountPaise > availableBalance) { toast.error(`Insufficient balance. Available: ₹${(availableBalance / 100).toLocaleString('en-IN')}`); return; }
   902→    if (method === 'upi' && !upiId) { toast.error('Enter UPI ID'); return; }
   903→    if (method === 'bank_transfer' && (!bankName || !accountNumber || !ifscCode || !accountHolder)) { toast.error('Fill all bank details'); return; }
   904→
   905→    setSubmitting(true);
   906→    try {
   907→      await api.withdrawals.create({
   908→        userId, caregiverId, amount: amountPaise, method,
   909→        upiId: method === 'upi' ? upiId : undefined,
   910→        bankName: method === 'bank_transfer' ? bankName : undefined,
   911→        accountNumber: method === 'bank_transfer' ? accountNumber : undefined,
   912→        ifscCode: method === 'bank_transfer' ? ifscCode : undefined,
   913→        accountHolder: method === 'bank_transfer' ? accountHolder : undefined,
   914→      });
   915→      toast.success('Withdrawal request submitted! Admin will process it soon.');
   916→      setOpen(false);
   917→      resetForm();
   918→    } catch (err: any) {
   919→      toast.error(err.message || 'Failed to submit withdrawal');
   920→    } finally {
   921→      setSubmitting(false);
   922→    }
   923→  };
   924→
   925→  return (
   926→    <>
   927→      <Button onClick={() => setOpen(true)} className="btn-green text-sm gap-2 rounded-xl">
   928→        <Wallet className="h-4 w-4" /> Withdraw Funds
   929→      </Button>
   930→
   931→      <Dialog open={open} onOpenChange={(o) => { setOpen(o); if (!o) resetForm(); }}>
   932→        <DialogContent className="max-w-md rounded-2xl">
   933→          <DialogHeader>
   934→            <DialogTitle className="flex items-center gap-2"><Wallet className="h-5 w-5 text-forest-800" /> Withdraw Funds</DialogTitle>
   935→            <DialogDescription>Available: <span className="font-semibold text-forest-700">₹{(availableBalance / 100).toLocaleString('en-IN')}</span></DialogDescription>
   936→          </DialogHeader>
   937→          <form onSubmit={handleSubmit} className="space-y-4 mt-3">
   938→            <div>
   939→              <Label className="text-xs font-medium text-gray-600">Amount (INR) *</Label>
   940→              <Input type="number" min="1" step="1" value={amount} onChange={e => setAmount(e.target.value)} placeholder="Enter amount" className="mt-1 rounded-xl" required />
   941→            </div>
   942→
   943→            {/* Method Toggle */}
   944→            <div>
   945→              <Label className="text-xs font-medium text-gray-600 mb-2 block">Withdrawal Method</Label>
   946→              <div className="grid grid-cols-2 gap-2">
   947→                <button type="button" onClick={() => setMethod('upi')} className={`flex items-center justify-center gap-2 p-3 rounded-xl border-2 text-sm font-medium transition-all ${method === 'upi' ? 'border-forest-500 bg-forest-50 text-forest-800' : 'border-gray-200 text-gray-500 hover:border-gray-300'}`}>
