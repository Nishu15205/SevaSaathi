     6→     6→import { useAuthStore } from "@/stores/authStore";

     9→     9→
    10→    10→interface LoginModalProps {
    11→    11→  isOpen: boolean;

    14→    14→

    16→    16→type UserRole = "FAMILY" | "CAREGIVER";
    17→    17→

    20→    20→  const [showPassword, setShowPassword] = useState(false);
    21→    21→  const [loading, setLoading] = useState(false);
    22→    22→  const [error, setError] = useState("");
    23→    23→  const [success, setSuccess] = useState("");
    24→    24→
    25→    25→  // Login form
    26→    26→  const [loginEmail, setLoginEmail] = useState("");
    27→    27→  const [loginPassword, setLoginPassword] = useState("");
    28→    28→
    29→    29→  // Register form
    30→    30→  const [regName, setRegName] = useState("");
    31→    31→  const [regEmail, setRegEmail] = useState("");
    32→    32→  const [regPhone, setRegPhone] = useState("");
    33→    33→  const [regPassword, setRegPassword] = useState("");
    34→    34→  const [regRole, setRegRole] = useState<UserRole>("FAMILY");
    35→    35→
    36→    36→  // Reset form

    38→    38→  const [resetNewPassword, setResetNewPassword] = useState("");

    41→    41→

    45→    45→  const [otpSent, setOtpSent] = useState(false);

    49→    49→  const [otpPurpose, setOtpPurpose] = useState<"REGISTER" | "RESET_PASSWORD">("REGISTER");
    50→    50→  const [demoOtp, setDemoOtp] = useState("");
    51→    51→  const [otpTimer, setOtpTimer] = useState(0);
    52→    52→  const [pendingRegisterData, setPendingRegisterData] = useState<any>(null);
    53→    53→  const otpInputRef = useRef<HTMLDivElement>(null);

    56→    56→

    61→    61→    setRegName("");

    63→    63→    setRegPhone("");

   100→   100→      setError("Something went wrong. Please try again.");
   101→   101→    } finally { setLoading(false); }
   102→   102→  };

   106→   106→    setOtpError("");

   108→   108→      const res = await fetch("/api/auth/send-otp", {
   109→   109→        method: "POST",
   110→   110→        headers: { "Content-Type": "application/json" },
   111→   111→        body: JSON.stringify({ email, purpose }),
   112→   112→      });
   113→   113→      const data = await res.json();
   114→   114→      if (!res.ok) { setOtpError(data.error || "Failed to send OTP"); return; }
   115→   115→      setDemoOtp(data.otp);
   116→   116→      setOtpSent(true);
   117→   117→      setOtpTimer(120); // 2 minute cooldown for resend
   118→   118→    } catch {
   119→   119→      setOtpError("Failed to send OTP. Please try again.");
   120→   120→    } finally { setOtpLoading(false); }
   121→   121→  };
   122→   122→
   123→   123→  const handleVerifyOtp = async () => {
   124→   124→    if (otpValue.length !== 6) { setOtpError("Please enter the 6-digit OTP"); return; }
   125→   125→    setOtpLoading(true);
   126→   126→    setOtpError("");
   127→   127→    try {
   128→   128→      const res = await fetch("/api/auth/verify-otp", {
   129→   129→        method: "POST",
   130→   130→        headers: { "Content-Type": "application/json" },
   131→   131→        body: JSON.stringify({ email: otpEmail, otp: otpValue, purpose: otpPurpose }),
   132→   132→      });
   133→   133→      const data = await res.json();
   134→   134→      if (!res.ok) { setOtpError(data.error || "Invalid OTP"); return; }
   135→   135→      setOtpVerified(true);
   136→   136→
   137→   137→      // After OTP verified, proceed with the original action
   138→   138→      if (otpPurpose === "REGISTER" && pendingRegisterData) {
   139→   139→        // Complete registration by logging in
   140→   140→        setLoading(true);
   141→   141→        try {
   142→   142→          const fullUser = await api.auth.me(pendingRegisterData.id);
   143→   143→          if (fullUser.user) useAuthStore.getState().setAuth(fullUser.user);
   144→   144→          setSuccess("Account verified! Welcome to SevaSaathi.");
   145→   145→          setTimeout(() => { onClose(); }, 800);
   146→   146→        } catch {
   147→   147→          setError("Account created but failed to load profile. Please login.");
   148→   148→          setTab("login");
   149→   149→        } finally { setLoading(false); }
   150→   150→      }
   151→   151→    } catch {
   152→   152→      setOtpError("Something went wrong. Please try again.");
   153→   153→    } finally { setOtpLoading(false); }
   154→   154→  };
   155→   155→
   156→   156→  const handleRegister = async (e: React.FormEvent) => {
   157→   157→    e.preventDefault();
   158→   158→    clearMessages();
   159→   159→    setLoading(true);
   160→   160→    try {
   161→   161→      // Step 1: Create account
   162→   162→      const res = await fetch("/api/auth/register", {
   163→   163→        method: "POST",
   164→   164→        headers: { "Content-Type": "application/json" },
   165→   165→        body: JSON.stringify({ name: regName, email: regEmail, phone: regPhone, password: regPassword, role: regRole }),
   166→   166→      });
   167→   167→      const data = await res.json();
   168→   168→      if (!res.ok) { setError(data.error || "Registration failed"); return; }
   169→   169→
   170→   170→      // Step 2: Send OTP for email verification
   171→   171→      setPendingRegisterData(data.user);
   172→   172→      setOtpEmail(regEmail);
   173→   173→      setOtpPurpose("REGISTER");
   174→   174→      setOtpValue("");
   175→   175→      setOtpSent(false);
   176→   176→      setOtpVerified(false);
   177→   177→      setOtpTimer(0);
   178→   178→      setTab("otp-verify");
   179→   179→
   180→   180→      // Auto-send OTP when entering OTP tab
   181→   181→      await handleSendOtp(regEmail, "REGISTER");
   182→   182→    } catch {
   183→   183→      setError("Something went wrong. Please try again.");
   184→   184→    } finally { setLoading(false); }
   185→   185→  };
   186→   186→
   187→   187→  const handleResetStep1 = async (e: React.FormEvent) => {
   188→   188→    e.preventDefault();
   189→   189→    clearMessages();
   190→   190→    if (resetNewPassword !== resetConfirmPassword) {
   191→   191→      setError("Passwords do not match");
   192→   192→      return;
   193→   193→    }
   194→   194→    setLoading(true);
   195→   195→    try {
   196→   196→      // Step 1: Send OTP
   197→   197→      setOtpEmail(resetEmail);
   198→   198→      setOtpPurpose("RESET_PASSWORD");
   199→   199→      setOtpValue("");
   200→   200→      setOtpSent(false);
   201→   201→      setOtpVerified(false);
   202→   202→      setOtpTimer(0);
   203→   203→      setTab("otp-verify");
   204→   204→
   205→   205→      await handleSendOtp(resetEmail, "RESET_PASSWORD");
   206→   206→    } catch {
   207→   207→      setError("Something went wrong. Please try again.");
   208→   208→    } finally { setLoading(false); }
   209→   209→  };
   210→   210→
   211→   211→  const handleResetStep2 = async () => {
   212→   212→    if (!otpVerified) { setOtpError("Please verify your email first"); return; }
   213→   213→    setLoading(true);
   214→   214→    try {
   215→   215→      const res = await fetch("/api/auth/reset-password", {
   216→   216→        method: "POST",
   217→   217→        headers: { "Content-Type": "application/json" },
   218→   218→        body: JSON.stringify({ email: resetEmail, newPassword: resetNewPassword, otpVerified: true }),
   219→   219→      });
   220→   220→      const data = await res.json();
   221→   221→      if (!res.ok) { setError(data.error || "Reset failed"); return; }
   222→   222→      setResetDone(true);
   223→   223→      setTab("reset");
   224→   224→    } catch {
   225→   225→      setError("Something went wrong. Please try again.");
   226→   226→    } finally { setLoading(false); }
   227→   227→  };
   228→   228→
   229→   229→  const handleClose = () => {

   231→   231→    setTab("login");
   232→   232→    onClose();
   233→   233→  };
   234→   234→
   235→   235→  const goToReset = (email?: string) => {

   238→   238→    if (email) setResetEmail(email);
   239→   239→    setTab("reset");
   240→   240→  };
   241→   241→
   242→   242→  const goToLogin = () => {
   243→   243→    clearMessages();
   244→   244→    setResetDone(false);
   245→   245→    setTab("login");
   246→   246→  };
   247→   247→
   248→   248→  // Watch for OTP verification in reset flow
   249→   249→  useEffect(() => {
   250→   250→    if (otpVerified && otpPurpose === "RESET_PASSWORD") {
   251→   251→      // Will be handled by the user clicking "Reset Password" button
   252→   252→    }
   253→   253→  }, [otpVerified, otpPurpose]);
   254→   254→
   255→   255→  const headerTitle = tab === "login"
   256→   256→    ? "Welcome back! Sign in to continue."
   257→   257→    : tab === "register"
   258→   258→    ? "Create your account to get started."
   259→   259→    : tab === "otp-verify"
   260→   260→    ? `Verify your email address`
   261→   261→    : "Reset your password";
   262→   262→
   263→   263→  const headerSubtitle = tab === "otp-verify"
   264→   264→    ? `We've sent a 6-digit code to ${otpEmail}`
   265→   265→    : undefined;
   266→   266→
   267→   267→  return (
   268→   268→    <AnimatePresence>
   269→   269→      {isOpen && (
   270→   270→        <motion.div
   271→   271→          initial={{ opacity: 0 }}
   272→   272→          animate={{ opacity: 1 }}
   273→   273→          exit={{ opacity: 0 }}
   274→   274→          className="fixed inset-0 z-[100] flex items-center justify-center p-4"
   275→   275→          onClick={handleClose}
   276→   276→        >
   277→   277→          <div className="absolute inset-0 bg-black/40 backdrop-blur-sm" />
   278→   278→          <motion.div
   279→   279→            initial={{ opacity: 0, scale: 0.95, y: 20 }}

   284→   284→            onClick={(e) => e.stopPropagation()}
   285→   285→          >
   286→   286→            {/* Green header */}

   289→   289→                <X className="h-4 w-4 text-white" />
   290→   290→              </button>
   291→   291→              <div className="flex items-center gap-2 mb-1">
   292→   292→                <Heart className="h-5 w-5 text-white fill-white" />
   293→   293→                <span className="text-white font-bold text-lg">SevaSaathi</span>
   294→   294→              </div>
   295→   295→              <p className="text-white/80 text-sm">{headerTitle}</p>
   296→   296→              {headerSubtitle && (
   297→   297→                <p className="text-white/60 text-xs mt-1 truncate">{headerSubtitle}</p>
   298→   298→              )}
   299→   299→            </div>
   300→   300→
   301→   301→            <div className="p-6">
   302→   302→              {/* Error / Success */}
   303→   303→              <AnimatePresence mode="wait">
   304→   304→                {error && (
   305→   305→                  <motion.div initial={{ opacity: 0, y: -10 }} animate={{ opacity: 1, y: 0 }} exit={{ opacity: 0, y: -10 }} className="mb-4 p-3 bg-red-50 text-red-700 text-sm rounded-2xl">{error}</motion.div>
   306→   306→                )}
   307→   307→                {success && (
   308→   308→                  <motion.div initial={{ opacity: 0, y: -10 }} animate={{ opacity: 1, y: 0 }} exit={{ opacity: 0, y: -10 }} className="mb-4 p-3 bg-green-50 text-green-700 text-sm rounded-2xl">{success}</motion.div>
   309→   309→                )}
   310→   310→              </AnimatePresence>
   311→   311→
   312→   312→              {/* ============ OTP VERIFICATION TAB ============ */}

   314→   314→                <motion.div initial={{ opacity: 0, y: 10 }} animate={{ opacity: 1, y: 0 }} className="space-y-5">
   315→   315→                  <div className="text-center pt-2">
   316→   316→                    <div className="w-16 h-16 rounded-full bg-forest-50 flex items-center justify-center mx-auto mb-4">
   317→   317→                      <Mail className="h-8 w-8 text-forest-700" />
   318→   318→                    </div>

   320→   320→                    <p className="text-sm text-gray-500 mt-1">

   323→   323→                        : "Verify your email to reset password"}
   324→   324→                    </p>
   325→   325→                  </div>
   326→   326→
   327→   327→                  {/* Demo OTP display */}

   330→   330→                      <p className="text-xs text-amber-600 font-medium mb-1">Demo OTP (would be sent to your email)</p>

   332→   332→                    </div>
   333→   333→                  )}
   334→   334→

   336→   336→                  <div ref={otpInputRef} className="flex justify-center">

   342→   342→                    >

   346→   346→                        <InputOTPSlot index={2} className="h-12 w-12 rounded-xl border-2 border-gray-200 text-lg font-bold" />
   347→   347→                      </InputOTPGroup>
   348→   348→                      <InputOTPSeparator className="text-gray-300" />

   359→   359→                  )}
   360→   360→
   361→   361→                  {otpVerified ? (
   362→   362→                    <motion.div initial={{ opacity: 0 }} animate={{ opacity: 1 }} className="text-center space-y-3">
   363→   363→                      <div className="w-12 h-12 rounded-full bg-green-50 flex items-center justify-center mx-auto">
   364→   364→                        <CheckCircle2 className="h-6 w-6 text-green-600" />

   373→   373→                          disabled={loading}

   379→   379→                    </motion.div>
   380→   380→                  ) : (
   381→   381→                    <>
   382→   382→                      <button
   383→   383→                        onClick={handleVerifyOtp}

   389→   389→
   390→   390→                      <div className="flex items-center justify-center gap-3 text-sm">

   393→   393→                        ) : (
   394→   394→                          <button
   395→   395→                            onClick={() => handleSendOtp(otpEmail, otpPurpose)}

   397→   397→                            className="text-forest-800 font-medium hover:text-forest-600 disabled:opacity-50"
   398→   398→                          >
   399→   399→                            {otpLoading ? "Sending..." : "Resend OTP"}
   400→   400→                          </button>
   401→   401→                        )}

   403→   403→
   404→   404→                      <button
   405→   405→                        type="button"

   408→   408→                          else { setTab("reset"); }
   409→   409→                        }}
   410→   410→                        className="flex items-center gap-1.5 text-sm text-gray-500 hover:text-forest-800 transition-colors"
   411→   411→                      >
   412→   412→                        <ArrowLeft className="h-4 w-4" /> Back

   414→   414→                    </>
   415→   415→                  )}
   416→   416→                </motion.div>
   417→   417→              )}
   418→   418→
   419→   419→              {/* ============ RESET PASSWORD TAB ============ */}
   420→   420→              {tab === "reset" && (
   421→   421→                <>
   422→   422→                  {resetDone ? (

   424→   424→                      <div className="w-16 h-16 rounded-full bg-green-50 flex items-center justify-center mx-auto mb-4">
   425→   425→                        <CheckCircle2 className="h-8 w-8 text-green-600" />
