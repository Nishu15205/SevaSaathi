   204→   204→function parseJsonSafe(jsonStr: string | null | undefined): string[] {
   205→   205→  if (!jsonStr) return [];
   206→   206→  try {
   207→   207→    const parsed = JSON.parse(jsonStr);
   208→   208→    return Array.isArray(parsed) ? parsed : [];
   209→   209→  } catch {
   210→   210→    return jsonStr.split(',').map((s) => s.trim()).filter(Boolean);
   211→   211→  }
   212→   212→}
   213→   213→
   214→   214→function formatDate(dateStr: string): string {
   215→   215→  try {
   216→   216→    return new Date(dateStr).toLocaleDateString('en-IN', {
   217→   217→      day: 'numeric',
   218→   218→      month: 'short',
   219→   219→      year: 'numeric',
   220→   220→    });
   221→   221→  } catch {
   222→   222→    return dateStr;
   223→   223→  }
   224→   224→}
   225→   225→
   226→   226→function getTodayDate(): string {
   227→   227→  const now = new Date();
   228→   228→  const year = now.getFullYear();
   229→   229→  const month = String(now.getMonth() + 1).padStart(2, '0');
   230→   230→  const day = String(now.getDate()).padStart(2, '0');
   231→   231→  return `${year}-${month}-${day}`;
   232→   232→}

   791→   791→
   792→   792→  const handleFileSelect = async (e: React.ChangeEvent<HTMLInputElement>) => {
   793→   793→    const file = e.target.files?.[0];
   794→   794→    if (!file) return;
   795→   795→
   796→   796→    // Show preview
   797→   797→    const reader = new FileReader();
   798→   798→    reader.onload = (ev) => {
   799→   799→      setImagePreview(ev.target?.result as string);
   800→   800→      setAadharResult(null);
   801→   801→    };

   803→   803→  };
   804→   804→

   806→   806→    if (!imagePreview) {
   807→   807→      toast.error('Please upload an Aadhar card image first');
   808→   808→      return;
   809→   809→    }

   812→   812→      const result = await api.verifyAadhar(imagePreview);
   813→   813→      setAadharResult(result);
   814→   814→      if (result.verified) {
   815→   815→        toast.success('Aadhar card verified successfully!');
   816→   816→      } else {
   817→   817→        toast.error(result.error || 'Aadhar verification failed');
   818→   818→      }
   819→   819→    } catch (err: any) {
   820→   820→      toast.error(err.message || 'Verification failed');
   821→   821→      setAadharResult({ verified: false, error: err.message });
   822→   822→    } finally {
   823→   823→      setVerifying(false);
   824→   824→    }
   825→   825→  };
   826→   826→
   827→   827→  return (
   828→   828→    <div className="mt-6">
   829→   829→      <div className="flex items-center gap-2 mb-3">

   834→   834→        <p className="text-xs text-gray-500 mb-4">Upload your Aadhar card for AI-powered instant verification. Your data is processed securely and not stored.</p>

   836→   836→        <div className="space-y-4">
   837→   837→          {/* Upload Area */}
   838→   838→          {!imagePreview ? (
