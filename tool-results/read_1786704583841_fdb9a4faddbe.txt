   196→   196→function parseJsonSafe(jsonStr: string | null | undefined): string[] {
   197→   197→  if (!jsonStr) return [];
   198→   198→  try {
   199→   199→    const parsed = JSON.parse(jsonStr);
   200→   200→    return Array.isArray(parsed) ? parsed : [];
   201→   201→  } catch {
   202→   202→    return jsonStr.split(',').map((s) => s.trim()).filter(Boolean);
   203→   203→  }
   204→   204→}
   205→   205→
   206→   206→function formatDate(dateStr: string): string {
   207→   207→  try {
   208→   208→    return new Date(dateStr).toLocaleDateString('en-IN', {
   209→   209→      day: 'numeric',
   210→   210→      month: 'short',
   211→   211→      year: 'numeric',
   212→   212→    });
   213→   213→  } catch {
   214→   214→    return dateStr;
   215→   215→  }
   216→   216→}
   217→   217→
   218→   218→function getTodayDate(): string {
   219→   219→  const now = new Date();
   220→   220→  const year = now.getFullYear();
   221→   221→  const month = String(now.getMonth() + 1).padStart(2, '0');
   222→   222→  const day = String(now.getDate()).padStart(2, '0');
   223→   223→  return `${year}-${month}-${day}`;
   224→   224→}
