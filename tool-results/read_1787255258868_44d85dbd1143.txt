   147→function OverviewTab() {
   148→  const user = useAuthStore((s) => s.user);
   149→  const [stats, setStats] = useState({ patients: 0, activeBookings: 0, reports: 0, pendingReviews: 0 });
   150→  const [recentBookings, setRecentBookings] = useState<any[]>([]);
   151→  const [loading, setLoading] = useState(true);
   152→  const [error, setError] = useState('');
   153→
   154→  const fetchData = useCallback(async () => {
   155→    if (!user?.id) return;
   156→    setLoading(true);
   157→    setError('');
   158→    try {
   159→      const [patientsRes, bookingsRes] = await Promise.all([
   160→        api.patients.list(user.id),
   161→        api.bookings.list({ familyId: user.id }),
   162→      ]);
   163→
   164→      const patients = patientsRes.patients || [];
   165→      const bookings = bookingsRes.bookings || [];
   166→
   167→      const activeBookings = bookings.filter((b: any) => ['PENDING', 'CONFIRMED', 'IN_PROGRESS'].includes(b.status));
   168→      const totalReports = bookings.reduce((acc: number, b: any) => acc + (b.careReports?.length || 0), 0);
   169→      const completedWithoutReview = bookings.filter(
   170→        (b: any) => b.status === 'COMPLETED' && (!b.reviews || b.reviews.length === 0)
   171→      ).length;
   172→
   173→      setStats({
   174→        patients: patients.length,
   175→        activeBookings: activeBookings.length,
   176→        reports: totalReports,
   177→        pendingReviews: completedWithoutReview,
   178→      });
   179→      setRecentBookings(bookings.slice(0, 5));
   180→    } catch (err: any) {
   181→      setError(err.message || 'Failed to load dashboard data');
   182→    } finally {
   183→      setLoading(false);
   184→    }
   185→  }, [user?.id]);
   186→
   187→  useEffect(() => {
   188→    fetchData();
   189→  }, [fetchData]);
   190→
   191→  if (loading) {
   192→    return (
   193→      <div className="space-y-6">
   194→        <LoadingGrid count={4} />
   195→        <div>
   196→          <Skeleton className="h-6 w-40 mb-3" />
   197→          <LoadingCards count={3} />
   198→        </div>
   199→      </div>
   200→    );
   201→  }
   202→
   203→  if (error) {
   204→    return <ErrorState message={error} onRetry={fetchData} />;
   205→  }
   206→
   207→  const statCards = [
   208→    { label: 'Total Members', value: stats.patients, icon: <Users className="h-5 w-5" />, color: 'bg-forest-50 text-forest-700' },
   209→    { label: 'Active Bookings', value: stats.activeBookings, icon: <CalendarCheck className="h-5 w-5" />, color: 'bg-blue-50 text-blue-700' },
   210→    { label: 'Care Reports', value: stats.reports, icon: <FileText className="h-5 w-5" />, color: 'bg-amber-50 text-amber-700' },
   211→    { label: 'Pending Reviews', value: stats.pendingReviews, icon: <Star className="h-5 w-5" />, color: 'bg-lime-50 text-lime-700' },
   212→  ];
   213→
   214→  return (
   215→    <div className="space-y-6">
   216→      {/* Welcome */}
   217→      <div>
   218→        <h2 className="text-2xl font-bold text-gray-900">
   219→          Welcome back, {user?.name?.split(' ')[0]}!
   220→        </h2>
   221→        <p className="text-sm text-gray-400 mt-1">Here's what's happening with your care requests.</p>
   222→      </div>
   223→
   224→      {/* Stat Cards */}
   225→      <div className="grid grid-cols-1 sm:grid-cols-2 lg:grid-cols-4 gap-4">
   226→        {statCards.map((stat, i) => (
   227→          <motion.div
   228→            key={stat.label}
   229→            initial={{ opacity: 0, y: 12 }}
   230→            animate={{ opacity: 1, y: 0 }}
   231→            transition={{ delay: i * 0.05 }}
   232→          >
   233→            <Card className="rounded-2xl border-gray-100 hover:shadow-md transition-shadow">
   234→              <CardContent className="p-5">
   235→                <div className="flex items-center justify-between">
   236→                  <div>
   237→                    <p className="text-sm text-gray-500 font-medium">{stat.label}</p>
   238→                    <p className="text-3xl font-bold text-gray-900 mt-1">{stat.value}</p>
   239→                  </div>
   240→                  <div className={`w-11 h-11 rounded-2xl flex items-center justify-center ${stat.color}`}>
   241→                    {stat.icon}
   242→                  </div>
   243→                </div>
   244→              </CardContent>
   245→            </Card>
   246→          </motion.div>
   247→        ))}
   248→      </div>
   249→
   250→      {/* Quick Actions for new users */}
   251→      {!loading && stats.activeBookings === 0 && stats.patients === 0 && (
   252→        <Card className="rounded-2xl border-dashed border-2 border-forest-200 bg-forest-50/50">
   253→          <CardContent className="p-6 text-center">
   254→            <Heart className="h-8 w-8 text-forest-400 mx-auto mb-3" />
   255→            <h3 className="text-base font-semibold text-gray-800 mb-1">Get started with SevaSaathi</h3>
   256→            <p className="text-sm text-gray-500">Add a member profile from the Members tab, then search and book a caregiver.</p>
   257→          </CardContent>
   258→        </Card>
   259→      )}
   260→
   261→      {/* Recent Bookings */}
   262→      <div>
   263→        <h3 className="text-base font-semibold text-gray-800 mb-3">Recent Bookings</h3>
   264→        {recentBookings.length === 0 ? (
   265→          <EmptyState
   266→            icon={<CalendarCheck className="h-7 w-7" />}
   267→            title="No bookings yet"
   268→            description="Find a caregiver and book your first care session."
   269→          />
   270→        ) : (
   271→          <div className="space-y-3">
   272→            {recentBookings.map((booking: any) => (
   273→              <Card key={booking.id} className="rounded-2xl border-gray-100">
   274→                <CardContent className="p-4">
   275→                  <div className="flex items-center justify-between gap-4">
   276→                    <div className="flex items-center gap-3 min-w-0">
   277→                      <div className="w-10 h-10 rounded-xl bg-forest-50 flex items-center justify-center shrink-0">
   278→                        <UserRound className="h-5 w-5 text-forest-700" />
   279→                      </div>
   280→                      <div className="min-w-0">
   281→                        <p className="text-sm font-semibold text-gray-800 truncate">
   282→                          {booking.patient?.name || 'Member'}
   283→                        </p>
   284→                        <p className="text-xs text-gray-400">
   285→                          {booking.caregiver?.user?.name || 'Caregiver'} &middot; {booking.shiftType?.replace(/_/g, ' ')}
   286→                        </p>
   287→                      </div>
   288→                    </div>
   289→                    <div className="text-right shrink-0">
   290→                      <Badge variant="outline" className={`text-[11px] rounded-full ${statusColors[booking.status] || ''}`}>
   291→                        {booking.status}
   292→                      </Badge>
   293→                      <p className="text-xs text-gray-400 mt-1">
   294→                        {booking.totalAmount ? `₹${booking.totalAmount}` : '--'}
   295→                      </p>
   296→                    </div>
   297→                  </div>
   298→                </CardContent>
   299→              </Card>
   300→            ))}
   301→          </div>
   302→        )}
   303→      </div>
   304→    </div>
   305→  );
   306→}

   311→function PatientsTab() {
   312→  const user = useAuthStore((s) => s.user);
   313→  const [patients, setPatients] = useState<any[]>([]);
   314→  const [loading, setLoading] = useState(true);
   315→  const [error, setError] = useState('');
   316→  const [modalOpen, setModalOpen] = useState(false);
   317→  const [submitting, setSubmitting] = useState(false);
   318→
   319→  // Form state
   320→  const [form, setForm] = useState({
   321→    name: '', age: '', gender: '', relationship: '', address: '', city: '',
   322→    pincode: '', mobilityStatus: '', medicalHistory: '', careRequirements: '',
   323→    emergencyName: '', emergencyPhone: '',
   324→  });
   325→
   326→  const fetchPatients = useCallback(async () => {
   327→    if (!user?.id) return;
   328→    setLoading(true);
   329→    setError('');
   330→    try {
   331→      const res = await api.patients.list(user.id);
   332→      setPatients(res.patients || []);
   333→    } catch (err: any) {
   334→      setError(err.message || 'Failed to load patients');
   335→    } finally {
   336→      setLoading(false);
   337→    }
   338→  }, [user?.id]);
   339→
   340→  useEffect(() => {
   341→    fetchPatients();
   342→  }, [fetchPatients]);
   343→
   344→  const updateForm = (field: string, value: string) => {
   345→    setForm((prev) => ({ ...prev, [field]: value }));
   346→  };
   347→
   348→  const handleSubmit = async (e: React.FormEvent) => {
   349→    e.preventDefault();
   350→    if (!user?.id) return;
   351→    setSubmitting(true);
   352→    try {
   353→      await api.patients.create({
   354→        familyId: user.id,
   355→        name: form.name,
   356→        age: parseInt(form.age, 10) || 0,
   357→        gender: form.gender,
   358→        relationship: form.relationship,
   359→        address: form.address,
   360→        city: form.city,
   361→        pincode: form.pincode || undefined,
   362→        mobilityStatus: form.mobilityStatus || 'mobile',
   363→        medicalHistory: form.medicalHistory,
   364→        careRequirements: form.careRequirements,
   365→        emergencyContact: JSON.stringify({ name: form.emergencyName, phone: form.emergencyPhone }),
   366→      });
   367→      toast.success('Member profile created successfully!');
   368→      setModalOpen(false);
   369→      setForm({
   370→        name: '', age: '', gender: '', relationship: '', address: '', city: '',
   371→        pincode: '', mobilityStatus: '', medicalHistory: '', careRequirements: '',
   372→        emergencyName: '', emergencyPhone: '',
   373→      });
   374→      fetchPatients();
   375→    } catch (err: any) {
   376→      toast.error(err.message || 'Failed to create member');
   377→    } finally {
   378→      setSubmitting(false);
   379→    }
   380→  };
   381→
   382→  if (loading) {
   383→    return <LoadingCards count={3} />;
   384→  }
   385→
   386→  if (error) {
   387→    return <ErrorState message={error} onRetry={fetchPatients} />;
   388→  }
   389→
   390→  const mobilityLabel: Record<string, string> = {
   391→    mobile: 'Mobile', bedridden: 'Bedridden', limited: 'Limited Mobility', wheelchair: 'Wheelchair',
   392→  };
   393→
   394→  return (
   395→    <div className="space-y-6">
   396→      <div className="flex items-center justify-between">
   397→        <div>
   398→          <h2 className="text-xl font-bold text-gray-900">Member Profiles</h2>
   399→          <p className="text-sm text-gray-400">Manage care recipients in your family.</p>
   400→        </div>
   401→        <Dialog open={modalOpen} onOpenChange={setModalOpen}>
   402→          <DialogTrigger asChild>
   403→            <Button className="btn-black text-sm gap-2">
   404→              <Plus className="h-4 w-4" /> Add Member
   405→            </Button>
   406→          </DialogTrigger>
   407→          <DialogContent className="max-w-lg max-h-[90vh] overflow-y-auto rounded-2xl">
   408→            <DialogHeader>
   409→              <DialogTitle>Add New Member</DialogTitle>
   410→            </DialogHeader>
   411→            <form onSubmit={handleSubmit} className="space-y-4 mt-2">
   412→              <div className="grid grid-cols-2 gap-3">
   413→                <div className="col-span-2">
   414→                  <Label className="text-xs font-medium text-gray-600">Full Name *</Label>
   415→                  <Input
   416→                    required
   417→                    value={form.name}
   418→                    onChange={(e) => updateForm('name', e.target.value)}
   419→                    placeholder="Member name"
   420→                    className="mt-1 rounded-xl"
   421→                  />
   422→                </div>
   423→                <div>
   424→                  <Label className="text-xs font-medium text-gray-600">Age *</Label>
   425→                  <Input
   426→                    required
