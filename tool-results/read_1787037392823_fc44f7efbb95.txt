     4→import {

   124→/* ================================================================== */
   125→/*  MAIN COMPONENT                                                     */
   126→/* ================================================================== */
   127→export default function LoginModal({ isOpen, onClose, defaultTab }: LoginModalProps) {
   128→  /* ---- state ---- */
   129→  const [tab, setTab] = useState<Tab>(defaultTab === "register" ? "register" : "login");
   130→  const [showPassword, setShowPassword] = useState(false);
   131→  const [loading, setLoading] = useState(false);
   132→  const [googleLoading, setGoogleLoading] = useState(false);
   133→  const [error, setError] = useState("");
   134→  const [resetOpen, setResetOpen] = useState(false);
   135→
   136→  // Login form
   137→  const [loginEmail, setLoginEmail] = useState("");
   138→  const [loginPassword, setLoginPassword] = useState("");
   139→
   140→  // Register form
   141→  const [regName, setRegName] = useState("");
   142→  const [regEmail, setRegEmail] = useState("");
   143→  const [regPhone, setRegPhone] = useState("");
   144→  const [regPassword, setRegPassword] = useState("");
   145→  const [regRole, setRegRole] = useState<UserRole>("FAMILY");
   146→
   147→  // Reset form (simplified — no OTP)
   148→  const [resetEmail, setResetEmail] = useState("");
   149→  const [resetNewPassword, setResetNewPassword] = useState("");
   150→  const [resetLoading, setResetLoading] = useState(false);
   151→  const [resetError, setResetError] = useState("");
   152→  const [resetDone, setResetDone] = useState(false);
   153→
   154→  const { setAuth } = useAuthStore();
   155→  const { toast } = useToast();
   156→
   157→  // Google popup form
   158→  const [googleOpen, setGoogleOpen] = useState(false);
   159→  const [googleEmail, setGoogleEmail] = useState("");
   160→  const [googleName, setGoogleName] = useState("");
   161→  const [googleError, setGoogleError] = useState("");
   162→  const [googleSubmitting, setGoogleSubmitting] = useState(false);
   163→
   164→  /* ---- reset form fields when switching tabs ---- */
   165→  useEffect(() => {
   166→    setError("");
   167→    setLoginPassword("");
   168→  }, [tab]);
   169→
   170→  /* ---- reset entire state when modal closes ---- */
   171→  const handleClose = useCallback(() => {
   172→    setTab(defaultTab === "register" ? "register" : "login");
   173→    setError("");
   174→    setShowPassword(false);
   175→    setLoading(false);
   176→    setGoogleLoading(false);
   177→    setLoginEmail("");
   178→    setLoginPassword("");
   179→    setRegName("");
   180→    setRegEmail("");
   181→    setRegPhone("");
   182→    setRegPassword("");
   183→    setRegRole("FAMILY");
   184→    setResetOpen(false);
   185→    setResetEmail("");
   186→    setResetNewPassword("");
   187→    setResetLoading(false);
   188→    setResetError("");
   189→    setResetDone(false);
   190→    setGoogleOpen(false);
   191→    setGoogleEmail("");
   192→    setGoogleName("");
   193→    setGoogleError("");
   194→    setGoogleSubmitting(false);
   195→    onClose();
   196→  }, [onClose, defaultTab]);
   197→
   198→  /* ================================================================ */
   199→  /*  HANDLERS                                                         */
   200→  /* ================================================================ */
   201→
   202→  /* ---- Google sign-in — opens simulated Google popup ---- */
   203→  const handleGoogleSignIn = () => {
   204→    setGoogleOpen(true);
   205→    setGoogleError("");
   206→  };
   207→
   208→  /* ---- Google popup submit ---- */
   209→  const handleGoogleSubmit = async (e: React.FormEvent) => {
   210→    e.preventDefault();
   211→    if (!googleEmail.trim() || !googleName.trim()) {
   212→      setGoogleError("Please enter your name and email.");
   213→      return;
   214→    }
   215→    if (!googleEmail.trim().includes("@")) {
   216→      setGoogleError("Please enter a valid email address.");
   217→      return;
   218→    }
   219→    setGoogleSubmitting(true);
   220→    setGoogleError("");
   221→    try {
   222→      // 1. Call the simulate endpoint (sets the session cookie)
   223→      const res = await fetch("/api/auth/google-simulate", {
   224→        method: "POST",
   225→        headers: { "Content-Type": "application/json" },
   226→        body: JSON.stringify({ email: googleEmail.trim(), name: googleName.trim() }),
   227→      });
   228→      const data = await res.json();
   229→      if (!res.ok) {
   230→        setGoogleError(data.error || "Google sign-in failed.");
   231→        return;
   232→      }
   233→
   234→      // 2. Fetch the NextAuth session (cookie is already set by step 1)
   235→      const sessionRes = await fetch("/api/auth/session");
   236→      const sessionData = await sessionRes.json();
   237→
   238→      if (sessionData?.user) {
   239→        // 3. Directly update the Zustand store — no page reload needed
   240→        setAuth({
   241→          id: sessionData.user.id,
   242→          email: sessionData.user.email,
   243→          name: sessionData.user.name,
   244→          phone: "",
   245→          role: sessionData.user.role,
   246→          avatarUrl: sessionData.user.image || null,
   247→          subscription: "NONE" as any,
   248→          patientProfiles: [],
   249→          caregiverProfile: null,
   250→        });
   251→        setGoogleOpen(false);
   252→        toast({
   253→          title: "Welcome" + (data.user.isNewUser ? " to SevaSaathi!" : " back!") + " \ud83c\udf89",
   254→          description: "Signed in as " + data.user.email,
   255→        });
   256→        handleClose();
   257→      } else {
   258→        // Fallback: reload the page so NextAuth picks up the cookie
   259→        window.location.href = window.location.origin;
   260→      }
   261→    } catch {
   262→      setGoogleError("Something went wrong. Please try again.");
   263→    } finally {
   264→      setGoogleSubmitting(false);
   265→    }
   266→  };
   267→
   268→  /* ---- Email / password login ---- */
   269→  const handleLogin = async (e: React.FormEvent) => {
   270→    e.preventDefault();
   271→    if (!loginEmail.trim() || !loginPassword.trim()) {
   272→      setError("Please enter your email and password.");
   273→      return;
   274→    }
   275→    setLoading(true);
   276→    setError("");
   277→    try {
   278→      const result = await signIn("credentials", {
   279→        email: loginEmail.trim(),
   280→        password: loginPassword,
   281→        redirect: false,
   282→      });
   283→
   284→      if (result?.error) {
   285→        setError("Invalid email or password. Please try again.");
   286→        return;
   287→      }
   288→
   289→      // AuthSync in AuthProvider will automatically sync session → Zustand store
   290→      toast({
   291→        title: "Welcome back! 🎉",
   292→        description: "You have successfully logged in.",
   293→      });
   294→      handleClose();
   295→    } catch {
   296→      setError("Something went wrong. Please try again.");
   297→    } finally {
   298→      setLoading(false);
   299→    }
   300→  };
   301→
   302→  /* ---- Registration ---- */
   303→  const handleRegister = async (e: React.FormEvent) => {
   304→    e.preventDefault();
   305→    if (!regName.trim() || !regEmail.trim() || !regPhone.trim() || !regPassword.trim()) {
   306→      setError("Please fill in all fields.");
   307→      return;
   308→    }
   309→    if (regPassword.length < 6) {
   310→      setError("Password must be at least 6 characters.");
   311→      return;
   312→    }
   313→    setLoading(true);
   314→    setError("");
   315→    try {
   316→      // 1. Register
   317→      const regRes = await api.auth.register({
   318→        name: regName.trim(),
   319→        email: regEmail.trim(),
   320→        phone: regPhone.trim(),
   321→        password: regPassword,
   322→        role: regRole,
   323→      });
   324→
   325→      // 2. Auto-login with credentials
   326→      const loginResult = await signIn("credentials", {
   327→        email: regEmail.trim(),
   328→        password: regPassword,
   329→        redirect: false,
   330→      });
   331→
   332→      if (loginResult?.error) {
   333→        // Registration succeeded but auto-login failed — that's ok, just notify
   334→        toast({ title: "Account created!", description: "Please log in with your credentials." });
   335→        setTab("login");
   336→        setLoginEmail(regEmail.trim());
   337→        return;
   338→      }
   339→
   340→      toast({ title: "Welcome to SevaSaathi! 🎉", description: "Your account has been created." });
   341→      handleClose();
   342→    } catch (err: any) {
   343→      setError(err?.message || "Registration failed. Please try again.");
   344→    } finally {
   345→      setLoading(false);
   346→    }
   347→  };
   348→
   349→  /* ---- Reset password (simplified) ---- */
   350→  const handleReset = async (e: React.FormEvent) => {
   351→    e.preventDefault();
   352→    if (!resetEmail.trim() || !resetNewPassword.trim()) {
   353→      setResetError("Please enter your email and new password.");
   354→      return;
   355→    }
   356→    if (resetNewPassword.length < 6) {
   357→      setResetError("Password must be at least 6 characters.");
   358→      return;
   359→    }
   360→    setResetLoading(true);
   361→    setResetError("");
   362→    try {
   363→      const res = await fetch("/api/auth/reset-password", {
   364→        method: "POST",
   365→        headers: { "Content-Type": "application/json" },
   366→        body: JSON.stringify({ email: resetEmail.trim(), newPassword: resetNewPassword }),
   367→      });
   368→      const data = await res.json();
   369→      if (!res.ok) {
   370→        setResetError(data.error || "Failed to reset password.");
   371→        return;
   372→      }
   373→      setResetDone(true);
   374→      toast({ title: "Password reset successfully!", description: "You can now log in with your new password." });
   375→    } catch {
   376→      setResetError("Something went wrong. Please try again.");
   377→    } finally {
   378→      setResetLoading(false);
   379→    }
   380→  };
   381→
   382→  /* ================================================================ */
   383→  /*  RENDER                                                           */
   384→  /* ================================================================ */
   385→  return (
   386→    <Dialog open={isOpen} onOpenChange={(open) => !open && handleClose()}>
   387→      <DialogContent className="sm:max-w-md p-0 overflow-hidden border-forest-200/40">
   388→        {/* Custom X button to override default position */}
   389→        <button
   390→          onClick={handleClose}
   391→          className="absolute top-3 right-3 z-50 rounded-full p-1.5 text-gray-400 hover:text-gray-600 hover:bg-gray-100 transition-colors"
   392→          aria-label="Close"
   393→        >
   394→          <X className="w-5 h-5" />
   395→        </button>
   396→
   397→        <div className="px-6 pt-6 pb-2">
   398→          <DialogHeader className="space-y-1 mb-1">
   399→            <div className="flex items-center gap-2 justify-center">
   400→              <div className="w-9 h-9 rounded-xl bg-forest-900 flex items-center justify-center">
   401→                <Heart className="w-5 h-5 text-lime-400" />
   402→              </div>
   403→              <DialogTitle className="text-xl font-bold text-forest-900">
   404→                SevaSaathi
   405→              </DialogTitle>
   406→            </div>
   407→            <DialogDescription className="text-sm text-gray-500 mt-1">
   408→              {tab === "login"
   409→                ? "Sign in to your account"
   410→                : "Create your account to get started"}
   411→            </DialogDescription>
   412→          </DialogHeader>
   413→        </div>
   414→
   415→        {/* ---- Error banner ---- */}
   416→        <AnimatePresence>
   417→          {error && (
   418→            <motion.div
   419→              initial={{ height: 0, opacity: 0 }}
   420→              animate={{ height: "auto", opacity: 1 }}
   421→              exit={{ height: 0, opacity: 0 }}
   422→              className="overflow-hidden"
   423→            >
   424→              <div className="mx-6 mt-1 px-3 py-2 rounded-lg bg-red-50 border border-red-200 text-red-700 text-sm flex items-center gap-2">
   425→                <X className="w-4 h-4 flex-shrink-0" />
   426→                {error}
   427→              </div>
   428→            </motion.div>
   429→          )}
   430→        </AnimatePresence>
   431→
   432→        {/* ---- Tab switcher ---- */}
   433→        <div className="px-6 pt-3">
   434→          <div className="flex bg-gray-100 rounded-lg p-1">
   435→            <button
   436→              onClick={() => setTab("login")}
   437→              className={`flex-1 py-2 px-4 rounded-md text-sm font-medium transition-all duration-200 ${
   438→                tab === "login"
   439→                  ? "bg-white text-forest-900 shadow-sm"
   440→                  : "text-gray-500 hover:text-gray-700"
   441→              }`}
   442→            >
   443→              Sign In
   444→            </button>
   445→            <button
   446→              onClick={() => setTab("register")}
   447→              className={`flex-1 py-2 px-4 rounded-md text-sm font-medium transition-all duration-200 ${
   448→                tab === "register"
   449→                  ? "bg-white text-forest-900 shadow-sm"
   450→                  : "text-gray-500 hover:text-gray-700"
   451→              }`}
   452→            >
   453→              Create Account
   454→            </button>
   455→          </div>
   456→        </div>
   457→
   458→        {/* ---- Animated content ---- */}
   459→        <div className="px-6 py-4 overflow-hidden">
   460→          <AnimatePresence mode="wait" custom={tab}>
   461→            {/* ========== LOGIN TAB ========== */}
   462→            {tab === "login" && (
   463→              <motion.div
   464→                key="login"
   465→                custom={tab}
   466→                variants={contentVariants}
   467→                initial="enter"
   468→                animate="center"
   469→                exit="exit"
   470→                className="space-y-4"
   471→              >
   472→                {/* Google button */}
   473→                <button
   474→                  type="button"
   475→                  onClick={handleGoogleSignIn}
   476→                  disabled={googleLoading}
   477→                  className="w-full flex items-center justify-center gap-3 h-11 px-4 rounded-lg border border-gray-300 bg-white hover:bg-gray-50 active:bg-gray-100 transition-all duration-150 text-sm font-medium text-gray-700 disabled:opacity-60 disabled:cursor-not-allowed shadow-sm"
   478→                >
   479→                  {googleLoading ? (
   480→                    <Loader2 className="w-5 h-5 animate-spin text-gray-500" />
   481→                  ) : (
   482→                    <GoogleIcon />
   483→                  )}
   484→                  Continue with Google
   485→                </button>
   486→
   487→                {/* Divider */}
   488→                <div className="flex items-center gap-3">
   489→                  <div className="flex-1 h-px bg-gray-200" />
   490→                  <span className="text-xs text-gray-400 font-medium uppercase tracking-wider">
   491→                    or continue with email
   492→                  </span>
   493→                  <div className="flex-1 h-px bg-gray-200" />
   494→                </div>
   495→
   496→                {/* Email / password form */}
   497→                <form onSubmit={handleLogin} className="space-y-3">
   498→                  <div className="space-y-1.5">
   499→                    <Label htmlFor="login-email" className="text-sm font-medium text-gray-700">
   500→                      Email
   501→                    </Label>
   502→                    <div className="relative">
   503→                      <Mail className="absolute left-3 top-1/2 -translate-y-1/2 w-4 h-4 text-gray-400" />
   504→                      <Input
   505→                        id="login-email"
   506→                        type="email"
   507→                        placeholder="you@example.com"
   508→                        value={loginEmail}
   509→                        onChange={(e) => setLoginEmail(e.target.value)}
   510→                        className="pl-10 h-11 border-gray-300 focus:border-forest-500 focus:ring-forest-500"
   511→                        autoComplete="email"
   512→                        disabled={loading}
   513→                      />
   514→                    </div>
   515→                  </div>
   516→
   517→                  <div className="space-y-1.5">
   518→                    <div className="flex items-center justify-between">
   519→                      <Label htmlFor="login-password" className="text-sm font-medium text-gray-700">
   520→                        Password
   521→                      </Label>
   522→                      <button
   523→                        type="button"
   524→                        onClick={() => setResetOpen(true)}
   525→                        className="text-xs font-medium text-forest-700 hover:text-forest-900 transition-colors"
   526→                      >
   527→                        Forgot password?
   528→                      </button>
   529→                    </div>
   530→                    <div className="relative">
   531→                      <Lock className="absolute left-3 top-1/2 -translate-y-1/2 w-4 h-4 text-gray-400" />
   532→                      <Input
   533→                        id="login-password"
   534→                        type={showPassword ? "text" : "password"}
   535→                        placeholder="Enter your password"
   536→                        value={loginPassword}
   537→                        onChange={(e) => setLoginPassword(e.target.value)}
   538→                        className="pl-10 pr-10 h-11 border-gray-300 focus:border-forest-500 focus:ring-forest-500"
   539→                        autoComplete="current-password"
   540→                        disabled={loading}
   541→                      />
   542→                      <button
   543→                        type="button"
   544→                        onClick={() => setShowPassword(!showPassword)}
   545→                        className="absolute right-3 top-1/2 -translate-y-1/2 text-gray-400 hover:text-gray-600 transition-colors"
   546→                        tabIndex={-1}
   547→                      >
   548→                        {showPassword ? <EyeOff className="w-4 h-4" /> : <Eye className="w-4 h-4" />}
   549→                      </button>
   550→                    </div>
   551→                  </div>
   552→
   553→                  <Button
   554→                    type="submit"
   555→                    disabled={loading}
   556→                    className="w-full h-11 bg-forest-900 hover:bg-forest-800 text-white font-semibold rounded-lg transition-all duration-150 shadow-sm hover:shadow-md"
   557→                  >
   558→                    {loading ? (
   559→                      <>
   560→                        <Loader2 className="w-4 h-4 mr-2 animate-spin" />
   561→                        Signing in...
   562→                      </>
   563→                    ) : (
   564→                      <>
   565→                        Sign In
   566→                        <ArrowRight className="w-4 h-4 ml-2" />
   567→                      </>
   568→                    )}
   569→                  </Button>
   570→                </form>
   571→
   572→                {/* Switch to register */}
   573→                <p className="text-center text-sm text-gray-500 pt-1">
   574→                  Don&apos;t have an account?{" "}
   575→                  <button
   576→                    onClick={() => setTab("register")}
   577→                    className="font-semibold text-forest-700 hover:text-forest-900 transition-colors"
   578→                  >
   579→                    Create one
   580→                  </button>
   581→                </p>
   582→              </motion.div>
   583→            )}
   584→
   585→            {/* ========== REGISTER TAB ========== */}
   586→            {tab === "register" && (
   587→              <motion.div
   588→                key="register"
   589→                custom={tab}
   590→                variants={contentVariants}
   591→                initial="enter"
   592→                animate="center"
   593→                exit="exit"
   594→                className="space-y-4"
   595→              >
   596→                {/* Google button */}
   597→                <button
