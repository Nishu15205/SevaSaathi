   124→/* ================================================================== */
   125→/*  MAIN COMPONENT                                                     */
   126→/* ================================================================== */
   127→export default function LoginModal({ isOpen, onClose, defaultTab }: LoginModalProps) {
   128→  /* ---- state ---- */
   129→  const [tab, setTab] = useState<Tab>(defaultTab === "register" ? "register" : "login");
   130→  const [showPassword, setShowPassword] = useState(false);
   131→  const [loading, setLoading] = useState(false);
   132→  const [googleLoading, setGoogleLoading] = useState(false);
   133→  const [error, setError] = useState("");
   134→  const [resetOpen, setResetOpen] = useState(false);
   135→
   136→  // Login form
   137→  const [loginEmail, setLoginEmail] = useState("");
   138→  const [loginPassword, setLoginPassword] = useState("");
   139→
   140→  // Register form
   141→  const [regName, setRegName] = useState("");
   142→  const [regEmail, setRegEmail] = useState("");
   143→  const [regPhone, setRegPhone] = useState("");
   144→  const [regPassword, setRegPassword] = useState("");
   145→  const [regRole, setRegRole] = useState<UserRole>("FAMILY");
   146→
   147→  // Reset form (simplified — no OTP)
   148→  const [resetEmail, setResetEmail] = useState("");
   149→  const [resetNewPassword, setResetNewPassword] = useState("");
   150→  const [resetLoading, setResetLoading] = useState(false);
   151→  const [resetError, setResetError] = useState("");
   152→  const [resetDone, setResetDone] = useState(false);
   153→
   154→  const { setAuth } = useAuthStore();
   155→  const { toast } = useToast();
   156→
   157→  /* ---- reset form fields when switching tabs ---- */
   158→  useEffect(() => {
   159→    setError("");
   160→    setLoginPassword("");
   161→  }, [tab]);
   162→
   163→  /* ---- reset entire state when modal closes ---- */
   164→  const handleClose = useCallback(() => {
   165→    setTab(defaultTab === "register" ? "register" : "login");
   166→    setError("");
   167→    setShowPassword(false);
   168→    setLoading(false);
   169→    setGoogleLoading(false);
   170→    setLoginEmail("");
   171→    setLoginPassword("");
   172→    setRegName("");
   173→    setRegEmail("");
   174→    setRegPhone("");
   175→    setRegPassword("");
   176→    setRegRole("FAMILY");
   177→    setResetOpen(false);
   178→    setResetEmail("");
   179→    setResetNewPassword("");
   180→    setResetLoading(false);
   181→    setResetError("");
   182→    setResetDone(false);
   183→    onClose();
   184→  }, [onClose, defaultTab]);
   185→
   186→  /* ================================================================ */
   187→  /*  HANDLERS                                                         */
   188→  /* ================================================================ */
   189→
   190→  /* ---- Google sign-in ---- */
   191→  const handleGoogleSignIn = async () => {
   192→    setGoogleLoading(true);
   193→    // Google OAuth requires a production domain with registered API credentials.
   194→    // In this environment the callback URL (localhost) is not accessible from your browser,
   195→    // so the OAuth flow cannot complete. Show a helpful message instead.
   196→    setTimeout(() => {
   197→      toast({
   198→        title: "Google Sign-In — Coming Soon",
   199→        description: "Please use email/password to login. Google login will work when deployed to production with a registered domain.",
   200→        variant: "destructive",
   201→      });
   202→      setGoogleLoading(false);
   203→    }, 600);
   204→  };
   205→
   206→  /* ---- Email / password login ---- */
   207→  const handleLogin = async (e: React.FormEvent) => {
   208→    e.preventDefault();
   209→    if (!loginEmail.trim() || !loginPassword.trim()) {
   210→      setError("Please enter your email and password.");
   211→      return;
   212→    }
   213→    setLoading(true);
   214→    setError("");
   215→    try {
   216→      const result = await signIn("credentials", {
   217→        email: loginEmail.trim(),
   218→        password: loginPassword,
   219→        redirect: false,
   220→      });
   221→
   222→      if (result?.error) {
   223→        setError("Invalid email or password. Please try again.");
   224→        return;
   225→      }
   226→
   227→      // AuthSync in AuthProvider will automatically sync session → Zustand store
   228→      toast({
   229→        title: "Welcome back! 🎉",
   230→        description: "You have successfully logged in.",
   231→      });
   232→      handleClose();
   233→    } catch {
   234→      setError("Something went wrong. Please try again.");
   235→    } finally {
   236→      setLoading(false);
   237→    }
   238→  };
   239→
   240→  /* ---- Registration ---- */
   241→  const handleRegister = async (e: React.FormEvent) => {
   242→    e.preventDefault();
   243→    if (!regName.trim() || !regEmail.trim() || !regPhone.trim() || !regPassword.trim()) {
   244→      setError("Please fill in all fields.");
   245→      return;
   246→    }
   247→    if (regPassword.length < 6) {
   248→      setError("Password must be at least 6 characters.");
   249→      return;
   250→    }
   251→    setLoading(true);
   252→    setError("");
   253→    try {
   254→      // 1. Register
   255→      const regRes = await api.auth.register({
   256→        name: regName.trim(),
   257→        email: regEmail.trim(),
   258→        phone: regPhone.trim(),
   259→        password: regPassword,
   260→        role: regRole,
   261→      });
   262→
   263→      // 2. Auto-login with credentials
   264→      const loginResult = await signIn("credentials", {
   265→        email: regEmail.trim(),
   266→        password: regPassword,
   267→        redirect: false,
   268→      });
   269→
   270→      if (loginResult?.error) {
   271→        // Registration succeeded but auto-login failed — that's ok, just notify
   272→        toast({ title: "Account created!", description: "Please log in with your credentials." });
   273→        setTab("login");
   274→        setLoginEmail(regEmail.trim());
   275→        return;
   276→      }
   277→
   278→      toast({ title: "Welcome to SevaSaathi! 🎉", description: "Your account has been created." });
   279→      handleClose();
   280→    } catch (err: any) {
   281→      setError(err?.message || "Registration failed. Please try again.");
   282→    } finally {
   283→      setLoading(false);
   284→    }
   285→  };
   286→
   287→  /* ---- Reset password (simplified) ---- */
   288→  const handleReset = async (e: React.FormEvent) => {
   289→    e.preventDefault();
   290→    if (!resetEmail.trim() || !resetNewPassword.trim()) {
   291→      setResetError("Please enter your email and new password.");
   292→      return;
   293→    }
   294→    if (resetNewPassword.length < 6) {
   295→      setResetError("Password must be at least 6 characters.");
   296→      return;
   297→    }
   298→    setResetLoading(true);
   299→    setResetError("");
   300→    try {
   301→      const res = await fetch("/api/auth/reset-password", {
   302→        method: "POST",
   303→        headers: { "Content-Type": "application/json" },
   304→        body: JSON.stringify({ email: resetEmail.trim(), newPassword: resetNewPassword }),
   305→      });
   306→      const data = await res.json();
   307→      if (!res.ok) {
   308→        setResetError(data.error || "Failed to reset password.");
   309→        return;
   310→      }
   311→      setResetDone(true);
   312→      toast({ title: "Password reset successfully!", description: "You can now log in with your new password." });
   313→    } catch {
   314→      setResetError("Something went wrong. Please try again.");
   315→    } finally {
   316→      setResetLoading(false);
   317→    }
   318→  };
   319→
   320→  /* ================================================================ */
   321→  /*  RENDER                                                           */
   322→  /* ================================================================ */
   323→  return (
   324→    <Dialog open={isOpen} onOpenChange={(open) => !open && handleClose()}>
   325→      <DialogContent className="sm:max-w-md p-0 overflow-hidden border-forest-200/40">
   326→        {/* Custom X button to override default position */}
   327→        <button
   328→          onClick={handleClose}
   329→          className="absolute top-3 right-3 z-50 rounded-full p-1.5 text-gray-400 hover:text-gray-600 hover:bg-gray-100 transition-colors"
   330→          aria-label="Close"
   331→        >
   332→          <X className="w-5 h-5" />
   333→        </button>
   334→
   335→        <div className="px-6 pt-6 pb-2">
   336→          <DialogHeader className="space-y-1 mb-1">
   337→            <div className="flex items-center gap-2 justify-center">
   338→              <div className="w-9 h-9 rounded-xl bg-forest-900 flex items-center justify-center">
   339→                <Heart className="w-5 h-5 text-lime-400" />
   340→              </div>
   341→              <DialogTitle className="text-xl font-bold text-forest-900">
   342→                SevaSaathi
   343→              </DialogTitle>
   344→            </div>
   345→            <DialogDescription className="text-sm text-gray-500 mt-1">
   346→              {tab === "login"
   347→                ? "Sign in to your account"
   348→                : "Create your account to get started"}
   349→            </DialogDescription>
   350→          </DialogHeader>
   351→        </div>
   352→
   353→        {/* ---- Error banner ---- */}
   354→        <AnimatePresence>
   355→          {error && (
   356→            <motion.div
   357→              initial={{ height: 0, opacity: 0 }}
   358→              animate={{ height: "auto", opacity: 1 }}
   359→              exit={{ height: 0, opacity: 0 }}
   360→              className="overflow-hidden"
   361→            >
   362→              <div className="mx-6 mt-1 px-3 py-2 rounded-lg bg-red-50 border border-red-200 text-red-700 text-sm flex items-center gap-2">
   363→                <X className="w-4 h-4 flex-shrink-0" />
   364→                {error}
   365→              </div>
   366→            </motion.div>
   367→          )}
   368→        </AnimatePresence>
   369→
   370→        {/* ---- Tab switcher ---- */}
   371→        <div className="px-6 pt-3">
   372→          <div className="flex bg-gray-100 rounded-lg p-1">
   373→            <button
   374→              onClick={() => setTab("login")}
   375→              className={`flex-1 py-2 px-4 rounded-md text-sm font-medium transition-all duration-200 ${
   376→                tab === "login"
   377→                  ? "bg-white text-forest-900 shadow-sm"
   378→                  : "text-gray-500 hover:text-gray-700"
   379→              }`}
   380→            >
   381→              Sign In
   382→            </button>
   383→            <button
   384→              onClick={() => setTab("register")}
   385→              className={`flex-1 py-2 px-4 rounded-md text-sm font-medium transition-all duration-200 ${
   386→                tab === "register"
   387→                  ? "bg-white text-forest-900 shadow-sm"
   388→                  : "text-gray-500 hover:text-gray-700"
   389→              }`}
   390→            >
   391→              Create Account
   392→            </button>
   393→          </div>
   394→        </div>
   395→
   396→        {/* ---- Animated content ---- */}
   397→        <div className="px-6 py-4 overflow-hidden">
   398→          <AnimatePresence mode="wait" custom={tab}>
   399→            {/* ========== LOGIN TAB ========== */}
   400→            {tab === "login" && (
   401→              <motion.div
   402→                key="login"
   403→                custom={tab}
   404→                variants={contentVariants}
   405→                initial="enter"
   406→                animate="center"
   407→                exit="exit"
   408→                className="space-y-4"
   409→              >
   410→                {/* Google button */}
   411→                <div className="relative">
   412→                  <button
   413→                    type="button"
   414→                    onClick={handleGoogleSignIn}
   415→                    disabled={googleLoading}
   416→                    className="w-full flex items-center justify-center gap-3 h-11 px-4 rounded-lg border border-gray-300 bg-white hover:bg-gray-50 active:bg-gray-100 transition-all duration-150 text-sm font-medium text-gray-700 disabled:opacity-60 disabled:cursor-not-allowed shadow-sm"
   417→                  >
   418→                    {googleLoading ? (
   419→                      <Loader2 className="w-5 h-5 animate-spin text-gray-500" />
   420→                    ) : (
   421→                      <GoogleIcon />
   422→                    )}
   423→                    Continue with Google
   424→                    <span className="text-[10px] font-semibold bg-amber-100 text-amber-700 px-1.5 py-0.5 rounded-full leading-none">Coming Soon</span>
   425→                  </button>
   426→                </div>
   427→
   428→                {/* Divider */}
   429→                <div className="flex items-center gap-3">
   430→                  <div className="flex-1 h-px bg-gray-200" />
   431→                  <span className="text-xs text-gray-400 font-medium uppercase tracking-wider">
   432→                    or continue with email
   433→                  </span>
   434→                  <div className="flex-1 h-px bg-gray-200" />
   435→                </div>
   436→
   437→                {/* Email / password form */}
   438→                <form onSubmit={handleLogin} className="space-y-3">
   439→                  <div className="space-y-1.5">
   440→                    <Label htmlFor="login-email" className="text-sm font-medium text-gray-700">
   441→                      Email
   442→                    </Label>
   443→                    <div className="relative">
   444→                      <Mail className="absolute left-3 top-1/2 -translate-y-1/2 w-4 h-4 text-gray-400" />
   445→                      <Input
   446→                        id="login-email"
   447→                        type="email"
   448→                        placeholder="you@example.com"
   449→                        value={loginEmail}
   450→                        onChange={(e) => setLoginEmail(e.target.value)}
   451→                        className="pl-10 h-11 border-gray-300 focus:border-forest-500 focus:ring-forest-500"
   452→                        autoComplete="email"
   453→                        disabled={loading}
   454→                      />
   455→                    </div>
   456→                  </div>
   457→
   458→                  <div className="space-y-1.5">
   459→                    <div className="flex items-center justify-between">
   460→                      <Label htmlFor="login-password" className="text-sm font-medium text-gray-700">
   461→                        Password
   462→                      </Label>
   463→                      <button
   464→                        type="button"
   465→                        onClick={() => setResetOpen(true)}
   466→                        className="text-xs font-medium text-forest-700 hover:text-forest-900 transition-colors"
   467→                      >
   468→                        Forgot password?
   469→                      </button>
   470→                    </div>
   471→                    <div className="relative">
   472→                      <Lock className="absolute left-3 top-1/2 -translate-y-1/2 w-4 h-4 text-gray-400" />
   473→                      <Input
   474→                        id="login-password"
   475→                        type={showPassword ? "text" : "password"}
   476→                        placeholder="Enter your password"
   477→                        value={loginPassword}
   478→                        onChange={(e) => setLoginPassword(e.target.value)}
   479→                        className="pl-10 pr-10 h-11 border-gray-300 focus:border-forest-500 focus:ring-forest-500"
   480→                        autoComplete="current-password"
   481→                        disabled={loading}
   482→                      />
   483→                      <button
   484→                        type="button"
   485→                        onClick={() => setShowPassword(!showPassword)}
   486→                        className="absolute right-3 top-1/2 -translate-y-1/2 text-gray-400 hover:text-gray-600 transition-colors"
   487→                        tabIndex={-1}
   488→                      >
   489→                        {showPassword ? <EyeOff className="w-4 h-4" /> : <Eye className="w-4 h-4" />}
   490→                      </button>
   491→                    </div>
   492→                  </div>
   493→
   494→                  <Button
   495→                    type="submit"
   496→                    disabled={loading}
   497→                    className="w-full h-11 bg-forest-900 hover:bg-forest-800 text-white font-semibold rounded-lg transition-all duration-150 shadow-sm hover:shadow-md"
   498→                  >
   499→                    {loading ? (
   500→                      <>
   501→                        <Loader2 className="w-4 h-4 mr-2 animate-spin" />
   502→                        Signing in...
   503→                      </>
   504→                    ) : (
   505→                      <>
   506→                        Sign In
   507→                        <ArrowRight className="w-4 h-4 ml-2" />
   508→                      </>
   509→                    )}
   510→                  </Button>
   511→                </form>
   512→
   513→                {/* Switch to register */}
   514→                <p className="text-center text-sm text-gray-500 pt-1">
   515→                  Don&apos;t have an account?{" "}
   516→                  <button
   517→                    onClick={() => setTab("register")}
   518→                    className="font-semibold text-forest-700 hover:text-forest-900 transition-colors"
   519→                  >
   520→                    Create one
   521→                  </button>
   522→                </p>
   523→              </motion.div>
   524→            )}
   525→
   526→            {/* ========== REGISTER TAB ========== */}
   527→            {tab === "register" && (
   528→              <motion.div
   529→                key="register"
   530→                custom={tab}
   531→                variants={contentVariants}
   532→                initial="enter"
   533→                animate="center"
   534→                exit="exit"
   535→                className="space-y-4"
   536→              >
   537→                {/* Google button */}
   538→                <div className="relative">
   539→                  <button
   540→                    type="button"
   541→                    onClick={handleGoogleSignIn}
   542→                    disabled={googleLoading}
   543→                    className="w-full flex items-center justify-center gap-3 h-11 px-4 rounded-lg border border-gray-300 bg-white hover:bg-gray-50 active:bg-gray-100 transition-all duration-150 text-sm font-medium text-gray-700 disabled:opacity-60 disabled:cursor-not-allowed shadow-sm"
   544→                  >
   545→                    {googleLoading ? (
   546→                      <Loader2 className="w-5 h-5 animate-spin text-gray-500" />
   547→                    ) : (
   548→                      <GoogleIcon />
   549→                    )}
   550→                    Sign up with Google
   551→                    <span className="text-[10px] font-semibold bg-amber-100 text-amber-700 px-1.5 py-0.5 rounded-full leading-none">Coming Soon</span>
   552→                  </button>
   553→                </div>
   554→
   555→                {/* Divider */}
   556→                <div className="flex items-center gap-3">
   557→                  <div className="flex-1 h-px bg-gray-200" />
   558→                  <span className="text-xs text-gray-400 font-medium uppercase tracking-wider">
   559→                    or register with email
   560→                  </span>
   561→                  <div className="flex-1 h-px bg-gray-200" />
   562→                </div>
   563→
   564→                {/* Registration form */}
   565→                <form onSubmit={handleRegister} className="space-y-3">
   566→                  {/* Name */}
   567→                  <div className="space-y-1.5">
   568→                    <Label htmlFor="reg-name" className="text-sm font-medium text-gray-700">
   569→                      Full Name
   570→                    </Label>
   571→                    <div className="relative">
   572→                      <User className="absolute left-3 top-1/2 -translate-y-1/2 w-4 h-4 text-gray-400" />
   573→                      <Input
   574→                        id="reg-name"
   575→                        type="text"
   576→                        placeholder="Your full name"
   577→                        value={regName}
   578→                        onChange={(e) => setRegName(e.target.value)}
   579→                        className="pl-10 h-11 border-gray-300 focus:border-forest-500 focus:ring-forest-500"
   580→                        autoComplete="name"
   581→                        disabled={loading}
   582→                      />
   583→                    </div>
   584→                  </div>
   585→
