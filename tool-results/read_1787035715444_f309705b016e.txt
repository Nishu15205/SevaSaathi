    84→function useDebounce(value: string, delay: number) {
    85→  const [debounced, setDebounced] = useState(value);
    86→  useEffect(() => {
    87→    const timer = setTimeout(() => setDebounced(value), delay);
    88→    return () => clearTimeout(timer);
    89→  }, [value, delay]);
    90→  return debounced;
    91→}

   229→/* ─────────── USERS TAB ─────────── */
   230→function UsersTab({ onViewUser }: { onViewUser: (userId: string, userName: string) => void }) {
   231→  const [role, setRole] = useState<string>('');
   232→  const [search, setSearch] = useState('');
   233→  const [users, setUsers] = useState<any[]>([]);
   234→  const [loading, setLoading] = useState(true);
   235→  const [page, setPage] = useState(1);
   236→  const [totalPages, setTotalPages] = useState(1);
   237→  const [deleting, setDeleting] = useState<string | null>(null);
   238→
   239→  const debouncedSearch = useDebounce(search, 300);
   240→
   241→  useEffect(() => {
   242→    setPage(1);
   243→  }, [debouncedSearch, role]);
   244→
   245→  const fetchUsers = useCallback(async () => {
   246→    setLoading(true);
   247→    try {
   248→      const res = await api.admin.users({ role: role || undefined, page, limit: 20, search: debouncedSearch || undefined });
   249→      setUsers(res.users ?? []);
   250→      setTotalPages(res.pagination?.totalPages ?? 1);
   251→    } catch {
   252→      toast.error('Failed to load users');
   253→    } finally {
   254→      setLoading(false);
   255→    }
   256→  }, [role, page, debouncedSearch]);
   257→
   258→  useEffect(() => { fetchUsers(); }, [fetchUsers]);
   259→
   260→  const handleDelete = async (userId: string, userName: string) => {
   261→    if (!window.confirm(`Are you sure you want to delete user "${userName}"? This action cannot be undone.`)) return;
   262→    setDeleting(userId);
   263→    try {
   264→      await api.admin.deleteUser(userId);
   265→      toast.success(`User "${userName}" deleted successfully`);
   266→      fetchUsers();
   267→    } catch {
   268→      toast.error('Failed to delete user');
   269→    } finally {
   270→      setDeleting(null);
   271→    }
   272→  };
   273→
   274→  const roleFilters = [
   275→    { label: 'All', value: '' },
   276→    { label: 'Family', value: 'FAMILY' },
   277→    { label: 'Caregiver', value: 'CAREGIVER' },
   278→  ];
   279→
   280→  return (
   281→    <div className="space-y-4">
   282→      <div className="flex flex-col sm:flex-row gap-3 sm:items-center sm:justify-between">
   283→        <SearchBar value={search} onChange={setSearch} placeholder="Search users by name, email, phone..." />
   284→        <div className="flex gap-2">
   285→          {roleFilters.map((f) => (
   286→            <Button
   287→              key={f.value}
   288→              variant={role === f.value ? 'default' : 'outline'}
   289→              size="sm"
   290→              className={role === f.value ? 'bg-[#14532d] hover:bg-[#14532d]/90 text-white rounded-full' : 'rounded-full'}
   291→              onClick={() => { setRole(f.value); setPage(1); }}
   292→            >
   293→              {f.label}
   294→            </Button>
   295→          ))}
   296→        </div>
   297→      </div>
   298→
   299→      {loading ? (
   300→        <div className="space-y-3">{Array.from({ length: 5 }).map((_, i) => <Skeleton key={i} className="h-20 rounded-2xl" />)}</div>
   301→      ) : users.length === 0 ? (
   302→        <Card className="rounded-2xl border-0 shadow-sm"><CardContent className="py-12 text-center text-muted-foreground">No users found.</CardContent></Card>
   303→      ) : (
   304→        <div className="space-y-3">
   305→          {users.map((u) => (
   306→            <motion.div key={u.id} variants={fadeVariants} initial="hidden" animate="visible">
   307→              <Card className="rounded-2xl card-hover border-0 shadow-sm">
   308→                <CardContent className="p-4 flex flex-col sm:flex-row sm:items-center gap-3">
   309→                  <div className="flex-1 min-w-0">
   310→                    <div className="flex items-center gap-2 flex-wrap">
   311→                      <UserNameButton id={u.id} name={u.name} onClick={onViewUser} />
   312→                      <Badge className={statusColors[u.role] ?? 'bg-gray-100 text-gray-700'}>{u.role}</Badge>
   313→                      {u.role === 'CAREGIVER' && u.isVerified && <BadgeCheck className="w-4 h-4 text-[#14532d]" />}
   314→                    </div>
   315→                    <div className="flex items-center gap-4 mt-1 text-xs text-muted-foreground flex-wrap">
   316→                      <span>{u.email}</span>
   317→                      <span>{u.phone}</span>
   318→                      <span>Joined {u.createdAt ? new Date(u.createdAt).toLocaleDateString('en-IN', { day: 'numeric', month: 'short', year: 'numeric' }) : '—'}</span>
   319→                    </div>
   320→                    {u.role === 'CAREGIVER' && (
   321→                      <div className="flex items-center gap-3 mt-1 text-xs text-muted-foreground">
   322→                        {u.city && <span className="flex items-center gap-1"><MapPin className="w-3 h-3" />{u.city}</span>}
   323→                        {u.rating != null && <span className="flex items-center gap-1"><Star className="w-3 h-3 text-amber-500" />{Number(u.rating).toFixed(1)}</span>}
   324→                      </div>
   325→                    )}
   326→                    {u.role === 'FAMILY' && u.patientCount != null && (
   327→                      <div className="mt-1 text-xs text-muted-foreground"><Users className="w-3 h-3 inline mr-1" />{u.patientCount} patient{u.patientCount !== 1 ? 's' : ''}</div>
   328→                    )}
   329→                  </div>
   330→                  <Button
   331→                    size="sm"
   332→                    variant="outline"
   333→                    className="text-red-500 hover:text-red-600 hover:bg-red-50 border-red-200 shrink-0"
   334→                    disabled={deleting === u.id}
   335→                    onClick={() => handleDelete(u.id, u.name)}
   336→                  >
   337→                    {deleting === u.id ? <Loader2 className="w-4 h-4 animate-spin" /> : <Trash2 className="w-4 h-4" />}
   338→                  </Button>
   339→                </CardContent>
   340→              </Card>
   341→            </motion.div>
   342→          ))}
   343→        </div>
   344→      )}
   345→
   346→      {totalPages > 1 && (
   347→        <div className="flex justify-center gap-2 pt-2">
   348→          <Button variant="outline" size="sm" className="rounded-full" disabled={page <= 1} onClick={() => setPage(page - 1)}>Previous</Button>
   349→          <span className="flex items-center text-sm text-muted-foreground px-3">Page {page} of {totalPages}</span>
   350→          <Button variant="outline" size="sm" className="rounded-full" disabled={page >= totalPages} onClick={() => setPage(page + 1)}>Next</Button>
   351→        </div>
   352→      )}
   353→    </div>
   354→  );
   355→}

   357→/* ─────────── VERIFICATIONS TAB ─────────── */
   358→function VerificationsTab({ onViewUser }: { onViewUser: (userId: string, userName: string) => void }) {
   359→  const [search, setSearch] = useState('');
   360→  const [verifications, setVerifications] = useState<any[]>([]);
   361→  const [loading, setLoading] = useState(true);
   362→  const [rejecting, setRejecting] = useState<string | null>(null);
   363→  const [approving, setApproving] = useState<string | null>(null);
   364→  const [reasons, setReasons] = useState<Record<string, string>>({});
   365→
   366→  const debouncedSearch = useDebounce(search, 300);
   367→
   368→  const fetchVerifications = useCallback(async () => {
   369→    setLoading(true);
   370→    try {
   371→      const res = await api.admin.verifications.list({ search: debouncedSearch || undefined });
   372→      setVerifications(res.verifications ?? []);
   373→    } catch {
   374→      toast.error('Failed to load verifications');
   375→    } finally {
   376→      setLoading(false);
   377→    }
   378→  }, [debouncedSearch]);
   379→
   380→  useEffect(() => { fetchVerifications(); }, [fetchVerifications]);
   381→
   382→  const handleApprove = async (id: string) => {
   383→    setApproving(id);
   384→    try {
   385→      await api.admin.verifications.update(id, { status: 'APPROVED' });
   386→      toast.success('Verification approved successfully');
   387→      fetchVerifications();
   388→    } catch {
   389→      toast.error('Failed to approve verification');
   390→    } finally {
   391→      setApproving(null);
   392→    }
   393→  };
   394→
   395→  const handleReject = async (id: string) => {
   396→    const reason = reasons[id]?.trim();
   397→    if (!reason) { toast.error('Please provide a rejection reason'); return; }
   398→    setRejecting(id);
   399→    try {
   400→      await api.admin.verifications.update(id, { status: 'REJECTED', rejectionReason: reason });
   401→      toast.success('Verification rejected');
   402→      setReasons((prev) => { const n = { ...prev }; delete n[id]; return n; });
   403→      fetchVerifications();
   404→    } catch {
   405→      toast.error('Failed to reject verification');
   406→    } finally {
   407→      setRejecting(null);
   408→    }
   409→  };
   410→
   411→  if (loading) {
   412→    return <div className="space-y-3">{Array.from({ length: 3 }).map((_, i) => <Skeleton key={i} className="h-24 rounded-2xl" />)}</div>;
   413→  }
   414→
   415→  if (verifications.length === 0) {
   416→    return (
   417→      <div className="space-y-4">
   418→        <SearchBar value={search} onChange={setSearch} placeholder="Search verifications..." />
   419→        <Card className="rounded-2xl border-0 shadow-sm"><CardContent className="py-16 text-center"><ShieldCheck className="w-12 h-12 mx-auto text-[#14532d]/30 mb-3" /><p className="text-muted-foreground">No pending verifications.</p></CardContent></Card>
   420→      </div>
   421→    );
   422→  }
   423→
   424→  return (
   425→    <div className="space-y-4">
   426→      <SearchBar value={search} onChange={setSearch} placeholder="Search verifications..." />
   427→      {verifications.map((v) => (
   428→        <motion.div key={v.id} variants={fadeVariants} initial="hidden" animate="visible">
   429→          <Card className="rounded-2xl card-hover border-0 shadow-sm">
   430→            <CardContent className="p-4 space-y-3">
   431→              <div className="flex flex-col sm:flex-row sm:items-center justify-between gap-2">
   432→                <div className="flex items-center gap-2 flex-wrap">
   433→                  {v.caregiver?.user?.id ? (
   434→                    <UserNameButton id={v.caregiver.user.id} name={v.caregiver?.user?.name ?? 'Caregiver'} onClick={onViewUser} />
   435→                  ) : (
   436→                    <span className="font-semibold text-sm">{v.caregiver?.user?.name ?? 'Caregiver'}</span>
   437→                  )}
   438→                  <Badge className={docTypeBadge[v.docType] ?? 'bg-gray-100 text-gray-700'}>{v.docType ?? 'DOC'}</Badge>
   439→                </div>
   440→                <span className="text-xs text-muted-foreground flex items-center gap-1">
   441→                  <Clock className="w-3 h-3" />{v.createdAt ? new Date(v.createdAt).toLocaleDateString('en-IN', { day: 'numeric', month: 'short', year: 'numeric' }) : '—'}
   442→                </span>
   443→              </div>
   444→              {v.docNumber && <p className="text-xs text-muted-foreground font-mono bg-muted/50 rounded-lg px-3 py-1.5 w-fit">{v.docNumber}</p>}
   445→              <div className="flex items-center gap-2 pt-1">
   446→                <Button
   447→                  size="sm"
   448→                  className="bg-[#14532d] hover:bg-[#14532d]/90 text-white rounded-full"
   449→                  disabled={approving === v.id}
   450→                  onClick={() => handleApprove(v.id)}
   451→                >
   452→                  {approving === v.id ? <Loader2 className="w-4 h-4 animate-spin" /> : <><CheckCircle2 className="w-4 h-4 mr-1" />Approve</>}
   453→                </Button>
   454→                <Button
   455→                  size="sm"

   890→        </div>
