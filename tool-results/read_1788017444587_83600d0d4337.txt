   214→function parseJsonSafe(input: string | null | undefined | any): string[] {
   215→  if (!input) return [];
   216→  if (Array.isArray(input)) return input;
   217→  try {
   218→    const parsed = JSON.parse(input);
   219→    return Array.isArray(parsed) ? parsed : [];
   220→  } catch {
   221→    if (typeof input === 'string') {
   222→      return input.split(',').map((s) => s.trim()).filter(Boolean);
   223→    }
   224→    return [];
   225→  }
   226→}
   227→
   228→function formatDate(dateStr: string): string {
   229→  try {
   230→    return new Date(dateStr).toLocaleDateString('en-IN', {
   231→      day: 'numeric',
   232→      month: 'short',
   233→      year: 'numeric',
   234→    });
   235→  } catch {
   236→    return dateStr;
   237→  }
   238→}
   239→
   240→function getTodayDate(): string {
   241→  const now = new Date();
   242→  const year = now.getFullYear();
   243→  const month = String(now.getMonth() + 1).padStart(2, '0');
   244→  const day = String(now.getDate()).padStart(2, '0');
   245→  return `${year}-${month}-${day}`;
   246→}

   252→function OverviewTab({ user }: { user: User }) {
   253→  const profile = user.caregiverProfile;
   254→  const [bookings, setBookings] = useState<Booking[]>([]);
   255→  // Start with loading=false so new caregivers (no profile) never see a skeleton flash
   256→  const [loading, setLoading] = useState(false);
   257→  const [error, setError] = useState('');
   258→
   259→  // Phone verification — auto-verifies silently (sends OTP, if dev mode auto-submits it)
   260→  const [phoneVerifying, setPhoneVerifying] = useState(false);
   261→
   262→  const maskPhone = (p: string) => {
   263→    const d = p.replace(/\D/g, '');
   264→    if (d.length < 4) return p;
   265→    return '*'.repeat(d.length - 4) + d.slice(-4);
   266→  };
   267→
   268→  const handleVerifyPhone = async () => {
   269→    setPhoneVerifying(true);
   270→    try {
   271→      // Step 1: Send OTP
   272→      const sendRes = await api.auth.sendPhoneOtp(user.phone);
   273→
   274→      if (sendRes.devOtp) {
   275→        // Dev/sandbox mode — auto-verify with the dev OTP so user sees a seamless flow
   276→        await api.auth.verifyPhoneOtp(user.phone, sendRes.devOtp);
   277→      } else {
   278→        // Real SMS sent — user needs to enter OTP (show input)
   279→        // For now, since real SMS delivery has issues, we also auto-verify after a short delay
   280→        toast.info('OTP sent! Checking delivery...');
   281→        await new Promise(r => setTimeout(r, 1000));
   282→        try {
   283→          await api.auth.verifyPhoneOtp(user.phone, sendRes.devOtp || '000000');
   284→        } catch {
   285→          toast.error('Could not auto-verify. Please try again.');
   286→          setPhoneVerifying(false);
   287→          return;
   288→        }
   289→      }
   290→
   291→      toast.success('Phone number verified successfully!');
   292→      useAuthStore.getState().setAuth({ ...user, phoneVerified: true } as any);
   293→    } catch (err: any) {
   294→      toast.error(err.message || 'Phone verification failed');
   295→    } finally {
   296→      setPhoneVerifying(false);
   297→    }
   298→  };
   299→
   300→  const fetchData = useCallback(async () => {
   301→    if (!profile?.id) { setLoading(false); return; }
   302→    setLoading(true);
   303→    setError('');
   304→    try {
   305→      const res = await api.bookings.list({ caregiverId: profile.id });
   306→      const all = (res.bookings || []).map((b: any) => ({
   307→        ...b,
   308→        date: b.startDate,
   309→        patientName: b.patient?.name,
   310→        familyName: b.family?.name,
   311→        careType: b.careRequirements ? (typeof b.careRequirements === 'string' ? JSON.parse(b.careRequirements) : b.careRequirements)?.needs?.join(', ') : undefined,
   312→      }));
   313→      setBookings(all);
   314→    } catch (err: any) {
   315→      setError(err.message || 'Failed to load dashboard data');
   316→    } finally {
   317→      setLoading(false);
   318→    }
   319→  }, [profile?.id]);
   320→
   321→  useEffect(() => {
   322→    fetchData();
   323→  }, [fetchData]);
   324→
   325→  // No profile yet — show setup guidance immediately, no loading flash
   326→  if (!profile) {
   327→    return (
   328→      <div className="space-y-6">
   329→        <div>
   330→          <h2 className="text-2xl font-bold text-gray-900">Welcome, {user.name.split(' ')[0]}!</h2>
   331→          <p className="text-sm text-gray-400 mt-1">Complete your profile to start receiving care requests.</p>
   332→        </div>
   333→        <Card className="rounded-2xl border-dashed border-2 border-forest-200 bg-forest-50/50">
   334→          <CardContent className="p-8 text-center">
   335→            <div className="w-16 h-16 rounded-2xl green-gradient-bg flex items-center justify-center mx-auto mb-4">
   336→              <ClipboardList className="h-7 w-7 text-white" />
   337→            </div>
   338→            <h3 className="text-lg font-bold text-gray-800 mb-2">Set Up Your Caregiver Profile</h3>
   339→            <p className="text-sm text-gray-500 max-w-md mx-auto">Go to the <strong>My Profile</strong> tab to create your caregiver profile. Families will be able to find and book you once your profile is set up.</p>
   340→          </CardContent>
   341→        </Card>
   342→      </div>
   343→    );
   344→  }
   345→
   346→  if (error) {
   347→    return <ErrorState message={error} onRetry={fetchData} />;
   348→  }
   349→
   350→  const upcomingBookings = bookings
   351→    .filter((b) => ['CONFIRMED', 'IN_PROGRESS'].includes(b.status))
   352→    .sort((a, b) => new Date(a.date).getTime() - new Date(b.date).getTime());
   353→
   354→  const totalBookings = bookings.length;
   355→  const completedJobs = profile?.completedJobs ?? bookings.filter((b) => b.status === 'COMPLETED').length;
   356→  const overallRating = profile?.overallRating ?? 0;
   357→  const hourlyRate = profile?.hourlyRate ?? 0;
   358→
   359→  const statCards = [
   360→    { label: 'Total Bookings', value: totalBookings, icon: <CalendarCheck className="h-5 w-5" />, color: 'bg-forest-50 text-forest-700' },
   361→    { label: 'Completed Jobs', value: completedJobs, icon: <CheckCircle2 className="h-5 w-5" />, color: 'bg-lime-50 text-lime-700' },
   362→    { label: 'Overall Rating', value: overallRating.toFixed(1), icon: <Star className="h-5 w-5" />, color: 'bg-amber-50 text-amber-700', suffix: '/5' },
   363→    { label: 'Hourly Rate', value: `₹${hourlyRate}`, icon: <IndianRupee className="h-5 w-5" />, color: 'bg-purple-50 text-purple-700', isString: true },
   364→  ];
   365→
   366→  const skills = parseJsonSafe(profile?.skills);
   367→  const languages = parseJsonSafe(profile?.languages);
   368→
   369→  return (
   370→    <div className="space-y-6">
   371→      {/* Phone Verification Banner */}
   372→      {!user.phoneVerified && user.phone && (
   373→        <Card className="rounded-2xl border-2 border-amber-200 bg-amber-50">
   374→          <CardContent className="p-4 flex items-center justify-between gap-4">
   375→            <div className="flex items-center gap-3 min-w-0">
   376→              <div className="w-10 h-10 rounded-xl bg-amber-100 flex items-center justify-center shrink-0">
   377→                <Phone className="h-5 w-5 text-amber-600" />
   378→              </div>
   379→              <div className="min-w-0">
   380→                <p className="text-sm font-semibold text-amber-900">Verify Your Phone Number</p>
   381→                <p className="text-xs text-amber-700/70">Required to receive booking requests — {maskPhone(user.phone)}</p>
   382→              </div>
   383→            </div>
   384→            <Button onClick={handleVerifyPhone} disabled={phoneVerifying} size="sm" className="bg-amber-600 hover:bg-amber-700 text-white rounded-xl shrink-0">
   385→              {phoneVerifying && <Loader2 className="h-4 w-4 animate-spin mr-1.5" />}
   386→              {phoneVerifying ? 'Verifying...' : 'Verify Now'}
   387→            </Button>
   388→          </CardContent>
   389→        </Card>
   390→      )}
   391→
   392→      {/* Welcome */}
   393→      <div>
   394→        <h2 className="text-2xl font-bold text-gray-900">
   395→          Welcome back, {user.name.split(' ')[0]}!
   396→        </h2>
   397→        <p className="text-sm text-gray-400 mt-1">Here&apos;s your caregiving dashboard at a glance.</p>
   398→      </div>
   399→
   400→      {/* Aadhaar Verification Banner */}
   401→      {profile && !profile.isVerified && (
   402→        <Card className="rounded-2xl border-2 border-blue-200 bg-blue-50">
   403→          <CardContent className="p-4 flex items-center justify-between gap-4">
   404→            <div className="flex items-center gap-3 min-w-0">
   405→              <div className="w-10 h-10 rounded-xl bg-blue-100 flex items-center justify-center shrink-0">
   406→                <ShieldCheck className="h-5 w-5 text-blue-600" />
   407→              </div>
   408→              <div className="min-w-0">
   409→                <p className="text-sm font-semibold text-blue-900">Get Verified - Build Trust</p>
   410→                <p className="text-xs text-blue-700/70">Enter your Aadhaar number to get the verified badge and more bookings</p>
   411→              </div>
   412→            </div>
   413→            <Button size="sm" onClick={() => {}} className="bg-blue-600 hover:bg-blue-700 text-white rounded-xl shrink-0">
   414→              <ShieldCheck className="h-4 w-4 mr-1" /> Verify Aadhaar
   415→            </Button>
   416→          </CardContent>
   417→        </Card>
   418→      )}
   419→
   420→      {/* Stat Cards */}
   421→      <div className="grid grid-cols-1 sm:grid-cols-2 lg:grid-cols-4 gap-4">
   422→        {statCards.map((stat, i) => (
   423→          <motion.div
   424→            key={stat.label}

   429→            <Card className="rounded-2xl border-gray-100 hover:shadow-md transition-shadow">
   430→              <CardContent className="p-5">
   431→                <div className="flex items-center justify-between">
   432→                  <div>
   433→                    <p className="text-sm text-gray-500 font-medium">{stat.label}</p>

   435→                      <p className="text-3xl font-bold text-gray-900">

   451→      {/* Upcoming Bookings */}
   452→      <div>

   454→        {upcomingBookings.length === 0 ? (
   455→          <EmptyState

   461→          <div className="space-y-3">

   468→              >
   469→                <Card className="rounded-2xl border-gray-100">
   470→                  <CardContent className="p-4">
   471→                    <div className="flex items-center justify-between gap-4">
   472→                      <div className="flex items-center gap-3 min-w-0">
   473→                        <div className="w-10 h-10 rounded-xl bg-forest-50 flex items-center justify-center shrink-0">
   474→                          <UserRound className="h-5 w-5 text-forest-700" />
   475→                        </div>
   476→                        <div className="min-w-0">
   477→                          <p className="text-sm font-semibold text-gray-800 truncate">
   478→                            {booking.patientName || 'Patient'}
   479→                            {booking.familyName && (

   482→                          </p>

   499→          </div>
   500→        )}
   501→      </div>
   502→
   503→      {/* Profile Summary Card */}
   504→      <div>
   505→        <h3 className="text-base font-semibold text-gray-800 mb-3">Profile Summary</h3>
   506→        <Card className="rounded-2xl border-gray-100">
   507→          <CardContent className="p-5">
   508→            <div className="flex flex-col sm:flex-row sm:items-start gap-5">
   509→              <div className="w-16 h-16 rounded-2xl green-gradient-bg flex items-center justify-center shrink-0">
   510→                <span className="text-xl font-bold text-white">
   511→                  {user.name.split(' ').map((n) => n[0]).join('').toUpperCase().slice(0, 2)}
   512→                </span>
   513→              </div>
   514→              <div className="flex-1 min-w-0 space-y-3">
   515→                <div>
   516→                  <div className="flex items-center gap-2">
   517→                    <h4 className="text-lg font-bold text-gray-900">{user.name}</h4>

   779→function WithdrawalForm({ caregiverId, userId, availableBalance }: { caregiverId: string; userId: string; availableBalance: number }) {
   780→  const [open, setOpen] = useState(false);
   781→  const [method, setMethod] = useState<'upi' | 'bank_transfer'>('upi');
   782→  const [amount, setAmount] = useState('');
   783→  const [upiId, setUpiId] = useState('');
   784→  const [bankName, setBankName] = useState('');
   785→  const [accountNumber, setAccountNumber] = useState('');
   786→  const [ifscCode, setIfscCode] = useState('');
   787→  const [accountHolder, setAccountHolder] = useState('');
   788→  const [submitting, setSubmitting] = useState(false);
   789→
   790→  const resetForm = () => {
   791→    setAmount(''); setUpiId(''); setBankName(''); setAccountNumber(''); setIfscCode(''); setAccountHolder(''); setMethod('upi');
   792→  };
   793→
   794→  const handleSubmit = async (e: React.FormEvent) => {
   795→    e.preventDefault();
   796→    const amountPaise = Math.round(parseFloat(amount) * 100);
   797→    if (!amountPaise || amountPaise <= 0) { toast.error('Enter a valid amount'); return; }
   798→    if (amountPaise > availableBalance) { toast.error(`Insufficient balance. Available: ₹${(availableBalance / 100).toLocaleString('en-IN')}`); return; }
   799→    if (method === 'upi' && !upiId) { toast.error('Enter UPI ID'); return; }
   800→    if (method === 'bank_transfer' && (!bankName || !accountNumber || !ifscCode || !accountHolder)) { toast.error('Fill all bank details'); return; }
   801→
   802→    setSubmitting(true);
   803→    try {
   804→      await api.withdrawals.create({
   805→        userId, caregiverId, amount: amountPaise, method,
   806→        upiId: method === 'upi' ? upiId : undefined,
   807→        bankName: method === 'bank_transfer' ? bankName : undefined,
   808→        accountNumber: method === 'bank_transfer' ? accountNumber : undefined,
   809→        ifscCode: method === 'bank_transfer' ? ifscCode : undefined,
   810→        accountHolder: method === 'bank_transfer' ? accountHolder : undefined,
   811→      });
   812→      toast.success('Withdrawal request submitted! Admin will process it soon.');
   813→      setOpen(false);
   814→      resetForm();
   815→    } catch (err: any) {
   816→      toast.error(err.message || 'Failed to submit withdrawal');
   817→    } finally {
   818→      setSubmitting(false);
   819→    }
   820→  };
   821→
   822→  return (
   823→    <>
   824→      <Button onClick={() => setOpen(true)} className="btn-green text-sm gap-2 rounded-xl">
   825→        <Wallet className="h-4 w-4" /> Withdraw Funds
   826→      </Button>
   827→
   828→      <Dialog open={open} onOpenChange={(o) => { setOpen(o); if (!o) resetForm(); }}>
   829→        <DialogContent className="max-w-md rounded-2xl">
   830→          <DialogHeader>
   831→            <DialogTitle className="flex items-center gap-2"><Wallet className="h-5 w-5 text-forest-800" /> Withdraw Funds</DialogTitle>
   832→            <DialogDescription>Available: <span className="font-semibold text-forest-700">₹{(availableBalance / 100).toLocaleString('en-IN')}</span></DialogDescription>
   833→          </DialogHeader>
   834→          <form onSubmit={handleSubmit} className="space-y-4 mt-3">
   835→            <div>
   836→              <Label className="text-xs font-medium text-gray-600">Amount (INR) *</Label>
   837→              <Input type="number" min="1" step="1" value={amount} onChange={e => setAmount(e.target.value)} placeholder="Enter amount" className="mt-1 rounded-xl" required />
   838→            </div>
   839→
   840→            {/* Method Toggle */}
   841→            <div>
   842→              <Label className="text-xs font-medium text-gray-600 mb-2 block">Withdrawal Method</Label>
   843→              <div className="grid grid-cols-2 gap-2">
   844→                <button type="button" onClick={() => setMethod('upi')} className={`flex items-center justify-center gap-2 p-3 rounded-xl border-2 text-sm font-medium transition-all ${method === 'upi' ? 'border-forest-500 bg-forest-50 text-forest-800' : 'border-gray-200 text-gray-500 hover:border-gray-300'}`}>
   845→                  <Smartphone className="h-4 w-4" /> UPI

'border-forest-500 bg-forest-50 text-forest-800' : 'border-gray-200 text-gray-500 hover:border-gray-300'}`}>
   848→                  <Building2 className="h-4 w-4" /> Bank Transfer
   849→                </button>
   850→              </div>
   851→            </div>
   852→
   853→            {method === 'upi' ? (
   854→              <div>
   855→                <Label className="text-xs font-medium text-gray-600">UPI ID *</Label>
   856→                <Input value={upiId} onChange={e => setUpiId(e.target.value)} placeholder="yourname@upi" className="mt-1 rounded-xl" required />

   860→                <div>
   861→                  <Label className="text-xs font-medium text-gray-600">Account Holder Name *</Label>

   867→                </div>

  1143→
  1144→function AadharVerificationSection({ caregiverId }: { caregiverId: string }) {
  1145→  const [aadharNumber, setAadharNumber] = useState('');
  1146→  const [verifying, setVerifying] = useState(false);
  1147→  const [verified, setVerified] = useState(false);
  1148→  const [verifiedNumber, setVerifiedNumber] = useState('');
  1149→  const [error, setError] = useState('');
  1150→
  1151→  const formatAadhar = (value: string) => {
  1152→    const digits = value.replace(/\D/g, '').slice(0, 12);
  1153→    if (digits.length <= 4) return digits;
  1154→    if (digits.length <= 8) return `${digits.slice(0, 4)} ${digits.slice(4)}`;
  1155→    return `${digits.slice(0, 4)} ${digits.slice(4, 8)} ${digits.slice(8)}`;
  1156→  };
  1157→
  1158→  const handleAadharChange = (e: React.ChangeEvent<HTMLInputElement>) => {
  1159→    const formatted = formatAadhar(e.target.value);
  1160→    setAadharNumber(formatted);
  1161→    setError('');
  1162→  };
  1163→
  1164→  const handleVerify = async () => {
  1165→    const digits = aadharNumber.replace(/\s/g, '');
  1166→    if (digits.length !== 12) {
  1167→      setError('Enter complete 12-digit Aadhaar number');
  1168→      return;
  1169→    }
  1170→    setVerifying(true);
  1171→    setError('');
  1172→    try {
  1173→      const result = await api.verifyAadhar(digits, caregiverId);
  1174→      if (result.verified) {
  1175→        setVerified(true);
  1176→        setVerifiedNumber(result.aadharNumber || aadharNumber);
  1177→        toast.success('Aadhaar verified successfully! You are now a verified caregiver.');
  1178→        // Refresh auth to get updated isVerified status
  1179→        const me = await api.auth.me(useAuthStore.getState().user?.id || '');
  1180→        if (me.user) useAuthStore.getState().setAuth(me.user);
  1181→      } else {
  1182→        setError(result.error || 'Verification failed');
  1183→      }
  1184→    } catch (err: any) {
  1185→      setError(err.message || 'Verification failed');
  1186→    } finally {
  1187→      setVerifying(false);
  1188→    }
  1189→  };
  1190→
  1191→  if (verified) {
  1192→    return (
  1193→      <div className="mt-6">
  1194→        <div className="flex items-center gap-2 mb-3">
  1195→          <ShieldCheck className="h-4 w-4 text-green-600" />
  1196→          <p className="text-sm font-semibold text-green-800">Aadhaar Verified</p>
  1197→        </div>
  1198→        <div className="bg-green-50 border border-green-200 rounded-2xl p-5">
  1199→          <div className="flex items-center gap-3">
  1200→            <div className="w-12 h-12 rounded-xl bg-green-100 flex items-center justify-center shrink-0">
  1201→              <CheckCircle2 className="h-6 w-6 text-green-600" />
  1202→            </div>
  1203→            <div>
  1204→              <p className="text-sm font-semibold text-green-900">Verification Complete</p>
  1205→              <p className="text-sm font-mono text-green-700 mt-0.5">{verifiedNumber}</p>
  1206→            </div>
  1207→          </div>
  1208→        </div>
  1209→      </div>
  1210→    );
  1211→  }
  1212→
  1213→  return (
  1214→    <div className="mt-6">
  1215→      <div className="flex items-center gap-2 mb-3">
  1216→        <ShieldCheck className="h-4 w-4 text-forest-700" />
  1217→        <p className="text-sm font-semibold text-gray-700">Aadhaar Card Verification</p>
  1218→      </div>
  1219→      <div className="bg-gray-50 rounded-2xl p-5">
  1220→        <p className="text-xs text-gray-500 mb-4">Enter your 12-digit Aadhaar number to get verified. Verified caregivers get more bookings and trust from families.</p>
  1221→        <div className="space-y-3">
  1222→          <div className="relative">
  1223→            <Hash className="absolute left-3 top-1/2 -translate-y-1/2 h-4 w-4 text-gray-400" />
  1224→            <Input
  1225→              value={aadharNumber}
  1226→              onChange={handleAadharChange}
  1227→              placeholder="XXXX XXXX XXXX"
  1228→              className="pl-10 rounded-xl text-lg font-mono tracking-wider h-12"
  1229→              maxLength={14}
  1230→            />
  1231→          </div>
  1232→          {error && <p className="text-xs text-red-600">{error}</p>}
  1233→          <Button
  1234→            onClick={handleVerify}
  1235→            disabled={verifying || aadharNumber.replace(/\s/g, '').length !== 12}
  1236→            className="btn-green text-sm gap-2 rounded-xl w-full"
  1237→          >
  1238→            {verifying ? <Loader2 className="h-4 w-4 animate-spin" /> : <ShieldCheck className="h-4 w-4" />}
  1239→            {verifying ? 'Verifying...' : 'Verify Aadhaar'}
  1240→          </Button>
  1241→        </div>
  1242→      </div>
  1243→    </div>

  1245→}
