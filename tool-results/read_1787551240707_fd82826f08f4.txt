     4→import {

    10→  Lock,

    23→import { api } from "@/lib/api";

    28→  DialogTitle,
    29→  DialogDescription,
    30→} from "@/components/ui/dialog";

    69→const modalVariants = {

    76→  center: { x: 0, opacity: 1, transition: { type: "spring", stiffness: 300, damping: 28 } },
    77→  exit: (tab: string) => ({ x: tab === "register" ? -40 : 40, opacity: 0, transition: { duration: 0.15 } }),

    80→/* ================================================================== */
    81→/*  MAIN COMPONENT                                                     */
    82→/* ================================================================== */
    83→export default function LoginModal({ isOpen, onClose, defaultTab }: LoginModalProps) {
    84→  /* ---- state ---- */
    85→  const [tab, setTab] = useState<Tab>(defaultTab === "register" ? "register" : "login");
    86→  const [showPassword, setShowPassword] = useState(false);
    87→  const [loading, setLoading] = useState(false);
    88→  const [error, setError] = useState("");
    89→  const [resetOpen, setResetOpen] = useState(false);
    90→
    91→  // Login form
    92→  const [loginEmail, setLoginEmail] = useState("");
    93→  const [loginPassword, setLoginPassword] = useState("");
    94→
    95→  // Register form
    96→  const [regName, setRegName] = useState("");
    97→  const [regEmail, setRegEmail] = useState("");
    98→  const [regPhone, setRegPhone] = useState("");
    99→  const [regPassword, setRegPassword] = useState("");
   100→  const [regRole, setRegRole] = useState<UserRole>("FAMILY");
   101→
   102→  // Reset form
   103→  const [resetEmail, setResetEmail] = useState("");
   104→  const [resetNewPassword, setResetNewPassword] = useState("");
   105→  const [resetLoading, setResetLoading] = useState(false);
   106→  const [resetError, setResetError] = useState("");
   107→  const [resetDone, setResetDone] = useState(false);
   108→
   109→  const { setAuth } = useAuthStore();
   110→  const { toast } = useToast();
   111→
   112→  // Google popup form
   113→  const [googleOpen, setGoogleOpen] = useState(false);
   114→  const [googleEmail, setGoogleEmail] = useState("");
   115→  const [googleName, setGoogleName] = useState("");
   116→  const [googleError, setGoogleError] = useState("");
   117→  // googleSubmitting removed - redirect-based flow doesn't need it
   118→
   119→  /* ---- reset form fields when switching tabs ---- */
   120→  useEffect(() => { setError(""); setLoginPassword(""); }, [tab]);
   121→
   122→  /* ---- reset entire state when modal closes ---- */
   123→  const handleClose = useCallback(() => {
   124→    setTab(defaultTab === "register" ? "register" : "login");
   125→    setError(""); setShowPassword(false); setLoading(false);
   126→    setLoginEmail(""); setLoginPassword("");
   127→    setRegName(""); setRegEmail(""); setRegPhone(""); setRegPassword(""); setRegRole("FAMILY");
   128→    setResetOpen(false); setResetEmail(""); setResetNewPassword("");
   129→    setResetLoading(false); setResetError(""); setResetDone(false);
   130→    setGoogleOpen(false); setGoogleEmail(""); setGoogleName(""); setGoogleError(""); setGoogleSubmitting(false);
   131→    onClose();
   132→  }, [onClose, defaultTab]);
   133→
   134→  /* ================================================================ */
   135→  /*  HANDLERS                                                         */
   136→/* ================================================================ */
   137→
   138→  const handleGoogleSignIn = async () => {
   139→    setGoogleError("");
   140→    try {
   141→      const checkRes = await fetch("/api/auth/google-configured");
   142→      const { configured } = await checkRes.json();
   143→      if (configured) {
   144→        const origin = window.location.origin;
   145→        const role = tab === "register" ? regRole : "";
   146→        window.location.href = `/api/auth/google-go?origin=${encodeURIComponent(origin)}&role=${encodeURIComponent(role)}`;
   147→        return;
   148→      }
   149→    } catch { /* fall through */ }
   150→    setGoogleOpen(true);
   151→  };
   152→
   153→  const handleGoogleSubmit = (e: React.FormEvent) => {
   154→    e.preventDefault();
   155→    if (!googleEmail.trim() || !googleName.trim()) { setGoogleError("Please enter your name and email."); return; }
   156→    if (!googleEmail.trim().includes("@")) { setGoogleError("Please enter a valid email address."); return; }
   157→    setGoogleError("");
   158→    // Use redirect-based flow so cookies survive the proxy chain
   159→    const role = tab === "register" ? regRole : "";
   160→    const params = new URLSearchParams({
   161→      email: googleEmail.trim(),
   162→      name: googleName.trim(),
   163→    });
   164→    if (role) params.set("role", role);
   165→    window.location.href = `/api/auth/google-simulate?${params.toString()}`;
   166→  };
   167→
   168→  const handleLogin = async (e: React.FormEvent) => {
   169→    e.preventDefault();
   170→    if (!loginEmail.trim() || !loginPassword.trim()) { setError("Please enter your email and password."); return; }
   171→    setLoading(true); setError("");
   172→    try {
   173→      const result = await signIn("credentials", { email: loginEmail.trim(), password: loginPassword, redirect: false });
   174→      if (result?.error) { setError("Invalid email or password. Please try again."); return; }
   175→      try {
   176→        const sessionRes = await fetch("/api/auth/session");
   177→        const sessionData = await sessionRes.json();
   178→        if (sessionData?.user) {
   179→          const u = sessionData.user as any;
   180→          setAuth({ id: u.id || "", email: u.email || "", name: u.name || "", phone: "", role: u.role || "FAMILY", avatarUrl: u.image || null, subscription: "NONE", patientProfiles: [], caregiverProfile: null });
   181→        }
   182→      } catch { /* AuthSync will pick it up */ }
   183→      toast({ title: "Welcome back! \ud83c\udf89", description: "You have successfully logged in." });
   184→      handleClose();
   185→    } catch { setError("Something went wrong. Please try again."); } finally { setLoading(false); }
   186→  };
   187→
   188→  const handleRegister = async (e: React.FormEvent) => {
   189→    e.preventDefault();
   190→    if (!regName.trim() || !regEmail.trim() || !regPhone.trim() || !regPassword.trim()) { setError("Please fill in all fields."); return; }
   191→    if (regPassword.length < 6) { setError("Password must be at least 6 characters."); return; }
   192→    setLoading(true); setError("");
   193→    try {
   194→      await api.auth.register({ name: regName.trim(), email: regEmail.trim(), phone: regPhone.trim(), password: regPassword, role: regRole });
   195→      const loginResult = await signIn("credentials", { email: regEmail.trim(), password: regPassword, redirect: false });
   196→      if (loginResult?.error) {
   197→        toast({ title: "Account created!", description: "Please log in with your credentials." });
   198→        setTab("login"); setLoginEmail(regEmail.trim()); return;
   199→      }
   200→      try {
   201→        const sessionRes = await fetch("/api/auth/session");
   202→        const sessionData = await sessionRes.json();
   203→        if (sessionData?.user) {
   204→          const u = sessionData.user as any;
   205→          setAuth({ id: u.id || "", email: u.email || "", name: u.name || "", phone: regPhone.trim(), role: u.role || regRole, avatarUrl: u.image || null, subscription: "NONE", patientProfiles: [], caregiverProfile: null });
   206→        }
   207→      } catch { /* AuthSync will pick it up */ }
   208→      toast({ title: "Welcome to SevaSaathi! \ud83c\udf89", description: "Your account has been created." });
   209→      handleClose();
   210→    } catch (err: any) { setError(err?.message || "Registration failed. Please try again."); } finally { setLoading(false); }
   211→  };
   212→
   213→  const handleReset = async (e: React.FormEvent) => {
   214→    e.preventDefault();
   215→    if (!resetEmail.trim() || !resetNewPassword.trim()) { setResetError("Please enter your email and new password."); return; }
   216→    if (resetNewPassword.length < 6) { setResetError("Password must be at least 6 characters."); return; }
   217→    setResetLoading(true); setResetError("");
   218→    try {
   219→      const res = await fetch("/api/auth/reset-password", { method: "POST", headers: { "Content-Type": "application/json" }, body: JSON.stringify({ email: resetEmail.trim(), newPassword: resetNewPassword }) });
   220→      const data = await res.json();
   221→      if (!res.ok) { setResetError(data.error || "Failed to reset password."); return; }
   222→      setResetDone(true);
   223→      toast({ title: "Password reset successfully!", description: "You can now log in with your new password." });
   224→    } catch { setResetError("Something went wrong. Please try again."); } finally { setResetLoading(false); }
   225→  };
   226→
   227→  /* ================================================================ */
   228→  /*  RENDER                                                           */
   229→/* ================================================================ */
   230→  return (
   231→    <Dialog open={isOpen} onOpenChange={(open) => !open && handleClose()}>
   232→      <DialogContent showCloseButton={false} className="sm:max-w-md p-0 overflow-hidden border-forest-200/40">
   233→        <button onClick={handleClose} className="absolute top-3 right-3 z-50 rounded-full p-1.5 text-gray-400 hover:text-gray-600 hover:bg-gray-100 transition-colors" aria-label="Close">
   234→          <X className="w-5 h-5" />
   235→        </button>
   236→
   237→        <div className="px-6 pt-6 pb-2">
   238→          <DialogHeader className="space-y-1 mb-1">
   239→            <div className="flex items-center gap-2 justify-center">
   240→              <div className="w-9 h-9 rounded-xl bg-forest-900 flex items-center justify-center">
   241→                <Heart className="w-5 h-5 text-lime-400" />
   242→              </div>
   243→              <DialogTitle className="text-xl font-bold text-forest-900">SevaSaathi</DialogTitle>
   244→            </div>
   245→            <DialogDescription className="text-sm text-gray-500 mt-1">
   246→              {tab === "login" ? "Sign in to your account" : "Create your account to get started"}
   247→            </DialogDescription>
   248→          </DialogHeader>
   249→        </div>
   250→
   251→        <AnimatePresence>
   252→          {error && (
   253→            <motion.div initial={{ height: 0, opacity: 0 }} animate={{ height: "auto", opacity: 1 }} exit={{ height: 0, opacity: 0 }} className="overflow-hidden">
   254→              <div className="mx-6 mt-1 px-3 py-2 rounded-lg bg-red-50 border border-red-200 text-red-700 text-sm flex items-center gap-2">
   255→                <X className="w-4 h-4 flex-shrink-0" />{error}
   256→              </div>
   257→            </motion.div>
   258→          )}
   259→        </AnimatePresence>
   260→
   261→        <div className="px-6 pt-3">
   262→          <div className="flex bg-gray-100 rounded-lg p-1">
   263→            <button onClick={() => setTab("login")} className={`flex-1 py-2 px-4 rounded-md text-sm font-medium transition-all ${tab === "login" ? "bg-white text-forest-900 shadow-sm" : "text-gray-500 hover:text-gray-700"}`}>Sign In</button>
   264→            <button onClick={() => setTab("register")} className={`flex-1 py-2 px-4 rounded-md text-sm font-medium transition-all ${tab === "register" ? "bg-white text-forest-900 shadow-sm" : "text-gray-500 hover:text-gray-700"}`}>Create Account</button>
   265→          </div>
   266→        </div>
   267→
   268→        <div className="px-6 py-4 overflow-y-auto max-h-[55vh]">
   269→          <AnimatePresence mode="wait" custom={tab}>
   270→            {tab === "login" && (
   271→              <motion.div key="login" custom={tab} variants={contentVariants} initial="enter" animate="center" exit="exit" className="space-y-4">
   272→                <button type="button" onClick={handleGoogleSignIn} className="w-full flex items-center justify-center gap-3 h-11 px-4 rounded-lg border border-gray-300 bg-white hover:bg-gray-50 transition-all text-sm font-medium text-gray-700 shadow-sm">
   273→                  <GoogleIcon /> Continue with Google
   274→                </button>
   275→                <div className="flex items-center gap-3">
   276→                  <div className="flex-1 h-px bg-gray-200" />
   277→                  <span className="text-xs text-gray-400 font-medium uppercase tracking-wider">or continue with email</span>
   278→                  <div className="flex-1 h-px bg-gray-200" />
   279→                </div>
   280→                <form onSubmit={handleLogin} className="space-y-3">
   281→                  <div className="space-y-1.5">
   282→                    <Label htmlFor="login-email" className="text-sm font-medium text-gray-700">Email</Label>
   283→                    <div className="relative">
   284→                      <Mail className="absolute left-3 top-1/2 -translate-y-1/2 w-4 h-4 text-gray-400" />
   285→                      <Input id="login-email" type="email" placeholder="you@example.com" value={loginEmail} onChange={(e) => setLoginEmail(e.target.value)} className="pl-10 h-11 border-gray-300" autoComplete="email" disabled={loading} />
   286→                    </div>
   287→                  </div>
   288→                  <div className="space-y-1.5">
   289→                    <div className="flex items-center justify-between">
   290→                      <Label htmlFor="login-password" className="text-sm font-medium text-gray-700">Password</Label>
   291→                      <button type="button" onClick={() => setResetOpen(true)} className="text-xs font-medium text-forest-700 hover:text-forest-900">Forgot password?</button>
   292→                    </div>
   293→                    <div className="relative">
   294→                      <Lock className="absolute left-3 top-1/2 -translate-y-1/2 w-4 h-4 text-gray-400" />
   295→                      <Input id="login-password" type={showPassword ? "text" : "password"} placeholder="Enter your password" value={loginPassword} onChange={(e) => setLoginPassword(e.target.value)} className="pl-10 pr-10 h-11 border-gray-300" autoComplete="current-password" disabled={loading} />
   296→                      <button type="button" onClick={() => setShowPassword(!showPassword)} className="absolute right-3 top-1/2 -translate-y-1/2 text-gray-400 hover:text-gray-600" tabIndex={-1}>
   297→                        {showPassword ? <EyeOff className="w-4 h-4" /> : <Eye className="w-4 h-4" />}
   298→                      </button>
   299→                    </div>
   300→                  </div>
   301→                  <Button type="submit" disabled={loading} className="w-full h-11 bg-forest-900 hover:bg-forest-800 text-white font-semibold rounded-lg shadow-sm hover:shadow-md">
   302→                    {loading ? <><Loader2 className="w-4 h-4 mr-2 animate-spin" />Signing in...</> : <>Sign In <ArrowRight className="w-4 h-4 ml-2" /></>}
   303→                  </Button>
   304→                </form>
   305→                <p className="text-center text-sm text-gray-500 pt-1">Don&apos;t have an account? <button onClick={() => setTab("register")} className="font-semibold text-forest-700 hover:text-forest-900">Create one</button></p>
   306→              </motion.div>
   307→            )}
   308→
   309→            {tab === "register" && (
   310→              <motion.div key="register" custom={tab} variants={contentVariants} initial="enter" animate="center" exit="exit" className="space-y-4">
   311→                {/* Role selector */}
   312→                <div className="space-y-2">
   313→                  <Label className="text-sm font-semibold text-gray-800">How would you like to use SevaSaathi?</Label>
   314→                  <div className="grid grid-cols-1 gap-2">
   315→                    <button type="button" onClick={() => setRegRole("FAMILY")} className={`flex items-start gap-3 p-3 rounded-xl border-2 text-left transition-all ${regRole === "FAMILY" ? "border-forest-500 bg-forest-50" : "border-gray-200 bg-white hover:border-gray-300"}`}>
   316→                      <div className={`w-9 h-9 rounded-lg flex items-center justify-center shrink-0 mt-0.5 ${regRole === "FAMILY" ? "bg-forest-100" : "bg-gray-100"}`}>
   317→                        <Heart className={`w-4.5 h-4.5 ${regRole === "FAMILY" ? "text-forest-700" : "text-gray-400"}`} />
   318→                      </div>
   319→                      <div className="min-w-0">
   320→                        <p className={`text-sm font-semibold ${regRole === "FAMILY" ? "text-forest-900" : "text-gray-700"}`}>I need care for my family</p>
   321→                        <p className={`text-xs mt-0.5 ${regRole === "FAMILY" ? "text-forest-600/80" : "text-gray-400"}`}>Find trusted caregivers for parents, elders, or family members at home</p>
   322→                      </div>
   323→                    </button>
   324→                    <button type="button" onClick={() => setRegRole("CAREGIVER")} className={`flex items-start gap-3 p-3 rounded-xl border-2 text-left transition-all ${regRole === "CAREGIVER" ? "border-forest-500 bg-forest-50" : "border-gray-200 bg-white hover:border-gray-300"}`}>
   325→                      <div className={`w-9 h-9 rounded-lg flex items-center justify-center shrink-0 mt-0.5 ${regRole === "CAREGIVER" ? "bg-forest-100" : "bg-gray-100"}`}>
   326→                        <ShieldCheck className={`w-4.5 h-4.5 ${regRole === "CAREGIVER" ? "text-forest-700" : "text-gray-400"}`} />
   327→                      </div>
   328→                      <div className="min-w-0">
   329→                        <p className={`text-sm font-semibold ${regRole === "CAREGIVER" ? "text-forest-900" : "text-gray-700"}`}>I want to provide care services</p>
   330→                        <p className={`text-xs mt-0.5 ${regRole === "CAREGIVER" ? "text-forest-600/80" : "text-gray-400"}`}>Join as a caregiver, get bookings, and earn money</p>
   331→                      </div>
   332→                    </button>
   333→                  </div>
   334→                </div>
   335→
   336→                <button type="button" onClick={handleGoogleSignIn} className="w-full flex items-center justify-center gap-3 h-11 px-4 rounded-lg border border-gray-300 bg-white hover:bg-gray-50 transition-all text-sm font-medium text-gray-700 shadow-sm">
   337→                  <GoogleIcon /> Sign up with Google as {regRole === "CAREGIVER" ? "Caregiver" : "Family Member"}
   338→                </button>
   339→                <div className="flex items-center gap-3">
   340→                  <div className="flex-1 h-px bg-gray-200" />
   341→                  <span className="text-xs text-gray-400 font-medium uppercase tracking-wider">or register with email</span>
   342→                  <div className="flex-1 h-px bg-gray-200" />
   343→                </div>
   344→
   345→                <form onSubmit={handleRegister} className="space-y-3">
   346→                  <div className="space-y-1.5">
   347→                    <Label htmlFor="reg-name" className="text-sm font-medium text-gray-700">Full Name</Label>
   348→                    <div className="relative">
   349→                      <User className="absolute left-3 top-1/2 -translate-y-1/2 w-4 h-4 text-gray-400" />
   350→                      <Input id="reg-name" type="text" placeholder="Your full name" value={regName} onChange={(e) => setRegName(e.target.value)} className="pl-10 h-11 border-gray-300" autoComplete="name" disabled={loading} />
   351→                    </div>
   352→                  </div>
   353→                  <div className="space-y-1.5">
   354→                    <Label htmlFor="reg-email" className="text-sm font-medium text-gray-700">Email</Label>
   355→                    <div className="relative">
   356→                      <Mail className="absolute left-3 top-1/2 -translate-y-1/2 w-4 h-4 text-gray-400" />
   357→                      <Input id="reg-email" type="email" placeholder="you@example.com" value={regEmail} onChange={(e) => setRegEmail(e.target.value)} className="pl-10 h-11 border-gray-300" autoComplete="email" disabled={loading} />
   358→                    </div>
   359→                  </div>
   360→                  <div className="space-y-1.5">
   361→                    <Label htmlFor="reg-phone" className="text-sm font-medium text-gray-700">Phone Number</Label>
   362→                    <div className="relative">
   363→                      <Phone className="absolute left-3 top-1/2 -translate-y-1/2 w-4 h-4 text-gray-400" />
   364→                      <Input id="reg-phone" type="tel" placeholder="10-digit mobile number" value={regPhone} onChange={(e) => setRegPhone(e.target.value)} className="pl-10 h-11 border-gray-300" autoComplete="tel" disabled={loading} />
   365→                    </div>
   366→                  </div>
   367→                  <div className="space-y-1.5">
   368→                    <Label htmlFor="reg-password" className="text-sm font-medium text-gray-700">Password</Label>
   369→                    <div className="relative">
   370→                      <Lock className="absolute left-3 top-1/2 -translate-y-1/2 w-4 h-4 text-gray-400" />
   371→                      <Input id="reg-password" type={showPassword ? "text" : "password"} placeholder="Min 6 characters" value={regPassword} onChange={(e) => setRegPassword(e.target.value)} className="pl-10 pr-10 h-11 border-gray-300" autoComplete="new-password" disabled={loading} />
   372→                      <button type="button" onClick={() => setShowPassword(!showPassword)} className="absolute right-3 top-1/2 -translate-y-1/2 text-gray-400 hover:text-gray-600" tabIndex={-1}>
   373→                        {showPassword ? <EyeOff className="w-4 h-4" /> : <Eye className="w-4 h-4" />}
   374→                      </button>
