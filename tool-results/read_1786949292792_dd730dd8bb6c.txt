   204→function parseJsonSafe(jsonStr: string | null | undefined): string[] {
   205→  if (!jsonStr) return [];
   206→  try {
   207→    const parsed = JSON.parse(jsonStr);
   208→    return Array.isArray(parsed) ? parsed : [];
   209→  } catch {
   210→    return jsonStr.split(',').map((s) => s.trim()).filter(Boolean);
   211→  }
   212→}
   213→
   214→function formatDate(dateStr: string): string {
   215→  try {
   216→    return new Date(dateStr).toLocaleDateString('en-IN', {
   217→      day: 'numeric',
   218→      month: 'short',
   219→      year: 'numeric',
   220→    });
   221→  } catch {
   222→    return dateStr;
   223→  }
   224→}
   225→
   226→function getTodayDate(): string {
   227→  const now = new Date();
   228→  const year = now.getFullYear();
   229→  const month = String(now.getMonth() + 1).padStart(2, '0');
   230→  const day = String(now.getDate()).padStart(2, '0');
   231→  return `${year}-${month}-${day}`;
   232→}

   529→        <h2 className="text-2xl font-bold text-gray-900">Create Your Profile</h2>

   787→function AadharVerificationSection({ caregiverId }: { caregiverId: string }) {
   788→  const [aadharResult, setAadharResult] = useState<any>(null);
   789→  const [verifying, setVerifying] = useState(false);
   790→  const [imagePreview, setImagePreview] = useState<string | null>(null);
   791→
   792→  const handleFileSelect = async (e: React.ChangeEvent<HTMLInputElement>) => {
   793→    const file = e.target.files?.[0];
   794→    if (!file) return;
   795→
   796→    // Show preview
   797→    const reader = new FileReader();
   798→    reader.onload = (ev) => {
   799→      setImagePreview(ev.target?.result as string);
   800→      setAadharResult(null);
   801→    };
   802→    reader.readAsDataURL(file);
   803→  };
   804→
   805→  const handleVerify = async () => {
   806→    if (!imagePreview) {
   807→      toast.error('Please upload an Aadhar card image first');
   808→      return;
   809→    }
   810→    setVerifying(true);
   811→    try {
   812→      const result = await api.verifyAadhar(imagePreview);
   813→      setAadharResult(result);
   814→      if (result.verified) {
   815→        toast.success('Aadhar card verified successfully!');
   816→      } else {
   817→        toast.error(result.error || 'Aadhar verification failed');
   818→      }
   819→    } catch (err: any) {
   820→      toast.error(err.message || 'Verification failed');
   821→      setAadharResult({ verified: false, error: err.message });
   822→    } finally {
   823→      setVerifying(false);
   824→    }
   825→  };
   826→
   827→  return (
   828→    <div className="mt-6">
   829→      <div className="flex items-center gap-2 mb-3">
   830→        <ShieldCheck className="h-4 w-4 text-forest-700" />
   831→        <p className="text-sm font-semibold text-gray-700">Aadhar Card Verification</p>
   832→      </div>
   833→      <div className="bg-gray-50 rounded-2xl p-5">
   834→        <p className="text-xs text-gray-500 mb-4">Upload your Aadhar card for AI-powered instant verification. Your data is processed securely and not stored.</p>
   835→        
   836→        <div className="space-y-4">
   837→          {/* Upload Area */}
   838→          {!imagePreview ? (
   839→            <label className="flex flex-col items-center justify-center border-2 border-dashed border-gray-300 rounded-xl p-8 cursor-pointer hover:border-forest-400 hover:bg-forest-50/50 transition-colors">
   840→              <Camera className="h-10 w-10 text-gray-400 mb-3" />
   841→              <p className="text-sm font-medium text-gray-600">Upload Aadhar Card</p>
   842→              <p className="text-xs text-gray-400 mt-1">JPG, PNG - Clear photo of your Aadhar card</p>
   843→              <input
   844→                type="file"
   845→                accept="image/*"
   846→                onChange={handleFileSelect}
   847→                className="hidden"
   848→              />
   849→            </label>
   850→          ) : (
   851→            <div className="space-y-3">
   852→              <div className="relative rounded-xl overflow-hidden border border-gray-200">
   853→                <img src={imagePreview} alt="Aadhar preview" className="w-full max-h-48 object-contain bg-white" />
   854→                <button
   855→                  onClick={() => { setImagePreview(null); setAadharResult(null); }}
   856→                  className="absolute top-2 right-2 w-7 h-7 rounded-full bg-black/60 text-white flex items-center justify-center hover:bg-black/80"
   857→                >
   858→                  <XCircle className="h-4 w-4" />
   859→                </button>
   860→              </div>
   861→              <div className="flex gap-2">
   862→                <Button
   863→                  onClick={handleVerify}
   864→                  disabled={verifying}
   865→                  className="btn-green text-sm gap-2"
   866→                >
   867→                  {verifying ? <Loader2 className="h-4 w-4 animate-spin" /> : <ShieldCheck className="h-4 w-4" />}
   868→                  {verifying ? 'Verifying...' : 'Verify Aadhar'}
   869→                </Button>
   870→                <Button variant="outline" size="sm" onClick={() => { setImagePreview(null); setAadharResult(null); }} className="rounded-full">
   871→                  Change Image
   872→                </Button>
   873→              </div>
   874→            </div>
   875→          )}
   876→
   877→          {/* Result */}
   878→          {aadharResult && (
   879→            <motion.div
   880→              initial={{ opacity: 0, y: 8 }}
   881→              animate={{ opacity: 1, y: 0 }}
   882→              className={`rounded-xl p-4 ${
   883→                aadharResult.verified
   884→                  ? 'bg-green-50 border border-green-200'
   885→                  : 'bg-red-50 border border-red-200'
   886→              }`}
   887→            >
   888→              <div className="flex items-center gap-2 mb-2">
   889→                {aadharResult.verified ? (
   890→                  <><CheckCircle2 className="h-5 w-5 text-green-600" /><p className="text-sm font-semibold text-green-800">Verification Successful</p></>
   891→                ) : (
   892→                  <><XCircle className="h-5 w-5 text-red-600" /><p className="text-sm font-semibold text-red-800">Verification Failed</p></>

   896→                <div className="grid grid-cols-1 sm:grid-cols-2 gap-2 mt-3">
   897→                  {aadharResult.name && (
   898→                    <div><p className="text-[10px] text-green-600 font-medium">Name</p><p className="text-sm font-semibold text-green-900">{aadharResult.name}</p></div>

   928→function BookingsTab({ user }: { user: User }) {

   930→  const [bookings, setBookings] = useState<Booking[]>([]);
   931→  const [loading, setLoading] = useState(true);

   933→  const [completingId, setCompletingId] = useState<string | null>(null);
   934→  const [statusFilter, setStatusFilter] = useState('');
   935→
   936→  const fetchBookings = useCallback(async () => {
   937→    if (!profile?.id) return;
   938→    setLoading(true);
   939→    setError('');
   940→    try {
   941→      const params: any = { caregiverId: profile.id };
   942→      if (statusFilter) params.status = statusFilter;
   943→      const res = await api.bookings.list(params);
   944→      const mapped = (res.bookings || []).map((b: any) => ({
   945→        ...b,
   946→        date: b.startDate,
   947→        patientName: b.patient?.name,
   948→        familyName: b.family?.name,
   949→        careType: b.careRequirements ? (typeof b.careRequirements === 'string' ? JSON.parse(b.careRequirements) : b.careRequirements)?.needs?.join(', ') : undefined,
   950→      }));
   951→      setBookings(mapped);
   952→    } catch (err: any) {
   953→      setError(err.message || 'Failed to load bookings');
   954→    } finally {
   955→      setLoading(false);
   956→    }
   957→  }, [profile?.id, statusFilter]);
   958→
   959→  useEffect(() => {
   960→    fetchBookings();
   961→  }, [fetchBookings]);
   962→
   963→  const handleStatusChange = async (bookingId: string, status: string, reason?: string) => {
   964→    setCompletingId(bookingId);
   965→    try {
   966→      await api.bookings.updateStatus(bookingId, { status, cancellationReason: reason });
   967→      const labels: Record<string, string> = {
   968→        CONFIRMED: 'Booking accepted!',
   969→        IN_PROGRESS: 'Care started!',
   970→        COMPLETED: 'Booking completed!',
   971→        CANCELLED: 'Booking declined.',
   972→      };
   973→      toast.success(labels[status] || `Booking updated`);
   974→      fetchBookings();
   975→    } catch (err: any) {
   976→      toast.error(err.message || 'Failed to update booking');
   977→    } finally {
   978→      setCompletingId(null);
   979→    }
   980→  };
   981→
   982→  if (loading) {
   983→    return (
   984→      <div className="space-y-6">
   985→        <Skeleton className="h-8 w-48" />
   986→        <LoadingCards count={4} />
   987→      </div>
   988→    );
   989→  }
   990→
   991→  if (error) {
   992→    return <ErrorState message={error} onRetry={fetchBookings} />;
   993→  }
   994→
   995→  if (bookings.length === 0) {
   996→    return (
   997→      <div className="space-y-6">
   998→        <div>
   999→          <h2 className="text-2xl font-bold text-gray-900">My Bookings</h2>
  1000→          <p className="text-sm text-gray-400 mt-1">Manage all your care bookings.</p>
  1001→        </div>
  1002→        <EmptyState
  1003→          icon={<CalendarCheck className="h-7 w-7" />}
  1004→          title="No bookings yet"
  1005→          description="When families book your services, their requests will appear here."
  1006→        />
  1007→      </div>
  1008→    );
  1009→  }
  1010→
