   215→function parseJsonSafe(input: string | null | undefined | any): string[] {
   216→  if (!input) return [];
   217→  if (Array.isArray(input)) return input;
   218→  try {
   219→    const parsed = JSON.parse(input);
   220→    return Array.isArray(parsed) ? parsed : [];
   221→  } catch {
   222→    if (typeof input === 'string') {
   223→      return input.split(',').map((s) => s.trim()).filter(Boolean);
   224→    }
   225→    return [];
   226→  }
   227→}

   229→function formatDate(dateStr: string): string {
   230→  try {
   231→    return new Date(dateStr).toLocaleDateString('en-IN', {
   232→      day: 'numeric',
   233→      month: 'short',
   234→      year: 'numeric',
   235→    });
   236→  } catch {
   237→    return dateStr;
   238→  }
   239→}
   240→
   241→function getTodayDate(): string {
   242→  const now = new Date();
   243→  const year = now.getFullYear();
   244→  const month = String(now.getMonth() + 1).padStart(2, '0');
   245→  const day = String(now.getDate()).padStart(2, '0');
   246→  return `${year}-${month}-${day}`;
   247→}

   253→function OverviewTab({ user }: { user: User }) {
   254→  const profile = user.caregiverProfile;
   255→  const [bookings, setBookings] = useState<Booking[]>([]);
   256→  // Start with loading=false so new caregivers (no profile) never see a skeleton flash
   257→  const [loading, setLoading] = useState(false);
   258→  const [error, setError] = useState('');
   259→
   260→  // Phone verification — Firebase SMS only
   261→  const [phoneVerifying, setPhoneVerifying] = useState(false);
   262→  const [showOtpDialog, setShowOtpDialog] = useState(false);
   263→  const [otpSent, setOtpSent] = useState(false);
   264→  const [firebaseError, setFirebaseError] = useState<string | null>(null);
   265→  const [otpValue, setOtpValue] = useState("");
   266→  const firebase = useFirebasePhoneAuth();
   267→
   268→  const maskPhone = (p: string) => {
   269→    const d = p.replace(/\D/g, '');
   270→    if (d.length < 4) return p;
   271→    return '*'.repeat(d.length - 4) + d.slice(-4);
   272→  };
   273→
   274→  const handleSendOtp = async () => {
   275→    setPhoneVerifying(true);
   276→    setShowOtpDialog(true);
   277→    setOtpSent(false);
   278→    setFirebaseError(null);
   279→    setOtpValue('');
   280→
   281→    if (!firebase.isReady) {
   282→      setFirebaseError('Firebase is not configured. Please check your Firebase settings.');
   283→      setPhoneVerifying(false);
   284→      return;
   285→    }
   286→
   287→    try {
   288→      const result = await firebase.sendOtp(user.phone);
   289→      if (result === true) {
   290→        setOtpSent(true);
   291→        toast.success('OTP sent to your phone via SMS!');
   292→      } else {
   293→        // result is the error string (returned synchronously)
   294→        console.error('Phone OTP send failed:', result);
   295→        if (result.includes('operation-not-allowed') || result.includes('not-allowed')) {
   296→          setFirebaseError('operation-not-allowed');
   297→        } else {
   298→          setFirebaseError(result);
   299→        }
   300→      }
   301→    } catch (e: any) {
   302→      console.error('Phone OTP unexpected error:', e);
   303→      setFirebaseError(e?.message || 'Something went wrong. Please try again.');
   304→    } finally {
   305→      setPhoneVerifying(false);
   306→    }
   307→  };
   308→
   309→  const handleVerifyOtp = async () => {
   310→    if (!otpValue || otpValue.length < 6) {
   311→      toast.error('Enter the 6-digit OTP');
   312→      return;
   313→    }
   314→    try {
   315→      const result = await firebase.verifyOtp(otpValue);
   316→      if (!result) {
   317→        toast.error(firebase.error || 'Invalid OTP. Please try again.');
   318→        return;
   319→      }
   320→
   321→      const res = await fetch('/api/auth/verify-phone-otp', {
   322→        method: 'POST',
   323→        headers: { 'Content-Type': 'application/json' },
   324→        body: JSON.stringify({
   325→          phone: user.phone,
   326→          firebaseToken: result.firebaseToken,
   327→        }),
   328→      });
   329→      const data = await res.json();
   330→      if (!res.ok) {
   331→        toast.error(data.error || 'Verification failed');
   332→        return;
   333→      }
   334→      toast.success('Phone number verified successfully!');
   335→      useAuthStore.getState().setAuth({ ...user, phoneVerified: true } as any);
   336→      setShowOtpDialog(false);
   337→      setOtpValue('');
   338→      firebase.reset();
   339→    } catch (err: any) {
   340→      toast.error(err?.message || 'Verification failed');
   341→    }
   342→  };
   343→
   344→  const fetchData = useCallback(async () => {
   345→    if (!profile?.id) { setLoading(false); return; }
   346→    setLoading(true);
   347→    setError('');
   348→    try {
   349→      const res = await api.bookings.list({ caregiverId: profile.id });
   350→      const all = (res.bookings || []).map((b: any) => ({
   351→        ...b,
   352→        date: b.startDate,
   353→        patientName: b.patient?.name,
   354→        familyName: b.family?.name,
   355→        careType: b.careRequirements ? (typeof b.careRequirements === 'string' ? JSON.parse(b.careRequirements) : b.careRequirements)?.needs?.join(', ') : undefined,
   356→      }));
   357→      setBookings(all);
   358→    } catch (err: any) {
   359→      setError(err.message || 'Failed to load dashboard data');
   360→    } finally {
   361→      setLoading(false);
   362→    }
   363→  }, [profile?.id]);
   364→
   365→  useEffect(() => {
   366→    fetchData();
   367→  }, [fetchData]);
   368→
   369→  // No profile yet — show setup guidance immediately, no loading flash
   370→  if (!profile) {
   371→    return (
   372→      <div className="space-y-6">
   373→        <div>
   374→          <h2 className="text-2xl font-bold text-gray-900">Welcome, {user.name.split(' ')[0]}!</h2>
   375→          <p className="text-sm text-gray-400 mt-1">Complete your profile to start receiving care requests.</p>
   376→        </div>
   377→        <Card className="rounded-2xl border-dashed border-2 border-forest-200 bg-forest-50/50">
   378→          <CardContent className="p-8 text-center">
   379→            <div className="w-16 h-16 rounded-2xl green-gradient-bg flex items-center justify-center mx-auto mb-4">
   380→              <ClipboardList className="h-7 w-7 text-white" />
   381→            </div>
   382→            <h3 className="text-lg font-bold text-gray-800 mb-2">Set Up Your Caregiver Profile</h3>
   383→            <p className="text-sm text-gray-500 max-w-md mx-auto">Go to the <strong>My Profile</strong> tab to create your caregiver profile. Families will be able to find and book you once your profile is set up.</p>
   384→          </CardContent>
   385→        </Card>
   386→      </div>
   387→    );
   388→  }
   389→
   390→  if (error) {
   391→    return <ErrorState message={error} onRetry={fetchData} />;
   392→  }
   393→
   394→  const upcomingBookings = bookings
   395→    .filter((b) => ['CONFIRMED', 'IN_PROGRESS'].includes(b.status))
   396→    .sort((a, b) => new Date(a.date).getTime() - new Date(b.date).getTime());
   397→
   398→  const totalBookings = bookings.length;
   399→  const completedJobs = profile?.completedJobs ?? bookings.filter((b) => b.status === 'COMPLETED').length;
   400→  const overallRating = profile?.overallRating ?? 0;
   401→  const hourlyRate = profile?.hourlyRate ?? 0;
   402→
   403→  const statCards = [
   404→    { label: 'Total Bookings', value: totalBookings, icon: <CalendarCheck className="h-5 w-5" />, color: 'bg-forest-50 text-forest-700' },
   405→    { label: 'Completed Jobs', value: completedJobs, icon: <CheckCircle2 className="h-5 w-5" />, color: 'bg-lime-50 text-lime-700' },
   406→    { label: 'Overall Rating', value: overallRating.toFixed(1), icon: <Star className="h-5 w-5" />, color: 'bg-amber-50 text-amber-700', suffix: '/5' },
   407→    { label: 'Hourly Rate', value: `₹${hourlyRate}`, icon: <IndianRupee className="h-5 w-5" />, color: 'bg-purple-50 text-purple-700', isString: true },
   408→  ];
   409→
   410→  const skills = parseJsonSafe(profile?.skills);
   411→  const languages = parseJsonSafe(profile?.languages);
   412→
   413→  return (
   414→    <div className="space-y-6">
   415→      {/* Phone Verification Banner */}
   416→      {!user.phoneVerified && user.phone && (
   417→        <Card className="rounded-2xl border-2 border-amber-200 bg-gradient-to-r from-amber-50 to-orange-50">
   418→          <CardContent className="p-4 flex items-center justify-between gap-4">
   419→            <div className="flex items-center gap-3 min-w-0">
   420→              <div className="w-10 h-10 rounded-xl bg-gradient-to-br from-amber-400 to-orange-500 flex items-center justify-center shrink-0 shadow-sm">
   421→                <Phone className="h-5 w-5 text-white" />
   422→              </div>
   423→              <div className="min-w-0">
   424→                <p className="text-sm font-semibold text-amber-900">Verify Your Phone Number</p>
   425→                <p className="text-xs text-amber-700/70">Required to receive booking requests — {maskPhone(user.phone)}</p>
   426→              </div>
   427→            </div>
   428→            <Button onClick={handleSendOtp} disabled={phoneVerifying} size="sm" className="bg-gradient-to-r from-forest-700 to-forest-900 hover:from-forest-800 hover:to-forest-950 text-white rounded-xl shrink-0 shadow-sm">
   429→              {phoneVerifying && <Loader2 className="h-4 w-4 animate-spin mr-1.5" />}
   430→              {phoneVerifying ? 'Sending...' : 'Send OTP'}
   431→            </Button>
   432→          </CardContent>
   433→        </Card>
   434→      )}
   435→
   436→      {/* Phone OTP Dialog */}
   437→      <Dialog open={showOtpDialog} onOpenChange={(open) => { if (!open) { setShowOtpDialog(false); setOtpValue(''); setFirebaseError(null); firebase.reset(); } }}>
   438→        <DialogContent className="sm:max-w-lg">
   439→          <DialogHeader>
   440→            <DialogTitle>Verify Phone Number</DialogTitle>
   441→            <DialogDescription>Enter the OTP sent via SMS to {maskPhone(user.phone)}</DialogDescription>
   442→          </DialogHeader>
   443→          <div className="space-y-4 pt-2">
   444→            {/* Firebase operation-not-allowed — show fix guide */}
   445→            {firebaseError === 'operation-not-allowed' && (
   446→              <div className="bg-gradient-to-br from-red-50 to-orange-50 border border-red-200 rounded-2xl p-4 space-y-3">
   447→                <div className="flex items-start gap-2.5">
   448→                  <div className="w-8 h-8 rounded-full bg-red-500 flex items-center justify-center shrink-0 mt-0.5">
   449→                    <AlertTriangle className="h-4 w-4 text-white" />
   450→                  </div>
   451→                  <div className="space-y-2">
   452→                    <p className="text-sm font-semibold text-red-800">SMS Service Not Active</p>
   453→                    <p className="text-xs text-red-600 leading-relaxed">
   454→                      Firebase Phone Auth is enabled, but the Google Cloud Identity Toolkit API needs to be activated.
   455→                    </p>
   456→                  </div>
   457→                </div>
   458→                <div className="bg-white/70 rounded-xl p-3 space-y-2.5">
   459→                  <p className="text-xs font-bold text-gray-700">🔧 Fix it in 2 minutes:</p>
   460→                  <ol className="text-xs text-gray-600 space-y-1.5 list-decimal list-inside">
   461→                    <li>Open <strong>Google Cloud Console → APIs &amp; Services → Library</strong></li>
   462→                    <li>Search: <strong>"Identity Toolkit API"</strong></li>
   463→                    <li>Click <strong>Enable</strong></li>
   464→                    <li>Wait 2 minutes, then try again</li>
   465→                  </ol>
   466→                  <a href="https://console.cloud.google.com/apis/library/identitytoolkit.googleapis.com" target="_blank" rel="noopener noreferrer" className="inline-flex items-center gap-1 text-xs font-semibold text-blue-600 hover:text-blue-800 hover:underline mt-1">
   467→                    Open Google Cloud Console ↗
   468→                  </a>
   469→                </div>
   470→                <div className="bg-white/70 rounded-xl p-3 space-y-2">
   471→                  <p className="text-xs font-bold text-gray-700">⚡ Quick Alternative (Works Instantly):</p>
   472→                  <p className="text-xs text-gray-600 leading-relaxed">Add a <strong>test phone number</strong> in Firebase Console. OTPs for test numbers are shown on screen — no real SMS needed.</p>
   473→                  <ol className="text-xs text-gray-600 space-y-1.5 list-decimal list-inside">
   474→                    <li>Firebase Console → <strong>Authentication → Sign-in method</strong></li>
   475→                    <li>Click <strong>Phone → Phone numbers for testing</strong></li>
   476→                    <li>Add your number + any 6-digit OTP (e.g. 123456)</li>
   477→                    <li>Use that OTP to verify instantly</li>
   478→                  </ol>
   479→                </div>
   480→                <Button
   481→                  onClick={async () => { setFirebaseError(null); setOtpValue(''); firebase.reset(); setShowOtpDialog(false); await handleSendOtp(); }}
   482→                  variant="outline"
   483→                  size="sm"
   484→                  className="w-full rounded-xl gap-2 text-xs border-red-200 text-red-700 hover:bg-red-50"
   485→                >
   486→                  <RefreshCw className="h-3 w-3" />
   487→                  I Fixed It — Try Again
   488→                </Button>
   489→              </div>
   490→            )}
   491→
   492→            {/* Other Firebase errors */}
   493→            {firebaseError && firebaseError !== 'operation-not-allowed' && (
   494→              <div className="flex items-start gap-2 bg-red-50 border border-red-200 rounded-xl p-3">
   495→                <AlertTriangle className="h-4 w-4 text-red-500 shrink-0 mt-0.5" />
   496→                <p className="text-xs text-red-700">{firebaseError}</p>
   497→              </div>
   498→            )}
   499→
   500→            {/* OTP sent success */}
   501→            {otpSent && !firebaseError && (
   502→              <div className="flex items-center gap-2 bg-green-50 border border-green-200 rounded-xl p-3">
   503→                <div className="w-6 h-6 rounded-full bg-green-500 flex items-center justify-center shrink-0">
   504→                  <span className="text-white text-xs font-bold">✓</span>
   505→                </div>
   506→                <p className="text-xs text-green-700 font-medium">OTP sent to {maskPhone(user.phone)} via SMS</p>
   507→              </div>
   508→            )}
   509→
   510→            {/* OTP input */}
   511→            {otpSent && !firebaseError && (
   512→              <>

  1325→/* VERIFICATION PROGRESS SECTION                                   */

  1328→type DocStatus = 'NOT_UPLOADED' | 'PENDING' | 'APPROVED' | 'REJECTED';
  1329→
  1330→function VerificationProgressSection({ caregiverId, user }: { caregiverId: string; user: User }) {
  1331→  const [aadhaarStatus, setAadhaarStatus] = useState<DocStatus>('NOT_UPLOADED');
  1332→  const [idCardStatus, setIdCardStatus] = useState<DocStatus>('NOT_UPLOADED');
  1333→  const [aadhaarFile, setAadhaarFile] = useState<File | null>(null);
  1334→  const [idCardFile, setIdCardFile] = useState<File | null>(null);
  1335→  const [aadhaarNumber, setAadhaarNumber] = useState('');
  1336→  const [uploading, setUploading] = useState<string | null>(null);
  1337→
  1338→  const phoneDone = !!user.phoneVerified;
  1339→  const aadhaarDone = aadhaarStatus === 'APPROVED';
  1340→  const idCardDone = idCardStatus === 'APPROVED';
  1341→  const completedCount = [phoneDone, aadhaarDone, idCardDone].filter(Boolean).length;
  1342→  const allDone = completedCount === 3;
  1343→
  1344→  // Fetch verification statuses on mount
  1345→  useEffect(() => {
  1346→    if (user.caregiverProfile?.isVerified) {
  1347→      setAadhaarStatus('APPROVED');
  1348→      setIdCardStatus('APPROVED');
  1349→      return;
  1350→    }
  1351→    (async () => {
  1352→      try {
  1353→        const res = await fetch(`/api/caregivers/${caregiverId}`);
  1354→        if (!res.ok) return;
  1355→        const data = await res.json();
  1356→        const verifications = data.caregiver?.verifications || [];
  1357→        for (const v of verifications) {
  1358→          if (v.docType === 'AADHAAR') setAadhaarStatus(v.status as DocStatus);
  1359→          if (v.docType === 'ID_CARD') setIdCardStatus(v.status as DocStatus);
  1360→        }
  1361→      } catch {
  1362→        // silently fail — local state stays NOT_UPLOADED
  1363→      }
  1364→    })();
  1365→  }, [caregiverId, user.caregiverProfile?.isVerified]);
  1366→
  1367→  const formatAadhar = (value: string) => {
  1368→    const digits = value.replace(/\D/g, '').slice(0, 12);
  1369→    if (digits.length <= 4) return digits;
  1370→    if (digits.length <= 8) return `${digits.slice(0, 4)} ${digits.slice(4)}`;
  1371→    return `${digits.slice(0, 4)} ${digits.slice(4, 8)} ${digits.slice(8)}`;
  1372→  };
  1373→
  1374→  const handleUpload = async (docType: 'AADHAAR' | 'ID_CARD') => {
  1375→    const file = docType === 'AADHAAR' ? aadhaarFile : idCardFile;
  1376→    if (!file) {
  1377→      toast.error('Please select an image first');
  1378→      return;
  1379→    }
  1380→    if (docType === 'AADHAAR' && aadhaarNumber.replace(/\s/g, '').length !== 12) {
  1381→      toast.error('Enter complete 12-digit Aadhaar number');
  1382→      return;
  1383→    }
  1384→    setUploading(docType);
  1385→    try {
  1386→      const formData = new FormData();
  1387→      formData.append('caregiverId', caregiverId);
  1388→      formData.append('docType', docType);
  1389→      formData.append('file', file);
  1390→      if (docType === 'AADHAAR') formData.append('docNumber', aadhaarNumber.replace(/\s/g, ''));
  1391→      const res = await fetch('/api/caregiver/upload-document', { method: 'POST', body: formData });
  1392→      if (!res.ok) {
  1393→        const err = await res.json();
  1394→        throw new Error(err.error || 'Upload failed');
  1395→      }
  1396→      toast.success(`${docType === 'AADHAAR' ? 'Aadhaar card' : 'ID card'} uploaded successfully! It will be reviewed shortly.`);
  1397→      if (docType === 'AADHAAR') {
  1398→        setAadhaarStatus('PENDING');
  1399→        setAadhaarFile(null);
  1400→      } else {
  1401→        setIdCardStatus('PENDING');
  1402→        setIdCardFile(null);
  1403→      }
  1404→    } catch (err: any) {
  1405→      toast.error(err.message || 'Upload failed');
  1406→    } finally {
  1407→      setUploading(null);
  1408→    }
  1409→  };
