   126→   126→/* ================================================================== */
   127→   127→export default function LoginModal({ isOpen, onClose, defaultTab }: LoginModalProps) {
   128→   128→  /* ---- state ---- */
   129→   129→  const [tab, setTab] = useState<Tab>(defaultTab === "register" ? "register" : "login");
   130→   130→  const [showPassword, setShowPassword] = useState(false);
   131→   131→  const [loading, setLoading] = useState(false);
   132→   132→  const [googleLoading, setGoogleLoading] = useState(false);
   133→   133→  const [error, setError] = useState("");
   134→   134→  const [resetOpen, setResetOpen] = useState(false);
   135→   135→
   136→   136→  // Login form
   137→   137→  const [loginEmail, setLoginEmail] = useState("");
   138→   138→  const [loginPassword, setLoginPassword] = useState("");
   139→   139→
   140→   140→  // Register form
   141→   141→  const [regName, setRegName] = useState("");
   142→   142→  const [regEmail, setRegEmail] = useState("");
   143→   143→  const [regPhone, setRegPhone] = useState("");
   144→   144→  const [regPassword, setRegPassword] = useState("");
   145→   145→  const [regRole, setRegRole] = useState<UserRole>("FAMILY");
   146→   146→
   147→   147→  // Reset form (simplified — no OTP)
   148→   148→  const [resetEmail, setResetEmail] = useState("");
   149→   149→  const [resetNewPassword, setResetNewPassword] = useState("");
   150→   150→  const [resetLoading, setResetLoading] = useState(false);
   151→   151→  const [resetError, setResetError] = useState("");
   152→   152→  const [resetDone, setResetDone] = useState(false);
   153→   153→
   154→   154→  const { setAuth } = useAuthStore();
   155→   155→  const { toast } = useToast();
   156→   156→
   157→   157→  /* ---- reset form fields when switching tabs ---- */
   158→   158→  useEffect(() => {
   159→   159→    setError("");
   160→   160→    setLoginPassword("");
   161→   161→  }, [tab]);
   162→   162→
   163→   163→  /* ---- reset entire state when modal closes ---- */
   164→   164→  const handleClose = useCallback(() => {
   165→   165→    setTab(defaultTab === "register" ? "register" : "login");
   166→   166→    setError("");
   167→   167→    setShowPassword(false);
   168→   168→    setLoading(false);
   169→   169→    setGoogleLoading(false);
   170→   170→    setLoginEmail("");
   171→   171→    setLoginPassword("");
   172→   172→    setRegName("");
   173→   173→    setRegEmail("");
   174→   174→    setRegPhone("");
   175→   175→    setRegPassword("");
   176→   176→    setRegRole("FAMILY");
   177→   177→    setResetOpen(false);
   178→   178→    setResetEmail("");
   179→   179→    setResetNewPassword("");
   180→   180→    setResetLoading(false);
   181→   181→    setResetError("");
   182→   182→    setResetDone(false);
   183→   183→    onClose();
   184→   184→  }, [onClose, defaultTab]);
   185→   185→
   186→   186→  /* ================================================================ */
   187→   187→  /*  HANDLERS                                                         */
   188→   188→  /* ================================================================ */
   189→   189→
   190→   190→  /* ---- Google sign-in ---- */
   191→   191→  const handleGoogleSignIn = async () => {
   192→   192→    setGoogleLoading(true);
   193→   193→    setError("");
   194→   194→    try {
   195→   195→      // Check if Google OAuth is configured on the server
   196→   196→      const statusRes = await fetch("/api/auth/google-status");
   197→   197→      const status = await statusRes.json();
   198→   198→
   199→   199→      if (!status.configured) {
   200→   200→        toast({
   201→   201→          title: "Google OAuth Setup Required",
   202→   202→          description: "Google Client ID and Secret are not configured. Please add them in .env.local to enable Google login.",
   203→   203→          variant: "destructive",
   204→   204→        });
   205→   205→        setGoogleLoading(false);
   206→   206→        return;
   207→   207→      }
   208→   208→
   209→   209→      // Credentials are configured — proceed with real OAuth flow
   210→   210→      await signIn("google", {
   211→   211→        callbackUrl: window.location.origin,
   212→   212→      });
   213→   213→    } catch (err: any) {
   214→   214→      toast({
   215→   215→        title: "Google Sign-In Failed",
   216→   216→        description: err?.message || "Could not start Google sign-in. Please try email/password.",
   217→   217→        variant: "destructive",
   218→   218→      });
   219→   219→    } finally {
   220→   220→      setGoogleLoading(false);
   221→   221→    }
   222→   222→  };
   223→   223→
   224→   224→  /* ---- Email / password login ---- */
   225→   225→  const handleLogin = async (e: React.FormEvent) => {
   226→   226→    e.preventDefault();
   227→   227→    if (!loginEmail.trim() || !loginPassword.trim()) {
   228→   228→      setError("Please enter your email and password.");
   229→   229→      return;
   230→   230→    }
   231→   231→    setLoading(true);
   232→   232→    setError("");
   233→   233→    try {
   234→   234→      const result = await signIn("credentials", {
   235→   235→        email: loginEmail.trim(),
   236→   236→        password: loginPassword,
   237→   237→        redirect: false,
   238→   238→      });
   239→   239→
   240→   240→      if (result?.error) {
   241→   241→        setError("Invalid email or password. Please try again.");
   242→   242→        return;
   243→   243→      }
   244→   244→
   245→   245→      // AuthSync in AuthProvider will automatically sync session → Zustand store
   246→   246→      toast({
   247→   247→        title: "Welcome back! 🎉",
   248→   248→        description: "You have successfully logged in.",
   249→   249→      });
   250→   250→      handleClose();
   251→   251→    } catch {
   252→   252→      setError("Something went wrong. Please try again.");
   253→   253→    } finally {
   254→   254→      setLoading(false);
   255→   255→    }
   256→   256→  };
   257→   257→
   258→   258→  /* ---- Registration ---- */
   259→   259→  const handleRegister = async (e: React.FormEvent) => {
   260→   260→    e.preventDefault();
   261→   261→    if (!regName.trim() || !regEmail.trim() || !regPhone.trim() || !regPassword.trim()) {
   262→   262→      setError("Please fill in all fields.");
   263→   263→      return;
   264→   264→    }
   265→   265→    if (regPassword.length < 6) {
   266→   266→      setError("Password must be at least 6 characters.");
   267→   267→      return;
   268→   268→    }
   269→   269→    setLoading(true);
   270→   270→    setError("");
   271→   271→    try {
   272→   272→      // 1. Register
   273→   273→      const regRes = await api.auth.register({
   274→   274→        name: regName.trim(),
   275→   275→        email: regEmail.trim(),
   276→   276→        phone: regPhone.trim(),
   277→   277→        password: regPassword,
   278→   278→        role: regRole,
   279→   279→      });
   280→   280→
   281→   281→      // 2. Auto-login with credentials
   282→   282→      const loginResult = await signIn("credentials", {
   283→   283→        email: regEmail.trim(),
   284→   284→        password: regPassword,
   285→   285→        redirect: false,
   286→   286→      });
   287→   287→
   288→   288→      if (loginResult?.error) {
   289→   289→        // Registration succeeded but auto-login failed — that's ok, just notify
   290→   290→        toast({ title: "Account created!", description: "Please log in with your credentials." });
   291→   291→        setTab("login");
   292→   292→        setLoginEmail(regEmail.trim());
   293→   293→        return;
   294→   294→      }
   295→   295→
   296→   296→      toast({ title: "Welcome to SevaSaathi! 🎉", description: "Your account has been created." });
   297→   297→      handleClose();
   298→   298→    } catch (err: any) {
   299→   299→      setError(err?.message || "Registration failed. Please try again.");
   300→   300→    } finally {
   301→   301→      setLoading(false);
   302→   302→    }
   303→   303→  };
   304→   304→
   305→   305→  /* ---- Reset password (simplified) ---- */
   306→   306→  const handleReset = async (e: React.FormEvent) => {
   307→   307→    e.preventDefault();
   308→   308→    if (!resetEmail.trim() || !resetNewPassword.trim()) {
   309→   309→      setResetError("Please enter your email and new password.");
   310→   310→      return;
   311→   311→    }
   312→   312→    if (resetNewPassword.length < 6) {
   313→   313→      setResetError("Password must be at least 6 characters.");
   314→   314→      return;
   315→   315→    }
   316→   316→    setResetLoading(true);
   317→   317→    setResetError("");
   318→   318→    try {
   319→   319→      const res = await fetch("/api/auth/reset-password", {
   320→   320→        method: "POST",
   321→   321→        headers: { "Content-Type": "application/json" },
   322→   322→        body: JSON.stringify({ email: resetEmail.trim(), newPassword: resetNewPassword }),
   323→   323→      });
   324→   324→      const data = await res.json();
   325→   325→      if (!res.ok) {
   326→   326→        setResetError(data.error || "Failed to reset password.");
   327→   327→        return;
   328→   328→      }
   329→   329→      setResetDone(true);
   330→   330→      toast({ title: "Password reset successfully!", description: "You can now log in with your new password." });
   331→   331→    } catch {
   332→   332→      setResetError("Something went wrong. Please try again.");
   333→   333→    } finally {
   334→   334→      setResetLoading(false);
   335→   335→    }
   336→   336→  };
   337→   337→
   338→   338→  /* ================================================================ */
   339→   339→  /*  RENDER                                                           */
   340→   340→  /* ================================================================ */
   341→   341→  return (
   342→   342→    <Dialog open={isOpen} onOpenChange={(open) => !open && handleClose()}>
   343→   343→      <DialogContent className="sm:max-w-md p-0 overflow-hidden border-forest-200/40">
   344→   344→        {/* Custom X button to override default position */}
   345→   345→        <button
   346→   346→          onClick={handleClose}
   347→   347→          className="absolute top-3 right-3 z-50 rounded-full p-1.5 text-gray-400 hover:text-gray-600 hover:bg-gray-100 transition-colors"
   348→   348→          aria-label="Close"
   349→   349→        >
   350→   350→          <X className="w-5 h-5" />
   351→   351→        </button>
   352→   352→
   353→   353→        <div className="px-6 pt-6 pb-2">
   354→   354→          <DialogHeader className="space-y-1 mb-1">
   355→   355→            <div className="flex items-center gap-2 justify-center">
   356→   356→              <div className="w-9 h-9 rounded-xl bg-forest-900 flex items-center justify-center">
   357→   357→                <Heart className="w-5 h-5 text-lime-400" />
   358→   358→              </div>
   359→   359→              <DialogTitle className="text-xl font-bold text-forest-900">
   360→   360→                SevaSaathi
   361→   361→              </DialogTitle>
   362→   362→            </div>
   363→   363→            <DialogDescription className="text-sm text-gray-500 mt-1">
   364→   364→              {tab === "login"
   365→   365→                ? "Sign in to your account"
   366→   366→                : "Create your account to get started"}
   367→   367→            </DialogDescription>
   368→   368→          </DialogHeader>
   369→   369→        </div>
   370→   370→
   371→   371→        {/* ---- Error banner ---- */}
   372→   372→        <AnimatePresence>
   373→   373→          {error && (
   374→   374→            <motion.div
   375→   375→              initial={{ height: 0, opacity: 0 }}
   376→   376→              animate={{ height: "auto", opacity: 1 }}
   377→   377→              exit={{ height: 0, opacity: 0 }}
   378→   378→              className="overflow-hidden"
   379→   379→            >
   380→   380→              <div className="mx-6 mt-1 px-3 py-2 rounded-lg bg-red-50 border border-red-200 text-red-700 text-sm flex items-center gap-2">
   381→   381→                <X className="w-4 h-4 flex-shrink-0" />
   382→   382→                {error}
   383→   383→              </div>
   384→   384→            </motion.div>
   385→   385→          )}
   386→   386→        </AnimatePresence>
   387→   387→
   388→   388→        {/* ---- Tab switcher ---- */}
   389→   389→        <div className="px-6 pt-3">
   390→   390→          <div className="flex bg-gray-100 rounded-lg p-1">
   391→   391→            <button
   392→   392→              onClick={() => setTab("login")}
   393→   393→              className={`flex-1 py-2 px-4 rounded-md text-sm font-medium transition-all duration-200 ${
   394→   394→                tab === "login"
   395→   395→                  ? "bg-white text-forest-900 shadow-sm"
   396→   396→                  : "text-gray-500 hover:text-gray-700"
   397→   397→              }`}
   398→   398→            >
   399→   399→              Sign In
   400→   400→            </button>
   401→   401→            <button
   402→   402→              onClick={() => setTab("register")}
   403→   403→              className={`flex-1 py-2 px-4 rounded-md text-sm font-medium transition-all duration-200 ${
   404→   404→                tab === "register"
   405→   405→                  ? "bg-white text-forest-900 shadow-sm"
   406→   406→                  : "text-gray-500 hover:text-gray-700"
   407→   407→              }`}
   408→   408→            >
   409→   409→              Create Account
   410→   410→            </button>
   411→   411→          </div>
   412→   412→        </div>
   413→   413→
   414→   414→        {/* ---- Animated content ---- */}
   415→   415→        <div className="px-6 py-4 overflow-hidden">
   416→   416→          <AnimatePresence mode="wait" custom={tab}>
   417→   417→            {/* ========== LOGIN TAB ========== */}
   418→   418→            {tab === "login" && (
   419→   419→              <motion.div
   420→   420→                key="login"
   421→   421→                custom={tab}
   422→   422→                variants={contentVariants}
   423→   423→                initial="enter"
   424→   424→                animate="center"
   425→   425→                exit="exit"
   426→   426→                className="space-y-4"
   427→   427→              >
   428→   428→                {/* Google button */}
   429→   429→                <button
   430→   430→                  type="button"
   431→   431→                  onClick={handleGoogleSignIn}
   432→   432→                  disabled={googleLoading}
   433→   433→                  className="w-full flex items-center justify-center gap-3 h-11 px-4 rounded-lg border border-gray-300 bg-white hover:bg-gray-50 active:bg-gray-100 transition-all duration-150 text-sm font-medium text-gray-700 disabled:opacity-60 disabled:cursor-not-allowed shadow-sm"
   434→   434→                >
   435→   435→                  {googleLoading ? (
   436→   436→                    <Loader2 className="w-5 h-5 animate-spin text-gray-500" />
   437→   437→                  ) : (
   438→   438→                    <GoogleIcon />
   439→   439→                  )}
   440→   440→                  Continue with Google
   441→   441→                </button>
   442→   442→
   443→   443→                {/* Divider */}
   444→   444→                <div className="flex items-center gap-3">
   445→   445→                  <div className="flex-1 h-px bg-gray-200" />
   446→   446→                  <span className="text-xs text-gray-400 font-medium uppercase tracking-wider">
   447→   447→                    or continue with email
   448→   448→                  </span>
   449→   449→                  <div className="flex-1 h-px bg-gray-200" />
   450→   450→                </div>
   451→   451→
   452→   452→                {/* Email / password form */}
   453→   453→                <form onSubmit={handleLogin} className="space-y-3">
   454→   454→                  <div className="space-y-1.5">
   455→   455→                    <Label htmlFor="login-email" className="text-sm font-medium text-gray-700">
   456→   456→                      Email
   457→   457→                    </Label>
   458→   458→                    <div className="relative">
   459→   459→                      <Mail className="absolute left-3 top-1/2 -translate-y-1/2 w-4 h-4 text-gray-400" />
   460→   460→                      <Input
   461→   461→                        id="login-email"
   462→   462→                        type="email"
   463→   463→                        placeholder="you@example.com"
   464→   464→                        value={loginEmail}
   465→   465→                        onChange={(e) => setLoginEmail(e.target.value)}
   466→   466→                        className="pl-10 h-11 border-gray-300 focus:border-forest-500 focus:ring-forest-500"
   467→   467→                        autoComplete="email"
   468→   468→                        disabled={loading}
   469→   469→                      />
   470→   470→                    </div>
   471→   471→                  </div>
   472→   472→
   473→   473→                  <div className="space-y-1.5">
   474→   474→                    <div className="flex items-center justify-between">

   478→   478→                      <button
   479→   479→                        type="button"
   480→   480→                        onClick={() => setResetOpen(true)}
   481→   481→                        className="text-xs font-medium text-forest-700 hover:text-forest-900 transition-colors"
   482→   482→                      >
   483→   483→                        Forgot password?
   484→   484→                      </button>
   485→   485→                    </div>
   486→   486→                    <div className="relative">
   487→   487→                      <Lock className="absolute left-3 top-1/2 -translate-y-1/2 w-4 h-4 text-gray-400" />
   488→   488→                      <Input
   489→   489→                        id="login-password"
   490→   490→                        type={showPassword ? "text" : "password"}
