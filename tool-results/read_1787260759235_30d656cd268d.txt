   205→function parseJsonSafe(jsonStr: string | null | undefined): string[] {
   206→  if (!jsonStr) return [];
   207→  try {
   208→    const parsed = JSON.parse(jsonStr);
   209→    return Array.isArray(parsed) ? parsed : [];
   210→  } catch {
   211→    return jsonStr.split(',').map((s) => s.trim()).filter(Boolean);
   212→  }
   213→}
   214→
   215→function formatDate(dateStr: string): string {
   216→  try {
   217→    return new Date(dateStr).toLocaleDateString('en-IN', {
   218→      day: 'numeric',
   219→      month: 'short',
   220→      year: 'numeric',
   221→    });
   222→  } catch {
   223→    return dateStr;
   224→  }
   225→}
   226→
   227→function getTodayDate(): string {
   228→  const now = new Date();
   229→  const year = now.getFullYear();
   230→  const month = String(now.getMonth() + 1).padStart(2, '0');
   231→  const day = String(now.getDate()).padStart(2, '0');
   232→  return `${year}-${month}-${day}`;
   233→}

   473→function CreateProfileForm({ userId, onCreated }: { userId: string; onCreated: (profile: any) => void }) {

   480→  const toggleSkill = (skill: string) => {
   481→    setSelectedSkills(prev => prev.includes(skill) ? prev.filter(s => s !== skill) : [...prev, skill]);
   482→  };
   483→
   484→  const handleSubmit = async (e: React.FormEvent) => {
   485→    e.preventDefault();
   486→    if (!form.city || !form.yearsExperience || !form.hourlyRate) {
   487→      toast.error('Please fill in City, Experience, and Hourly Rate.');
   488→      return;
   489→    }
   490→    if (selectedSkills.length === 0) {
   491→      toast.error('Please select at least one skill.');
   492→      return;
   493→    }
   494→    setSaving(true);
   495→    try {
   496→      const res = await api.caregivers.create({
   497→        userId,
   498→        city: form.city,
   499→        yearsExperience: parseInt(form.yearsExperience),
   500→        hourlyRate: parseInt(form.hourlyRate),
   501→        skills: JSON.stringify(selectedSkills),
   502→        qualifications: form.qualifications ? JSON.stringify(form.qualifications.split(',').map((s: string) => s.trim()).filter(Boolean)) : JSON.stringify(['Caregiver']),
   503→        languages: JSON.stringify(form.languages.split(',').map((s: string) => s.trim()).filter(Boolean)),
   504→        bio: form.bio,
   505→      });
   506→      toast.success('Profile created successfully!');
   507→      onCreated(res.caregiver);
   508→    } catch (err: any) {
   509→      toast.error(err.message || 'Failed to create profile');
   510→    } finally {
   511→      setSaving(false);
   512→    }
   513→  };
   514→
   515→  return (
   516→    <div className="space-y-6 max-w-2xl">
   517→      <div>
   518→        <h2 className="text-2xl font-bold text-gray-900">Create Your Profile</h2>
   519→        <p className="text-sm text-gray-400 mt-1">Set up your caregiver profile to start receiving booking requests from families.</p>
   520→      </div>
   521→      <Card className="rounded-2xl border-gray-100">
   522→        <CardContent className="p-6">
   523→          <form onSubmit={handleSubmit} className="space-y-4">
   524→            <div className="grid grid-cols-1 sm:grid-cols-2 gap-4">
