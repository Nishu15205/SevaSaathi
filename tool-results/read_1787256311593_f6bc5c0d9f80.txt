   205→function parseJsonSafe(jsonStr: string | null | undefined): string[] {
   206→  if (!jsonStr) return [];
   207→  try {
   208→    const parsed = JSON.parse(jsonStr);
   209→    return Array.isArray(parsed) ? parsed : [];
   210→  } catch {
   211→    return jsonStr.split(',').map((s) => s.trim()).filter(Boolean);
   212→  }
   213→}
   214→
   215→function formatDate(dateStr: string): string {
   216→  try {
   217→    return new Date(dateStr).toLocaleDateString('en-IN', {
   218→      day: 'numeric',
   219→      month: 'short',
   220→      year: 'numeric',
   221→    });
   222→  } catch {
   223→    return dateStr;
   224→  }
   225→}
   226→
   227→function getTodayDate(): string {
   228→  const now = new Date();
   229→  const year = now.getFullYear();
   230→  const month = String(now.getMonth() + 1).padStart(2, '0');
   231→  const day = String(now.getDate()).padStart(2, '0');
   232→  return `${year}-${month}-${day}`;
   233→}

   472→function CreateProfileForm({ userId, onCreated }: { userId: string; onCreated: (profile: any) => void }) {

   474→  const [selectedSkills, setSelectedSkills] = useState<string[]>(['elderly-care']);

   477→  });
   478→
   479→  const toggleSkill = (skill: string) => {
   480→    setSelectedSkills(prev => prev.includes(skill) ? prev.filter(s => s !== skill) : [...prev, skill]);
   481→  };
   482→
   483→  const handleSubmit = async (e: React.FormEvent) => {
   484→    e.preventDefault();
   485→    if (!form.city || !form.yearsExperience || !form.hourlyRate) {
   486→      toast.error('Please fill in City, Experience, and Hourly Rate.');
   487→      return;
   488→    }
   489→    if (selectedSkills.length === 0) {
   490→      toast.error('Please select at least one skill.');
   491→      return;
   492→    }
   493→    setSaving(true);
   494→    try {
   495→      const res = await api.caregivers.create({
   496→        userId,
   497→        city: form.city,
   498→        yearsExperience: parseInt(form.yearsExperience),
   499→        hourlyRate: parseInt(form.hourlyRate),
   500→        skills: JSON.stringify(selectedSkills),
   501→        qualifications: form.qualifications ? JSON.stringify(form.qualifications.split(',').map((s: string) => s.trim()).filter(Boolean)) : JSON.stringify(['Caregiver']),
   502→        languages: JSON.stringify(form.languages.split(',').map((s: string) => s.trim()).filter(Boolean)),
   503→        bio: form.bio,
   504→      });
   505→      toast.success('Profile created successfully!');
   506→      onCreated(res.caregiver);
   507→    } catch (err: any) {
   508→      toast.error(err.message || 'Failed to create profile');
   509→    } finally {
   510→      setSaving(false);
   511→    }
   512→  };
   513→
   514→  return (
   515→    <div className="space-y-6 max-w-2xl">
   516→      <div>
   517→        <h2 className="text-2xl font-bold text-gray-900">Create Your Profile</h2>
   518→        <p className="text-sm text-gray-400 mt-1">Set up your caregiver profile to start receiving booking requests from families.</p>
   519→      </div>
   520→      <Card className="rounded-2xl border-gray-100">
   521→        <CardContent className="p-6">
   522→          <form onSubmit={handleSubmit} className="space-y-4">
   523→            <div className="grid grid-cols-1 sm:grid-cols-2 gap-4">

   775→function AadharVerificationSection({ caregiverId }: { caregiverId: string }) {
   776→  const [aadharResult, setAadharResult] = useState<any>(null);
   777→  const [verifying, setVerifying] = useState(false);
   778→  const [imagePreview, setImagePreview] = useState<string | null>(null);
   779→
   780→  const handleFileSelect = async (e: React.ChangeEvent<HTMLInputElement>) => {
   781→    const file = e.target.files?.[0];
   782→    if (!file) return;
   783→
   784→    // Show preview
   785→    const reader = new FileReader();
   786→    reader.onload = (ev) => {
   787→      setImagePreview(ev.target?.result as string);
   788→      setAadharResult(null);
   789→    };
   790→    reader.readAsDataURL(file);
   791→  };
   792→
   793→  const handleVerify = async () => {
   794→    if (!imagePreview) {
   795→      toast.error('Please upload an Aadhar card image first');
   796→      return;
   797→    }
   798→    setVerifying(true);
   799→    try {
   800→      const result = await api.verifyAadhar(imagePreview);
   801→      setAadharResult(result);
   802→      if (result.verified) {
   803→        toast.success('Aadhar card verified successfully!');
   804→      } else {
   805→        toast.error(result.error || 'Aadhar verification failed');
   806→      }
   807→    } catch (err: any) {
   808→      toast.error(err.message || 'Verification failed');
   809→      setAadharResult({ verified: false, error: err.message });
   810→    } finally {
   811→      setVerifying(false);
   812→    }
   813→  };
   814→
   815→  return (
   816→    <div className="mt-6">
   817→      <div className="flex items-center gap-2 mb-3">
   818→        <ShieldCheck className="h-4 w-4 text-forest-700" />
   819→        <p className="text-sm font-semibold text-gray-700">Aadhar Card Verification</p>
   820→      </div>
   821→      <div className="bg-gray-50 rounded-2xl p-5">
   822→        <p className="text-xs text-gray-500 mb-4">Upload your Aadhar card for AI-powered instant verification. Your data is processed securely and not stored.</p>
   823→        
   824→        <div className="space-y-4">
   825→          {/* Upload Area */}
   826→          {!imagePreview ? (
   827→            <label className="flex flex-col items-center justify-center border-2 border-dashed border-gray-300 rounded-xl p-8 cursor-pointer hover:border-forest-400 hover:bg-forest-50/50 transition-colors">
   828→              <Camera className="h-10 w-10 text-gray-400 mb-3" />
   829→              <p className="text-sm font-medium text-gray-600">Upload Aadhar Card</p>
   830→              <p className="text-xs text-gray-400 mt-1">JPG, PNG - Clear photo of your Aadhar card</p>
   831→              <input
   832→                type="file"
   833→                accept="image/*"
   834→                onChange={handleFileSelect}
   835→                className="hidden"
   836→              />
   837→            </label>
   838→          ) : (
   839→            <div className="space-y-3">
   840→              <div className="relative rounded-xl overflow-hidden border border-gray-200">
   841→                <img src={imagePreview} alt="Aadhar preview" className="w-full max-h-48 object-contain bg-white" />
   842→                <button
   843→                  onClick={() => { setImagePreview(null); setAadharResult(null); }}
   844→                  className="absolute top-2 right-2 w-7 h-7 rounded-full bg-black/60 text-white flex items-center justify-center hover:bg-black/80"
   845→                >
   846→                  <XCircle className="h-4 w-4" />
   847→                </button>
   848→              </div>
   849→              <div className="flex gap-2">
   850→                <Button
   851→                  onClick={handleVerify}
   852→                  disabled={verifying}
   853→                  className="btn-green text-sm gap-2"
   854→                >
   855→                  {verifying ? <Loader2 className="h-4 w-4 animate-spin" /> : <ShieldCheck className="h-4 w-4" />}
   856→                  {verifying ? 'Verifying...' : 'Verify Aadhar'}
   857→                </Button>
   858→                <Button variant="outline" size="sm" onClick={() => { setImagePreview(null); setAadharResult(null); }} className="rounded-full">
   859→                  Change Image
   860→                </Button>
   861→              </div>
   862→            </div>
   863→          )}
   864→
   865→          {/* Result */}
   866→          {aadharResult && (
   867→            <motion.div
   868→              initial={{ opacity: 0, y: 8 }}
   869→              animate={{ opacity: 1, y: 0 }}
   870→              className={`rounded-xl p-4 ${
   871→                aadharResult.verified
   872→                  ? 'bg-green-50 border border-green-200'
   873→                  : 'bg-red-50 border border-red-200'
   874→              }`}
   875→            >
   876→              <div className="flex items-center gap-2 mb-2">
   877→                {aadharResult.verified ? (
   878→                  <><CheckCircle2 className="h-5 w-5 text-green-600" /><p className="text-sm font-semibold text-green-800">Verification Successful</p></>
   879→                ) : (
   880→                  <><XCircle className="h-5 w-5 text-red-600" /><p className="text-sm font-semibold text-red-800">Verification Failed</p></>
   881→                )}
   882→              </div>
   883→              {aadharResult.verified ? (
   884→                <div className="grid grid-cols-1 sm:grid-cols-2 gap-2 mt-3">
   885→                  {aadharResult.name && (
   886→                    <div><p className="text-[10px] text-green-600 font-medium">Name</p><p className="text-sm font-semibold text-green-900">{aadharResult.name}</p></div>
