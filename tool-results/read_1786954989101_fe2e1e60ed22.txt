   126→   126→/* ================================================================== */
   127→   127→export default function LoginModal({ isOpen, onClose, defaultTab }: LoginModalProps) {
   128→   128→  /* ---- state ---- */
   129→   129→  const [tab, setTab] = useState<Tab>(defaultTab === "register" ? "register" : "login");
   130→   130→  const [showPassword, setShowPassword] = useState(false);
   131→   131→  const [loading, setLoading] = useState(false);
   132→   132→  const [googleLoading, setGoogleLoading] = useState(false);
   133→   133→  const [error, setError] = useState("");
   134→   134→  const [resetOpen, setResetOpen] = useState(false);
   135→   135→
   136→   136→  // Login form
   137→   137→  const [loginEmail, setLoginEmail] = useState("");
   138→   138→  const [loginPassword, setLoginPassword] = useState("");
   139→   139→
   140→   140→  // Register form
   141→   141→  const [regName, setRegName] = useState("");
   142→   142→  const [regEmail, setRegEmail] = useState("");
   143→   143→  const [regPhone, setRegPhone] = useState("");
   144→   144→  const [regPassword, setRegPassword] = useState("");
   145→   145→  const [regRole, setRegRole] = useState<UserRole>("FAMILY");
   146→   146→
   147→   147→  // Reset form (simplified — no OTP)
   148→   148→  const [resetEmail, setResetEmail] = useState("");
   149→   149→  const [resetNewPassword, setResetNewPassword] = useState("");
   150→   150→  const [resetLoading, setResetLoading] = useState(false);
   151→   151→  const [resetError, setResetError] = useState("");
   152→   152→  const [resetDone, setResetDone] = useState(false);
   153→   153→
   154→   154→  const { setAuth } = useAuthStore();
   155→   155→
   156→   156→  /* ---- reset form fields when switching tabs ---- */
   157→   157→  useEffect(() => {
   158→   158→    setError("");
   159→   159→    setLoginPassword("");
   160→   160→  }, [tab]);
   161→   161→
   162→   162→  /* ---- reset entire state when modal closes ---- */
   163→   163→  const handleClose = useCallback(() => {
   164→   164→    setTab(defaultTab === "register" ? "register" : "login");
   165→   165→    setError("");
   166→   166→    setShowPassword(false);
   167→   167→    setLoading(false);
   168→   168→    setGoogleLoading(false);
   169→   169→    setLoginEmail("");
   170→   170→    setLoginPassword("");
   171→   171→    setRegName("");
   172→   172→    setRegEmail("");
   173→   173→    setRegPhone("");
   174→   174→    setRegPassword("");
   175→   175→    setRegRole("FAMILY");
   176→   176→    setResetOpen(false);
   177→   177→    setResetEmail("");
   178→   178→    setResetNewPassword("");
   179→   179→    setResetLoading(false);
   180→   180→    setResetError("");
   181→   181→    setResetDone(false);
   182→   182→    onClose();
   183→   183→  }, [onClose, defaultTab]);
   184→   184→
   185→   185→  /* ================================================================ */
   186→   186→  /*  HANDLERS                                                         */
   187→   187→  /* ================================================================ */
   188→   188→
   189→   189→  /* ---- Google sign-in ---- */
   190→   190→  const handleGoogleSignIn = async () => {
   191→   191→    setGoogleLoading(true);
   192→   192→    // Google OAuth requires a production domain with registered API credentials.
   193→   193→    // In this environment the callback URL (localhost) is not accessible from your browser,
   194→   194→    // so the OAuth flow cannot complete. Show a helpful message instead.
   195→   195→    setTimeout(() => {
   196→   196→      toast.error("Google Sign-In is not available in this environment.", {
   197→   197→        duration: 6000,
   198→   198→        description: "Please use email/password to login. Google login will work when deployed to production with a registered domain.",
   199→   199→      });
   200→   200→      setGoogleLoading(false);
   201→   201→    }, 600);
   202→   202→  };
   203→   203→
   204→   204→  /* ---- Email / password login ---- */
   205→   205→  const handleLogin = async (e: React.FormEvent) => {
   206→   206→    e.preventDefault();
   207→   207→    if (!loginEmail.trim() || !loginPassword.trim()) {
   208→   208→      setError("Please enter your email and password.");
   209→   209→      return;
   210→   210→    }
   211→   211→    setLoading(true);
   212→   212→    setError("");
   213→   213→    try {
   214→   214→      const result = await signIn("credentials", {
   215→   215→        email: loginEmail.trim(),
   216→   216→        password: loginPassword,
   217→   217→        redirect: false,
   218→   218→      });
   219→   219→
   220→   220→      if (result?.error) {
   221→   221→        setError("Invalid email or password. Please try again.");
   222→   222→        return;
   223→   223→      }
   224→   224→
   225→   225→      // AuthSync in AuthProvider will automatically sync session → Zustand store
   226→   226→      toast.success("Welcome back! 🎉");
   227→   227→      handleClose();
   228→   228→    } catch {
   229→   229→      setError("Something went wrong. Please try again.");
   230→   230→    } finally {
   231→   231→      setLoading(false);
   232→   232→    }
   233→   233→  };
   234→   234→
   235→   235→  /* ---- Registration ---- */
   236→   236→  const handleRegister = async (e: React.FormEvent) => {
   237→   237→    e.preventDefault();
   238→   238→    if (!regName.trim() || !regEmail.trim() || !regPhone.trim() || !regPassword.trim()) {
   239→   239→      setError("Please fill in all fields.");
   240→   240→      return;
   241→   241→    }
   242→   242→    if (regPassword.length < 6) {
   243→   243→      setError("Password must be at least 6 characters.");
   244→   244→      return;
   245→   245→    }
   246→   246→    setLoading(true);
   247→   247→    setError("");
   248→   248→    try {
   249→   249→      // 1. Register
   250→   250→      const regRes = await api.auth.register({
   251→   251→        name: regName.trim(),
   252→   252→        email: regEmail.trim(),
   253→   253→        phone: regPhone.trim(),
   254→   254→        password: regPassword,
   255→   255→        role: regRole,
   256→   256→      });
   257→   257→
   258→   258→      // 2. Auto-login with credentials
   259→   259→      const loginResult = await signIn("credentials", {
   260→   260→        email: regEmail.trim(),
   261→   261→        password: regPassword,
   262→   262→        redirect: false,
   263→   263→      });
   264→   264→
   265→   265→      if (loginResult?.error) {
   266→   266→        // Registration succeeded but auto-login failed — that's ok, just notify
   267→   267→        toast.success("Account created! Please log in.");
   268→   268→        setTab("login");
   269→   269→        setLoginEmail(regEmail.trim());
   270→   270→        return;
   271→   271→      }
   272→   272→
   273→   273→      toast.success("Welcome to SevaSaathi! 🎉");
   274→   274→      handleClose();
   275→   275→    } catch (err: any) {
   276→   276→      setError(err?.message || "Registration failed. Please try again.");
   277→   277→    } finally {
   278→   278→      setLoading(false);
   279→   279→    }
   280→   280→  };
   281→   281→
   282→   282→  /* ---- Reset password (simplified) ---- */
   283→   283→  const handleReset = async (e: React.FormEvent) => {
   284→   284→    e.preventDefault();
   285→   285→    if (!resetEmail.trim() || !resetNewPassword.trim()) {
   286→   286→      setResetError("Please enter your email and new password.");
   287→   287→      return;
   288→   288→    }
   289→   289→    if (resetNewPassword.length < 6) {
   290→   290→      setResetError("Password must be at least 6 characters.");
   291→   291→      return;
   292→   292→    }
   293→   293→    setResetLoading(true);
   294→   294→    setResetError("");
   295→   295→    try {
   296→   296→      const res = await fetch("/api/auth/reset-password", {
   297→   297→        method: "POST",
   298→   298→        headers: { "Content-Type": "application/json" },
   299→   299→        body: JSON.stringify({ email: resetEmail.trim(), newPassword: resetNewPassword }),
   300→   300→      });
   301→   301→      const data = await res.json();
   302→   302→      if (!res.ok) {
   303→   303→        setResetError(data.error || "Failed to reset password.");
   304→   304→        return;
   305→   305→      }
   306→   306→      setResetDone(true);
   307→   307→      toast.success("Password reset successfully!");
   308→   308→    } catch {
   309→   309→      setResetError("Something went wrong. Please try again.");
   310→   310→    } finally {
   311→   311→      setResetLoading(false);
   312→   312→    }
   313→   313→  };
   314→   314→
   315→   315→  /* ================================================================ */
   316→   316→  /*  RENDER                                                           */
   317→   317→  /* ================================================================ */
   318→   318→  return (
   319→   319→    <Dialog open={isOpen} onOpenChange={(open) => !open && handleClose()}>
   320→   320→      <DialogContent className="sm:max-w-md p-0 overflow-hidden border-forest-200/40">
   321→   321→        {/* Custom X button to override default position */}
   322→   322→        <button
   323→   323→          onClick={handleClose}
   324→   324→          className="absolute top-3 right-3 z-50 rounded-full p-1.5 text-gray-400 hover:text-gray-600 hover:bg-gray-100 transition-colors"
   325→   325→          aria-label="Close"
   326→   326→        >
   327→   327→          <X className="w-5 h-5" />
   328→   328→        </button>
   329→   329→
   330→   330→        <div className="px-6 pt-6 pb-2">
   331→   331→          <DialogHeader className="space-y-1 mb-1">
   332→   332→            <div className="flex items-center gap-2 justify-center">
   333→   333→              <div className="w-9 h-9 rounded-xl bg-forest-900 flex items-center justify-center">
   334→   334→                <Heart className="w-5 h-5 text-lime-400" />
   335→   335→              </div>
   336→   336→              <DialogTitle className="text-xl font-bold text-forest-900">
   337→   337→                SevaSaathi
   338→   338→              </DialogTitle>
   339→   339→            </div>
   340→   340→            <DialogDescription className="text-sm text-gray-500 mt-1">
   341→   341→              {tab === "login"
   342→   342→                ? "Sign in to your account"
   343→   343→                : "Create your account to get started"}
   344→   344→            </DialogDescription>
   345→   345→          </DialogHeader>
   346→   346→        </div>
   347→   347→
   348→   348→        {/* ---- Error banner ---- */}
   349→   349→        <AnimatePresence>
   350→   350→          {error && (
   351→   351→            <motion.div
   352→   352→              initial={{ height: 0, opacity: 0 }}
   353→   353→              animate={{ height: "auto", opacity: 1 }}
   354→   354→              exit={{ height: 0, opacity: 0 }}
   355→   355→              className="overflow-hidden"
   356→   356→            >
   357→   357→              <div className="mx-6 mt-1 px-3 py-2 rounded-lg bg-red-50 border border-red-200 text-red-700 text-sm flex items-center gap-2">
   358→   358→                <X className="w-4 h-4 flex-shrink-0" />
   359→   359→                {error}
   360→   360→              </div>
   361→   361→            </motion.div>
   362→   362→          )}
   363→   363→        </AnimatePresence>
   364→   364→
   365→   365→        {/* ---- Tab switcher ---- */}
   366→   366→        <div className="px-6 pt-3">
   367→   367→          <div className="flex bg-gray-100 rounded-lg p-1">
   368→   368→            <button
   369→   369→              onClick={() => setTab("login")}
   370→   370→              className={`flex-1 py-2 px-4 rounded-md text-sm font-medium transition-all duration-200 ${
   371→   371→                tab === "login"
   372→   372→                  ? "bg-white text-forest-900 shadow-sm"
   373→   373→                  : "text-gray-500 hover:text-gray-700"
   374→   374→              }`}
   375→   375→            >
   376→   376→              Sign In
   377→   377→            </button>
   378→   378→            <button
   379→   379→              onClick={() => setTab("register")}
   380→   380→              className={`flex-1 py-2 px-4 rounded-md text-sm font-medium transition-all duration-200 ${
   381→   381→                tab === "register"
   382→   382→                  ? "bg-white text-forest-900 shadow-sm"
   383→   383→                  : "text-gray-500 hover:text-gray-700"
   384→   384→              }`}
   385→   385→            >
   386→   386→              Create Account
   387→   387→            </button>
   388→   388→          </div>
   389→   389→        </div>
   390→   390→
   391→   391→        {/* ---- Animated content ---- */}
   392→   392→        <div className="px-6 py-4 overflow-hidden">
   393→   393→          <AnimatePresence mode="wait" custom={tab}>
   394→   394→            {/* ========== LOGIN TAB ========== */}
   395→   395→            {tab === "login" && (
   396→   396→              <motion.div
   397→   397→                key="login"
   398→   398→                custom={tab}
   399→   399→                variants={contentVariants}
   400→   400→                initial="enter"
   401→   401→                animate="center"
   402→   402→                exit="exit"
   403→   403→                className="space-y-4"
   404→   404→              >
   405→   405→                {/* Google button */}
   406→   406→                <button
   407→   407→                  type="button"
   408→   408→                  onClick={handleGoogleSignIn}
   409→   409→                  disabled={googleLoading}
   410→   410→                  className="w-full flex items-center justify-center gap-3 h-11 px-4 rounded-lg border border-gray-300 bg-white hover:bg-gray-50 active:bg-gray-100 transition-all duration-150 text-sm font-medium text-gray-700 disabled:opacity-60 disabled:cursor-not-allowed shadow-sm"
   411→   411→                >
   412→   412→                  {googleLoading ? (
   413→   413→                    <Loader2 className="w-5 h-5 animate-spin text-gray-500" />
   414→   414→                  ) : (
   415→   415→                    <GoogleIcon />
   416→   416→                  )}
   417→   417→                  Continue with Google
   418→   418→                </button>
   419→   419→
   420→   420→                {/* Divider */}
   421→   421→                <div className="flex items-center gap-3">
   422→   422→                  <div className="flex-1 h-px bg-gray-200" />
   423→   423→                  <span className="text-xs text-gray-400 font-medium uppercase tracking-wider">
   424→   424→                    or continue with email
   425→   425→                  </span>
   426→   426→                  <div className="flex-1 h-px bg-gray-200" />
   427→   427→                </div>
   428→   428→
   429→   429→                {/* Email / password form */}
   430→   430→                <form onSubmit={handleLogin} className="space-y-3">
   431→   431→                  <div className="space-y-1.5">
   432→   432→                    <Label htmlFor="login-email" className="text-sm font-medium text-gray-700">
   433→   433→                      Email
   434→   434→                    </Label>
   435→   435→                    <div className="relative">
   436→   436→                      <Mail className="absolute left-3 top-1/2 -translate-y-1/2 w-4 h-4 text-gray-400" />
   437→   437→                      <Input
   438→   438→                        id="login-email"
   439→   439→                        type="email"
   440→   440→                        placeholder="you@example.com"
   441→   441→                        value={loginEmail}
   442→   442→                        onChange={(e) => setLoginEmail(e.target.value)}
   443→   443→                        className="pl-10 h-11 border-gray-300 focus:border-forest-500 focus:ring-forest-500"
   444→   444→                        autoComplete="email"
   445→   445→                        disabled={loading}
   446→   446→                      />
   447→   447→                    </div>
   448→   448→                  </div>
   449→   449→
   450→   450→                  <div className="space-y-1.5">
   451→   451→                    <div className="flex items-center justify-between">
   452→   452→                      <Label htmlFor="login-password" className="text-sm font-medium text-gray-700">
   453→   453→                        Password
   454→   454→                      </Label>
   455→   455→                      <button
   456→   456→                        type="button"
   457→   457→                        onClick={() => setResetOpen(true)}
   458→   458→                        className="text-xs font-medium text-forest-700 hover:text-forest-900 transition-colors"
   459→   459→                      >
   460→   460→                        Forgot password?
   461→   461→                      </button>
   462→   462→                    </div>
   463→   463→                    <div className="relative">
   464→   464→                      <Lock className="absolute left-3 top-1/2 -translate-y-1/2 w-4 h-4 text-gray-400" />
   465→   465→                      <Input
   466→   466→                        id="login-password"
   467→   467→                        type={showPassword ? "text" : "password"}
   468→   468→                        placeholder="Enter your password"
   469→   469→                        value={loginPassword}
   470→   470→                        onChange={(e) => setLoginPassword(e.target.value)}
   471→   471→                        className="pl-10 pr-10 h-11 border-gray-300 focus:border-forest-500 focus:ring-forest-500"
   472→   472→                        autoComplete="current-password"
   473→   473→                        disabled={loading}
   474→   474→                      />
   475→   475→                      <button
   476→   476→                        type="button"
   477→   477→                        onClick={() => setShowPassword(!showPassword)}
   478→   478→                        className="absolute right-3 top-1/2 -translate-y-1/2 text-gray-400 hover:text-gray-600 transition-colors"
   479→   479→                        tabIndex={-1}
   480→   480→                      >
   481→   481→                        {showPassword ? <EyeOff className="w-4 h-4" /> : <Eye className="w-4 h-4" />}
   482→   482→                      </button>
   483→   483→                    </div>
   484→   484→                  </div>
   485→   485→
   486→   486→                  <Button
