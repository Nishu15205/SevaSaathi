   135→
   136→  // Login form

   138→  const [loginPassword, setLoginPassword] = useState("");
   139→
   140→  // Register form
   141→  const [regName, setRegName] = useState("");
   142→  const [regEmail, setRegEmail] = useState("");
   143→  const [regPhone, setRegPhone] = useState("");
   144→  const [regPassword, setRegPassword] = useState("");
   145→  const [regRole, setRegRole] = useState<UserRole>("FAMILY");
   146→
   147→  // Reset form (simplified — no OTP)
   148→  const [resetEmail, setResetEmail] = useState("");
   149→  const [resetNewPassword, setResetNewPassword] = useState("");
   150→  const [resetLoading, setResetLoading] = useState(false);
   151→  const [resetError, setResetError] = useState("");
   152→  const [resetDone, setResetDone] = useState(false);
   153→
   154→  const { setAuth } = useAuthStore();
   155→  const { toast } = useToast();
   156→
   157→  // Google popup form
   158→  const [googleOpen, setGoogleOpen] = useState(false);
   159→  const [googleEmail, setGoogleEmail] = useState("");
   160→  const [googleName, setGoogleName] = useState("");
   161→  const [googleError, setGoogleError] = useState("");
   162→  const [googleSubmitting, setGoogleSubmitting] = useState(false);
   163→

   190→    setGoogleOpen(false);
   191→    setGoogleEmail("");
   192→    setGoogleName("");
   193→    setGoogleError("");

   197→
   198→  /* ================================================================ */
   199→  /*  HANDLERS                                                         */
   200→  /* ================================================================ */
   201→
   202→  /* ---- Google sign-in — real OAuth popup, fallback to simulated ---- */
   203→  const handleGoogleSignIn = async () => {
   204→    setGoogleLoading(true);
   205→    setGoogleError("");
   206→    try {
   207→      // Check if real Google OAuth is configured
   208→      const checkRes = await fetch("/api/auth/google-configured");
   209→      const { configured } = await checkRes.json();
   210→
   211→      if (configured) {
   212→        // Real Google OAuth — use custom API to get correct redirect_uri
   213→        // Pass window.location.origin so the server uses the browser's actual URL
   214→        const origin = window.location.origin;
   215→        const goRes = await fetch(`/api/auth/google-go?origin=${encodeURIComponent(origin)}`);
   216→        const goData = await goRes.json();
   217→
   218→        if (!goData.url) {
   219→          setGoogleError("Failed to start Google sign-in. Please try again.");
   220→          setGoogleLoading(false);
   221→          return;
   222→        }
   223→
   224→        const width = 500;
   225→        const height = 650;
   226→        const left = (window.screen.width - width) / 2;
   227→        const top = (window.screen.height - height) / 2;
   228→
   229→        const popup = window.open(
   230→          goData.url,
   231→          "google-auth",
   232→          `width=${width},height=${height},left=${left},top=${top},scrollbars=yes,resizable=yes`
   233→        );
   234→
   235→        if (!popup) {
   236→          setGoogleError(
   237→            "Popup blocked! Please allow popups for this site and try again."
   238→          );
   239→          setGoogleLoading(false);
   240→          return;
   241→        }
   242→
   243→        // Listen for postMessage from popup callback
   244→        let settled = false;
   245→        const cleanup = () => {
   246→          if (settled) return;
   247→          settled = true;
   248→          window.removeEventListener("message", handleMessage);
   249→          clearInterval(checkInterval);
   250→        };
   251→
   252→        const completeLogin = async () => {
   253→          cleanup();
   254→          const sessionRes = await fetch("/api/auth/session");
   255→          const session = await sessionRes.json();
   256→          if (session?.user) {
   257→            setAuth({
   258→              id: session.user.id || "",
   259→              email: session.user.email || "",
   260→              name: session.user.name || "",
   261→              phone: "",
   262→              role: (session.user as any).role || "FAMILY",
   263→              avatarUrl: session.user.image || null,
   264→              subscription: "NONE",
   265→              patientProfiles: [],
   266→              caregiverProfile: null,
   267→            });
   268→            toast({
   269→              title: "Welcome! 🎉",
   270→              description: "Signed in with Google as " + session.user.email,
   271→            });
   272→            setGoogleLoading(false);
   273→            handleClose();
   274→          } else {
   275→            setGoogleLoading(false);
   276→          }
   277→        };
   278→
   279→        const handleMessage = async (event: MessageEvent) => {
   280→          if (event.data?.type === "google-oauth-callback") {
   281→            cleanup();
   282→            if (event.data.success) {
   283→              await completeLogin();
   284→            } else {
   285→              setGoogleError("Google sign-in was cancelled or failed.");
   286→              setGoogleLoading(false);
   287→            }
   288→          }
   289→        };
   290→        window.addEventListener("message", handleMessage);
   291→
   292→        // Fallback: poll for session if postMessage doesn't fire
   293→        const checkInterval = setInterval(async () => {
   294→          if (popup.closed) {
   295→            cleanup();
   296→            setGoogleLoading(false);
   297→            return;
   298→          }
   299→          try {
   300→            const res = await fetch("/api/auth/session");
   301→            const session = await res.json();
   302→            if (session?.user) {
   303→              await completeLogin();
   304→            }
   305→          } catch {
   306→            // keep polling
   307→          }
   308→        }, 2000);
   309→
   310→        // Safety timeout — 5 minutes
   311→        setTimeout(() => {
   312→          cleanup();
   313→          setGoogleLoading(false);
   314→        }, 5 * 60 * 1000);
   315→        return;
   316→      }
   317→    } catch {
   318→      // If check fails, fall through to simulated flow
   319→    }
   320→    setGoogleLoading(false);
   321→    // Google OAuth not configured — open simulated sign-in
   322→    setGoogleOpen(true);
   323→  };
   324→
   325→  /* ---- Google popup submit ---- */
   326→  const handleGoogleSubmit = async (e: React.FormEvent) => {
   327→    e.preventDefault();
   328→    if (!googleEmail.trim() || !googleName.trim()) {
   329→      setGoogleError("Please enter your name and email.");
   330→      return;
   331→    }
   332→    if (!googleEmail.trim().includes("@")) {
   333→      setGoogleError("Please enter a valid email address.");
   334→      return;
   335→    }
   336→    setGoogleSubmitting(true);
   337→    setGoogleError("");
   338→    try {
   339→      // 1. Call the simulate endpoint (sets the session cookie)
   340→      const res = await fetch("/api/auth/google-simulate", {
   341→        method: "POST",
   342→        headers: { "Content-Type": "application/json" },
   343→        body: JSON.stringify({ email: googleEmail.trim(), name: googleName.trim() }),
   344→      });
   345→      const data = await res.json();
   346→      if (!res.ok) {
   347→        setGoogleError(data.error || "Google sign-in failed.");
   348→        return;
   349→      }
   350→
   351→      // 2. Fetch the NextAuth session (cookie is already set by step 1)
   352→      const sessionRes = await fetch("/api/auth/session");
   353→      const sessionData = await sessionRes.json();
   354→
   355→      if (sessionData?.user) {
   356→        // 3. Directly update the Zustand store — no page reload needed
   357→        setAuth({
   358→          id: sessionData.user.id,
   359→          email: sessionData.user.email,
   360→          name: sessionData.user.name,
   361→          phone: "",
   362→          role: sessionData.user.role,
   363→          avatarUrl: sessionData.user.image || null,
   364→          subscription: "NONE" as any,
   365→          patientProfiles: [],
   366→          caregiverProfile: null,
   367→        });
   368→        setGoogleOpen(false);
   369→        toast({
   370→          title: "Welcome" + (data.user.isNewUser ? " to SevaSaathi!" : " back!") + " \ud83c\udf89",
   371→          description: "Signed in as " + data.user.email,
   372→        });
   373→        handleClose();
   374→      } else {
   375→        // Fallback: reload the page so NextAuth picks up the cookie
   376→        window.location.href = window.location.origin;
   377→      }
   378→    } catch {
   379→      setGoogleError("Something went wrong. Please try again.");
   380→    } finally {
   381→      setGoogleSubmitting(false);
   382→    }
   383→  };
   384→
   385→  /* ---- Email / password login ---- */
   386→  const handleLogin = async (e: React.FormEvent) => {
   387→    e.preventDefault();
   388→    if (!loginEmail.trim() || !loginPassword.trim()) {
   389→      setError("Please enter your email and password.");
   390→      return;
   391→    }
   392→    setLoading(true);
   393→    setError("");
   394→    try {
   395→      const result = await signIn("credentials", {
   396→        email: loginEmail.trim(),
   397→        password: loginPassword,
   398→        redirect: false,
   399→      });
   400→
   401→      if (result?.error) {
   402→        setError("Invalid email or password. Please try again.");
   403→        return;
   404→      }
   405→
   406→      // Fetch session to get user data, then set auth explicitly
   407→      try {
   408→        const sessionRes = await fetch("/api/auth/session");
   409→        const sessionData = await sessionRes.json();
   410→        if (sessionData?.user) {
   411→          const u = sessionData.user as any;
   412→          setAuth({
   413→            id: u.id || "",
   414→            email: u.email || "",
   415→            name: u.name || "",
   416→            phone: "",
   417→            role: u.role || "FAMILY",
   418→            avatarUrl: u.image || null,
   419→            subscription: "NONE",
   420→            patientProfiles: [],
   421→            caregiverProfile: null,
   422→          });
   423→        }
   424→      } catch {
   425→        // AuthSync will pick it up as fallback
   426→      }
   427→
   428→      toast({
   429→        title: "Welcome back! 🎉",
   430→        description: "You have successfully logged in.",
   431→      });
   432→      handleClose();
   433→    } catch {
   434→      setError("Something went wrong. Please try again.");
   435→    } finally {
   436→      setLoading(false);
   437→    }
   438→  };
   439→
   440→  /* ---- Registration ---- */
   441→  const handleRegister = async (e: React.FormEvent) => {
   442→    e.preventDefault();
   443→    if (!regName.trim() || !regEmail.trim() || !regPhone.trim() || !regPassword.trim()) {
   444→      setError("Please fill in all fields.");
   445→      return;
   446→    }
   447→    if (regPassword.length < 6) {
   448→      setError("Password must be at least 6 characters.");
   449→      return;
   450→    }
   451→    setLoading(true);
   452→    setError("");
   453→    try {
   454→      // 1. Register
   455→      const regRes = await api.auth.register({
   456→        name: regName.trim(),
   457→        email: regEmail.trim(),
   458→        phone: regPhone.trim(),
   459→        password: regPassword,
   460→        role: regRole,
   461→      });
   462→
   463→      // 2. Auto-login with credentials
   464→      const loginResult = await signIn("credentials", {
   465→        email: regEmail.trim(),
   466→        password: regPassword,
   467→        redirect: false,
   468→      });
   469→
   470→      if (loginResult?.error) {
   471→        // Registration succeeded but auto-login failed — that's ok, just notify
   472→        toast({ title: "Account created!", description: "Please log in with your credentials." });
   473→        setTab("login");
   474→        setLoginEmail(regEmail.trim());
   475→        return;
   476→      }
   477→
   478→      // Fetch session to get user data, then set auth explicitly
   479→      try {
   480→        const sessionRes = await fetch("/api/auth/session");
   481→        const sessionData = await sessionRes.json();
   482→        if (sessionData?.user) {
   483→          const u = sessionData.user as any;
   484→          setAuth({
   485→            id: u.id || "",
   486→            email: u.email || "",
   487→            name: u.name || "",
   488→            phone: regPhone.trim(),
   489→            role: u.role || regRole,
   490→            avatarUrl: u.image || null,
   491→            subscription: "NONE",
   492→            patientProfiles: [],
   493→            caregiverProfile: null,
   494→          });
   495→        }
   496→      } catch {
   497→        // AuthSync will pick it up as fallback
   498→      }
   499→
   500→      toast({ title: "Welcome to SevaSaathi! 🎉", description: "Your account has been created." });
   501→      handleClose();
   502→    } catch (err: any) {
   503→      setError(err?.message || "Registration failed. Please try again.");
   504→    } finally {
   505→      setLoading(false);
   506→    }
   507→  };
   508→
   509→  /* ---- Reset password (simplified) ---- */
   510→  const handleReset = async (e: React.FormEvent) => {
   511→    e.preventDefault();
   512→    if (!resetEmail.trim() || !resetNewPassword.trim()) {
   513→      setResetError("Please enter your email and new password.");
   514→      return;
   515→    }
   516→    if (resetNewPassword.length < 6) {
   517→      setResetError("Password must be at least 6 characters.");
   518→      return;
   519→    }
   520→    setResetLoading(true);
   521→    setResetError("");
   522→    try {
   523→      const res = await fetch("/api/auth/reset-password", {
   524→        method: "POST",
   525→        headers: { "Content-Type": "application/json" },
   526→        body: JSON.stringify({ email: resetEmail.trim(), newPassword: resetNewPassword }),
   527→      });
   528→      const data = await res.json();
   529→      if (!res.ok) {
   530→        setResetError(data.error || "Failed to reset password.");
   531→        return;
   532→      }
   533→      setResetDone(true);
   534→      toast({ title: "Password reset successfully!", description: "You can now log in with your new password." });
   535→    } catch {
   536→      setResetError("Something went wrong. Please try again.");
   537→    } finally {
   538→      setResetLoading(false);
   539→    }
   540→  };
   541→

   546→    <Dialog open={isOpen} onOpenChange={(open) => !open && handleClose()}>
   547→      <DialogContent showCloseButton={false} className="sm:max-w-md p-0 overflow-hidden border-forest-200/40">
   548→        {/* Single custom close button */}
   549→        <button
   550→          onClick={handleClose}
   551→          className="absolute top-3 right-3 z-50 rounded-full p-1.5 text-gray-400 hover:text-gray-600 hover:bg-gray-100 transition-colors"
   552→          aria-label="Close"
   553→        >
   554→          <X className="w-5 h-5" />
   555→        </button>
   556→
   557→        <div className="px-6 pt-6 pb-2">
   558→          <DialogHeader className="space-y-1 mb-1">
   559→            <div className="flex items-center gap-2 justify-center">
   560→              <div className="w-9 h-9 rounded-xl bg-forest-900 flex items-center justify-center">
   561→                <Heart className="w-5 h-5 text-lime-400" />
   562→              </div>
   563→              <DialogTitle className="text-xl font-bold text-forest-900">
   564→                SevaSaathi
   565→              </DialogTitle>
   566→            </div>
   567→            <DialogDescription className="text-sm text-gray-500 mt-1">
   568→              {tab === "login"
   569→                ? "Sign in to your account"
   570→                : "Create your account to get started"}
   571→            </DialogDescription>
   572→          </DialogHeader>
   573→        </div>
   574→
   575→        {/* ---- Error banner ---- */}
   576→        <AnimatePresence>
   577→          {error && (
   578→            <motion.div
   579→              initial={{ height: 0, opacity: 0 }}
   580→              animate={{ height: "auto", opacity: 1 }}
   581→              exit={{ height: 0, opacity: 0 }}
   582→              className="overflow-hidden"
   583→            >
   584→              <div className="mx-6 mt-1 px-3 py-2 rounded-lg bg-red-50 border border-red-200 text-red-700 text-sm flex items-center gap-2">
   585→                <X className="w-4 h-4 flex-shrink-0" />
   586→                {error}
   587→              </div>
   588→            </motion.div>
   589→          )}
   590→        </AnimatePresence>
   591→
   592→        {/* ---- Tab switcher ---- */}
   593→        <div className="px-6 pt-3">
   594→          <div className="flex bg-gray-100 rounded-lg p-1">
   595→            <button
   596→              onClick={() => setTab("login")}
   597→              className={`flex-1 py-2 px-4 rounded-md text-sm font-medium transition-all duration-200 ${
   598→                tab === "login"
   599→                  ? "bg-white text-forest-900 shadow-sm"
   600→                  : "text-gray-500 hover:text-gray-700"
   601→              }`}
   602→            >
   603→              Sign In
   604→            </button>
   605→            <button
   606→              onClick={() => setTab("register")}
   607→              className={`flex-1 py-2 px-4 rounded-md text-sm font-medium transition-all duration-200 ${
   608→                tab === "register"
   609→                  ? "bg-white text-forest-900 shadow-sm"
   610→                  : "text-gray-500 hover:text-gray-700"
   611→              }`}
   612→            >
   613→              Create Account
   614→            </button>
   615→          </div>
   616→        </div>
   617→
