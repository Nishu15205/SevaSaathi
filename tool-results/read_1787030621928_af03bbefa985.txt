   131→  const [loading, setLoading] = useState(false);

   176→    setRegRole("FAMILY");

   206→        return;
   207→      }
   208→
   209→      // Credentials are configured — proceed with real OAuth flow
   210→      await signIn("google", {
   211→        callbackUrl: window.location.origin,
   212→      });
   213→    } catch (err: any) {

   215→        title: "Google Sign-In Failed",

   284→        password: regPassword,
   285→        redirect: false,
   286→      });

   298→    } catch (err: any) {
   299→      setError(err?.message || "Registration failed. Please try again.");

   302→    }

   388→        {/* ---- Tab switcher ---- */}
