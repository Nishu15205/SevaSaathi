   124→/* ================================================================== */
   125→/*  MAIN COMPONENT                                                     */
   126→/* ================================================================== */
   127→export default function LoginModal({ isOpen, onClose, defaultTab }: LoginModalProps) {
   128→  /* ---- state ---- */
   129→  const [tab, setTab] = useState<Tab>(defaultTab === "register" ? "register" : "login");
   130→  const [showPassword, setShowPassword] = useState(false);
   131→  const [loading, setLoading] = useState(false);
   132→  const [googleLoading, setGoogleLoading] = useState(false);
   133→  const [error, setError] = useState("");
   134→  const [resetOpen, setResetOpen] = useState(false);
   135→
   136→  // Login form
   137→  const [loginEmail, setLoginEmail] = useState("");
   138→  const [loginPassword, setLoginPassword] = useState("");
   139→
   140→  // Register form
   141→  const [regName, setRegName] = useState("");
   142→  const [regEmail, setRegEmail] = useState("");
   143→  const [regPhone, setRegPhone] = useState("");
   144→  const [regPassword, setRegPassword] = useState("");
   145→  const [regRole, setRegRole] = useState<UserRole>("FAMILY");
   146→
   147→  // Reset form (simplified — no OTP)
   148→  const [resetEmail, setResetEmail] = useState("");
   149→  const [resetNewPassword, setResetNewPassword] = useState("");
   150→  const [resetLoading, setResetLoading] = useState(false);
   151→  const [resetError, setResetError] = useState("");
   152→  const [resetDone, setResetDone] = useState(false);
   153→
   154→  const { setAuth } = useAuthStore();
   155→
   156→  /* ---- reset form fields when switching tabs ---- */
   157→  useEffect(() => {
   158→    setError("");
   159→    setLoginPassword("");
   160→  }, [tab]);
   161→
   162→  /* ---- reset entire state when modal closes ---- */
   163→  const handleClose = useCallback(() => {
   164→    setError("");
   165→    setShowPassword(false);
   166→    setLoading(false);
   167→    setGoogleLoading(false);
   168→    setLoginEmail("");
   169→    setLoginPassword("");
   170→    setRegName("");
   171→    setRegEmail("");
   172→    setRegPhone("");
   173→    setRegPassword("");
   174→    setRegRole("FAMILY");
   175→    setResetOpen(false);
   176→    setResetEmail("");
   177→    setResetNewPassword("");
   178→    setResetLoading(false);
   179→    setResetError("");
   180→    setResetDone(false);
   181→    onClose();
   182→  }, [onClose]);
   183→
   184→  /* ================================================================ */
   185→  /*  HANDLERS                                                         */
   186→  /* ================================================================ */
   187→
   188→  /* ---- Google sign-in ---- */
   189→  const handleGoogleSignIn = async () => {
   190→    setGoogleLoading(true);
   191→    setError("");
   192→    try {
   193→      await signIn("google", { redirect: false });
   194→      toast.success("Welcome to SevaSaathi!");
   195→      handleClose();
   196→    } catch {
   197→      setError("Google sign-in failed. Please try again.");
   198→    } finally {
   199→      setGoogleLoading(false);
   200→    }
   201→  };
   202→
   203→  /* ---- Email / password login ---- */
   204→  const handleLogin = async (e: React.FormEvent) => {
   205→    e.preventDefault();
   206→    if (!loginEmail.trim() || !loginPassword.trim()) {
   207→      setError("Please enter your email and password.");
   208→      return;
   209→    }
   210→    setLoading(true);
   211→    setError("");
   212→    try {
   213→      const result = await signIn("credentials", {
   214→        email: loginEmail.trim(),
   215→        password: loginPassword,
   216→        redirect: false,
   217→      });
   218→
   219→      if (result?.error) {
   220→        setError("Invalid email or password. Please try again.");
   221→        return;
   222→      }
   223→
   224→      // AuthSync in AuthProvider will automatically sync session → Zustand store
   225→      toast.success("Welcome back! 🎉");
   226→      handleClose();
   227→    } catch {
   228→      setError("Something went wrong. Please try again.");
   229→    } finally {
   230→      setLoading(false);
   231→    }
   232→  };
   233→
   234→  /* ---- Registration ---- */
   235→  const handleRegister = async (e: React.FormEvent) => {
   236→    e.preventDefault();
   237→    if (!regName.trim() || !regEmail.trim() || !regPhone.trim() || !regPassword.trim()) {
   238→      setError("Please fill in all fields.");
   239→      return;
   240→    }
   241→    if (regPassword.length < 6) {
   242→      setError("Password must be at least 6 characters.");
   243→      return;
   244→    }
   245→    setLoading(true);
   246→    setError("");
   247→    try {
   248→      // 1. Register
   249→      const regRes = await api.auth.register({
   250→        name: regName.trim(),
   251→        email: regEmail.trim(),
   252→        phone: regPhone.trim(),
   253→        password: regPassword,
   254→        role: regRole,
   255→      });
   256→
   257→      // 2. Auto-login with credentials
   258→      const loginResult = await signIn("credentials", {
   259→        email: regEmail.trim(),
   260→        password: regPassword,
   261→        redirect: false,
   262→      });
   263→
   264→      if (loginResult?.error) {
   265→        // Registration succeeded but auto-login failed — that's ok, just notify
   266→        toast.success("Account created! Please log in.");
   267→        setTab("login");
   268→        setLoginEmail(regEmail.trim());
   269→        return;
   270→      }
   271→
   272→      toast.success("Welcome to SevaSaathi! 🎉");
   273→      handleClose();
   274→    } catch (err: any) {
   275→      setError(err?.message || "Registration failed. Please try again.");
   276→    } finally {
   277→      setLoading(false);
   278→    }
   279→  };
   280→
   281→  /* ---- Reset password (simplified) ---- */
   282→  const handleReset = async (e: React.FormEvent) => {
   283→    e.preventDefault();
   284→    if (!resetEmail.trim() || !resetNewPassword.trim()) {
   285→      setResetError("Please enter your email and new password.");
   286→      return;
   287→    }
   288→    if (resetNewPassword.length < 6) {
   289→      setResetError("Password must be at least 6 characters.");
   290→      return;
   291→    }
   292→    setResetLoading(true);
   293→    setResetError("");
   294→    try {
   295→      const res = await fetch("/api/auth/reset-password", {
   296→        method: "POST",
   297→        headers: { "Content-Type": "application/json" },
   298→        body: JSON.stringify({ email: resetEmail.trim(), newPassword: resetNewPassword }),
   299→      });
   300→      const data = await res.json();
   301→      if (!res.ok) {
   302→        setResetError(data.error || "Failed to reset password.");
   303→        return;
   304→      }
   305→      setResetDone(true);
   306→      toast.success("Password reset successfully!");
   307→    } catch {
   308→      setResetError("Something went wrong. Please try again.");
   309→    } finally {
   310→      setResetLoading(false);
   311→    }
   312→  };
   313→
   314→  /* ================================================================ */
   315→  /*  RENDER                                                           */
   316→  /* ================================================================ */
   317→  return (
   318→    <Dialog open={isOpen} onOpenChange={(open) => !open && handleClose()}>
   319→      <DialogContent className="sm:max-w-md p-0 overflow-hidden border-forest-200/40">
   320→        {/* Custom X button to override default position */}
   321→        <button
   322→          onClick={handleClose}
   323→          className="absolute top-3 right-3 z-50 rounded-full p-1.5 text-gray-400 hover:text-gray-600 hover:bg-gray-100 transition-colors"
   324→          aria-label="Close"
   325→        >
   326→          <X className="w-5 h-5" />
   327→        </button>
   328→
   329→        <div className="px-6 pt-6 pb-2">
   330→          <DialogHeader className="space-y-1 mb-1">
   331→            <div className="flex items-center gap-2 justify-center">
   332→              <div className="w-9 h-9 rounded-xl bg-forest-900 flex items-center justify-center">
   333→                <Heart className="w-5 h-5 text-lime-400" />
   334→              </div>
   335→              <DialogTitle className="text-xl font-bold text-forest-900">
   336→                SevaSaathi
   337→              </DialogTitle>
   338→            </div>
   339→            <DialogDescription className="text-sm text-gray-500 mt-1">
   340→              {tab === "login"
   341→                ? "Sign in to your account"
   342→                : "Create your account to get started"}
   343→            </DialogDescription>
   344→          </DialogHeader>
   345→        </div>
   346→
   347→        {/* ---- Error banner ---- */}
   348→        <AnimatePresence>
   349→          {error && (
   350→            <motion.div
   351→              initial={{ height: 0, opacity: 0 }}
   352→              animate={{ height: "auto", opacity: 1 }}
   353→              exit={{ height: 0, opacity: 0 }}
   354→              className="overflow-hidden"
   355→            >
   356→              <div className="mx-6 mt-1 px-3 py-2 rounded-lg bg-red-50 border border-red-200 text-red-700 text-sm flex items-center gap-2">
   357→                <X className="w-4 h-4 flex-shrink-0" />
   358→                {error}
   359→              </div>
   360→            </motion.div>
   361→          )}
   362→        </AnimatePresence>
   363→
   364→        {/* ---- Tab switcher ---- */}
   365→        <div className="px-6 pt-3">
   366→          <div className="flex bg-gray-100 rounded-lg p-1">
   367→            <button
   368→              onClick={() => setTab("login")}
   369→              className={`flex-1 py-2 px-4 rounded-md text-sm font-medium transition-all duration-200 ${
   370→                tab === "login"
   371→                  ? "bg-white text-forest-900 shadow-sm"
   372→                  : "text-gray-500 hover:text-gray-700"
   373→              }`}
   374→            >
   375→              Sign In
   376→            </button>
   377→            <button
   378→              onClick={() => setTab("register")}
   379→              className={`flex-1 py-2 px-4 rounded-md text-sm font-medium transition-all duration-200 ${
   380→                tab === "register"
   381→                  ? "bg-white text-forest-900 shadow-sm"
   382→                  : "text-gray-500 hover:text-gray-700"
   383→              }`}
   384→            >
   385→              Create Account
   386→            </button>
   387→          </div>
   388→        </div>
   389→
   390→        {/* ---- Animated content ---- */}
   391→        <div className="px-6 py-4 overflow-hidden">
   392→          <AnimatePresence mode="wait" custom={tab}>
   393→            {/* ========== LOGIN TAB ========== */}
   394→            {tab === "login" && (
   395→              <motion.div
   396→                key="login"
   397→                custom={tab}
   398→                variants={contentVariants}
   399→                initial="enter"
   400→                animate="center"
   401→                exit="exit"
   402→                className="space-y-4"
   403→              >
   404→                {/* Google button */}
   405→                <button
   406→                  type="button"
   407→                  onClick={handleGoogleSignIn}
   408→                  disabled={googleLoading}
   409→                  className="w-full flex items-center justify-center gap-3 h-11 px-4 rounded-lg border border-gray-300 bg-white hover:bg-gray-50 active:bg-gray-100 transition-all duration-150 text-sm font-medium text-gray-700 disabled:opacity-60 disabled:cursor-not-allowed shadow-sm"
   410→                >
   411→                  {googleLoading ? (
   412→                    <Loader2 className="w-5 h-5 animate-spin text-gray-500" />
   413→                  ) : (
   414→                    <GoogleIcon />
   415→                  )}
   416→                  Continue with Google
   417→                </button>
   418→
   419→                {/* Divider */}
   420→                <div className="flex items-center gap-3">
   421→                  <div className="flex-1 h-px bg-gray-200" />
   422→                  <span className="text-xs text-gray-400 font-medium uppercase tracking-wider">
   423→                    or continue with email
   424→                  </span>
   425→                  <div className="flex-1 h-px bg-gray-200" />
   426→                </div>
   427→
   428→                {/* Email / password form */}
   429→                <form onSubmit={handleLogin} className="space-y-3">
   430→                  <div className="space-y-1.5">
   431→                    <Label htmlFor="login-email" className="text-sm font-medium text-gray-700">
   432→                      Email
   433→                    </Label>
   434→                    <div className="relative">
   435→                      <Mail className="absolute left-3 top-1/2 -translate-y-1/2 w-4 h-4 text-gray-400" />
   436→                      <Input
   437→                        id="login-email"
   438→                        type="email"
   439→                        placeholder="you@example.com"
   440→                        value={loginEmail}
   441→                        onChange={(e) => setLoginEmail(e.target.value)}
   442→                        className="pl-10 h-11 border-gray-300 focus:border-forest-500 focus:ring-forest-500"
   443→                        autoComplete="email"
   444→                        disabled={loading}
   445→                      />
   446→                    </div>
   447→                  </div>
   448→
   449→                  <div className="space-y-1.5">
   450→                    <div className="flex items-center justify-between">
   451→                      <Label htmlFor="login-password" className="text-sm font-medium text-gray-700">
   452→                        Password
   453→                      </Label>
   454→                      <button
   455→                        type="button"
   456→                        onClick={() => setResetOpen(true)}
   457→                        className="text-xs font-medium text-forest-700 hover:text-forest-900 transition-colors"
   458→                      >
   459→                        Forgot password?
   460→                      </button>
   461→                    </div>
   462→                    <div className="relative">
   463→                      <Lock className="absolute left-3 top-1/2 -translate-y-1/2 w-4 h-4 text-gray-400" />
   464→                      <Input
   465→                        id="login-password"
   466→                        type={showPassword ? "text" : "password"}
   467→                        placeholder="Enter your password"
   468→                        value={loginPassword}
   469→                        onChange={(e) => setLoginPassword(e.target.value)}
   470→                        className="pl-10 pr-10 h-11 border-gray-300 focus:border-forest-500 focus:ring-forest-500"
   471→                        autoComplete="current-password"
   472→                        disabled={loading}
   473→                      />
   474→                      <button
   475→                        type="button"
   476→                        onClick={() => setShowPassword(!showPassword)}
   477→                        className="absolute right-3 top-1/2 -translate-y-1/2 text-gray-400 hover:text-gray-600 transition-colors"
   478→                        tabIndex={-1}
   479→                      >
   480→                        {showPassword ? <EyeOff className="w-4 h-4" /> : <Eye className="w-4 h-4" />}
   481→                      </button>
   482→                    </div>
   483→                  </div>
   484→
   485→                  <Button
   486→                    type="submit"
   487→                    disabled={loading}
   488→                    className="w-full h-11 bg-forest-900 hover:bg-forest-800 text-white font-semibold rounded-lg transition-all duration-150 shadow-sm hover:shadow-md"
   489→                  >
   490→                    {loading ? (
   491→                      <>
   492→                        <Loader2 className="w-4 h-4 mr-2 animate-spin" />
   493→                        Signing in...
   494→                      </>
   495→                    ) : (
   496→                      <>
   497→                        Sign In
   498→                        <ArrowRight className="w-4 h-4 ml-2" />
   499→                      </>
   500→                    )}
   501→                  </Button>
   502→                </form>
   503→
   504→                {/* Switch to register */}
   505→                <p className="text-center text-sm text-gray-500 pt-1">
   506→                  Don&apos;t have an account?{" "}
   507→                  <button
   508→                    onClick={() => setTab("register")}
   509→                    className="font-semibold text-forest-700 hover:text-forest-900 transition-colors"
   510→                  >
   511→                    Create one
   512→                  </button>
   513→                </p>
   514→              </motion.div>
   515→            )}
   516→
   517→            {/* ========== REGISTER TAB ========== */}
   518→            {tab === "register" && (
   519→              <motion.div
   520→                key="register"
   521→                custom={tab}
   522→                variants={contentVariants}
   523→                initial="enter"
   524→                animate="center"
   525→                exit="exit"
   526→                className="space-y-4"
   527→              >
   528→                {/* Google button */}
   529→                <button
   530→                  type="button"
   531→                  onClick={handleGoogleSignIn}
   532→                  disabled={googleLoading}
   533→                  className="w-full flex items-center justify-center gap-3 h-11 px-4 rounded-lg border border-gray-300 bg-white hover:bg-gray-50 active:bg-gray-100 transition-all duration-150 text-sm font-medium text-gray-700 disabled:opacity-60 disabled:cursor-not-allowed shadow-sm"
   534→                >
   535→                  {googleLoading ? (
   536→                    <Loader2 className="w-5 h-5 animate-spin text-gray-500" />
   537→                  ) : (
   538→                    <GoogleIcon />
   539→                  )}
   540→                  Sign up with Google
   541→                </button>
   542→
   543→                {/* Divider */}
   544→                <div className="flex items-center gap-3">
   545→                  <div className="flex-1 h-px bg-gray-200" />
   546→                  <span className="text-xs text-gray-400 font-medium uppercase tracking-wider">
   547→                    or register with email
   548→                  </span>
   549→                  <div className="flex-1 h-px bg-gray-200" />
   550→                </div>
   551→
   552→                {/* Registration form */}
   553→                <form onSubmit={handleRegister} className="space-y-3">
   554→                  {/* Name */}
   555→                  <div className="space-y-1.5">
   556→                    <Label htmlFor="reg-name" className="text-sm font-medium text-gray-700">
   557→                      Full Name
   558→                    </Label>
   559→                    <div className="relative">
   560→                      <User className="absolute left-3 top-1/2 -translate-y-1/2 w-4 h-4 text-gray-400" />
   561→                      <Input
   562→                        id="reg-name"
   563→                        type="text"
   564→                        placeholder="Your full name"
   565→                        value={regName}
   566→                        onChange={(e) => setRegName(e.target.value)}
   567→                        className="pl-10 h-11 border-gray-300 focus:border-forest-500 focus:ring-forest-500"
   568→                        autoComplete="name"
   569→                        disabled={loading}
   570→                      />
   571→                    </div>
   572→                  </div>
   573→
   574→                  {/* Email */}
   575→                  <div className="space-y-1.5">
   576→                    <Label htmlFor="reg-email" className="text-sm font-medium text-gray-700">
   577→                      Email
   578→                    </Label>
   579→                    <div className="relative">
   580→                      <Mail className="absolute left-3 top-1/2 -translate-y-1/2 w-4 h-4 text-gray-400" />
   581→                      <Input
   582→                        id="reg-email"
   583→                        type="email"
   584→                        placeholder="you@example.com"
   585→                        value={regEmail}
   586→                        onChange={(e) => setRegEmail(e.target.value)}
   587→                        className="pl-10 h-11 border-gray-300 focus:border-forest-500 focus:ring-forest-500"
   588→                        autoComplete="email"
   589→                        disabled={loading}
   590→                      />
   591→                    </div>
   592→                  </div>
   593→
   594→                  {/* Phone */}
   595→                  <div className="space-y-1.5">
