   209→function parseJsonSafe(jsonStr: string | null | undefined): string[] {
   210→  if (!jsonStr) return [];
   211→  try {
   212→    const parsed = JSON.parse(jsonStr);
   213→    return Array.isArray(parsed) ? parsed : [];
   214→  } catch {
   215→    return jsonStr.split(',').map((s) => s.trim()).filter(Boolean);
   216→  }
   217→}

   219→function formatDate(dateStr: string): string {
   220→  try {
   221→    return new Date(dateStr).toLocaleDateString('en-IN', {
   222→      day: 'numeric',
   223→      month: 'short',
   224→      year: 'numeric',
   225→    });
   226→  } catch {
   227→    return dateStr;
   228→  }
   229→}

   231→function getTodayDate(): string {
   232→  const now = new Date();
   233→  const year = now.getFullYear();
   234→  const month = String(now.getMonth() + 1).padStart(2, '0');
   235→  const day = String(now.getDate()).padStart(2, '0');
   236→  return `${year}-${month}-${day}`;
   237→}

   243→function AadhaarQuickVerifyDialog({ open, onClose, userId }: { open: boolean; onClose: () => void; userId: string }) {
   244→  const [aadharResult, setAadharResult] = useState<any>(null);
   245→  const [verifying, setVerifying] = useState(false);
   246→  const [imagePreview, setImagePreview] = useState<string | null>(null);
   247→
   248→  const handleFileSelect = async (e: React.ChangeEvent<HTMLInputElement>) => {
   249→    const file = e.target.files?.[0];
   250→    if (!file) return;
   251→    const reader = new FileReader();
   252→    reader.onload = (ev) => {
   253→      setImagePreview(ev.target?.result as string);
   254→      setAadharResult(null);
   255→    };
   256→    reader.readAsDataURL(file);
   257→  };
   258→
   259→  const handleVerify = async () => {
   260→    if (!imagePreview) { toast.error('Please upload an Aadhaar card image first'); return; }
   261→    setVerifying(true);
   262→    try {
   263→      const result = await api.verifyAadhar(imagePreview);
   264→      setAadharResult(result);
   265→      if (result.verified) {
   266→        toast.success('Aadhaar card verified successfully! Admin will review and confirm.');
   267→      } else {
   268→        toast.error(result.error || 'Aadhaar verification failed');
   269→      }
   270→    } catch (err: any) {
   271→      toast.error(err.message || 'Verification failed');
   272→      setAadharResult({ verified: false, error: err.message });
   273→    } finally {
   274→      setVerifying(false);
   275→    }
   276→  };
   277→
   278→  const handleClose = () => {
   279→    setImagePreview(null);
   280→    setAadharResult(null);
   281→    onClose();
   282→  };
   283→
   284→  return (
   285→    <Dialog open={open} onOpenChange={(o) => !o && handleClose()}>
   286→      <DialogContent className="max-w-sm rounded-2xl">
   287→        <DialogHeader>
   288→          <DialogTitle className="flex items-center gap-2">
   289→            <ShieldCheck className="h-5 w-5 text-blue-600" />
   290→            Aadhaar Card Verification
   291→          </DialogTitle>
   292→          <DialogDescription>Upload your Aadhaar card for AI-powered instant verification</DialogDescription>
   293→        </DialogHeader>
   294→
   295→        <div className="space-y-4 mt-2">
   296→          <p className="text-xs text-gray-500">Your data is processed securely by AI and is not stored on our servers.</p>
   297→
   298→          {!imagePreview ? (
   299→            <label className="flex flex-col items-center justify-center border-2 border-dashed border-gray-300 rounded-xl p-8 cursor-pointer hover:border-forest-400 hover:bg-forest-50/50 transition-colors">
   300→              <Camera className="h-10 w-10 text-gray-400 mb-3" />
   301→              <p className="text-sm font-medium text-gray-600">Upload Aadhaar Card</p>
   302→              <p className="text-xs text-gray-400 mt-1">JPG, PNG - Clear photo of your Aadhaar card</p>
   303→              <input type="file" accept="image/*" onChange={handleFileSelect} className="hidden" />
   304→            </label>
   305→          ) : (
   306→            <div className="space-y-3">
   307→              <div className="relative rounded-xl overflow-hidden border border-gray-200">
   308→                <img src={imagePreview} alt="Aadhaar preview" className="w-full max-h-48 object-contain bg-white" />
   309→                <button onClick={() => { setImagePreview(null); setAadharResult(null); }} className="absolute top-2 right-2 w-7 h-7 rounded-full bg-black/60 text-white flex items-center justify-center hover:bg-black/80">
   310→                  <XCircle className="h-4 w-4" />
   311→                </button>
   312→              </div>
   313→              <div className="flex gap-2">
   314→                <Button onClick={handleVerify} disabled={verifying} className="btn-green text-sm gap-2 flex-1">
   315→                  {verifying ? <Loader2 className="h-4 w-4 animate-spin" /> : <ShieldCheck className="h-4 w-4" />}
   316→                  {verifying ? 'Verifying...' : 'Verify Aadhaar'}
   317→                </Button>
   318→                <Button variant="outline" size="sm" onClick={() => { setImagePreview(null); setAadharResult(null); }} className="rounded-full">
   319→                  Change
   320→                </Button>
   321→              </div>
   322→            </div>
   323→          )}
   324→
   325→          {aadharResult && (
   326→            <motion.div initial={{ opacity: 0, y: 8 }} animate={{ opacity: 1, y: 0 }} className={`rounded-xl p-4 ${aadharResult.verified ? 'bg-green-50 border border-green-200' : 'bg-red-50 border border-red-200'}`}>
   327→              <div className="flex items-center gap-2 mb-2">
   328→                {aadharResult.verified ? (
   329→                  <><CheckCircle2 className="h-5 w-5 text-green-600" /><p className="text-sm font-semibold text-green-800">Verification Successful</p></>
   330→                ) : (
   331→                  <><AlertCircle className="h-5 w-5 text-red-600" /><p className="text-sm font-semibold text-red-800">Verification Failed</p></>
   332→                )}
   333→              </div>
   334→              {aadharResult.verified ? (
   335→                <div className="grid grid-cols-2 gap-3">
   336→                  {aadharResult.name && <div><p className="text-[10px] text-green-600 font-medium">Name</p><p className="text-sm font-semibold text-green-900">{aadharResult.name}</p></div>}
   337→                  {aadharResult.aadharNumber && <div><p className="text-[10px] text-green-600 font-medium">Aadhar Number</p><p className="text-sm font-semibold text-green-900 font-mono">{aadharResult.aadharNumber.replace(/(\d{4})/g, '$1 ').trim()}</p></div>}
   338→                  {aadharResult.dob && <div><p className="text-[10px] text-green-600 font-medium">Date of Birth</p><p className="text-sm font-semibold text-green-900">{aadharResult.dob}</p></div>}
   339→                  {aadharResult.gender && <div><p className="text-[10px] text-green-600 font-medium">Gender</p><p className="text-sm font-semibold text-green-900">{aadharResult.gender}</p></div>}
   340→                  {aadharResult.address && <div className="col-span-2"><p className="text-[10px] text-green-600 font-medium">Address</p><p className="text-sm font-semibold text-green-900">{aadharResult.address}</p></div>}
   341→                </div>
   342→              ) : (
   343→                <p className="text-sm text-red-700">{aadharResult.error || 'Could not verify the uploaded document.'}</p>
   344→              )}
   345→            </motion.div>
   346→          )}
   347→
   348→          {aadharResult?.verified && (
   349→            <Button onClick={handleClose} className="w-full rounded-xl">Done</Button>
   350→          )}
   351→        </div>
   352→      </DialogContent>
   353→    </Dialog>
   354→  );
   355→}

   361→function OverviewTab({ user }: { user: User }) {
   362→  const profile = user.caregiverProfile;
   363→  const [bookings, setBookings] = useState<Booking[]>([]);
   364→  // Start with loading=false so new caregivers (no profile) never see a skeleton flash
   365→  const [loading, setLoading] = useState(false);
   366→  const [error, setError] = useState('');
   367→
   368→  // Phone verification state
   369→  const [phoneDialogOpen, setPhoneDialogOpen] = useState(false);
   370→  const [phoneOtpSent, setPhoneOtpSent] = useState(false);
   371→  const [phoneOtp, setPhoneOtp] = useState('');
   372→  const [phoneSending, setPhoneSending] = useState(false);
   373→  const [phoneVerifying, setPhoneVerifying] = useState(false);
   374→  const [phoneDevOtp, setPhoneDevOtp] = useState('');
   375→
   376→  const sendPhoneOtp = async () => {
   377→    setPhoneSending(true);
   378→    try {
   379→      const res = await api.auth.sendPhoneOtp(user.phone);
   380→      setPhoneDevOtp(res.devOtp || '');
   381→      setPhoneOtpSent(true);
   382→      toast.success('OTP sent to your phone!');
   383→    } catch (err: any) {
   384→      toast.error(err.message || 'Failed to send OTP');
   385→    } finally {
   386→      setPhoneSending(false);
   387→    }
   388→  };
   389→
   390→  const verifyPhoneOtp = async () => {
   391→    if (phoneOtp.length !== 6) { toast.error('Enter 6-digit OTP'); return; }
   392→    setPhoneVerifying(true);
   393→    try {
   394→      await api.auth.verifyPhoneOtp(user.phone, phoneOtp);
   395→      toast.success('Phone verified successfully!');
   396→      setPhoneDialogOpen(false);
   397→      // Update local state
   398→      useAuthStore.getState().setAuth({ ...user, phoneVerified: true } as any);
   399→    } catch (err: any) {
   400→      toast.error(err.message || 'Verification failed');
   401→    } finally {
   402→      setPhoneVerifying(false);
   403→    }

   406→  // Aadhaar dialog state (works even before profile exists)

   408→

   416→        ...b,
   417→        date: b.startDate,
   418→        patientName: b.patient?.name,
   419→        familyName: b.family?.name,
   420→        careType: b.careRequirements ? (typeof b.careRequirements === 'string' ? JSON.parse(b.careRequirements) : b.careRequirements)?.needs?.join(', ') : undefined,
   421→      }));
   422→      setBookings(all);
   423→    } catch (err: any) {
   424→      setError(err.message || 'Failed to load dashboard data');
   425→    } finally {
   426→      setLoading(false);
   427→    }
   428→  }, [profile?.id]);
   429→
   430→  useEffect(() => {
   431→    fetchData();
   432→  }, [fetchData]);
   433→
   434→  // No profile yet — show setup guidance with Aadhaar verification
   435→  if (!profile) {
   436→    return (
   437→      <div className="space-y-6">
   438→        <div>
   439→          <h2 className="text-2xl font-bold text-gray-900">Welcome, {user.name.split(' ')[0]}!</h2>
   440→          <p className="text-sm text-gray-400 mt-1">Complete your profile and get verified to start receiving care requests.</p>
   441→        </div>
   442→
   443→        {/* Aadhaar Verification for new caregivers */}
   444→        <Card className="rounded-2xl border-2 border-blue-200 bg-gradient-to-br from-blue-50 to-indigo-50">
   445→          <CardContent className="p-6">
   446→            <div className="flex flex-col sm:flex-row sm:items-center gap-4">
   447→              <div className="w-14 h-14 rounded-2xl bg-blue-100 flex items-center justify-center shrink-0">
   448→                <ShieldCheck className="h-7 w-7 text-blue-600" />
   449→              </div>
   450→              <div className="flex-1 min-w-0">
   451→                <h3 className="text-base font-bold text-blue-900">Verify Your Identity with Aadhaar</h3>
   452→                <p className="text-sm text-blue-700/70 mt-1">Upload your Aadhaar card for AI-powered instant verification. Get the verified badge and attract more families.</p>
   453→              </div>
   454→              <Button onClick={() => setAadhaarDialogOpen(true)} size="lg" className="bg-blue-600 hover:bg-blue-700 text-white rounded-xl shrink-0">
   455→                <ShieldCheck className="h-4 w-4 mr-2" /> Verify Aadhaar
   456→              </Button>
   457→            </div>
   458→          </CardContent>
   459→        </Card>
   460→
   461→        {/* Phone verification for new caregivers */}
   462→        {!user.phoneVerified && user.phone && (
   463→          <Card className="rounded-2xl border-2 border-amber-200 bg-amber-50">
   464→            <CardContent className="p-4 flex items-center justify-between gap-4">
   465→              <div className="flex items-center gap-3 min-w-0">
   466→                <div className="w-10 h-10 rounded-xl bg-amber-100 flex items-center justify-center shrink-0">
   467→                  <Phone className="h-5 w-5 text-amber-600" />
   468→                </div>
   469→                <div className="min-w-0">
   470→                  <p className="text-sm font-semibold text-amber-900">Verify Your Phone Number</p>
   471→                  <p className="text-xs text-amber-700/70">Required to receive booking requests and OTPs</p>
   472→                </div>
   473→              </div>
   474→              <Button onClick={() => setPhoneDialogOpen(true)} size="sm" className="bg-amber-600 hover:bg-amber-700 text-white rounded-xl shrink-0">
   475→                Verify Now
   476→              </Button>
   477→            </CardContent>
   478→          </Card>

   484→            <div className="w-16 h-16 rounded-2xl green-gradient-bg flex items-center justify-center mx-auto mb-4">

   489→          </CardContent>

   491→
   492→        {/* Aadhaar Verification Dialog */}

   494→      </div>
   495→    );
   496→  }
   497→
   498→  if (error) {
   499→    return <ErrorState message={error} onRetry={fetchData} />;
