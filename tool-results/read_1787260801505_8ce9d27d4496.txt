   205→   205→function parseJsonSafe(jsonStr: string | null | undefined): string[] {
   206→   206→  if (!jsonStr) return [];
   207→   207→  try {
   208→   208→    const parsed = JSON.parse(jsonStr);
   209→   209→    return Array.isArray(parsed) ? parsed : [];
   210→   210→  } catch {
   211→   211→    return jsonStr.split(',').map((s) => s.trim()).filter(Boolean);
   212→   212→  }
   213→   213→}
   214→   214→
   215→   215→function formatDate(dateStr: string): string {
   216→   216→  try {
   217→   217→    return new Date(dateStr).toLocaleDateString('en-IN', {
   218→   218→      day: 'numeric',
   219→   219→      month: 'short',
   220→   220→      year: 'numeric',
   221→   221→    });
   222→   222→  } catch {
   223→   223→    return dateStr;
   224→   224→  }
   225→   225→}
   226→   226→
   227→   227→function getTodayDate(): string {
   228→   228→  const now = new Date();
   229→   229→  const year = now.getFullYear();
   230→   230→  const month = String(now.getMonth() + 1).padStart(2, '0');
   231→   231→  const day = String(now.getDate()).padStart(2, '0');
   232→   232→  return `${year}-${month}-${day}`;
   233→   233→}

   248→   248→    setLoading(true);

   251→   251→      const res = await api.bookings.list({ caregiverId: profile.id });

   256→   256→        familyName: b.family?.name,

   258→   258→      }));

   261→   261→      setError(err.message || 'Failed to load dashboard data');
   262→   262→    } finally {
   263→   263→      setLoading(false);
   264→   264→    }
   265→   265→  }, [profile?.id]);
   266→   266→
   267→   267→  useEffect(() => {
   268→   268→    fetchData();
   269→   269→  }, [fetchData]);

   271→   271→  // No profile yet — show setup guidance immediately, no loading flash

   290→   290→  }
   291→   291→
   292→   292→  if (error) {

   294→   294→  }

   296→   296→  const upcomingBookings = bookings
   297→   297→    .filter((b) => ['CONFIRMED', 'IN_PROGRESS'].includes(b.status))

   309→   309→    { label: 'Hourly Rate', value: `₹${hourlyRate}`, icon: <IndianRupee className="h-5 w-5" />, color: 'bg-purple-50 text-purple-700', isString: true },

   809→   809→      toast.error(err.message || 'Verification failed');
