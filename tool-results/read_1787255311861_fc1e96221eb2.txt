   127→   127→export default function LoginModal({ isOpen, onClose, defaultTab }: LoginModalProps) {
   128→   128→  /* ---- state ---- */
   129→   129→  const [tab, setTab] = useState<Tab>(defaultTab === "register" ? "register" : "login");
   130→   130→  const [showPassword, setShowPassword] = useState(false);
   131→   131→  const [loading, setLoading] = useState(false);
   132→   132→  const [error, setError] = useState("");
   133→   133→  const [resetOpen, setResetOpen] = useState(false);
   134→   134→
   135→   135→  // Login form
   136→   136→  const [loginEmail, setLoginEmail] = useState("");
   137→   137→  const [loginPassword, setLoginPassword] = useState("");
   138→   138→
   139→   139→  // Register form
   140→   140→  const [regName, setRegName] = useState("");
   141→   141→  const [regEmail, setRegEmail] = useState("");
   142→   142→  const [regPhone, setRegPhone] = useState("");
   143→   143→  const [regPassword, setRegPassword] = useState("");
   144→   144→  const [regRole, setRegRole] = useState<UserRole>("FAMILY");
   145→   145→
   146→   146→  // Reset form (simplified — no OTP)
   147→   147→  const [resetEmail, setResetEmail] = useState("");
   148→   148→  const [resetNewPassword, setResetNewPassword] = useState("");
   149→   149→  const [resetLoading, setResetLoading] = useState(false);
   150→   150→  const [resetError, setResetError] = useState("");
   151→   151→  const [resetDone, setResetDone] = useState(false);
   152→   152→
   153→   153→  const { setAuth } = useAuthStore();
   154→   154→  const { toast } = useToast();
   155→   155→
   156→   156→  // Google popup form
   157→   157→  const [googleOpen, setGoogleOpen] = useState(false);
   158→   158→  const [googleEmail, setGoogleEmail] = useState("");
   159→   159→  const [googleName, setGoogleName] = useState("");
   160→   160→  const [googleError, setGoogleError] = useState("");
   161→   161→  const [googleSubmitting, setGoogleSubmitting] = useState(false);
   162→   162→
   163→   163→  /* ---- reset form fields when switching tabs ---- */
   164→   164→  useEffect(() => {
   165→   165→    setError("");
   166→   166→    setLoginPassword("");
   167→   167→  }, [tab]);
   168→   168→
   169→   169→  /* ---- reset entire state when modal closes ---- */
   170→   170→  const handleClose = useCallback(() => {
   171→   171→    setTab(defaultTab === "register" ? "register" : "login");
   172→   172→    setError("");
   173→   173→    setShowPassword(false);
   174→   174→    setLoading(false);
   175→   175→    setLoginEmail("");
   176→   176→    setLoginPassword("");
   177→   177→    setRegName("");
   178→   178→    setRegEmail("");
   179→   179→    setRegPhone("");
   180→   180→    setRegPassword("");
   181→   181→    setRegRole("FAMILY");
   182→   182→    setResetOpen(false);
   183→   183→    setResetEmail("");
   184→   184→    setResetNewPassword("");
   185→   185→    setResetLoading(false);
   186→   186→    setResetError("");
   187→   187→    setResetDone(false);
   188→   188→    setGoogleOpen(false);
   189→   189→    setGoogleEmail("");
   190→   190→    setGoogleName("");
   191→   191→    setGoogleError("");

   195→   195→
   196→   196→  /* ================================================================ */
   197→   197→  /*  HANDLERS                                                         */
   198→   198→  /* ================================================================ */
   199→   199→
   200→   200→  /* ---- Google sign-in — redirect flow, fallback to simulated ---- */
   201→   201→  const handleGoogleSignIn = async () => {
   202→   202→    setGoogleError("");
   203→   203→    try {
   204→   204→      // Check if real Google OAuth is configured
   205→   205→      const checkRes = await fetch("/api/auth/google-configured");
   206→   206→      const { configured } = await checkRes.json();
   207→   207→
   208→   208→      if (configured) {
   209→   209→        // Real Google OAuth — full-page redirect to /api/auth/google-go
   210→   210→        const origin = window.location.origin;
   211→   211→        window.location.href = `/api/auth/google-go?origin=${encodeURIComponent(origin)}`;
   212→   212→        return;
   213→   213→      }
   214→   214→    } catch {
   215→   215→      // If check fails, fall through to simulated flow
   216→   216→    }
   217→   217→    // Google OAuth not configured — open simulated sign-in
   218→   218→    setGoogleOpen(true);
   219→   219→  };
   220→   220→
   221→   221→  /* ---- Google popup submit ---- */
   222→   222→  const handleGoogleSubmit = async (e: React.FormEvent) => {
   223→   223→    e.preventDefault();
   224→   224→    if (!googleEmail.trim() || !googleName.trim()) {
   225→   225→      setGoogleError("Please enter your name and email.");
   226→   226→      return;
   227→   227→    }
   228→   228→    if (!googleEmail.trim().includes("@")) {
   229→   229→      setGoogleError("Please enter a valid email address.");
   230→   230→      return;
   231→   231→    }
   232→   232→    setGoogleSubmitting(true);
   233→   233→    setGoogleError("");
   234→   234→    try {
   235→   235→      // 1. Call the simulate endpoint (sets the session cookie)
   236→   236→      const res = await fetch("/api/auth/google-simulate", {
   237→   237→        method: "POST",
   238→   238→        headers: { "Content-Type": "application/json" },
   239→   239→        body: JSON.stringify({ email: googleEmail.trim(), name: googleName.trim() }),
   240→   240→      });
   241→   241→      const data = await res.json();
   242→   242→      if (!res.ok) {
   243→   243→        setGoogleError(data.error || "Google sign-in failed.");
   244→   244→        return;
   245→   245→      }
   246→   246→
   247→   247→      // 2. Fetch the NextAuth session (cookie is already set by step 1)
   248→   248→      const sessionRes = await fetch("/api/auth/session");
   249→   249→      const sessionData = await sessionRes.json();
   250→   250→
   251→   251→      if (sessionData?.user) {
   252→   252→        // 3. Directly update the Zustand store — no page reload needed
   253→   253→        setAuth({
   254→   254→          id: sessionData.user.id,
   255→   255→          email: sessionData.user.email,
   256→   256→          name: sessionData.user.name,
   257→   257→          phone: "",
   258→   258→          role: sessionData.user.role,
   259→   259→          avatarUrl: sessionData.user.image || null,
   260→   260→          subscription: "NONE" as any,
   261→   261→          patientProfiles: [],
   262→   262→          caregiverProfile: null,
   263→   263→        });
   264→   264→        setGoogleOpen(false);
   265→   265→        toast({
   266→   266→          title: "Welcome" + (data.user.isNewUser ? " to SevaSaathi!" : " back!") + " \ud83c\udf89",
   267→   267→          description: "Signed in as " + data.user.email,
   268→   268→        });
   269→   269→        handleClose();
   270→   270→      } else {
   271→   271→        // Fallback: reload the page so NextAuth picks up the cookie
   272→   272→        window.location.href = window.location.origin;
   273→   273→      }
   274→   274→    } catch {
   275→   275→      setGoogleError("Something went wrong. Please try again.");
   276→   276→    } finally {
   277→   277→      setGoogleSubmitting(false);
   278→   278→    }
   279→   279→  };
   280→   280→
   281→   281→  /* ---- Email / password login ---- */
   282→   282→  const handleLogin = async (e: React.FormEvent) => {
   283→   283→    e.preventDefault();
   284→   284→    if (!loginEmail.trim() || !loginPassword.trim()) {
   285→   285→      setError("Please enter your email and password.");
   286→   286→      return;
   287→   287→    }
   288→   288→    setLoading(true);
   289→   289→    setError("");
   290→   290→    try {
   291→   291→      const result = await signIn("credentials", {
   292→   292→        email: loginEmail.trim(),
   293→   293→        password: loginPassword,
   294→   294→        redirect: false,
   295→   295→      });
   296→   296→
   297→   297→      if (result?.error) {
   298→   298→        setError("Invalid email or password. Please try again.");
   299→   299→        return;
   300→   300→      }
   301→   301→
   302→   302→      // Fetch session to get user data, then set auth explicitly
   303→   303→      try {
   304→   304→        const sessionRes = await fetch("/api/auth/session");
   305→   305→        const sessionData = await sessionRes.json();
   306→   306→        if (sessionData?.user) {
   307→   307→          const u = sessionData.user as any;
   308→   308→          setAuth({
   309→   309→            id: u.id || "",
   310→   310→            email: u.email || "",
   311→   311→            name: u.name || "",
   312→   312→            phone: "",
   313→   313→            role: u.role || "FAMILY",
   314→   314→            avatarUrl: u.image || null,
   315→   315→            subscription: "NONE",
   316→   316→            patientProfiles: [],
   317→   317→            caregiverProfile: null,
   318→   318→          });
   319→   319→        }
   320→   320→      } catch {
   321→   321→        // AuthSync will pick it up as fallback
   322→   322→      }
   323→   323→
   324→   324→      toast({
   325→   325→        title: "Welcome back! 🎉",
   326→   326→        description: "You have successfully logged in.",
   327→   327→      });
   328→   328→      handleClose();
   329→   329→    } catch {
   330→   330→      setError("Something went wrong. Please try again.");
   331→   331→    } finally {
   332→   332→      setLoading(false);
   333→   333→    }
   334→   334→  };
   335→   335→
   336→   336→  /* ---- Registration ---- */
   337→   337→  const handleRegister = async (e: React.FormEvent) => {
   338→   338→    e.preventDefault();
   339→   339→    if (!regName.trim() || !regEmail.trim() || !regPhone.trim() || !regPassword.trim()) {
   340→   340→      setError("Please fill in all fields.");
   341→   341→      return;
   342→   342→    }
   343→   343→    if (regPassword.length < 6) {
   344→   344→      setError("Password must be at least 6 characters.");
   345→   345→      return;
   346→   346→    }
   347→   347→    setLoading(true);
   348→   348→    setError("");
   349→   349→    try {
   350→   350→      // 1. Register
   351→   351→      const regRes = await api.auth.register({
   352→   352→        name: regName.trim(),
   353→   353→        email: regEmail.trim(),
   354→   354→        phone: regPhone.trim(),
   355→   355→        password: regPassword,
   356→   356→        role: regRole,
   357→   357→      });
   358→   358→
   359→   359→      // 2. Auto-login with credentials
   360→   360→      const loginResult = await signIn("credentials", {
   361→   361→        email: regEmail.trim(),
   362→   362→        password: regPassword,
   363→   363→        redirect: false,
   364→   364→      });
   365→   365→
   366→   366→      if (loginResult?.error) {
   367→   367→        // Registration succeeded but auto-login failed — that's ok, just notify
   368→   368→        toast({ title: "Account created!", description: "Please log in with your credentials." });
   369→   369→        setTab("login");
   370→   370→        setLoginEmail(regEmail.trim());
   371→   371→        return;
   372→   372→      }
   373→   373→
   374→   374→      // Fetch session to get user data, then set auth explicitly
   375→   375→      try {
   376→   376→        const sessionRes = await fetch("/api/auth/session");
   377→   377→        const sessionData = await sessionRes.json();
   378→   378→        if (sessionData?.user) {
   379→   379→          const u = sessionData.user as any;
   380→   380→          setAuth({
   381→   381→            id: u.id || "",
   382→   382→            email: u.email || "",
   383→   383→            name: u.name || "",
   384→   384→            phone: regPhone.trim(),
   385→   385→            role: u.role || regRole,
   386→   386→            avatarUrl: u.image || null,
   387→   387→            subscription: "NONE",
   388→   388→            patientProfiles: [],
   389→   389→            caregiverProfile: null,
   390→   390→          });
   391→   391→        }
   392→   392→      } catch {
   393→   393→        // AuthSync will pick it up as fallback
   394→   394→      }
   395→   395→
   396→   396→      toast({ title: "Welcome to SevaSaathi! 🎉", description: "Your account has been created." });
   397→   397→      handleClose();
   398→   398→    } catch (err: any) {
   399→   399→      setError(err?.message || "Registration failed. Please try again.");
   400→   400→    } finally {
   401→   401→      setLoading(false);
   402→   402→    }
   403→   403→  };
   404→   404→
   405→   405→  /* ---- Reset password (simplified) ---- */
   406→   406→  const handleReset = async (e: React.FormEvent) => {
   407→   407→    e.preventDefault();
   408→   408→    if (!resetEmail.trim() || !resetNewPassword.trim()) {
   409→   409→      setResetError("Please enter your email and new password.");
   410→   410→      return;
   411→   411→    }
   412→   412→    if (resetNewPassword.length < 6) {
   413→   413→      setResetError("Password must be at least 6 characters.");
   414→   414→      return;
   415→   415→    }
   416→   416→    setResetLoading(true);
   417→   417→    setResetError("");
   418→   418→    try {
   419→   419→      const res = await fetch("/api/auth/reset-password", {
   420→   420→        method: "POST",
   421→   421→        headers: { "Content-Type": "application/json" },
   422→   422→        body: JSON.stringify({ email: resetEmail.trim(), newPassword: resetNewPassword }),
   423→   423→      });
   424→   424→      const data = await res.json();
   425→   425→      if (!res.ok) {
   426→   426→        setResetError(data.error || "Failed to reset password.");
   427→   427→        return;
   428→   428→      }
   429→   429→      setResetDone(true);
   430→   430→      toast({ title: "Password reset successfully!", description: "You can now log in with your new password." });
   431→   431→    } catch {
   432→   432→      setResetError("Something went wrong. Please try again.");
   433→   433→    } finally {
   434→   434→      setResetLoading(false);
   435→   435→    }
   436→   436→  };
   437→   437→
   438→   438→  /* ================================================================ */
   439→   439→  /*  RENDER                                                           */
   440→   440→  /* ================================================================ */
   441→   441→  return (
   442→   442→    <Dialog open={isOpen} onOpenChange={(open) => !open && handleClose()}>
   443→   443→      <DialogContent showCloseButton={false} className="sm:max-w-md p-0 overflow-hidden border-forest-200/40">
   444→   444→        {/* Single custom close button */}
   445→   445→        <button
   446→   446→          onClick={handleClose}
   447→   447→          className="absolute top-3 right-3 z-50 rounded-full p-1.5 text-gray-400 hover:text-gray-600 hover:bg-gray-100 transition-colors"
   448→   448→          aria-label="Close"
   449→   449→        >
   450→   450→          <X className="w-5 h-5" />
   451→   451→        </button>
   452→   452→
   453→   453→        <div className="px-6 pt-6 pb-2">
   454→   454→          <DialogHeader className="space-y-1 mb-1">
   455→   455→            <div className="flex items-center gap-2 justify-center">
   456→   456→              <div className="w-9 h-9 rounded-xl bg-forest-900 flex items-center justify-center">
   457→   457→                <Heart className="w-5 h-5 text-lime-400" />
   458→   458→              </div>
   459→   459→              <DialogTitle className="text-xl font-bold text-forest-900">
   460→   460→                SevaSaathi
   461→   461→              </DialogTitle>
   462→   462→            </div>
   463→   463→            <DialogDescription className="text-sm text-gray-500 mt-1">
   464→   464→              {tab === "login"
   465→   465→                ? "Sign in to your account"
   466→   466→                : "Create your account to get started"}
   467→   467→            </DialogDescription>
   468→   468→          </DialogHeader>
   469→   469→        </div>
   470→   470→
   471→   471→        {/* ---- Error banner ---- */}
   472→   472→        <AnimatePresence>
   473→   473→          {error && (
   474→   474→            <motion.div
   475→   475→              initial={{ height: 0, opacity: 0 }}
   476→   476→              animate={{ height: "auto", opacity: 1 }}
   477→   477→              exit={{ height: 0, opacity: 0 }}
   478→   478→              className="overflow-hidden"
   479→   479→            >
   480→   480→              <div className="mx-6 mt-1 px-3 py-2 rounded-lg bg-red-50 border border-red-200 text-red-700 text-sm flex items-center gap-2">
   481→   481→                <X className="w-4 h-4 flex-shrink-0" />
   482→   482→                {error}
   483→   483→              </div>
   484→   484→            </motion.div>
   485→   485→          )}
   486→   486→        </AnimatePresence>
   487→   487→
   488→   488→        {/* ---- Tab switcher ---- */}
   489→   489→        <div className="px-6 pt-3">
   490→   490→          <div className="flex bg-gray-100 rounded-lg p-1">
   491→   491→            <button
   492→   492→              onClick={() => setTab("login")}
   493→   493→              className={`flex-1 py-2 px-4 rounded-md text-sm font-medium transition-all duration-200 ${
