   327→function PatientsTab() {
   328→  const user = useAuthStore((s) => s.user);
   329→  const [patients, setPatients] = useState<any[]>([]);
   330→  const [loading, setLoading] = useState(false);
   331→  const [error, setError] = useState('');
   332→  const [modalOpen, setModalOpen] = useState(false);
   333→  const [submitting, setSubmitting] = useState(false);
   334→
   335→  // Form state
   336→  const [form, setForm] = useState({
   337→    name: '', age: '', gender: '', relationship: '', address: '', city: '',
   338→    pincode: '', mobilityStatus: '', medicalHistory: '', careRequirements: '',
   339→    emergencyName: '', emergencyPhone: '',
   340→  });
   341→
   342→  const fetchPatients = useCallback(async () => {
   343→    if (!user?.id) return;
   344→    setLoading(true);
   345→    setError('');
   346→    try {
   347→      const res = await api.patients.list(user.id);
   348→      setPatients(res.patients || []);
   349→    } catch (err: any) {
   350→      setError(err.message || 'Failed to load patients');
   351→    } finally {
   352→      setLoading(false);
   353→    }
   354→  }, [user?.id]);
   355→
   356→  useEffect(() => {
   357→    fetchPatients();
   358→  }, [fetchPatients]);
   359→
   360→  const updateForm = (field: string, value: string) => {
   361→    setForm((prev) => ({ ...prev, [field]: value }));
   362→  };
   363→
   364→  const handleSubmit = async (e: React.FormEvent) => {
   365→    e.preventDefault();
   366→    if (!user?.id) return;
   367→    setSubmitting(true);
   368→    try {
   369→      await api.patients.create({
   370→        familyId: user.id,
   371→        name: form.name,
   372→        age: parseInt(form.age, 10) || 0,
   373→        gender: form.gender,
   374→        relationship: form.relationship,
   375→        address: form.address,
   376→        city: form.city,
   377→        pincode: form.pincode || undefined,
   378→        mobilityStatus: form.mobilityStatus || 'mobile',
   379→        medicalHistory: form.medicalHistory,
   380→        careRequirements: form.careRequirements,
   381→        emergencyContact: JSON.stringify({ name: form.emergencyName, phone: form.emergencyPhone }),
   382→      });
   383→      toast.success('Member profile created successfully!');
   384→      setModalOpen(false);
   385→      setForm({
   386→        name: '', age: '', gender: '', relationship: '', address: '', city: '',
   387→        pincode: '', mobilityStatus: '', medicalHistory: '', careRequirements: '',
   388→        emergencyName: '', emergencyPhone: '',
   389→      });
   390→      fetchPatients();
   391→    } catch (err: any) {
   392→      toast.error(err.message || 'Failed to create member');
   393→    } finally {
   394→      setSubmitting(false);
   395→    }
   396→  };
   397→
   398→  if (error) {
   399→    return <ErrorState message={error} onRetry={fetchPatients} />;
   400→  }
   401→
   402→  const mobilityLabel: Record<string, string> = {
   403→    mobile: 'Mobile', bedridden: 'Bedridden', limited: 'Limited Mobility', wheelchair: 'Wheelchair',
   404→  };
   405→
   406→  return (
   407→    <div className="space-y-6">
   408→      <div className="flex items-center justify-between">
   409→        <div>
   410→          <h2 className="text-xl font-bold text-gray-900">Member Profiles</h2>
   411→          <p className="text-sm text-gray-400">Manage care recipients in your family.</p>
   412→        </div>
   413→        <Dialog open={modalOpen} onOpenChange={setModalOpen}>
   414→          <DialogTrigger asChild>
   415→            <Button className="btn-black text-sm gap-2">
   416→              <Plus className="h-4 w-4" /> Add Member
   417→            </Button>
   418→          </DialogTrigger>
   419→          <DialogContent className="max-w-lg max-h-[90vh] overflow-y-auto rounded-2xl">
   420→            <DialogHeader>
   421→              <DialogTitle>Add New Member</DialogTitle>
   422→            </DialogHeader>
   423→            <form onSubmit={handleSubmit} className="space-y-4 mt-2">
   424→              <div className="grid grid-cols-2 gap-3">
   425→                <div className="col-span-2">
   426→                  <Label className="text-xs font-medium text-gray-600">Full Name *</Label>
   427→                  <Input
   428→                    required
   429→                    value={form.name}
   430→                    onChange={(e) => updateForm('name', e.target.value)}
   431→                    placeholder="Member name"
   432→                    className="mt-1 rounded-xl"
   433→                  />
   434→                </div>
   435→                <div>
   436→                  <Label className="text-xs font-medium text-gray-600">Age *</Label>
   437→                  <Input
   438→                    required
   439→                    type="number"
   440→                    min={0}
   441→                    max={150}
   442→                    value={form.age}
   443→                    onChange={(e) => updateForm('age', e.target.value)}
   444→                    placeholder="Age"
   445→                    className="mt-1 rounded-xl"
   446→                  />
   447→                </div>
   448→                <div>
   449→                  <Label className="text-xs font-medium text-gray-600">Gender *</Label>
   450→                  <Select value={form.gender} onValueChange={(v) => updateForm('gender', v)}>
