   124→/* ================================================================== */
   125→/*  MAIN COMPONENT                                                     */
   126→/* ================================================================== */
   127→export default function LoginModal({ isOpen, onClose, defaultTab }: LoginModalProps) {
   128→  /* ---- state ---- */
   129→  const [tab, setTab] = useState<Tab>(defaultTab === "register" ? "register" : "login");
   130→  const [showPassword, setShowPassword] = useState(false);
   131→  const [loading, setLoading] = useState(false);
   132→  const [error, setError] = useState("");
   133→  const [resetOpen, setResetOpen] = useState(false);
   134→
   135→  // Login form
   136→  const [loginEmail, setLoginEmail] = useState("");
   137→  const [loginPassword, setLoginPassword] = useState("");
   138→
   139→  // Register form
   140→  const [regName, setRegName] = useState("");
   141→  const [regEmail, setRegEmail] = useState("");
   142→  const [regPhone, setRegPhone] = useState("");
   143→  const [regPassword, setRegPassword] = useState("");
   144→  const [regRole, setRegRole] = useState<UserRole>("FAMILY");
   145→
   146→  // Reset form (simplified — no OTP)
   147→  const [resetEmail, setResetEmail] = useState("");
   148→  const [resetNewPassword, setResetNewPassword] = useState("");
   149→  const [resetLoading, setResetLoading] = useState(false);
   150→  const [resetError, setResetError] = useState("");
   151→  const [resetDone, setResetDone] = useState(false);
   152→
   153→  const { setAuth } = useAuthStore();
   154→  const { toast } = useToast();
   155→
   156→  // Google popup form
   157→  const [googleOpen, setGoogleOpen] = useState(false);
   158→  const [googleEmail, setGoogleEmail] = useState("");
   159→  const [googleName, setGoogleName] = useState("");
   160→  const [googleError, setGoogleError] = useState("");
   161→  const [googleSubmitting, setGoogleSubmitting] = useState(false);
   162→
   163→  /* ---- reset form fields when switching tabs ---- */
   164→  useEffect(() => {
   165→    setError("");
   166→    setLoginPassword("");
   167→  }, [tab]);
   168→
   169→  /* ---- reset entire state when modal closes ---- */
   170→  const handleClose = useCallback(() => {
   171→    setTab(defaultTab === "register" ? "register" : "login");
   172→    setError("");
   173→    setShowPassword(false);
   174→    setLoading(false);
   175→    setLoginEmail("");
   176→    setLoginPassword("");
   177→    setRegName("");
   178→    setRegEmail("");
   179→    setRegPhone("");
   180→    setRegPassword("");
   181→    setRegRole("FAMILY");
   182→    setResetOpen(false);
   183→    setResetEmail("");
   184→    setResetNewPassword("");
   185→    setResetLoading(false);
   186→    setResetError("");
   187→    setResetDone(false);
   188→    setGoogleOpen(false);
   189→    setGoogleEmail("");
   190→    setGoogleName("");
   191→    setGoogleError("");
   192→    setGoogleSubmitting(false);
   193→    onClose();
   194→  }, [onClose, defaultTab]);
   195→
   196→  /* ================================================================ */
   197→  /*  HANDLERS                                                         */
   198→  /* ================================================================ */
   199→
   200→  /* ---- Google sign-in — redirect flow, fallback to simulated ---- */
   201→  const handleGoogleSignIn = async () => {
   202→    setGoogleError("");
   203→    try {
   204→      // Check if real Google OAuth is configured
   205→      const checkRes = await fetch("/api/auth/google-configured");
   206→      const { configured } = await checkRes.json();
   207→
   208→      if (configured) {
   209→        // Real Google OAuth — full-page redirect to /api/auth/google-go
   210→        const origin = window.location.origin;
   211→        window.location.href = `/api/auth/google-go?origin=${encodeURIComponent(origin)}`;
   212→        return;
   213→      }
   214→    } catch {
   215→      // If check fails, fall through to simulated flow
   216→    }
   217→    // Google OAuth not configured — open simulated sign-in
   218→    setGoogleOpen(true);
   219→  };
   220→
   221→  /* ---- Google popup submit ---- */
   222→  const handleGoogleSubmit = async (e: React.FormEvent) => {
   223→    e.preventDefault();
   224→    if (!googleEmail.trim() || !googleName.trim()) {
   225→      setGoogleError("Please enter your name and email.");
   226→      return;
   227→    }
   228→    if (!googleEmail.trim().includes("@")) {
   229→      setGoogleError("Please enter a valid email address.");
   230→      return;
   231→    }
   232→    setGoogleSubmitting(true);
   233→    setGoogleError("");
   234→    try {
   235→      // 1. Call the simulate endpoint (sets the session cookie)
   236→      const res = await fetch("/api/auth/google-simulate", {
   237→        method: "POST",
   238→        headers: { "Content-Type": "application/json" },
   239→        body: JSON.stringify({ email: googleEmail.trim(), name: googleName.trim() }),
   240→      });
   241→      const data = await res.json();
   242→      if (!res.ok) {
   243→        setGoogleError(data.error || "Google sign-in failed.");
   244→        return;
   245→      }
   246→
   247→      // 2. Fetch the NextAuth session (cookie is already set by step 1)
   248→      const sessionRes = await fetch("/api/auth/session");
   249→      const sessionData = await sessionRes.json();
   250→
   251→      if (sessionData?.user) {
   252→        // 3. Directly update the Zustand store — no page reload needed
   253→        setAuth({
   254→          id: sessionData.user.id,
   255→          email: sessionData.user.email,
   256→          name: sessionData.user.name,
   257→          phone: "",
   258→          role: sessionData.user.role,
   259→          avatarUrl: sessionData.user.image || null,
   260→          subscription: "NONE" as any,
   261→          patientProfiles: [],
   262→          caregiverProfile: null,
   263→        });
   264→        setGoogleOpen(false);
   265→        toast({
   266→          title: "Welcome" + (data.user.isNewUser ? " to SevaSaathi!" : " back!") + " \ud83c\udf89",
   267→          description: "Signed in as " + data.user.email,
   268→        });
   269→        handleClose();
   270→      } else {
   271→        // Fallback: reload the page so NextAuth picks up the cookie
   272→        window.location.href = window.location.origin;
   273→      }
   274→    } catch {
   275→      setGoogleError("Something went wrong. Please try again.");
   276→    } finally {
   277→      setGoogleSubmitting(false);
   278→    }
   279→  };
   280→
   281→  /* ---- Email / password login ---- */
   282→  const handleLogin = async (e: React.FormEvent) => {
   283→    e.preventDefault();
   284→    if (!loginEmail.trim() || !loginPassword.trim()) {
   285→      setError("Please enter your email and password.");
   286→      return;
   287→    }
   288→    setLoading(true);
   289→    setError("");
   290→    try {
   291→      const result = await signIn("credentials", {
   292→        email: loginEmail.trim(),
   293→        password: loginPassword,
   294→        redirect: false,
   295→      });
   296→
   297→      if (result?.error) {
   298→        setError("Invalid email or password. Please try again.");
   299→        return;
   300→      }
   301→
   302→      // Fetch session to get user data, then set auth explicitly
   303→      try {
   304→        const sessionRes = await fetch("/api/auth/session");
   305→        const sessionData = await sessionRes.json();
   306→        if (sessionData?.user) {
   307→          const u = sessionData.user as any;
   308→          setAuth({
   309→            id: u.id || "",
   310→            email: u.email || "",
   311→            name: u.name || "",
   312→            phone: "",
   313→            role: u.role || "FAMILY",
   314→            avatarUrl: u.image || null,
   315→            subscription: "NONE",
   316→            patientProfiles: [],
   317→            caregiverProfile: null,
   318→          });
   319→        }
   320→      } catch {
   321→        // AuthSync will pick it up as fallback
   322→      }
   323→
   324→      toast({
   325→        title: "Welcome back! 🎉",
   326→        description: "You have successfully logged in.",
   327→      });
   328→      handleClose();
   329→    } catch {
   330→      setError("Something went wrong. Please try again.");
   331→    } finally {
   332→      setLoading(false);
   333→    }
   334→  };
   335→
   336→  /* ---- Registration ---- */
   337→  const handleRegister = async (e: React.FormEvent) => {
   338→    e.preventDefault();
   339→    if (!regName.trim() || !regEmail.trim() || !regPhone.trim() || !regPassword.trim()) {
   340→      setError("Please fill in all fields.");
   341→      return;
   342→    }
   343→    if (regPassword.length < 6) {
   344→      setError("Password must be at least 6 characters.");
   345→      return;
   346→    }
   347→    setLoading(true);
   348→    setError("");
   349→    try {
   350→      // 1. Register
   351→      const regRes = await api.auth.register({
   352→        name: regName.trim(),
   353→        email: regEmail.trim(),
   354→        phone: regPhone.trim(),
   355→        password: regPassword,
   356→        role: regRole,
   357→      });
   358→
   359→      // 2. Auto-login with credentials
   360→      const loginResult = await signIn("credentials", {
   361→        email: regEmail.trim(),
   362→        password: regPassword,
   363→        redirect: false,
   364→      });
   365→
   366→      if (loginResult?.error) {
   367→        // Registration succeeded but auto-login failed — that's ok, just notify
   368→        toast({ title: "Account created!", description: "Please log in with your credentials." });
   369→        setTab("login");
   370→        setLoginEmail(regEmail.trim());
   371→        return;
   372→      }
   373→
   374→      // Fetch session to get user data, then set auth explicitly
   375→      try {
   376→        const sessionRes = await fetch("/api/auth/session");
   377→        const sessionData = await sessionRes.json();
   378→        if (sessionData?.user) {
   379→          const u = sessionData.user as any;
   380→          setAuth({
   381→            id: u.id || "",
   382→            email: u.email || "",
   383→            name: u.name || "",
   384→            phone: regPhone.trim(),
   385→            role: u.role || regRole,
   386→            avatarUrl: u.image || null,
   387→            subscription: "NONE",
   388→            patientProfiles: [],
   389→            caregiverProfile: null,
   390→          });
   391→        }
   392→      } catch {
   393→        // AuthSync will pick it up as fallback
   394→      }
   395→
   396→      toast({ title: "Welcome to SevaSaathi! 🎉", description: "Your account has been created." });
   397→      handleClose();
   398→    } catch (err: any) {
   399→      setError(err?.message || "Registration failed. Please try again.");
   400→    } finally {
   401→      setLoading(false);
   402→    }
   403→  };
   404→
   405→  /* ---- Reset password (simplified) ---- */
   406→  const handleReset = async (e: React.FormEvent) => {
   407→    e.preventDefault();
   408→    if (!resetEmail.trim() || !resetNewPassword.trim()) {
   409→      setResetError("Please enter your email and new password.");
   410→      return;
   411→    }
   412→    if (resetNewPassword.length < 6) {
   413→      setResetError("Password must be at least 6 characters.");
   414→      return;
   415→    }
   416→    setResetLoading(true);
   417→    setResetError("");
   418→    try {
   419→      const res = await fetch("/api/auth/reset-password", {
   420→        method: "POST",
   421→        headers: { "Content-Type": "application/json" },
   422→        body: JSON.stringify({ email: resetEmail.trim(), newPassword: resetNewPassword }),
   423→      });
   424→      const data = await res.json();
   425→      if (!res.ok) {
   426→        setResetError(data.error || "Failed to reset password.");
   427→        return;
   428→      }
   429→      setResetDone(true);
   430→      toast({ title: "Password reset successfully!", description: "You can now log in with your new password." });
   431→    } catch {
   432→      setResetError("Something went wrong. Please try again.");
   433→    } finally {
   434→      setResetLoading(false);
   435→    }
   436→  };
   437→
   438→  /* ================================================================ */
   439→  /*  RENDER                                                           */
   440→  /* ================================================================ */
   441→  return (
   442→    <Dialog open={isOpen} onOpenChange={(open) => !open && handleClose()}>
   443→      <DialogContent showCloseButton={false} className="sm:max-w-md p-0 overflow-hidden border-forest-200/40">
   444→        {/* Single custom close button */}
   445→        <button
   446→          onClick={handleClose}
   447→          className="absolute top-3 right-3 z-50 rounded-full p-1.5 text-gray-400 hover:text-gray-600 hover:bg-gray-100 transition-colors"
   448→          aria-label="Close"
   449→        >
   450→          <X className="w-5 h-5" />
   451→        </button>
   452→
   453→        <div className="px-6 pt-6 pb-2">
   454→          <DialogHeader className="space-y-1 mb-1">
   455→            <div className="flex items-center gap-2 justify-center">
   456→              <div className="w-9 h-9 rounded-xl bg-forest-900 flex items-center justify-center">
   457→                <Heart className="w-5 h-5 text-lime-400" />
   458→              </div>
   459→              <DialogTitle className="text-xl font-bold text-forest-900">
   460→                SevaSaathi
   461→              </DialogTitle>
   462→            </div>
   463→            <DialogDescription className="text-sm text-gray-500 mt-1">
   464→              {tab === "login"
   465→                ? "Sign in to your account"
   466→                : "Create your account to get started"}
   467→            </DialogDescription>
   468→          </DialogHeader>
   469→        </div>
   470→
   471→        {/* ---- Error banner ---- */}
   472→        <AnimatePresence>
   473→          {error && (
   474→            <motion.div
   475→              initial={{ height: 0, opacity: 0 }}
   476→              animate={{ height: "auto", opacity: 1 }}
   477→              exit={{ height: 0, opacity: 0 }}
   478→              className="overflow-hidden"
   479→            >
   480→              <div className="mx-6 mt-1 px-3 py-2 rounded-lg bg-red-50 border border-red-200 text-red-700 text-sm flex items-center gap-2">
   481→                <X className="w-4 h-4 flex-shrink-0" />
   482→                {error}
   483→              </div>
   484→            </motion.div>
   485→          )}
   486→        </AnimatePresence>
   487→
   488→        {/* ---- Tab switcher ---- */}
   489→        <div className="px-6 pt-3">
   490→          <div className="flex bg-gray-100 rounded-lg p-1">
   491→            <button
   492→              onClick={() => setTab("login")}
   493→              className={`flex-1 py-2 px-4 rounded-md text-sm font-medium transition-all duration-200 ${
   494→                tab === "login"
   495→                  ? "bg-white text-forest-900 shadow-sm"
   496→                  : "text-gray-500 hover:text-gray-700"
   497→              }`}
   498→            >
   499→              Sign In
   500→            </button>
   501→            <button
   502→              onClick={() => setTab("register")}
   503→              className={`flex-1 py-2 px-4 rounded-md text-sm font-medium transition-all duration-200 ${
   504→                tab === "register"
   505→                  ? "bg-white text-forest-900 shadow-sm"
   506→                  : "text-gray-500 hover:text-gray-700"
   507→              }`}
   508→            >
   509→              Create Account
   510→            </button>
   511→          </div>
   512→        </div>
   513→
   514→        {/* ---- Animated content ---- */}
   515→        <div className="px-6 py-4 overflow-hidden">
   516→          <AnimatePresence mode="wait" custom={tab}>
   517→            {/* ========== LOGIN TAB ========== */}
   518→            {tab === "login" && (
   519→              <motion.div
   520→                key="login"
   521→                custom={tab}
   522→                variants={contentVariants}
   523→                initial="enter"
   524→                animate="center"
   525→                exit="exit"
   526→                className="space-y-4"
   527→              >
   528→                {/* Google button */}
   529→                <button
   530→                  type="button"
   531→                  onClick={handleGoogleSignIn}
   532→                  className="w-full flex items-center justify-center gap-3 h-11 px-4 rounded-lg border border-gray-300 bg-white hover:bg-gray-50 active:bg-gray-100 transition-all duration-150 text-sm font-medium text-gray-700 shadow-sm"
   533→                >
   534→                  <GoogleIcon />
   535→                  Continue with Google
   536→                </button>
   537→
   538→                {/* Divider */}
   539→                <div className="flex items-center gap-3">
   540→                  <div className="flex-1 h-px bg-gray-200" />
   541→                  <span className="text-xs text-gray-400 font-medium uppercase tracking-wider">
   542→                    or continue with email
   543→                  </span>
   544→                  <div className="flex-1 h-px bg-gray-200" />
   545→                </div>
   546→
   547→                {/* Email / password form */}
   548→                <form onSubmit={handleLogin} className="space-y-3">
   549→                  <div className="space-y-1.5">
   550→                    <Label htmlFor="login-email" className="text-sm font-medium text-gray-700">
   551→                      Email
   552→                    </Label>
   553→                    <div className="relative">
   554→                      <Mail className="absolute left-3 top-1/2 -translate-y-1/2 w-4 h-4 text-gray-400" />
   555→                      <Input
   556→                        id="login-email"
   557→                        type="email"
   558→                        placeholder="you@example.com"
   559→                        value={loginEmail}
   560→                        onChange={(e) => setLoginEmail(e.target.value)}
   561→                        className="pl-10 h-11 border-gray-300 focus:border-forest-500 focus:ring-forest-500"
   562→                        autoComplete="email"
   563→                        disabled={loading}
   564→                      />
   565→                    </div>
   566→                  </div>
   567→
   568→                  <div className="space-y-1.5">
   569→                    <div className="flex items-center justify-between">
   570→                      <Label htmlFor="login-password" className="text-sm font-medium text-gray-700">
   571→                        Password
   572→                      </Label>
   573→                      <button
   574→                        type="button"
   575→                        onClick={() => setResetOpen(true)}
   576→                        className="text-xs font-medium text-forest-700 hover:text-forest-900 transition-colors"
   577→                      >
   578→                        Forgot password?
   579→                      </button>
   580→                    </div>
   581→                    <div className="relative">
   582→                      <Lock className="absolute left-3 top-1/2 -translate-y-1/2 w-4 h-4 text-gray-400" />
   583→                      <Input
   584→                        id="login-password"
   585→                        type={showPassword ? "text" : "password"}
   586→                        placeholder="Enter your password"
   587→                        value={loginPassword}
   588→                        onChange={(e) => setLoginPassword(e.target.value)}
   589→                        className="pl-10 pr-10 h-11 border-gray-300 focus:border-forest-500 focus:ring-forest-500"
   590→                        autoComplete="current-password"
   591→                        disabled={loading}
   592→                      />
   593→                      <button
   594→                        type="button"
   595→                        onClick={() => setShowPassword(!showPassword)}
   596→                        className="absolute right-3 top-1/2 -translate-y-1/2 text-gray-400 hover:text-gray-600 transition-colors"
