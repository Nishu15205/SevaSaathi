     1→"use client";
     2→
     3→import { useState, useEffect, useCallback } from "react";
     4→import {
     5→  X,
     6→  Eye,
     7→  EyeOff,
     8→  Loader2,
     9→  Mail,
    10→  Lock,
    11→  User,
    12→  Phone,
    13→  Heart,
    14→  ShieldCheck,
    15→  ArrowRight,
    16→  KeyRound,
    17→  CheckCircle2,
    18→} from "lucide-react";
    19→import { motion, AnimatePresence } from "framer-motion";
    20→import { signIn } from "next-auth/react";
    21→import { useToast } from "@/hooks/use-toast";
    22→import { useAuthStore } from "@/stores/authStore";
    23→import { api } from "@/lib/api";
    24→import {
    25→  Dialog,
    26→  DialogContent,
    27→  DialogHeader,
    28→  DialogTitle,
    29→  DialogDescription,
    30→} from "@/components/ui/dialog";
    31→import { Input } from "@/components/ui/input";
    32→import { Label } from "@/components/ui/label";
    33→import { Button } from "@/components/ui/button";
    34→import { Separator } from "@/components/ui/separator";
    35→
    36→/* ------------------------------------------------------------------ */
    37→/*  Props                                                              */
    38→/* ------------------------------------------------------------------ */
    39→interface LoginModalProps {
    40→  isOpen: boolean;
    41→  onClose: () => void;
    42→  defaultTab?: string;
    43→}
    44→
    45→/* ------------------------------------------------------------------ */
    46→/*  Types                                                              */
    47→/* ------------------------------------------------------------------ */
    48→type Tab = "login" | "register";
    49→type UserRole = "FAMILY" | "CAREGIVER";
    50→
    51→/* ------------------------------------------------------------------ */
    52→/*  Google SVG icon                                                    */
    53→/* ------------------------------------------------------------------ */

    55→  return (
    56→    <svg viewBox="0 0 24 24" width="20" height="20" className={className}>
    57→      <path d="M22.56 12.25c0-.78-.07-1.53-.2-2.25H12v4.26h5.92a5.06 5.06 0 0 1-2.2 3.32v2.77h3.57c2.08-1.92 3.28-4.74 3.28-8.1z" fill="#4285F4" />
    58→      <path d="M12 23c2.97 0 5.46-.98 7.28-2.66l-3.57-2.77c-.98.66-2.23 1.06-3.71 1.06-2.86 0-5.29-1.93-6.16-4.53H2.18v2.84C3.99 20.53 7.7 23 12 23z" fill="#34A853" />
    59→      <path d="M5.84 14.09c-.22-.66-.35-1.36-.35-2.09s.13-1.43.35-2.09V7.07H2.18C1.43 8.55 1 10.22 1 12s.43 3.45 1.18 4.93l2.85-2.22.81-.62z" fill="#FBBC05" />
    60→      <path d="M12 5.38c1.62 0 3.06.56 4.21 1.64l3.15-3.15C17.45 2.09 14.97 1 12 1 7.7 1 3.99 3.47 2.18 7.07l3.66 2.84c.87-2.6 3.3-4.53 6.16-4.53z" fill="#EA4335" />
    61→    </svg>
    62→  );
    63→}

    65→/* ------------------------------------------------------------------ */
    66→/*  Animation variants                                                 */
    67→/* ------------------------------------------------------------------ */
    68→const overlayVariants = { hidden: { opacity: 0 }, visible: { opacity: 1 }, exit: { opacity: 0 } };
    69→const modalVariants = {
    70→  hidden: { opacity: 0, scale: 0.92, y: 20 },
    71→  visible: { opacity: 1, scale: 1, y: 0, transition: { type: "spring", stiffness: 350, damping: 30 } },
    72→  exit: { opacity: 0, scale: 0.92, y: 20, transition: { duration: 0.2 } },
    73→};
    74→const contentVariants = {
    75→  enter: (tab: string) => ({ x: tab === "register" ? 40 : -40, opacity: 0 }),
    76→  center: { x: 0, opacity: 1, transition: { type: "spring", stiffness: 300, damping: 28 } },
    77→  exit: (tab: string) => ({ x: tab === "register" ? -40 : 40, opacity: 0, transition: { duration: 0.15 } }),
    78→};
    79→
    80→/* ================================================================== */
    81→/*  MAIN COMPONENT                                                     */
    82→/* ================================================================== */
    83→export default function LoginModal({ isOpen, onClose, defaultTab }: LoginModalProps) {
    84→  /* ---- state ---- */
    85→  const [tab, setTab] = useState<Tab>(defaultTab === "register" ? "register" : "login");
    86→  const [showPassword, setShowPassword] = useState(false);
    87→  const [loading, setLoading] = useState(false);
    88→  const [error, setError] = useState("");
    89→  const [resetOpen, setResetOpen] = useState(false);
    90→
    91→  // Login form
    92→  const [loginEmail, setLoginEmail] = useState("");
    93→  const [loginPassword, setLoginPassword] = useState("");
    94→
    95→  // Register form
    96→  const [regName, setRegName] = useState("");
    97→  const [regEmail, setRegEmail] = useState("");
    98→  const [regPhone, setRegPhone] = useState("");
    99→  const [regPassword, setRegPassword] = useState("");
   100→  const [regRole, setRegRole] = useState<UserRole>("FAMILY");
   101→
   102→  // Reset form
   103→  const [resetEmail, setResetEmail] = useState("");
   104→  const [resetNewPassword, setResetNewPassword] = useState("");
   105→  const [resetLoading, setResetLoading] = useState(false);
   106→  const [resetError, setResetError] = useState("");
   107→  const [resetDone, setResetDone] = useState(false);
   108→
   109→  const { setAuth } = useAuthStore();
   110→  const { toast } = useToast();
   111→
   112→  // Google popup form
   113→  const [googleOpen, setGoogleOpen] = useState(false);
   114→  const [googleEmail, setGoogleEmail] = useState("");
   115→  const [googleName, setGoogleName] = useState("");
   116→  const [googleError, setGoogleError] = useState("");
   117→  const [googleSubmitting, setGoogleSubmitting] = useState(false);
   118→
   119→  /* ---- reset form fields when switching tabs ---- */
   120→  useEffect(() => { setError(""); setLoginPassword(""); }, [tab]);
   121→
   122→  /* ---- reset entire state when modal closes ---- */
   123→  const handleClose = useCallback(() => {
   124→    setTab(defaultTab === "register" ? "register" : "login");
   125→    setError(""); setShowPassword(false); setLoading(false);
   126→    setLoginEmail(""); setLoginPassword("");
   127→    setRegName(""); setRegEmail(""); setRegPhone(""); setRegPassword(""); setRegRole("FAMILY");
   128→    setResetOpen(false); setResetEmail(""); setResetNewPassword("");
   129→    setResetLoading(false); setResetError(""); setResetDone(false);
   130→    setGoogleOpen(false); setGoogleEmail(""); setGoogleName(""); setGoogleError(""); setGoogleSubmitting(false);
   131→    onClose();
   132→  }, [onClose, defaultTab]);
   133→
   134→  /* ================================================================ */
   135→  /*  HANDLERS                                                         */
   136→/* ================================================================ */
   137→
   138→  const handleGoogleSignIn = async () => {
   139→    setGoogleError("");
   140→    try {
   141→      const checkRes = await fetch("/api/auth/google-configured");
   142→      const { configured } = await checkRes.json();
   143→      if (configured) {
   144→        const origin = window.location.origin;
   145→        const role = tab === "register" ? regRole : "";
   146→        window.location.href = `/api/auth/google-go?origin=${encodeURIComponent(origin)}&role=${encodeURIComponent(role)}`;
   147→        return;
   148→      }
   149→    } catch { /* fall through */ }
   150→    setGoogleOpen(true);
   151→  };
   152→
   153→  const handleGoogleSubmit = async (e: React.FormEvent) => {
   154→    e.preventDefault();
   155→    if (!googleEmail.trim() || !googleName.trim()) { setGoogleError("Please enter your name and email."); return; }
   156→    if (!googleEmail.trim().includes("@")) { setGoogleError("Please enter a valid email address."); return; }
   157→    setGoogleSubmitting(true); setGoogleError("");
   158→    try {
   159→      const res = await fetch("/api/auth/google-simulate", {
   160→        method: "POST", headers: { "Content-Type": "application/json" },
   161→        body: JSON.stringify({ email: googleEmail.trim(), name: googleName.trim(), role: tab === "register" ? regRole : undefined }),
   162→      });
   163→      const data = await res.json();
   164→      if (!res.ok) { setGoogleError(data.error || "Google sign-in failed."); return; }
   165→      const sessionRes = await fetch("/api/auth/session");
   166→      const sessionData = await sessionRes.json();
   167→      if (sessionData?.user) {
   168→        const u = sessionData.user as any;
   169→        setAuth({ id: u.id, email: u.email, name: u.name, phone: "", role: u.role, avatarUrl: u.image || null, subscription: "NONE" as any, patientProfiles: [], caregiverProfile: null });
   170→        setGoogleOpen(false);
   171→        toast({ title: "Welcome" + (data.user.isNewUser ? " to SevaSaathi!" : " back!") + " \ud83c\udf89", description: "Signed in as " + data.user.email });
   172→        handleClose();
   173→      } else { window.location.href = window.location.origin; }
   174→    } catch { setGoogleError("Something went wrong. Please try again."); } finally { setGoogleSubmitting(false); }
   175→  };
   176→
   177→  const handleLogin = async (e: React.FormEvent) => {
   178→    e.preventDefault();
   179→    if (!loginEmail.trim() || !loginPassword.trim()) { setError("Please enter your email and password."); return; }
   180→    setLoading(true); setError("");
   181→    try {
   182→      const result = await signIn("credentials", { email: loginEmail.trim(), password: loginPassword, redirect: false });
   183→      if (result?.error) { setError("Invalid email or password. Please try again."); return; }
   184→      try {
   185→        const sessionRes = await fetch("/api/auth/session");
   186→        const sessionData = await sessionRes.json();
   187→        if (sessionData?.user) {
   188→          const u = sessionData.user as any;
   189→          setAuth({ id: u.id || "", email: u.email || "", name: u.name || "", phone: "", role: u.role || "FAMILY", avatarUrl: u.image || null, subscription: "NONE", patientProfiles: [], caregiverProfile: null });
   190→        }
   191→      } catch { /* AuthSync will pick it up */ }
   192→      toast({ title: "Welcome back! \ud83c\udf89", description: "You have successfully logged in." });
   193→      handleClose();
   194→    } catch { setError("Something went wrong. Please try again."); } finally { setLoading(false); }
   195→  };
   196→
   197→  const handleRegister = async (e: React.FormEvent) => {
   198→    e.preventDefault();
   199→    if (!regName.trim() || !regEmail.trim() || !regPhone.trim() || !regPassword.trim()) { setError("Please fill in all fields."); return; }
   200→    if (regPassword.length < 6) { setError("Password must be at least 6 characters."); return; }
   201→    setLoading(true); setError("");
   202→    try {
   203→      await api.auth.register({ name: regName.trim(), email: regEmail.trim(), phone: regPhone.trim(), password: regPassword, role: regRole });
   204→      const loginResult = await signIn("credentials", { email: regEmail.trim(), password: regPassword, redirect: false });
   205→      if (loginResult?.error) {
   206→        toast({ title: "Account created!", description: "Please log in with your credentials." });
   207→        setTab("login"); setLoginEmail(regEmail.trim()); return;
   208→      }
   209→      try {
   210→        const sessionRes = await fetch("/api/auth/session");
   211→        const sessionData = await sessionRes.json();
   212→        if (sessionData?.user) {
   213→          const u = sessionData.user as any;
   214→          setAuth({ id: u.id || "", email: u.email || "", name: u.name || "", phone: regPhone.trim(), role: u.role || regRole, avatarUrl: u.image || null, subscription: "NONE", patientProfiles: [], caregiverProfile: null });
   215→        }
   216→      } catch { /* AuthSync will pick it up */ }
   217→      toast({ title: "Welcome to SevaSaathi! \ud83c\udf89", description: "Your account has been created." });
   218→      handleClose();
   219→    } catch (err: any) { setError(err?.message || "Registration failed. Please try again."); } finally { setLoading(false); }
   220→  };
   221→
   222→  const handleReset = async (e: React.FormEvent) => {
   223→    e.preventDefault();
   224→    if (!resetEmail.trim() || !resetNewPassword.trim()) { setResetError("Please enter your email and new password."); return; }
   225→    if (resetNewPassword.length < 6) { setResetError("Password must be at least 6 characters."); return; }
   226→    setResetLoading(true); setResetError("");
   227→    try {
   228→      const res = await fetch("/api/auth/reset-password", { method: "POST", headers: { "Content-Type": "application/json" }, body: JSON.stringify({ email: resetEmail.trim(), newPassword: resetNewPassword }) });
   229→      const data = await res.json();
   230→      if (!res.ok) { setResetError(data.error || "Failed to reset password."); return; }
   231→      setResetDone(true);
   232→      toast({ title: "Password reset successfully!", description: "You can now log in with your new password." });
   233→    } catch { setResetError("Something went wrong. Please try again."); } finally { setResetLoading(false); }
   234→  };
   235→
   236→  /* ================================================================ */
   237→  /*  RENDER                                                           */
   238→/* ================================================================ */
   239→  return (
   240→    <Dialog open={isOpen} onOpenChange={(open) => !open && handleClose()}>
   241→      <DialogContent showCloseButton={false} className="sm:max-w-md p-0 overflow-hidden border-forest-200/40">
   242→        <button onClick={handleClose} className="absolute top-3 right-3 z-50 rounded-full p-1.5 text-gray-400 hover:text-gray-600 hover:bg-gray-100 transition-colors" aria-label="Close">
   243→          <X className="w-5 h-5" />
   244→        </button>
   245→
   246→        <div className="px-6 pt-6 pb-2">
   247→          <DialogHeader className="space-y-1 mb-1">
   248→            <div className="flex items-center gap-2 justify-center">
   249→              <div className="w-9 h-9 rounded-xl bg-forest-900 flex items-center justify-center">
   250→                <Heart className="w-5 h-5 text-lime-400" />
   251→              </div>
   252→              <DialogTitle className="text-xl font-bold text-forest-900">SevaSaathi</DialogTitle>
   253→            </div>
   254→            <DialogDescription className="text-sm text-gray-500 mt-1">
   255→              {tab === "login" ? "Sign in to your account" : "Create your account to get started"}
   256→            </DialogDescription>
   257→          </DialogHeader>
   258→        </div>
   259→
   260→        <AnimatePresence>
   261→          {error && (
   262→            <motion.div initial={{ height: 0, opacity: 0 }} animate={{ height: "auto", opacity: 1 }} exit={{ height: 0, opacity: 0 }} className="overflow-hidden">
   263→              <div className="mx-6 mt-1 px-3 py-2 rounded-lg bg-red-50 border border-red-200 text-red-700 text-sm flex items-center gap-2">
   264→                <X className="w-4 h-4 flex-shrink-0" />{error}
   265→              </div>
   266→            </motion.div>
   267→          )}
   268→        </AnimatePresence>
   269→
   270→        <div className="px-6 pt-3">
   271→          <div className="flex bg-gray-100 rounded-lg p-1">
   272→            <button onClick={() => setTab("login")} className={`flex-1 py-2 px-4 rounded-md text-sm font-medium transition-all ${tab === "login" ? "bg-white text-forest-900 shadow-sm" : "text-gray-500 hover:text-gray-700"}`}>Sign In</button>
   273→            <button onClick={() => setTab("register")} className={`flex-1 py-2 px-4 rounded-md text-sm font-medium transition-all ${tab === "register" ? "bg-white text-forest-900 shadow-sm" : "text-gray-500 hover:text-gray-700"}`}>Create Account</button>
   274→          </div>
   275→        </div>
   276→
   277→        <div className="px-6 py-4 overflow-y-auto max-h-[55vh]">
   278→          <AnimatePresence mode="wait" custom={tab}>
   279→            {tab === "login" && (
   280→              <motion.div key="login" custom={tab} variants={contentVariants} initial="enter" animate="center" exit="exit" className="space-y-4">
   281→                <button type="button" onClick={handleGoogleSignIn} className="w-full flex items-center justify-center gap-3 h-11 px-4 rounded-lg border border-gray-300 bg-white hover:bg-gray-50 transition-all text-sm font-medium text-gray-700 shadow-sm">
   282→                  <GoogleIcon /> Continue with Google
   283→                </button>
   284→                <div className="flex items-center gap-3">
   285→                  <div className="flex-1 h-px bg-gray-200" />
   286→                  <span className="text-xs text-gray-400 font-medium uppercase tracking-wider">or continue with email</span>
   287→                  <div className="flex-1 h-px bg-gray-200" />
   288→                </div>
   289→                <form onSubmit={handleLogin} className="space-y-3">
   290→                  <div className="space-y-1.5">
   291→                    <Label htmlFor="login-email" className="text-sm font-medium text-gray-700">Email</Label>
   292→                    <div className="relative">
   293→                      <Mail className="absolute left-3 top-1/2 -translate-y-1/2 w-4 h-4 text-gray-400" />
   294→                      <Input id="login-email" type="email" placeholder="you@example.com" value={loginEmail} onChange={(e) => setLoginEmail(e.target.value)} className="pl-10 h-11 border-gray-300" autoComplete="email" disabled={loading} />
   295→                    </div>
   296→                  </div>
   297→                  <div className="space-y-1.5">
   298→                    <div className="flex items-center justify-between">
   299→                      <Label htmlFor="login-password" className="text-sm font-medium text-gray-700">Password</Label>
   300→                      <button type="button" onClick={() => setResetOpen(true)} className="text-xs font-medium text-forest-700 hover:text-forest-900">Forgot password?</button>
   301→                    </div>
   302→                    <div className="relative">
   303→                      <Lock className="absolute left-3 top-1/2 -translate-y-1/2 w-4 h-4 text-gray-400" />
   304→                      <Input id="login-password" type={showPassword ? "text" : "password"} placeholder="Enter your password" value={loginPassword} onChange={(e) => setLoginPassword(e.target.value)} className="pl-10 pr-10 h-11 border-gray-300" autoComplete="current-password" disabled={loading} />
   305→                      <button type="button" onClick={() => setShowPassword(!showPassword)} className="absolute right-3 top-1/2 -translate-y-1/2 text-gray-400 hover:text-gray-600" tabIndex={-1}>
   306→                        {showPassword ? <EyeOff className="w-4 h-4" /> : <Eye className="w-4 h-4" />}
   307→                      </button>
   308→                    </div>
   309→                  </div>
   310→                  <Button type="submit" disabled={loading} className="w-full h-11 bg-forest-900 hover:bg-forest-800 text-white font-semibold rounded-lg shadow-sm hover:shadow-md">
   311→                    {loading ? <><Loader2 className="w-4 h-4 mr-2 animate-spin" />Signing in...</> : <>Sign In <ArrowRight className="w-4 h-4 ml-2" /></>}
   312→                  </Button>
   313→                </form>
   314→                <p className="text-center text-sm text-gray-500 pt-1">Don&apos;t have an account? <button onClick={() => setTab("register")} className="font-semibold text-forest-700 hover:text-forest-900">Create one</button></p>
   315→              </motion.div>
   316→            )}
   317→
   318→            {tab === "register" && (
   319→              <motion.div key="register" custom={tab} variants={contentVariants} initial="enter" animate="center" exit="exit" className="space-y-4">
   320→                {/* Role selector */}
   321→                <div className="space-y-2">
   322→                  <Label className="text-sm font-semibold text-gray-800">How would you like to use SevaSaathi?</Label>
   323→                  <div className="grid grid-cols-1 gap-2">
   324→                    <button type="button" onClick={() => setRegRole("FAMILY")} className={`flex items-start gap-3 p-3 rounded-xl border-2 text-left transition-all ${regRole === "FAMILY" ? "border-forest-500 bg-forest-50" : "border-gray-200 bg-white hover:border-gray-300"}`}>
   325→                      <div className={`w-9 h-9 rounded-lg flex items-center justify-center shrink-0 mt-0.5 ${regRole === "FAMILY" ? "bg-forest-100" : "bg-gray-100"}`}>
   326→                        <Heart className={`w-4.5 h-4.5 ${regRole === "FAMILY" ? "text-forest-700" : "text-gray-400"}`} />
   327→                      </div>
   328→                      <div className="min-w-0">
   329→                        <p className={`text-sm font-semibold ${regRole === "FAMILY" ? "text-forest-900" : "text-gray-700"}`}>I need care for my family</p>
   330→                        <p className={`text-xs mt-0.5 ${regRole === "FAMILY" ? "text-forest-600/80" : "text-gray-400"}`}>Find trusted caregivers for parents, elders, or family members at home</p>
   331→                      </div>
   332→                    </button>
   333→                    <button type="button" onClick={() => setRegRole("CAREGIVER")} className={`flex items-start gap-3 p-3 rounded-xl border-2 text-left transition-all ${regRole === "CAREGIVER" ? "border-forest-500 bg-forest-50" : "border-gray-200 bg-white hover:border-gray-300"}`}>
   334→                      <div className={`w-9 h-9 rounded-lg flex items-center justify-center shrink-0 mt-0.5 ${regRole === "CAREGIVER" ? "bg-forest-100" : "bg-gray-100"}`}>
   335→                        <ShieldCheck className={`w-4.5 h-4.5 ${regRole === "CAREGIVER" ? "text-forest-700" : "text-gray-400"}`} />
   336→                      </div>
   337→                      <div className="min-w-0">
   338→                        <p className={`text-sm font-semibold ${regRole === "CAREGIVER" ? "text-forest-900" : "text-gray-700"}`}>I want to provide care services</p>
   339→                        <p className={`text-xs mt-0.5 ${regRole === "CAREGIVER" ? "text-forest-600/80" : "text-gray-400"}`}>Join as a caregiver, get bookings, and earn money</p>
   340→                      </div>
   341→                    </button>
   342→                  </div>
   343→                </div>
   344→
   345→                <button type="button" onClick={handleGoogleSignIn} className="w-full flex items-center justify-center gap-3 h-11 px-4 rounded-lg border border-gray-300 bg-white hover:bg-gray-50 transition-all text-sm font-medium text-gray-700 shadow-sm">
   346→                  <GoogleIcon /> Sign up with Google as {regRole === "CAREGIVER" ? "Caregiver" : "Family Member"}
   347→                </button>
   348→                <div className="flex items-center gap-3">
   349→                  <div className="flex-1 h-px bg-gray-200" />
   350→                  <span className="text-xs text-gray-400 font-medium uppercase tracking-wider">or register with email</span>
   351→                  <div className="flex-1 h-px bg-gray-200" />
   352→                </div>
   353→
   354→                <form onSubmit={handleRegister} className="space-y-3">
   355→                  <div className="space-y-1.5">
   356→                    <Label htmlFor="reg-name" className="text-sm font-medium text-gray-700">Full Name</Label>
   357→                    <div className="relative">
   358→                      <User className="absolute left-3 top-1/2 -translate-y-1/2 w-4 h-4 text-gray-400" />
   359→                      <Input id="reg-name" type="text" placeholder="Your full name" value={regName} onChange={(e) => setRegName(e.target.value)} className="pl-10 h-11 border-gray-300" autoComplete="name" disabled={loading} />
   360→                    </div>
   361→                  </div>
   362→                  <div className="space-y-1.5">
   363→                    <Label htmlFor="reg-email" className="text-sm font-medium text-gray-700">Email</Label>
   364→                    <div className="relative">
   365→                      <Mail className="absolute left-3 top-1/2 -translate-y-1/2 w-4 h-4 text-gray-400" />
   366→                      <Input id="reg-email" type="email" placeholder="you@example.com" value={regEmail} onChange={(e) => setRegEmail(e.target.value)} className="pl-10 h-11 border-gray-300" autoComplete="email" disabled={loading} />
   367→                    </div>
   368→                  </div>
   369→                  <div className="space-y-1.5">
   370→                    <Label htmlFor="reg-phone" className="text-sm font-medium text-gray-700">Phone Number</Label>
   371→                    <div className="relative">
   372→                      <Phone className="absolute left-3 top-1/2 -translate-y-1/2 w-4 h-4 text-gray-400" />
   373→                      <Input id="reg-phone" type="tel" placeholder="10-digit mobile number" value={regPhone} onChange={(e) => setRegPhone(e.target.value)} className="pl-10 h-11 border-gray-300" autoComplete="tel" disabled={loading} />
   374→                    </div>
   375→                  </div>
   376→                  <div className="space-y-1.5">
