   198→function parseJsonSafe(jsonStr: string | null | undefined): string[] {
   199→  if (!jsonStr) return [];
   200→  try {
   201→    const parsed = JSON.parse(jsonStr);
   202→    return Array.isArray(parsed) ? parsed : [];
   203→  } catch {
   204→    return jsonStr.split(',').map((s) => s.trim()).filter(Boolean);
   205→  }
   206→}
   207→
   208→function formatDate(dateStr: string): string {
   209→  try {
   210→    return new Date(dateStr).toLocaleDateString('en-IN', {
   211→      day: 'numeric',
   212→      month: 'short',
   213→      year: 'numeric',
   214→    });
   215→  } catch {
   216→    return dateStr;
   217→  }
   218→}
   219→
   220→function getTodayDate(): string {
   221→  const now = new Date();
   222→  const year = now.getFullYear();
   223→  const month = String(now.getMonth() + 1).padStart(2, '0');
   224→  const day = String(now.getDate()).padStart(2, '0');
   225→  return `${year}-${month}-${day}`;
   226→}

   478→function CreateProfileForm({ userId, onCreated }: { userId: string; onCreated: (profile: any) => void }) {
   479→  const [saving, setSaving] = useState(false);
   480→  const [selectedSkills, setSelectedSkills] = useState<string[]>(['elderly-care']);
   481→  const [form, setForm] = useState({
   482→    city: '', yearsExperience: '', hourlyRate: '', qualifications: '', languages: 'Hindi, English', bio: '',
   483→  });
   484→
   485→  const toggleSkill = (skill: string) => {
   486→    setSelectedSkills(prev => prev.includes(skill) ? prev.filter(s => s !== skill) : [...prev, skill]);
   487→  };
   488→
   489→  const handleSubmit = async (e: React.FormEvent) => {
   490→    e.preventDefault();
   491→    if (!form.city || !form.yearsExperience || !form.hourlyRate) {
   492→      toast.error('Please fill in City, Experience, and Hourly Rate.');
   493→      return;
   494→    }
   495→    if (selectedSkills.length === 0) {
   496→      toast.error('Please select at least one skill.');
   497→      return;
   498→    }
   499→    setSaving(true);
   500→    try {
   501→      const res = await api.caregivers.create({
   502→        userId,
   503→        city: form.city,
   504→        yearsExperience: parseInt(form.yearsExperience),
   505→        hourlyRate: parseInt(form.hourlyRate),
   506→        skills: JSON.stringify(selectedSkills),
   507→        qualifications: form.qualifications ? JSON.stringify(form.qualifications.split(',').map((s: string) => s.trim()).filter(Boolean)) : JSON.stringify(['Caregiver']),
   508→        languages: JSON.stringify(form.languages.split(',').map((s: string) => s.trim()).filter(Boolean)),
   509→        bio: form.bio,
   510→      });
   511→      toast.success('Profile created successfully!');
   512→      onCreated(res.caregiver);
   513→    } catch (err: any) {
   514→      toast.error(err.message || 'Failed to create profile');
   515→    } finally {
   516→      setSaving(false);
   517→    }
   518→  };
   519→
   520→  return (
   521→    <div className="space-y-6 max-w-2xl">
   522→      <div>
   523→        <h2 className="text-2xl font-bold text-gray-900">Create Your Profile</h2>
   524→        <p className="text-sm text-gray-400 mt-1">Set up your caregiver profile to start receiving booking requests from families.</p>
   525→      </div>
   526→      <Card className="rounded-2xl border-gray-100">
   527→        <CardContent className="p-6">
