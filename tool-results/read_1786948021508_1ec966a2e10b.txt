   286→/* ─────────── VERIFICATIONS TAB ─────────── */
   287→function VerificationsTab() {
   288→  const [verifications, setVerifications] = useState<any[]>([]);
   289→  const [loading, setLoading] = useState(true);
   290→  const [rejecting, setRejecting] = useState<string | null>(null);
   291→  const [approving, setApproving] = useState<string | null>(null);
   292→  const [reasons, setReasons] = useState<Record<string, string>>({});
   293→
   294→  const fetchVerifications = useCallback(async () => {
   295→    setLoading(true);
   296→    try {
   297→      const res = await api.admin.verifications.list();
   298→      setVerifications(res.verifications ?? []);
   299→    } catch {
   300→      toast.error('Failed to load verifications');
   301→    } finally {
   302→      setLoading(false);
   303→    }
   304→  }, []);
   305→
   306→  useEffect(() => { fetchVerifications(); }, [fetchVerifications]);
   307→
   308→  const handleApprove = async (id: string) => {
   309→    setApproving(id);
   310→    try {
   311→      await api.admin.verifications.update(id, { status: 'APPROVED' });
   312→      toast.success('Verification approved successfully');
   313→      fetchVerifications();
   314→    } catch {
   315→      toast.error('Failed to approve verification');
   316→    } finally {
   317→      setApproving(null);
   318→    }
   319→  };
   320→
   321→  const handleReject = async (id: string) => {
   322→    const reason = reasons[id]?.trim();
   323→    if (!reason) { toast.error('Please provide a rejection reason'); return; }
   324→    setRejecting(id);
   325→    try {
   326→      await api.admin.verifications.update(id, { status: 'REJECTED', rejectionReason: reason });
   327→      toast.success('Verification rejected');
   328→      setReasons((prev) => { const n = { ...prev }; delete n[id]; return n; });
   329→      fetchVerifications();
   330→    } catch {
   331→      toast.error('Failed to reject verification');
   332→    } finally {
   333→      setRejecting(null);
   334→    }
   335→  };
   336→
   337→  if (loading) {
   338→    return <div className="space-y-3">{Array.from({ length: 3 }).map((_, i) => <Skeleton key={i} className="h-24 rounded-2xl" />)}</div>;
   339→  }
   340→
   341→  if (verifications.length === 0) {
   342→    return <Card className="rounded-2xl border-0 shadow-sm"><CardContent className="py-16 text-center"><ShieldCheck className="w-12 h-12 mx-auto text-[#14532d]/30 mb-3" /><p className="text-muted-foreground">No pending verifications.</p></CardContent></Card>;
   343→  }
   344→
   345→  return (
   346→    <div className="space-y-3">
   347→      {verifications.map((v) => (
   348→        <motion.div key={v.id} variants={fadeVariants} initial="hidden" animate="visible">
   349→          <Card className="rounded-2xl card-hover border-0 shadow-sm">
   350→            <CardContent className="p-4 space-y-3">
   351→              <div className="flex flex-col sm:flex-row sm:items-center justify-between gap-2">
   352→                <div className="flex items-center gap-2 flex-wrap">
   353→                  <span className="font-semibold text-sm">{v.caregiver?.user?.name ?? 'Caregiver'}</span>
   354→                  <Badge className={docTypeBadge[v.docType] ?? 'bg-gray-100 text-gray-700'}>{v.docType ?? 'DOC'}</Badge>
   355→                </div>
   356→                <span className="text-xs text-muted-foreground flex items-center gap-1">
   357→                  <Clock className="w-3 h-3" />{v.createdAt ? new Date(v.createdAt).toLocaleDateString('en-IN', { day: 'numeric', month: 'short', year: 'numeric' }) : '—'}
   358→                </span>
   359→              </div>
   360→              {v.docNumber && <p className="text-xs text-muted-foreground font-mono bg-muted/50 rounded-lg px-3 py-1.5 w-fit">{v.docNumber}</p>}
   361→              <div className="flex items-center gap-2 pt-1">
   362→                <Button
   363→                  size="sm"
   364→                  className="bg-[#14532d] hover:bg-[#14532d]/90 text-white rounded-full"
   365→                  disabled={approving === v.id}
   366→                  onClick={() => handleApprove(v.id)}
   367→                >
   368→                  {approving === v.id ? <Loader2 className="w-4 h-4 animate-spin" /> : <><CheckCircle2 className="w-4 h-4 mr-1" />Approve</>}
   369→                </Button>
   370→                <Button
   371→                  size="sm"
   372→                  variant="outline"
   373→                  className="text-red-600 border-red-200 hover:bg-red-50 rounded-full"
   374→                  disabled={rejecting === v.id}
   375→                  onClick={() => {
   376→                    if (!reasons[v.id]?.trim()) {
   377→                      setReasons((prev) => ({ ...prev, [v.id]: prev[v.id] ?? '' }));
   378→                    } else {
   379→                      handleReject(v.id);
   380→                    }
   381→                  }}
   382→                >
   383→                  {rejecting === v.id ? <Loader2 className="w-4 h-4 animate-spin" /> : <><XCircle className="w-4 h-4 mr-1" />Reject</>}
   384→                </Button>
   385→              </div>
   386→              {reasons[v.id] !== undefined && (
   387→                <motion.div initial={{ height: 0, opacity: 0 }} animate={{ height: 'auto', opacity: 1 }} className="space-y-2">
   388→                  <Textarea
   389→                    placeholder="Enter rejection reason..."
   390→                    value={reasons[v.id]}
   391→                    onChange={(e) => setReasons((prev) => ({ ...prev, [v.id]: e.target.value }))}
   392→                    className="text-sm rounded-xl min-h-[60px] resize-none"
   393→                  />
   394→                  <Button size="sm" variant="destructive" className="rounded-full" disabled={rejecting === v.id} onClick={() => handleReject(v.id)}>
   395→                    {rejecting === v.id ? <Loader2 className="w-4 h-4 animate-spin" /> : <Ban className="w-4 h-4 mr-1" />}Confirm Rejection
   396→                  </Button>
   397→                </motion.div>
   398→              )}
   399→            </CardContent>
   400→          </Card>
   401→        </motion.div>
   402→      ))}
   403→    </div>
   404→  );
   405→}

   410→  const [loading, setLoading] = useState(true);

   412→  useEffect(() => {

   416→        const caregivers = res.users ?? [];
   417→        const allBookings: any[] = [];
   418→        await Promise.all(
   419→          caregivers.map(async (c: any) => {
   420→            try {
   421→              const caregiverId = c.caregiverProfile?.id;
   422→              if (!caregiverId) return;
   423→              const bRes = await api.bookings.list({ caregiverId });
   424→              allBookings.push(...(bRes.bookings ?? []).map((b: any) => ({
   425→                ...b,
   426→                caregiverName: b.caregiver?.user?.name ?? c.name,
   427→                patientName: b.patient?.name ?? 'Patient',
   428→                date: b.startDate,
   429→                amount: b.totalAmount,
   430→              })));
   431→            } catch { /* skip */ }
   432→          })
   433→        );
   434→        allBookings.sort((a, b) => new Date(b.createdAt ?? 0).getTime() - new Date(a.createdAt ?? 0).getTime());
   435→        setBookings(allBookings);
   436→      } catch {
   437→        toast.error('Failed to load bookings');
   438→      } finally {
   439→        setLoading(false);
   440→      }
   441→    })();
   442→  }, []);
   443→
   444→  if (loading) {
   445→    return <div className="space-y-3">{Array.from({ length: 4 }).map((_, i) => <Skeleton key={i} className="h-20 rounded-2xl" />)}</div>;
   446→  }
   447→
   448→  return (
   449→    <div className="space-y-4">
   450→      <p className="text-sm text-muted-foreground">Showing recent platform bookings across caregivers.</p>
   451→      {bookings.length === 0 ? (
   452→        <Card className="rounded-2xl border-0 shadow-sm"><CardContent className="py-16 text-center"><CalendarCheck className="w-12 h-12 mx-auto text-[#14532d]/30 mb-3" /><p className="text-muted-foreground">No bookings found.</p></CardContent></Card>
   453→      ) : (
   454→        <div className="space-y-3 max-h-[600px] overflow-y-auto">

   554→/* ─────────── COMPLAINTS TAB ─────────── */
   555→function ComplaintsTab() {
   556→  const [complaints, setComplaints] = useState<any[]>([]);
   557→  const [loading, setLoading] = useState(true);
   558→  const [actionId, setActionId] = useState<string | null>(null);
   559→  const [expandedId, setExpandedId] = useState<string | null>(null);
   560→  const [resolutions, setResolutions] = useState<Record<string, string>>({});
   561→  const [statusFilter, setStatusFilter] = useState<string>('');
   562→
   563→  const fetchComplaints = useCallback(async () => {
   564→    setLoading(true);
   565→    try {
   566→      const params: any = {};
   567→      if (statusFilter) params.status = statusFilter;
   568→      const res = await api.complaints.list(params);
   569→      setComplaints(res.complaints ?? []);
   570→    } catch {
   571→      toast.error('Failed to load complaints');
   572→    } finally {
   573→      setLoading(false);
   574→    }
   575→  }, [statusFilter]);
   576→
   577→  useEffect(() => { fetchComplaints(); }, [fetchComplaints]);
   578→
   579→  const handleResolve = async (id: string) => {
   580→    const resolution = resolutions[id]?.trim();
   581→    if (!resolution) { toast.error('Please enter a resolution message'); return; }
   582→    setActionId(id);
   583→    try {
   584→      await api.complaints.update(id, { status: 'RESOLVED', resolution });
   585→      toast.success('Complaint resolved');
   586→      setResolutions((prev) => { const n = { ...prev }; delete n[id]; return n; });
   587→      setExpandedId(null);
   588→      fetchComplaints();
   589→    } catch (err: any) {
   590→      toast.error(err.message || 'Failed to resolve complaint');
   591→    } finally {
   592→      setActionId(null);
   593→    }
   594→  };
   595→
   596→  const handleDismiss = async (id: string) => {
   597→    const resolution = resolutions[id]?.trim() || 'Complaint dismissed by admin.';
   598→    setActionId(id);
   599→    try {
   600→      await api.complaints.update(id, { status: 'DISMISSED', resolution });
   601→      toast.success('Complaint dismissed');
   602→      setResolutions((prev) => { const n = { ...prev }; delete n[id]; return n; });
   603→      setExpandedId(null);
   604→      fetchComplaints();
   605→    } catch (err: any) {
   606→      toast.error(err.message || 'Failed to dismiss complaint');
   607→    } finally {
   608→      setActionId(null);
   609→    }
   610→  };
   611→
   612→  const handleInProgress = async (id: string) => {
   613→    setActionId(id);
   614→    try {
   615→      await api.complaints.update(id, { status: 'IN_PROGRESS' });
   616→      toast.success('Complaint marked as in progress');
   617→      fetchComplaints();
   618→    } catch (err: any) {
   619→      toast.error(err.message || 'Failed to update complaint');
   620→    } finally {
   621→      setActionId(null);
   622→    }
   623→  };
   624→
   625→  const openCount = complaints.filter(c => c.status === 'OPEN').length;
   626→
   627→  if (loading) {
   628→    return <div className="space-y-3">{Array.from({ length: 3 }).map((_, i) => <Skeleton key={i} className="h-24 rounded-2xl" />)}</div>;
   629→  }
   630→
   631→  return (
   632→    <div className="space-y-4">
   633→      <div className="flex items-center justify-between flex-wrap gap-3">
   634→        <div className="flex items-center gap-3">
   635→          <div className="w-10 h-10 rounded-xl bg-[#14532d]/10 flex items-center justify-center">
   636→            <MessageSquare className="w-5 h-5 text-[#14532d]" />
   637→          </div>
   638→          <div>
   639→            <p className="font-semibold text-sm">Complaints Management</p>
   640→            <p className="text-xs text-muted-foreground">{complaints.length} total · {openCount} open</p>
   641→          </div>
   642→        </div>
   643→      </div>
   644→
   645→      <div className="flex gap-2 flex-wrap">
   646→        {['', 'OPEN', 'IN_PROGRESS', 'RESOLVED', 'DISMISSED'].map((f) => (
   647→          <Button
   648→            key={f}
   649→            variant={statusFilter === f ? 'default' : 'outline'}
   650→            size="sm"
   651→            className={statusFilter === f ? 'bg-[#14532d] hover:bg-[#14532d]/90 text-white rounded-full' : 'rounded-full'}
   652→            onClick={() => setStatusFilter(f)}
   653→          >
   654→            {f ? f.replace(/_/g, ' ') : 'All'}
   655→          </Button>
   656→        ))}
   657→      </div>
   658→
   659→      {complaints.length === 0 ? (
   660→        <Card className="rounded-2xl border-0 shadow-sm"><CardContent className="py-16 text-center"><AlertTriangle className="w-12 h-12 mx-auto text-[#14532d]/30 mb-3" /><p className="text-muted-foreground">No complaints found.</p></CardContent></Card>
   661→      ) : (
   662→        <div className="space-y-3 max-h-[600px] overflow-y-auto">
   663→          {complaints.map((c, i) => (
   664→            <motion.div key={c.id} initial={{ opacity: 0, y: 6 }} animate={{ opacity: 1, y: 0 }} transition={{ delay: i * 0.03 }}>
   665→              <Card className="rounded-2xl card-hover border-0 shadow-sm">
   666→                <CardContent className="p-4 space-y-3">
   667→                  <div className="flex items-center justify-between flex-wrap gap-2">
   668→                    <div className="flex items-center gap-2 min-w-0">
   669→                      <span className="font-semibold text-sm truncate">{c.subject ?? 'Complaint'}</span>
   670→                      <Badge className={statusColors[c.status] ?? 'bg-gray-100 text-gray-700'}>{c.status?.replace(/_/g, ' ')}</Badge>
   671→                      {c.priority && <Badge className={`text-[10px] rounded-full ${priorityColors[c.priority] || ''}`}>{c.priority}</Badge>}
   672→                    </div>
   673→                    <div className="flex items-center gap-1 text-xs text-muted-foreground">
   674→                      <Clock className="w-3 h-3" />
   675→                      {c.createdAt ? new Date(c.createdAt).toLocaleDateString('en-IN', { day: 'numeric', month: 'short', year: 'numeric' }) : ''}
   676→                    </div>
   677→                  </div>
   678→
   679→                  <div className="flex items-center gap-4 text-xs text-muted-foreground">
   680→                    <span>From: {c.family?.name || 'Family'}</span>
   681→                    <span>Against: {c.caregiver?.user?.name || 'Caregiver'}</span>
   682→                    {c.booking && <span>Booking: {c.booking.startDate ? new Date(c.booking.startDate).toLocaleDateString('en-IN', { day: 'numeric', month: 'short' }) : c.booking.id.slice(0, 8)}</span>}
   683→                  </div>
   684→
   685→                  {c.description && <p className="text-sm text-muted-foreground line-clamp-2">{c.description}</p>}
   686→
   687→                  {c.resolution && (
   688→                    <div className="bg-green-50 rounded-xl p-3 text-xs text-green-700">
   689→                      <p className="font-medium">Resolution:</p>
   690→                      <p className="mt-0.5">{c.resolution}</p>
   691→                    </div>
   692→                  )}
   693→
   694→                  {/* Actions for OPEN complaints */
   695→                  {c.status === 'OPEN' && (
   696→                    <div className="flex items-center gap-2 flex-wrap pt-1">
   697→                      <Button
   698→                        size="sm"
   699→                        variant="outline"
   700→                        className="text-[#14532d] border-[#14532d]/30 hover:bg-[#14532d]/10 rounded-full"
   701→                        disabled={actionId === c.id}
   702→                        onClick={() => handleInProgress(c.id)}
   703→                      >
   704→                        {actionId === c.id ? <Loader2 className="w-3.5 h-3.5 animate-spin" /> : <Activity className="w-3.5 h-3.5 mr-1" />}
   705→                        In Progress
   706→                      </Button>
   707→                      <Button
   708→                        size="sm"
   709→                        className="bg-[#14532d] hover:bg-[#14532d]/90 text-white rounded-full"
   710→                        disabled={actionId === c.id}
   711→                        onClick={() => { setExpandedId(expandedId === c.id ? null : c.id); }}
   712→                      >
   713→                        <CheckCircle2 className="w-3.5 h-3.5 mr-1" />Resolve
   714→                      </Button>
   715→                      <Button
   716→                        size="sm"
   717→                        variant="outline"
   718→                        className="text-red-600 border-red-200 hover:bg-red-50 rounded-full"
   719→                        disabled={actionId === c.id}
   720→                        onClick={() => { setExpandedId(expandedId === c.id ? null : c.id); }}
   721→                      >
   722→                        <Ban className="w-3.5 h-3.5 mr-1" />Dismiss
   723→                      </Button>
   724→                    </div>
   725→                  )}
   726→
   727→                  {/* Expandable resolution form */
   728→                  {expandedId === c.id && (
   729→                    <motion.div initial={{ height: 0, opacity: 0 }} animate={{ height: 'auto', opacity: 1 }} className="space-y-2">
   730→                      <Textarea
   731→                        placeholder="Enter resolution message..."
   732→                        value={resolutions[c.id] || ''}
   733→                        onChange={(e) => setResolutions((prev) => ({ ...prev, [c.id]: e.target.value }))}
   734→                        className="text-sm rounded-xl min-h-[80px] resize-none"
   735→                      />
   736→                      <div className="flex gap-2">
   737→                        <Button size="sm" className="btn-green rounded-full" disabled={actionId === c.id} onClick={() => handleResolve(c.id)}>
   738→                          {actionId === c.id ? <Loader2 className="w-3.5 h-3.5 animate-spin" /> : <><CheckCircle2 className="w-3.5 h-3.5 mr-1" />Confirm Resolve</>}
   739→                        </Button>
   740→                        <Button size="sm" variant="destructive" className="rounded-full" disabled={actionId === c.id} onClick={() => handleDismiss(c.id)}>
   741→                          {actionId === c.id ? <Loader2 className="w-3.5 h-3.5 animate-spin" /> : <><Ban className="w-3.5 h-3.5 mr-1" />Confirm Dismiss</>}
   742→                        </Button>
   743→                        <Button size="sm" variant="ghost" onClick={() => setExpandedId(null)} className="rounded-full">Cancel</Button>
   744→                      </div>
   745→                    </motion.div>
   746→                  )}
   747→                </CardContent>
   748→              </Card>
   749→            </motion.div>
   750→          ))}
   751→        </div>
   752→      )}
   753→    </div>
   754→  );
