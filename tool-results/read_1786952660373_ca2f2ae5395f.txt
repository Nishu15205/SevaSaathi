   145→function OverviewTab() {
   146→  const user = useAuthStore((s) => s.user);
   147→  const [stats, setStats] = useState({ patients: 0, activeBookings: 0, reports: 0, pendingReviews: 0 });
   148→  const [recentBookings, setRecentBookings] = useState<any[]>([]);
   149→  const [loading, setLoading] = useState(true);
   150→  const [error, setError] = useState('');
   151→
   152→  const fetchData = useCallback(async () => {
   153→    if (!user?.id) return;
   154→    setLoading(true);
   155→    setError('');
   156→    try {
   157→      const [patientsRes, bookingsRes] = await Promise.all([
   158→        api.patients.list(user.id),
   159→        api.bookings.list({ familyId: user.id }),
   160→      ]);
   161→
   162→      const patients = patientsRes.patients || [];
   163→      const bookings = bookingsRes.bookings || [];
   164→
   165→      const activeBookings = bookings.filter((b: any) => ['PENDING', 'CONFIRMED', 'IN_PROGRESS'].includes(b.status));
   166→      const totalReports = bookings.reduce((acc: number, b: any) => acc + (b.careReports?.length || 0), 0);
   167→      const completedWithoutReview = bookings.filter(
   168→        (b: any) => b.status === 'COMPLETED' && (!b.reviews || b.reviews.length === 0)
   169→      ).length;
   170→
   171→      setStats({
   172→        patients: patients.length,
   173→        activeBookings: activeBookings.length,
   174→        reports: totalReports,
   175→        pendingReviews: completedWithoutReview,
   176→      });
   177→      setRecentBookings(bookings.slice(0, 5));
   178→    } catch (err: any) {
   179→      setError(err.message || 'Failed to load dashboard data');
   180→    } finally {
   181→      setLoading(false);
   182→    }
   183→  }, [user?.id]);
   184→
   185→  useEffect(() => {
   186→    fetchData();
   187→  }, [fetchData]);
   188→
   189→  if (loading) {
   190→    return (
   191→      <div className="space-y-6">
   192→        <LoadingGrid count={4} />
   193→        <div>
   194→          <Skeleton className="h-6 w-40 mb-3" />
   195→          <LoadingCards count={3} />
   196→        </div>
   197→      </div>
   198→    );
   199→  }
   200→
   201→  if (error) {
   202→    return <ErrorState message={error} onRetry={fetchData} />;
   203→  }
   204→
   205→  const statCards = [
   206→    { label: 'Total Members', value: stats.patients, icon: <Users className="h-5 w-5" />, color: 'bg-forest-50 text-forest-700' },
   207→    { label: 'Active Bookings', value: stats.activeBookings, icon: <CalendarCheck className="h-5 w-5" />, color: 'bg-blue-50 text-blue-700' },
   208→    { label: 'Care Reports', value: stats.reports, icon: <FileText className="h-5 w-5" />, color: 'bg-amber-50 text-amber-700' },
   209→    { label: 'Pending Reviews', value: stats.pendingReviews, icon: <Star className="h-5 w-5" />, color: 'bg-lime-50 text-lime-700' },
   210→  ];
   211→
   212→  return (
   213→    <div className="space-y-6">
   214→      {/* Welcome */}
   215→      <div>
   216→        <h2 className="text-2xl font-bold text-gray-900">
   217→          Welcome back, {user?.name?.split(' ')[0]}!
   218→        </h2>
   219→        <p className="text-sm text-gray-400 mt-1">Here's what's happening with your care requests.</p>
   220→      </div>
   221→
   222→      {/* Stat Cards */}
   223→      <div className="grid grid-cols-1 sm:grid-cols-2 lg:grid-cols-4 gap-4">
   224→        {statCards.map((stat, i) => (
   225→          <motion.div
   226→            key={stat.label}
   227→            initial={{ opacity: 0, y: 12 }}
   228→            animate={{ opacity: 1, y: 0 }}
   229→            transition={{ delay: i * 0.05 }}
   230→          >
   231→            <Card className="rounded-2xl border-gray-100 hover:shadow-md transition-shadow">
   232→              <CardContent className="p-5">
   233→                <div className="flex items-center justify-between">
   234→                  <div>
   235→                    <p className="text-sm text-gray-500 font-medium">{stat.label}</p>
   236→                    <p className="text-3xl font-bold text-gray-900 mt-1">{stat.value}</p>
   237→                  </div>
   238→                  <div className={`w-11 h-11 rounded-2xl flex items-center justify-center ${stat.color}`}>
   239→                    {stat.icon}
   240→                  </div>
   241→                </div>
   242→              </CardContent>
   243→            </Card>
   244→          </motion.div>
   245→        ))}
   246→      </div>
   247→
   248→      {/* Quick Actions for new users */}
   249→      {!loading && stats.activeBookings === 0 && stats.patients === 0 && (
   250→        <Card className="rounded-2xl border-dashed border-2 border-forest-200 bg-forest-50/50">
   251→          <CardContent className="p-6 text-center">
   252→            <Heart className="h-8 w-8 text-forest-400 mx-auto mb-3" />
   253→            <h3 className="text-base font-semibold text-gray-800 mb-1">Get started with SevaSaathi</h3>
   254→            <p className="text-sm text-gray-500">Add a member profile from the Members tab, then search and book a caregiver.</p>
   255→          </CardContent>
   256→        </Card>
   257→      )}
   258→
   259→      {/* Recent Bookings */}
   260→      <div>
   261→        <h3 className="text-base font-semibold text-gray-800 mb-3">Recent Bookings</h3>
   262→        {recentBookings.length === 0 ? (
   263→          <EmptyState
   264→            icon={<CalendarCheck className="h-7 w-7" />}
   265→            title="No bookings yet"
   266→            description="Find a caregiver and book your first care session."
   267→          />
   268→        ) : (
   269→          <div className="space-y-3">
   270→            {recentBookings.map((booking: any) => (
   271→              <Card key={booking.id} className="rounded-2xl border-gray-100">
   272→                <CardContent className="p-4">
   273→                  <div className="flex items-center justify-between gap-4">
   274→                    <div className="flex items-center gap-3 min-w-0">
   275→                      <div className="w-10 h-10 rounded-xl bg-forest-50 flex items-center justify-center shrink-0">
   276→                        <UserRound className="h-5 w-5 text-forest-700" />
   277→                      </div>
   278→                      <div className="min-w-0">
   279→                        <p className="text-sm font-semibold text-gray-800 truncate">
   280→                          {booking.patient?.name || 'Member'}
   281→                        </p>
   282→                        <p className="text-xs text-gray-400">
   283→                          {booking.caregiver?.user?.name || 'Caregiver'} &middot; {booking.shiftType?.replace(/_/g, ' ')}
   284→                        </p>
   285→                      </div>
   286→                    </div>
   287→                    <div className="text-right shrink-0">
   288→                      <Badge variant="outline" className={`text-[11px] rounded-full ${statusColors[booking.status] || ''}`}>
   289→                        {booking.status}
   290→                      </Badge>
   291→                      <p className="text-xs text-gray-400 mt-1">
   292→                        {booking.totalAmount ? `₹${booking.totalAmount}` : '--'}
   293→                      </p>
   294→                    </div>
   295→                  </div>
   296→                </CardContent>
   297→              </Card>
   298→            ))}
   299→          </div>
   300→        )}
   301→      </div>
   302→    </div>
   303→  );
   304→}

   309→function PatientsTab() {
   310→  const user = useAuthStore((s) => s.user);
   311→  const [patients, setPatients] = useState<any[]>([]);
   312→  const [loading, setLoading] = useState(true);
   313→  const [error, setError] = useState('');
   314→  const [modalOpen, setModalOpen] = useState(false);
   315→  const [submitting, setSubmitting] = useState(false);
   316→
   317→  // Form state
   318→  const [form, setForm] = useState({
   319→    name: '', age: '', gender: '', relationship: '', address: '', city: '',
   320→    pincode: '', mobilityStatus: '', medicalHistory: '', careRequirements: '',
   321→    emergencyName: '', emergencyPhone: '',
   322→  });
   323→
   324→  const fetchPatients = useCallback(async () => {
   325→    if (!user?.id) return;
   326→    setLoading(true);
   327→    setError('');
   328→    try {
   329→      const res = await api.patients.list(user.id);
   330→      setPatients(res.patients || []);
   331→    } catch (err: any) {
   332→      setError(err.message || 'Failed to load patients');
   333→    } finally {
   334→      setLoading(false);
   335→    }
   336→  }, [user?.id]);
   337→
   338→  useEffect(() => {
   339→    fetchPatients();
   340→  }, [fetchPatients]);
   341→
   342→  const updateForm = (field: string, value: string) => {
   343→    setForm((prev) => ({ ...prev, [field]: value }));
   344→  };
   345→
   346→  const handleSubmit = async (e: React.FormEvent) => {
   347→    e.preventDefault();
   348→    if (!user?.id) return;
   349→    setSubmitting(true);
   350→    try {
   351→      await api.patients.create({
   352→        familyId: user.id,
   353→        name: form.name,
   354→        age: parseInt(form.age, 10) || 0,
   355→        gender: form.gender,
   356→        relationship: form.relationship,
   357→        address: form.address,
   358→        city: form.city,
   359→        pincode: form.pincode || undefined,
   360→        mobilityStatus: form.mobilityStatus || 'mobile',
   361→        medicalHistory: form.medicalHistory,
   362→        careRequirements: form.careRequirements,
   363→        emergencyContact: JSON.stringify({ name: form.emergencyName, phone: form.emergencyPhone }),
   364→      });
   365→      toast.success('Member profile created successfully!');
   366→      setModalOpen(false);
   367→      setForm({
   368→        name: '', age: '', gender: '', relationship: '', address: '', city: '',
   369→        pincode: '', mobilityStatus: '', medicalHistory: '', careRequirements: '',
   370→        emergencyName: '', emergencyPhone: '',
   371→      });
   372→      fetchPatients();
   373→    } catch (err: any) {
   374→      toast.error(err.message || 'Failed to create member');
   375→    } finally {
   376→      setSubmitting(false);
   377→    }
   378→  };
   379→
   380→  if (loading) {
   381→    return <LoadingCards count={3} />;
   382→  }
   383→
   384→  if (error) {
   385→    return <ErrorState message={error} onRetry={fetchPatients} />;
   386→  }
   387→
   388→  const mobilityLabel: Record<string, string> = {
   389→    mobile: 'Mobile', bedridden: 'Bedridden', limited: 'Limited Mobility', wheelchair: 'Wheelchair',
   390→  };
   391→
   392→  return (
   393→    <div className="space-y-6">
   394→      <div className="flex items-center justify-between">
   395→        <div>
   396→          <h2 className="text-xl font-bold text-gray-900">Member Profiles</h2>
   397→          <p className="text-sm text-gray-400">Manage care recipients in your family.</p>
   398→        </div>
   399→        <Dialog open={modalOpen} onOpenChange={setModalOpen}>
   400→          <DialogTrigger asChild>
   401→            <Button className="btn-black text-sm gap-2">
   402→              <Plus className="h-4 w-4" /> Add Member
   403→            </Button>
   404→          </DialogTrigger>
   405→          <DialogContent className="max-w-lg max-h-[90vh] overflow-y-auto rounded-2xl">
   406→            <DialogHeader>
   407→              <DialogTitle>Add New Member</DialogTitle>
   408→            </DialogHeader>
   409→            <form onSubmit={handleSubmit} className="space-y-4 mt-2">
   410→              <div className="grid grid-cols-2 gap-3">
   411→                <div className="col-span-2">
   412→                  <Label className="text-xs font-medium text-gray-600">Full Name *</Label>
   413→                  <Input
   414→                    required
   415→                    value={form.name}
   416→                    onChange={(e) => updateForm('name', e.target.value)}
   417→                    placeholder="Member name"
   418→                    className="mt-1 rounded-xl"
   419→                  />
   420→                </div>
   421→                <div>
   422→                  <Label className="text-xs font-medium text-gray-600">Age *</Label>
   423→                  <Input
   424→                    required
   425→                    type="number"
   426→                    min={0}
   427→                    max={150}
   428→                    value={form.age}
   429→                    onChange={(e) => updateForm('age', e.target.value)}
   430→                    placeholder="Age"

   432→                  />
   433→                </div>

  1755→function ComplaintsTab() {

  1757→  const [complaints, setComplaints] = useState<any[]>([]);
  1758→  const [loading, setLoading] = useState(true);
  1759→  const [error, setError] = useState('');
  1760→  const [modalOpen, setModalOpen] = useState(false);

  1765→    subject: '', description: '', priority: 'medium', caregiverId: '',

  1769→    if (!user?.id) return;
  1770→    setLoading(true);
  1771→    setError('');
  1772→    try {
  1773→      const res = await api.complaints.list({ familyId: user.id });
  1774→      setComplaints(res.complaints || []);
  1775→      // Also fetch unique caregivers from bookings

  1778→        (bRes.bookings || []).forEach((b: any) => {
  1779→          if (b.caregiver && !uniqueCaregivers.has(b.caregiverId)) {
  1780→            uniqueCaregivers.set(b.caregiverId, b.caregiver);
  1781→          }
  1782→        });

  1785→    } catch (err: any) {
  1786→      setError(err.message || 'Failed to load complaints');
  1787→    } finally {
  1788→      setLoading(false);
  1789→    }
  1790→  }, [user?.id]);
  1791→
  1792→  useEffect(() => {
  1793→    fetchComplaints();

  1796→  const updateForm = (field: string, value: string) => {

  1824→    }

  1827→  const priorityLabel: Record<string, string> = {
  1828→    low: 'Low', medium: 'Medium', high: 'High', urgent: 'Urgent',
