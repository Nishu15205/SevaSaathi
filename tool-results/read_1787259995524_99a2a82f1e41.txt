   205→function parseJsonSafe(jsonStr: string | null | undefined): string[] {
   206→  if (!jsonStr) return [];
   207→  try {
   208→    const parsed = JSON.parse(jsonStr);
   209→    return Array.isArray(parsed) ? parsed : [];
   210→  } catch {
   211→    return jsonStr.split(',').map((s) => s.trim()).filter(Boolean);
   212→  }
   213→}
   214→
   215→function formatDate(dateStr: string): string {
   216→  try {
   217→    return new Date(dateStr).toLocaleDateString('en-IN', {
   218→      day: 'numeric',
   219→      month: 'short',
   220→      year: 'numeric',
   221→    });
   222→  } catch {
   223→    return dateStr;
   224→  }
   225→}
   226→
   227→function getTodayDate(): string {
   228→  const now = new Date();
   229→  const year = now.getFullYear();
   230→  const month = String(now.getMonth() + 1).padStart(2, '0');
   231→  const day = String(now.getDate()).padStart(2, '0');
   232→  return `${year}-${month}-${day}`;
   233→}

   473→function CreateProfileForm({ userId, onCreated }: { userId: string; onCreated: (profile: any) => void }) {
   474→  const [saving, setSaving] = useState(false);
   475→  const [selectedSkills, setSelectedSkills] = useState<string[]>(['elderly-care']);
   476→  const [form, setForm] = useState({

   478→  });
   479→
   480→  const toggleSkill = (skill: string) => {
   481→    setSelectedSkills(prev => prev.includes(skill) ? prev.filter(s => s !== skill) : [...prev, skill]);
   482→  };
   483→
   484→  const handleSubmit = async (e: React.FormEvent) => {
   485→    e.preventDefault();
   486→    if (!form.city || !form.yearsExperience || !form.hourlyRate) {
   487→      toast.error('Please fill in City, Experience, and Hourly Rate.');
   488→      return;
   489→    }
   490→    if (selectedSkills.length === 0) {
   491→      toast.error('Please select at least one skill.');
   492→      return;
   493→    }
   494→    setSaving(true);
   495→    try {
   496→      const res = await api.caregivers.create({
   497→        userId,
   498→        city: form.city,
   499→        yearsExperience: parseInt(form.yearsExperience),
   500→        hourlyRate: parseInt(form.hourlyRate),
   501→        skills: JSON.stringify(selectedSkills),
   502→        qualifications: form.qualifications ? JSON.stringify(form.qualifications.split(',').map((s: string) => s.trim()).filter(Boolean)) : JSON.stringify(['Caregiver']),
   503→        languages: JSON.stringify(form.languages.split(',').map((s: string) => s.trim()).filter(Boolean)),
   504→        bio: form.bio,
   505→      });
   506→      toast.success('Profile created successfully!');
   507→      onCreated(res.caregiver);
   508→    } catch (err: any) {
   509→      toast.error(err.message || 'Failed to create profile');
   510→    } finally {
   511→      setSaving(false);
   512→    }
   513→  };
   514→
   515→  return (
   516→    <div className="space-y-6 max-w-2xl">
   517→      <div>
   518→        <h2 className="text-2xl font-bold text-gray-900">Create Your Profile</h2>
   519→        <p className="text-sm text-gray-400 mt-1">Set up your caregiver profile to start receiving booking requests from families.</p>
   520→      </div>
   521→      <Card className="rounded-2xl border-gray-100">
   522→        <CardContent className="p-6">
   523→          <form onSubmit={handleSubmit} className="space-y-4">
   524→            <div className="grid grid-cols-1 sm:grid-cols-2 gap-4">

   776→function AadharVerificationSection({ caregiverId }: { caregiverId: string }) {
   777→  const [aadharResult, setAadharResult] = useState<any>(null);
   778→  const [verifying, setVerifying] = useState(false);
   779→  const [imagePreview, setImagePreview] = useState<string | null>(null);
   780→
   781→  const handleFileSelect = async (e: React.ChangeEvent<HTMLInputElement>) => {
   782→    const file = e.target.files?.[0];
   783→    if (!file) return;
   784→
   785→    // Show preview
   786→    const reader = new FileReader();
   787→    reader.onload = (ev) => {
   788→      setImagePreview(ev.target?.result as string);
   789→      setAadharResult(null);
   790→    };
   791→    reader.readAsDataURL(file);
   792→  };
   793→
   794→  const handleVerify = async () => {
   795→    if (!imagePreview) {
   796→      toast.error('Please upload an Aadhar card image first');
   797→      return;
   798→    }
   799→    setVerifying(true);
   800→    try {
   801→      const result = await api.verifyAadhar(imagePreview);
   802→      setAadharResult(result);
   803→      if (result.verified) {
   804→        toast.success('Aadhar card verified successfully!');
   805→      } else {
   806→        toast.error(result.error || 'Aadhar verification failed');
   807→      }
   808→    } catch (err: any) {
   809→      toast.error(err.message || 'Verification failed');
   810→      setAadharResult({ verified: false, error: err.message });
   811→    } finally {
   812→      setVerifying(false);
   813→    }
   814→  };
   815→
   816→  return (
   817→    <div className="mt-6">
   818→      <div className="flex items-center gap-2 mb-3">
   819→        <ShieldCheck className="h-4 w-4 text-forest-700" />
   820→        <p className="text-sm font-semibold text-gray-700">Aadhar Card Verification</p>
   821→      </div>
   822→      <div className="bg-gray-50 rounded-2xl p-5">
   823→        <p className="text-xs text-gray-500 mb-4">Upload your Aadhar card for AI-powered instant verification. Your data is processed securely and not stored.</p>
   824→        
   825→        <div className="space-y-4">
   826→          {/* Upload Area */}
   827→          {!imagePreview ? (
   828→            <label className="flex flex-col items-center justify-center border-2 border-dashed border-gray-300 rounded-xl p-8 cursor-pointer hover:border-forest-400 hover:bg-forest-50/50 transition-colors">
   829→              <Camera className="h-10 w-10 text-gray-400 mb-3" />
   830→              <p className="text-sm font-medium text-gray-600">Upload Aadhar Card</p>
   831→              <p className="text-xs text-gray-400 mt-1">JPG, PNG - Clear photo of your Aadhar card</p>
   832→              <input
   833→                type="file"
   834→                accept="image/*"
   835→                onChange={handleFileSelect}
   836→                className="hidden"
   837→              />
   838→            </label>
   839→          ) : (
   840→            <div className="space-y-3">
   841→              <div className="relative rounded-xl overflow-hidden border border-gray-200">
   842→                <img src={imagePreview} alt="Aadhar preview" className="w-full max-h-48 object-contain bg-white" />
   843→                <button
   844→                  onClick={() => { setImagePreview(null); setAadharResult(null); }}
   845→                  className="absolute top-2 right-2 w-7 h-7 rounded-full bg-black/60 text-white flex items-center justify-center hover:bg-black/80"
   846→                >
   847→                  <XCircle className="h-4 w-4" />
   848→                </button>
   849→              </div>
   850→              <div className="flex gap-2">
   851→                <Button
   852→                  onClick={handleVerify}
   853→                  disabled={verifying}
   854→                  className="btn-green text-sm gap-2"
   855→                >
   856→                  {verifying ? <Loader2 className="h-4 w-4 animate-spin" /> : <ShieldCheck className="h-4 w-4" />}
   857→                  {verifying ? 'Verifying...' : 'Verify Aadhar'}
   858→                </Button>
   859→                <Button variant="outline" size="sm" onClick={() => { setImagePreview(null); setAadharResult(null); }} className="rounded-full">
   860→                  Change Image
   861→                </Button>
   862→              </div>
   863→            </div>
   864→          )}
   865→
   866→          {/* Result */}
   867→          {aadharResult && (
   868→            <motion.div
   869→              initial={{ opacity: 0, y: 8 }}
   870→              animate={{ opacity: 1, y: 0 }}
   871→              className={`rounded-xl p-4 ${
   872→                aadharResult.verified
   873→                  ? 'bg-green-50 border border-green-200'
   874→                  : 'bg-red-50 border border-red-200'
   875→              }`}
   876→            >
   877→              <div className="flex items-center gap-2 mb-2">
   878→                {aadharResult.verified ? (
   879→                  <><CheckCircle2 className="h-5 w-5 text-green-600" /><p className="text-sm font-semibold text-green-800">Verification Successful</p></>
   880→                ) : (
   881→                  <><XCircle className="h-5 w-5 text-red-600" /><p className="text-sm font-semibold text-red-800">Verification Failed</p></>
   882→                )}
   883→              </div>
   884→              {aadharResult.verified ? (
   885→                <div className="grid grid-cols-1 sm:grid-cols-2 gap-2 mt-3">
   886→                  {aadharResult.name && (
   887→                    <div><p className="text-[10px] text-green-600 font-medium">Name</p><p className="text-sm font-semibold text-green-900">{aadharResult.name}</p></div>
