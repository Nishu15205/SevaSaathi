   215→function parseJsonSafe(input: string | null | undefined | any): string[] {
   216→  if (!input) return [];
   217→  if (Array.isArray(input)) return input;
   218→  try {
   219→    const parsed = JSON.parse(input);
   220→    return Array.isArray(parsed) ? parsed : [];
   221→  } catch {
   222→    if (typeof input === 'string') {
   223→      return input.split(',').map((s) => s.trim()).filter(Boolean);
   224→    }
   225→    return [];
   226→  }
   227→}

   229→function formatDate(dateStr: string): string {
   230→  try {
   231→    return new Date(dateStr).toLocaleDateString('en-IN', {
   232→      day: 'numeric',
   233→      month: 'short',
   234→      year: 'numeric',
   235→    });
   236→  } catch {
   237→    return dateStr;
   238→  }
   239→}
   240→
   241→function getTodayDate(): string {
   242→  const now = new Date();
   243→  const year = now.getFullYear();
   244→  const month = String(now.getMonth() + 1).padStart(2, '0');
   245→  const day = String(now.getDate()).padStart(2, '0');
   246→  return `${year}-${month}-${day}`;
   247→}

   253→function OverviewTab({ user }: { user: User }) {
   254→  const profile = user.caregiverProfile;
   255→  const [bookings, setBookings] = useState<Booking[]>([]);
   256→  // Start with loading=false so new caregivers (no profile) never see a skeleton flash
   257→  const [loading, setLoading] = useState(false);
   258→  const [error, setError] = useState('');
   259→
   260→  // Phone verification — real OTP via Firebase
   261→  const [phoneVerifying, setPhoneVerifying] = useState(false);
   262→  const [showOtpDialog, setShowOtpDialog] = useState(false);
   263→  const [otpValue, setOtpValue] = useState("");
   264→  const firebase = useFirebasePhoneAuth();
   265→
   266→  const maskPhone = (p: string) => {
   267→    const d = p.replace(/\D/g, '');
   268→    if (d.length < 4) return p;
   269→    return '*'.repeat(d.length - 4) + d.slice(-4);
   270→  };
   271→
   272→  const handleSendOtp = async () => {
   273→    if (!firebase.isReady) {
   274→      toast.error('Phone verification service is not available. Please contact support.');
   275→      return;
   276→    }
   277→    setPhoneVerifying(true);
   278→    try {
   279→      const sent = await firebase.sendOtp(user.phone);
   280→      if (sent) {
   281→        setShowOtpDialog(true);
   282→        toast.success('OTP sent to your phone!');
   283→      } else {
   284→        toast.error(firebase.error || 'Failed to send OTP. Please try again.');
   285→      }
   286→    } finally {
   287→      setPhoneVerifying(false);
   288→    }
   289→  };
   290→
   291→  const handleVerifyOtp = async () => {
   292→    if (!otpValue || otpValue.length !== 6) {
   293→      toast.error('Enter the 6-digit OTP sent to your phone');
   294→      return;
   295→    }
   296→    try {
   297→      const result = await firebase.verifyOtp(otpValue);
   298→      if (!result) {
   299→        toast.error(firebase.error || 'Invalid OTP. Please try again.');
   300→        return;
   301→      }
   302→      const res = await fetch('/api/auth/verify-phone-otp', {
   303→        method: 'POST',
   304→        headers: { 'Content-Type': 'application/json' },
   305→        body: JSON.stringify({ phone: user.phone, firebaseToken: result.firebaseToken }),
   306→      });
   307→      const data = await res.json();
   308→      if (!res.ok) {
   309→        toast.error(data.error || 'Verification failed');
   310→        return;
   311→      }
   312→      toast.success('Phone number verified successfully!');
   313→      useAuthStore.getState().setAuth({ ...user, phoneVerified: true } as any);
   314→      setShowOtpDialog(false);
   315→      setOtpValue('');
   316→      firebase.reset();
   317→    } catch (err: any) {
   318→      toast.error(err?.message || 'Verification failed');
   319→    }
   320→  };
   321→
   322→  const fetchData = useCallback(async () => {
   323→    if (!profile?.id) { setLoading(false); return; }
   324→    setLoading(true);
   325→    setError('');
   326→    try {
   327→      const res = await api.bookings.list({ caregiverId: profile.id });
   328→      const all = (res.bookings || []).map((b: any) => ({
   329→        ...b,
   330→        date: b.startDate,
   331→        patientName: b.patient?.name,
   332→        familyName: b.family?.name,
   333→        careType: b.careRequirements ? (typeof b.careRequirements === 'string' ? JSON.parse(b.careRequirements) : b.careRequirements)?.needs?.join(', ') : undefined,
   334→      }));
   335→      setBookings(all);
   336→    } catch (err: any) {
   337→      setError(err.message || 'Failed to load dashboard data');
   338→    } finally {
   339→      setLoading(false);
   340→    }
   341→  }, [profile?.id]);
   342→
   343→  useEffect(() => {
   344→    fetchData();
   345→  }, [fetchData]);
   346→
   347→  // No profile yet — show setup guidance immediately, no loading flash
   348→  if (!profile) {
   349→    return (
   350→      <div className="space-y-6">
   351→        <div>
   352→          <h2 className="text-2xl font-bold text-gray-900">Welcome, {user.name.split(' ')[0]}!</h2>
   353→          <p className="text-sm text-gray-400 mt-1">Complete your profile to start receiving care requests.</p>
   354→        </div>
   355→        <Card className="rounded-2xl border-dashed border-2 border-forest-200 bg-forest-50/50">
   356→          <CardContent className="p-8 text-center">
   357→            <div className="w-16 h-16 rounded-2xl green-gradient-bg flex items-center justify-center mx-auto mb-4">
   358→              <ClipboardList className="h-7 w-7 text-white" />
   359→            </div>
   360→            <h3 className="text-lg font-bold text-gray-800 mb-2">Set Up Your Caregiver Profile</h3>
   361→            <p className="text-sm text-gray-500 max-w-md mx-auto">Go to the <strong>My Profile</strong> tab to create your caregiver profile. Families will be able to find and book you once your profile is set up.</p>
   362→          </CardContent>
   363→        </Card>
   364→      </div>
   365→    );
   366→  }
   367→
   368→  if (error) {
   369→    return <ErrorState message={error} onRetry={fetchData} />;
   370→  }
   371→
   372→  const upcomingBookings = bookings
   373→    .filter((b) => ['CONFIRMED', 'IN_PROGRESS'].includes(b.status))
   374→    .sort((a, b) => new Date(a.date).getTime() - new Date(b.date).getTime());
   375→
   376→  const totalBookings = bookings.length;
   377→  const completedJobs = profile?.completedJobs ?? bookings.filter((b) => b.status === 'COMPLETED').length;
   378→  const overallRating = profile?.overallRating ?? 0;
   379→  const hourlyRate = profile?.hourlyRate ?? 0;
   380→
   381→  const statCards = [
   382→    { label: 'Total Bookings', value: totalBookings, icon: <CalendarCheck className="h-5 w-5" />, color: 'bg-forest-50 text-forest-700' },
   383→    { label: 'Completed Jobs', value: completedJobs, icon: <CheckCircle2 className="h-5 w-5" />, color: 'bg-lime-50 text-lime-700' },
   384→    { label: 'Overall Rating', value: overallRating.toFixed(1), icon: <Star className="h-5 w-5" />, color: 'bg-amber-50 text-amber-700', suffix: '/5' },
   385→    { label: 'Hourly Rate', value: `₹${hourlyRate}`, icon: <IndianRupee className="h-5 w-5" />, color: 'bg-purple-50 text-purple-700', isString: true },
   386→  ];
   387→
   388→  const skills = parseJsonSafe(profile?.skills);
   389→  const languages = parseJsonSafe(profile?.languages);
   390→
   391→  return (
   392→    <div className="space-y-6">
   393→      {/* Phone Verification Banner */}
   394→      {!user.phoneVerified && user.phone && (
   395→        <Card className="rounded-2xl border-2 border-amber-200 bg-amber-50">
   396→          <CardContent className="p-4 flex items-center justify-between gap-4">
   397→            <div className="flex items-center gap-3 min-w-0">
   398→              <div className="w-10 h-10 rounded-xl bg-amber-100 flex items-center justify-center shrink-0">
   399→                <Phone className="h-5 w-5 text-amber-600" />
   400→              </div>
   401→              <div className="min-w-0">
   402→                <p className="text-sm font-semibold text-amber-900">Verify Your Phone Number</p>
   403→                <p className="text-xs text-amber-700/70">Required to receive booking requests — {maskPhone(user.phone)}</p>
   404→              </div>
   405→            </div>
   406→            <Button onClick={handleSendOtp} disabled={phoneVerifying || !firebase.isReady} size="sm" className="bg-amber-600 hover:bg-amber-700 text-white rounded-xl shrink-0">
   407→              {phoneVerifying && <Loader2 className="h-4 w-4 animate-spin mr-1.5" />}
   408→              {phoneVerifying ? 'Sending OTP...' : !firebase.isReady ? 'Unavailable' : 'Send OTP'}
   409→            </Button>
   410→          </CardContent>
   411→        </Card>
   412→      )}
   413→
   414→      {/* Phone OTP Dialog */}
   415→      <Dialog open={showOtpDialog} onOpenChange={(open) => { if (!open) { setShowOtpDialog(false); setOtpValue(''); firebase.reset(); } }}>
   416→        <DialogContent className="sm:max-w-md">
   417→          <DialogHeader>
   418→            <DialogTitle>Verify Phone Number</DialogTitle>
   419→            <DialogDescription>Enter the 6-digit OTP sent to {maskPhone(user.phone)}</DialogDescription>
   420→          </DialogHeader>
   421→          <div className="space-y-4 pt-2">
   422→            <div>
   423→              <Label className="text-sm">OTP</Label>
   424→              <Input
   425→                value={otpValue}
   426→                onChange={(e) => setOtpValue(e.target.value.replace(/\D/g, '').slice(0, 6))}
   427→                placeholder="Enter 6-digit OTP"
   428→                className="mt-1"
   429→                maxLength={6}
   430→                autoFocus
   431→              />
   432→            </div>
   433→            <Button
   434→              onClick={handleVerifyOtp}
   435→              disabled={firebase.verifying || otpValue.length !== 6}
   436→              className="w-full bg-forest-900 hover:bg-forest-800 text-white rounded-xl"
   437→            >
   438→              {firebase.verifying && <Loader2 className="h-4 w-4 animate-spin mr-2" />}
   439→              Verify OTP
   440→            </Button>
   441→            <button
   442→              type="button"
   443→              onClick={async () => { setOtpValue(''); firebase.reset(); setShowOtpDialog(false); await handleSendOtp(); }}
   444→              disabled={firebase.sendingOtp}
   445→              className="w-full text-xs text-forest-700 hover:text-forest-900 font-medium text-center disabled:opacity-50"
   446→            >
   447→              {firebase.sendingOtp ? 'Resending...' : 'Resend OTP'}
   448→            </button>
   449→            {firebase.error && <p className="text-xs text-red-500 text-center">{firebase.error}</p>}
   450→          </div>
   451→        </DialogContent>
   452→      </Dialog>
   453→
   454→      {/* Welcome */}
   455→      <div>
   456→        <h2 className="text-2xl font-bold text-gray-900">
   457→          Welcome back, {user.name.split(' ')[0]}!
   458→        </h2>
   459→        <p className="text-sm text-gray-400 mt-1">Here&apos;s your caregiving dashboard at a glance.</p>
   460→      </div>
   461→
   462→      {/* Verification Progress Banner */}
   463→      {profile && (
   464→        profile.isVerified ? (
   465→          <div className="rounded-2xl bg-gradient-to-r from-green-500 to-emerald-600 p-4 flex items-center gap-3">
   466→            <div className="w-10 h-10 rounded-xl bg-white/20 flex items-center justify-center shrink-0">
   467→              <ShieldCheck className="h-5 w-5 text-white" />
   468→            </div>
   469→            <div>
   470→              <p className="text-sm font-semibold text-white">You're a Verified Caregiver!</p>
   471→              <p className="text-xs text-green-100">Your profile displays the trusted verified badge</p>
   472→            </div>
   473→            <CheckCircle2 className="h-6 w-6 text-white ml-auto shrink-0" />
   474→          </div>
   475→        ) : (
   476→          <Card className="rounded-2xl border-2 border-amber-200 bg-amber-50">
   477→            <CardContent className="p-4 flex items-center justify-between gap-4">
   478→              <div className="flex items-center gap-3 min-w-0">
   479→                <div className="w-10 h-10 rounded-xl bg-amber-100 flex items-center justify-center shrink-0">
   480→                  <ShieldCheck className="h-5 w-5 text-amber-600" />
   481→                </div>
   482→                <div className="min-w-0">
   483→                  <p className="text-sm font-semibold text-amber-900">Complete Verification — Get More Bookings</p>
   484→                  <p className="text-xs text-amber-700/70">Verified caregivers get 40% more visibility and a trusted badge</p>
   485→                </div>
   486→              </div>
   487→              <Button size="sm" onClick={() => toast.info('Go to My Profile to complete verification')} className="bg-amber-600 hover:bg-amber-700 text-white rounded-xl shrink-0">
   488→                <ShieldCheck className="h-4 w-4 mr-1" /> Verify Now
   489→              </Button>
   490→            </CardContent>
   491→          </Card>
   492→        )
   493→      )}
   494→
   495→      {/* Stat Cards */}
   496→      <div className="grid grid-cols-1 sm:grid-cols-2 lg:grid-cols-4 gap-4">
   497→        {statCards.map((stat, i) => (
   498→          <motion.div
   499→            key={stat.label}
   500→            initial={{ opacity: 0, y: 12 }}
   501→            animate={{ opacity: 1, y: 0 }}
   502→            transition={{ delay: i * 0.05 }}
   503→          >
   504→            <Card className="rounded-2xl border-gray-100 hover:shadow-md transition-shadow">
   505→              <CardContent className="p-5">
   506→                <div className="flex items-center justify-between">
   507→                  <div>
   508→                    <p className="text-sm text-gray-500 font-medium">{stat.label}</p>
   509→                    <div className="flex items-baseline gap-1 mt-1">
   510→                      <p className="text-3xl font-bold text-gray-900">
   511→                        {stat.isString ? stat.value : stat.value}
   512→                      </p>
   513→                      {stat.suffix && <span className="text-sm text-gray-400">{stat.suffix}</span>}
   514→                    </div>
   515→                  </div>
   516→                  <div className={`w-11 h-11 rounded-2xl flex items-center justify-center ${stat.color}`}>
   517→                    {stat.icon}
   518→                  </div>
   519→                </div>
   520→              </CardContent>
   521→            </Card>
   522→          </motion.div>
   523→        ))}
   524→      </div>
   525→

   854→function WithdrawalForm({ caregiverId, userId, availableBalance }: { caregiverId: string; userId: string; availableBalance: number }) {
   855→  const [open, setOpen] = useState(false);
   856→  const [method, setMethod] = useState<'upi' | 'bank_transfer'>('upi');
   857→  const [amount, setAmount] = useState('');
   858→  const [upiId, setUpiId] = useState('');
   859→  const [bankName, setBankName] = useState('');
   860→  const [accountNumber, setAccountNumber] = useState('');
   861→  const [ifscCode, setIfscCode] = useState('');
   862→  const [accountHolder, setAccountHolder] = useState('');
   863→  const [submitting, setSubmitting] = useState(false);
   864→
   865→  const resetForm = () => {
   866→    setAmount(''); setUpiId(''); setBankName(''); setAccountNumber(''); setIfscCode(''); setAccountHolder(''); setMethod('upi');
   867→  };
   868→
   869→  const handleSubmit = async (e: React.FormEvent) => {
   870→    e.preventDefault();
   871→    const amountPaise = Math.round(parseFloat(amount) * 100);
   872→    if (!amountPaise || amountPaise <= 0) { toast.error('Enter a valid amount'); return; }
   873→    if (amountPaise > availableBalance) { toast.error(`Insufficient balance. Available: ₹${(availableBalance / 100).toLocaleString('en-IN')}`); return; }
   874→    if (method === 'upi' && !upiId) { toast.error('Enter UPI ID'); return; }
   875→    if (method === 'bank_transfer' && (!bankName || !accountNumber || !ifscCode || !accountHolder)) { toast.error('Fill all bank details'); return; }
   876→
   877→    setSubmitting(true);
   878→    try {
   879→      await api.withdrawals.create({
   880→        userId, caregiverId, amount: amountPaise, method,
   881→        upiId: method === 'upi' ? upiId : undefined,
   882→        bankName: method === 'bank_transfer' ? bankName : undefined,
   883→        accountNumber: method === 'bank_transfer' ? accountNumber : undefined,
   884→        ifscCode: method === 'bank_transfer' ? ifscCode : undefined,
   885→        accountHolder: method === 'bank_transfer' ? accountHolder : undefined,
   886→      });
   887→      toast.success('Withdrawal request submitted! Admin will process it soon.');
   888→      setOpen(false);
   889→      resetForm();
   890→    } catch (err: any) {
   891→      toast.error(err.message || 'Failed to submit withdrawal');
   892→    } finally {
   893→      setSubmitting(false);
   894→    }
   895→  };
   896→
   897→  return (
   898→    <>
   899→      <Button onClick={() => setOpen(true)} className="btn-green text-sm gap-2 rounded-xl">
   900→        <Wallet className="h-4 w-4" /> Withdraw Funds

   905→          <DialogHeader>
   906→            <DialogTitle className="flex items-center gap-2"><Wallet className="h-5 w-5 text-forest-800" /> Withdraw Funds</DialogTitle>
   907→            <DialogDescription>Available: <span className="font-semibold text-forest-700">₹{(availableBalance / 100).toLocaleString('en-IN')}</span></DialogDescription>
   908→          </DialogHeader>
   909→          <form onSubmit={handleSubmit} className="space-y-4 mt-3">
   910→            <div>
   911→              <Label className="text-xs font-medium text-gray-600">Amount (INR) *</Label>
   912→              <Input type="number" min="1" step="1" value={amount} onChange={e => setAmount(e.target.value)} placeholder="Enter amount" className="mt-1 rounded-xl" required />
   913→            </div>
   914→
   915→            {/* Method Toggle */}
   916→            <div>
   917→              <Label className="text-xs font-medium text-gray-600 mb-2 block">Withdrawal Method</Label>

'border-forest-500 bg-forest-50 text-forest-800' : 'border-gray-200 text-gray-500 hover:border-gray-300'}`}>
   923→                  <Building2 className="h-4 w-4" /> Bank Transfer
   924→                </button>
   925→              </div>
   926→            </div>
   927→
   928→            {method === 'upi' ? (
