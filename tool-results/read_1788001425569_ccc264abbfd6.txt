    90→    90→function useDebounce(value: string, delay: number) {
    91→    91→  const [debounced, setDebounced] = useState(value);
    92→    92→  useEffect(() => {
    93→    93→    const timer = setTimeout(() => setDebounced(value), delay);
    94→    94→    return () => clearTimeout(timer);
    95→    95→  }, [value, delay]);
    96→    96→  return debounced;
    97→    97→}
