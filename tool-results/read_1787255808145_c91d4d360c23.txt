   324→function PatientsTab() {
   325→  const user = useAuthStore((s) => s.user);
   326→  const [patients, setPatients] = useState<any[]>([]);
   327→  const [loading, setLoading] = useState(true);
   328→  const [error, setError] = useState('');
   329→  const [modalOpen, setModalOpen] = useState(false);
   330→  const [submitting, setSubmitting] = useState(false);
   331→
   332→  // Form state
   333→  const [form, setForm] = useState({
   334→    name: '', age: '', gender: '', relationship: '', address: '', city: '',
   335→    pincode: '', mobilityStatus: '', medicalHistory: '', careRequirements: '',
   336→    emergencyName: '', emergencyPhone: '',
   337→  });
   338→
   339→  const fetchPatients = useCallback(async () => {
   340→    if (!user?.id) return;
   341→    setLoading(true);
   342→    setError('');
   343→    try {
   344→      const res = await api.patients.list(user.id);
   345→      setPatients(res.patients || []);
   346→    } catch (err: any) {
   347→      setError(err.message || 'Failed to load patients');
   348→    } finally {
   349→      setLoading(false);
   350→    }
   351→  }, [user?.id]);
   352→
   353→  useEffect(() => {
   354→    fetchPatients();
   355→  }, [fetchPatients]);
   356→
   357→  const updateForm = (field: string, value: string) => {
   358→    setForm((prev) => ({ ...prev, [field]: value }));
   359→  };
   360→
   361→  const handleSubmit = async (e: React.FormEvent) => {
   362→    e.preventDefault();
   363→    if (!user?.id) return;
   364→    setSubmitting(true);
   365→    try {
   366→      await api.patients.create({
   367→        familyId: user.id,
   368→        name: form.name,
   369→        age: parseInt(form.age, 10) || 0,
   370→        gender: form.gender,
   371→        relationship: form.relationship,
   372→        address: form.address,
   373→        city: form.city,
   374→        pincode: form.pincode || undefined,
   375→        mobilityStatus: form.mobilityStatus || 'mobile',
   376→        medicalHistory: form.medicalHistory,
   377→        careRequirements: form.careRequirements,
   378→        emergencyContact: JSON.stringify({ name: form.emergencyName, phone: form.emergencyPhone }),
   379→      });
   380→      toast.success('Member profile created successfully!');
   381→      setModalOpen(false);
   382→      setForm({
   383→        name: '', age: '', gender: '', relationship: '', address: '', city: '',
   384→        pincode: '', mobilityStatus: '', medicalHistory: '', careRequirements: '',
   385→        emergencyName: '', emergencyPhone: '',
   386→      });
   387→      fetchPatients();
   388→    } catch (err: any) {
   389→      toast.error(err.message || 'Failed to create member');
   390→    } finally {
   391→      setSubmitting(false);
   392→    }
   393→  };
   394→
   395→  if (loading) {
   396→    return <LoadingCards count={3} />;
   397→  }
   398→
   399→  if (error) {
   400→    return <ErrorState message={error} onRetry={fetchPatients} />;
   401→  }
   402→
   403→  const mobilityLabel: Record<string, string> = {
   404→    mobile: 'Mobile', bedridden: 'Bedridden', limited: 'Limited Mobility', wheelchair: 'Wheelchair',
   405→  };
   406→
   407→  return (
   408→    <div className="space-y-6">
   409→      <div className="flex items-center justify-between">
   410→        <div>
   411→          <h2 className="text-xl font-bold text-gray-900">Member Profiles</h2>
   412→          <p className="text-sm text-gray-400">Manage care recipients in your family.</p>
   413→        </div>
   414→        <Dialog open={modalOpen} onOpenChange={setModalOpen}>
   415→          <DialogTrigger asChild>
   416→            <Button className="btn-black text-sm gap-2">
   417→              <Plus className="h-4 w-4" /> Add Member
   418→            </Button>
   419→          </DialogTrigger>
   420→          <DialogContent className="max-w-lg max-h-[90vh] overflow-y-auto rounded-2xl">
   421→            <DialogHeader>
   422→              <DialogTitle>Add New Member</DialogTitle>
   423→            </DialogHeader>
   424→            <form onSubmit={handleSubmit} className="space-y-4 mt-2">
   425→              <div className="grid grid-cols-2 gap-3">
   426→                <div className="col-span-2">
   427→                  <Label className="text-xs font-medium text-gray-600">Full Name *</Label>
   428→                  <Input
   429→                    required
   430→                    value={form.name}
   431→                    onChange={(e) => updateForm('name', e.target.value)}
   432→                    placeholder="Member name"
   433→                    className="mt-1 rounded-xl"
   434→                  />
   435→                </div>
   436→                <div>
   437→                  <Label className="text-xs font-medium text-gray-600">Age *</Label>
   438→                  <Input
   439→                    required
   440→                    type="number"
   441→                    min={0}
   442→                    max={150}
   443→                    value={form.age}
   444→                    onChange={(e) => updateForm('age', e.target.value)}
   445→                    placeholder="Age"
   446→                    className="mt-1 rounded-xl"
   447→                  />
   448→                </div>
   449→                <div>
   450→                  <Label className="text-xs font-medium text-gray-600">Gender *</Label>
