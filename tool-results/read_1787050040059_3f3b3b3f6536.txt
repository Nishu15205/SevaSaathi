   144→   144→  const [regPassword, setRegPassword] = useState("");
   145→   145→  const [regRole, setRegRole] = useState<UserRole>("FAMILY");
   146→   146→
   147→   147→  // Reset form (simplified — no OTP)

   149→   149→  const [resetNewPassword, setResetNewPassword] = useState("");
   150→   150→  const [resetLoading, setResetLoading] = useState(false);
   151→   151→  const [resetError, setResetError] = useState("");

   159→   159→  const [googleEmail, setGoogleEmail] = useState("");
   160→   160→  const [googleName, setGoogleName] = useState("");
   161→   161→  const [googleError, setGoogleError] = useState("");
   162→   162→  const [googleSubmitting, setGoogleSubmitting] = useState(false);

   200→   200→  /* ================================================================ */
   201→   201→
   202→   202→  /* ---- Google sign-in — real OAuth popup, fallback to simulated ---- */
   203→   203→  const handleGoogleSignIn = async () => {
   204→   204→    setGoogleLoading(true);
   205→   205→    setGoogleError("");
   206→   206→    try {
   207→   207→      // Check if real Google OAuth is configured
   208→   208→      const checkRes = await fetch("/api/auth/google-configured");
   209→   209→      const { configured } = await checkRes.json();
   210→   210→
   211→   211→      if (configured) {
   212→   212→        // Real Google OAuth — use custom API to get correct redirect_uri
   213→   213→        // Pass window.location.origin so the server uses the browser's actual URL
   214→   214→        const origin = window.location.origin;
   215→   215→        const goRes = await fetch(`/api/auth/google-go?origin=${encodeURIComponent(origin)}`);
   216→   216→        const goData = await goRes.json();
   217→   217→
   218→   218→        if (!goData.url) {
   219→   219→          setGoogleError("Failed to start Google sign-in. Please try again.");
   220→   220→          setGoogleLoading(false);
   221→   221→          return;
   222→   222→        }
   223→   223→
   224→   224→        const width = 500;
   225→   225→        const height = 650;
   226→   226→        const left = (window.screen.width - width) / 2;
   227→   227→        const top = (window.screen.height - height) / 2;
   228→   228→
   229→   229→        const popup = window.open(
   230→   230→          goData.url,
   231→   231→          "google-auth",
   232→   232→          `width=${width},height=${height},left=${left},top=${top},scrollbars=yes,resizable=yes`
   233→   233→        );
   234→   234→
   235→   235→        if (!popup) {
   236→   236→          setGoogleError(
   237→   237→            "Popup blocked! Please allow popups for this site and try again."
   238→   238→          );
   239→   239→          setGoogleLoading(false);
   240→   240→          return;
   241→   241→        }
   242→   242→
   243→   243→        // Listen for postMessage from popup callback
   244→   244→        let settled = false;
   245→   245→        const cleanup = () => {
   246→   246→          if (settled) return;
   247→   247→          settled = true;
   248→   248→          window.removeEventListener("message", handleMessage);
   249→   249→          clearInterval(checkInterval);
   250→   250→        };
   251→   251→
   252→   252→        const completeLogin = async () => {
   253→   253→          cleanup();
   254→   254→          const sessionRes = await fetch("/api/auth/session");
   255→   255→          const session = await sessionRes.json();
   256→   256→          if (session?.user) {
   257→   257→            setAuth({
   258→   258→              id: session.user.id || "",
   259→   259→              email: session.user.email || "",
   260→   260→              name: session.user.name || "",
   261→   261→              phone: "",
   262→   262→              role: (session.user as any).role || "FAMILY",
   263→   263→              avatarUrl: session.user.image || null,
   264→   264→              subscription: "NONE",
   265→   265→              patientProfiles: [],
   266→   266→              caregiverProfile: null,
   267→   267→            });
   268→   268→            toast({
   269→   269→              title: "Welcome! 🎉",
   270→   270→              description: "Signed in with Google as " + session.user.email,
   271→   271→            });
   272→   272→            setGoogleLoading(false);
   273→   273→            handleClose();
   274→   274→          } else {
   275→   275→            setGoogleLoading(false);
   276→   276→          }
   277→   277→        };
   278→   278→
   279→   279→        const handleMessage = async (event: MessageEvent) => {
   280→   280→          if (event.data?.type === "google-oauth-callback") {
   281→   281→            cleanup();
   282→   282→            if (event.data.success) {
   283→   283→              await completeLogin();
   284→   284→            } else {
   285→   285→              setGoogleError("Google sign-in was cancelled or failed.");
   286→   286→              setGoogleLoading(false);
   287→   287→            }
   288→   288→          }
   289→   289→        };
   290→   290→        window.addEventListener("message", handleMessage);
   291→   291→
   292→   292→        // Fallback: poll for session if postMessage doesn't fire
   293→   293→        const checkInterval = setInterval(async () => {
   294→   294→          if (popup.closed) {
   295→   295→            cleanup();
   296→   296→            setGoogleLoading(false);
   297→   297→            return;
   298→   298→          }
   299→   299→          try {
   300→   300→            const res = await fetch("/api/auth/session");
   301→   301→            const session = await res.json();
   302→   302→            if (session?.user) {
   303→   303→              await completeLogin();
   304→   304→            }
   305→   305→          } catch {
   306→   306→            // keep polling
   307→   307→          }
   308→   308→        }, 2000);
   309→   309→
   310→   310→        // Safety timeout — 5 minutes
   311→   311→        setTimeout(() => {
   312→   312→          cleanup();
   313→   313→          setGoogleLoading(false);
   314→   314→        }, 5 * 60 * 1000);
   315→   315→        return;
   316→   316→      }
   317→   317→    } catch {
   318→   318→      // If check fails, fall through to simulated flow
   319→   319→    }
   320→   320→    setGoogleLoading(false);
   321→   321→    // Google OAuth not configured — open simulated sign-in
   322→   322→    setGoogleOpen(true);
   323→   323→  };
   324→   324→
   325→   325→  /* ---- Google popup submit ---- */
   326→   326→  const handleGoogleSubmit = async (e: React.FormEvent) => {
   327→   327→    e.preventDefault();
   328→   328→    if (!googleEmail.trim() || !googleName.trim()) {
   329→   329→      setGoogleError("Please enter your name and email.");
   330→   330→      return;
   331→   331→    }
   332→   332→    if (!googleEmail.trim().includes("@")) {
   333→   333→      setGoogleError("Please enter a valid email address.");
   334→   334→      return;
   335→   335→    }
   336→   336→    setGoogleSubmitting(true);
   337→   337→    setGoogleError("");
   338→   338→    try {
   339→   339→      // 1. Call the simulate endpoint (sets the session cookie)
   340→   340→      const res = await fetch("/api/auth/google-simulate", {
   341→   341→        method: "POST",
   342→   342→        headers: { "Content-Type": "application/json" },
   343→   343→        body: JSON.stringify({ email: googleEmail.trim(), name: googleName.trim() }),
   344→   344→      });
   345→   345→      const data = await res.json();
   346→   346→      if (!res.ok) {
   347→   347→        setGoogleError(data.error || "Google sign-in failed.");
   348→   348→        return;
   349→   349→      }
   350→   350→
   351→   351→      // 2. Fetch the NextAuth session (cookie is already set by step 1)
   352→   352→      const sessionRes = await fetch("/api/auth/session");
   353→   353→      const sessionData = await sessionRes.json();
   354→   354→
   355→   355→      if (sessionData?.user) {
   356→   356→        // 3. Directly update the Zustand store — no page reload needed
   357→   357→        setAuth({
   358→   358→          id: sessionData.user.id,
   359→   359→          email: sessionData.user.email,
   360→   360→          name: sessionData.user.name,
   361→   361→          phone: "",
   362→   362→          role: sessionData.user.role,
   363→   363→          avatarUrl: sessionData.user.image || null,
   364→   364→          subscription: "NONE" as any,
   365→   365→          patientProfiles: [],
   366→   366→          caregiverProfile: null,
   367→   367→        });
   368→   368→        setGoogleOpen(false);
   369→   369→        toast({
   370→   370→          title: "Welcome" + (data.user.isNewUser ? " to SevaSaathi!" : " back!") + " \ud83c\udf89",
   371→   371→          description: "Signed in as " + data.user.email,
   372→   372→        });
   373→   373→        handleClose();
   374→   374→      } else {
   375→   375→        // Fallback: reload the page so NextAuth picks up the cookie
   376→   376→        window.location.href = window.location.origin;
   377→   377→      }
   378→   378→    } catch {
   379→   379→      setGoogleError("Something went wrong. Please try again.");
   380→   380→    } finally {
   381→   381→      setGoogleSubmitting(false);
   382→   382→    }
   383→   383→  };
   384→   384→
   385→   385→  /* ---- Email / password login ---- */
   386→   386→  const handleLogin = async (e: React.FormEvent) => {
   387→   387→    e.preventDefault();
   388→   388→    if (!loginEmail.trim() || !loginPassword.trim()) {
   389→   389→      setError("Please enter your email and password.");
   390→   390→      return;
   391→   391→    }
   392→   392→    setLoading(true);
   393→   393→    setError("");
   394→   394→    try {
   395→   395→      const result = await signIn("credentials", {
   396→   396→        email: loginEmail.trim(),
   397→   397→        password: loginPassword,
   398→   398→        redirect: false,
   399→   399→      });
   400→   400→
   401→   401→      if (result?.error) {
   402→   402→        setError("Invalid email or password. Please try again.");
   403→   403→        return;
   404→   404→      }
   405→   405→
   406→   406→      // AuthSync in AuthProvider will automatically sync session → Zustand store
   407→   407→      toast({
   408→   408→        title: "Welcome back! 🎉",
   409→   409→        description: "You have successfully logged in.",
   410→   410→      });
   411→   411→      handleClose();
   412→   412→    } catch {
   413→   413→      setError("Something went wrong. Please try again.");
   414→   414→    } finally {
   415→   415→      setLoading(false);
   416→   416→    }
   417→   417→  };
   418→   418→
   419→   419→  /* ---- Registration ---- */
   420→   420→  const handleRegister = async (e: React.FormEvent) => {
   421→   421→    e.preventDefault();
   422→   422→    if (!regName.trim() || !regEmail.trim() || !regPhone.trim() || !regPassword.trim()) {
   423→   423→      setError("Please fill in all fields.");
   424→   424→      return;
   425→   425→    }
   426→   426→    if (regPassword.length < 6) {
   427→   427→      setError("Password must be at least 6 characters.");
   428→   428→      return;
   429→   429→    }
   430→   430→    setLoading(true);
   431→   431→    setError("");
   432→   432→    try {
   433→   433→      // 1. Register
   434→   434→      const regRes = await api.auth.register({
   435→   435→        name: regName.trim(),
   436→   436→        email: regEmail.trim(),
   437→   437→        phone: regPhone.trim(),
   438→   438→        password: regPassword,
   439→   439→        role: regRole,
   440→   440→      });
   441→   441→
   442→   442→      // 2. Auto-login with credentials
   443→   443→      const loginResult = await signIn("credentials", {
   444→   444→        email: regEmail.trim(),
   445→   445→        password: regPassword,
   446→   446→        redirect: false,
   447→   447→      });
   448→   448→
   449→   449→      if (loginResult?.error) {
   450→   450→        // Registration succeeded but auto-login failed — that's ok, just notify
   451→   451→        toast({ title: "Account created!", description: "Please log in with your credentials." });
   452→   452→        setTab("login");
   453→   453→        setLoginEmail(regEmail.trim());
   454→   454→        return;
   455→   455→      }
   456→   456→
   457→   457→      toast({ title: "Welcome to SevaSaathi! 🎉", description: "Your account has been created." });
   458→   458→      handleClose();
   459→   459→    } catch (err: any) {
   460→   460→      setError(err?.message || "Registration failed. Please try again.");
   461→   461→    } finally {
   462→   462→      setLoading(false);
   463→   463→    }
   464→   464→  };
   465→   465→
   466→   466→  /* ---- Reset password (simplified) ---- */
   467→   467→  const handleReset = async (e: React.FormEvent) => {
   468→   468→    e.preventDefault();
   469→   469→    if (!resetEmail.trim() || !resetNewPassword.trim()) {
   470→   470→      setResetError("Please enter your email and new password.");
   471→   471→      return;
   472→   472→    }
   473→   473→    if (resetNewPassword.length < 6) {
   474→   474→      setResetError("Password must be at least 6 characters.");
   475→   475→      return;
   476→   476→    }
   477→   477→    setResetLoading(true);
   478→   478→    setResetError("");
   479→   479→    try {
   480→   480→      const res = await fetch("/api/auth/reset-password", {
   481→   481→        method: "POST",
   482→   482→        headers: { "Content-Type": "application/json" },
   483→   483→        body: JSON.stringify({ email: resetEmail.trim(), newPassword: resetNewPassword }),
   484→   484→      });
   485→   485→      const data = await res.json();
   486→   486→      if (!res.ok) {
   487→   487→        setResetError(data.error || "Failed to reset password.");
   488→   488→        return;
   489→   489→      }
   490→   490→      setResetDone(true);
   491→   491→      toast({ title: "Password reset successfully!", description: "You can now log in with your new password." });
   492→   492→    } catch {
   493→   493→      setResetError("Something went wrong. Please try again.");
   494→   494→    } finally {
   495→   495→      setResetLoading(false);
   496→   496→    }
   497→   497→  };
   498→   498→
   499→   499→  /* ================================================================ */
   500→   500→  /*  RENDER                                                           */
   501→   501→  /* ================================================================ */
   502→   502→  return (
   503→   503→    <Dialog open={isOpen} onOpenChange={(open) => !open && handleClose()}>
   504→   504→      <DialogContent showCloseButton={false} className="sm:max-w-md p-0 overflow-hidden border-forest-200/40">
   505→   505→        {/* Single custom close button */}
