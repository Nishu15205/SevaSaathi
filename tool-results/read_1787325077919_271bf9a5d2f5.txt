     4→import {

    48→/* ------------------------------------------------------------------ */
    49→/*  Types                                                              */
    50→/* ------------------------------------------------------------------ */
    51→type Tab = "login" | "register";
    52→type RegStep = "form" | "otp";
    53→type UserRole = "FAMILY" | "CAREGIVER";
    54→

    62→      <path d="M12 23c2.97 0 5.46-.98 7.28-2.66l-3.57-2.77c-.98.66-2.23 1.06-3.71 1.06-2.86 0-5.29-1.93-6.16-4.53H2.18v2.84C3.99 20.53 7.7 23 12 23z" fill="#34A853" />

    84→/* ================================================================== */
    85→/*  MAIN COMPONENT                                                     */
    86→/* ================================================================== */
    87→export default function LoginModal({ isOpen, onClose, defaultTab }: LoginModalProps) {
    88→  /* ---- state ---- */
    89→  const [tab, setTab] = useState<Tab>(defaultTab === "register" ? "register" : "login");
    90→  const [showPassword, setShowPassword] = useState(false);
    91→  const [loading, setLoading] = useState(false);
    92→  const [error, setError] = useState("");
    93→  const [resetOpen, setResetOpen] = useState(false);
    94→
    95→  // Login form
    96→  const [loginEmail, setLoginEmail] = useState("");
    97→  const [loginPassword, setLoginPassword] = useState("");
    98→
    99→  // Register form
   100→  const [regName, setRegName] = useState("");
   101→  const [regEmail, setRegEmail] = useState("");
   102→  const [regPhone, setRegPhone] = useState("");
   103→  const [regPassword, setRegPassword] = useState("");
   104→  const [regRole, setRegRole] = useState<UserRole>("FAMILY");
   105→  const [regStep, setRegStep] = useState<RegStep>("form");
   106→  const [regOtp, setRegOtp] = useState("");
   107→  const [regDevOtp, setRegDevOtp] = useState("");
   108→  const [otpSending, setOtpSending] = useState(false);
   109→  const [otpVerifying, setOtpVerifying] = useState(false);
   110→  const [otpTimer, setOtpTimer] = useState(0);
   111→  const timerRef = useRef<ReturnType<typeof setInterval> | null>(null);
   112→
   113→  // Reset form
   114→  const [resetEmail, setResetEmail] = useState("");
   115→  const [resetNewPassword, setResetNewPassword] = useState("");
   116→  const [resetLoading, setResetLoading] = useState(false);
   117→  const [resetError, setResetError] = useState("");
   118→  const [resetDone, setResetDone] = useState(false);
   119→
   120→  const { setAuth } = useAuthStore();
   121→  const { toast } = useToast();
   122→
   123→  /* ---- OTP timer countdown ---- */
   124→  useEffect(() => {
   125→    if (otpTimer > 0) {
   126→      timerRef.current = setInterval(() => {
   127→        setOtpTimer((t) => {
   128→          if (t <= 1) {
   129→            if (timerRef.current) clearInterval(timerRef.current);
   130→            return 0;
   131→          }
   132→          return t - 1;
   133→        });
   134→      }, 1000);
   135→    }
   136→    return () => {
   137→      if (timerRef.current) clearInterval(timerRef.current);
   138→    };
   139→  }, [otpTimer > 0]);
   140→
   141→  /* ---- reset form fields when switching tabs ---- */
   142→  useEffect(() => {
   143→    setError("");
   144→    setLoginPassword("");
   145→    resetRegFlow();
   146→  }, [tab]);
   147→
   148→  /* ---- reset entire state when modal closes ---- */
   149→  const handleClose = useCallback(() => {
   150→    setTab(defaultTab === "register" ? "register" : "login");
   151→    setError(""); setShowPassword(false); setLoading(false);
   152→    setLoginEmail(""); setLoginPassword("");
   153→    resetRegFlow();
   154→    setResetOpen(false); setResetEmail(""); setResetNewPassword("");
   155→    setResetLoading(false); setResetError(""); setResetDone(false);
   156→    onClose();
   157→  }, [onClose, defaultTab]);
   158→
   159→  const resetRegFlow = () => {
   160→    setRegName(""); setRegEmail(""); setRegPhone(""); setRegPassword("");
   161→    setRegRole("FAMILY"); setRegStep("form"); setRegOtp(""); setRegDevOtp("");
   162→    setOtpSending(false); setOtpVerifying(false); setOtpTimer(0);
   163→    if (timerRef.current) clearInterval(timerRef.current);
   164→  };
   165→
   166→  /* ================================================================ */
   167→/*  HANDLERS                                                         */
   168→/* ================================================================ */
   169→
   170→  const handleGoogleSignIn = async () => {
   171→    setError("");
   172→    try {
   173→      const checkRes = await fetch("/api/auth/google-configured");
   174→      const { configured } = await checkRes.json();
   175→      if (configured) {
   176→        const origin = window.location.origin;
   177→        const role = tab === "register" ? regRole : "";
   178→        window.location.href = `/api/auth/google-go?origin=${encodeURIComponent(origin)}&role=${encodeURIComponent(role)}`;
   179→        return;
   180→      }
   181→    } catch { /* fall through */ }
   182→    toast({ title: "Google Sign-In", description: "Google OAuth is being configured. Please use email registration." });
   183→  };
   184→
   185→  const handleLogin = async (e: React.FormEvent) => {
   186→    e.preventDefault();
   187→    if (!loginEmail.trim() || !loginPassword.trim()) { setError("Please enter your email and password."); return; }
   188→    setLoading(true); setError("");
   189→    try {
   190→      const result = await signIn("credentials", { email: loginEmail.trim(), password: loginPassword, redirect: false });
   191→      if (result?.error) { setError("Invalid email or password. Please try again."); return; }
   192→      try {
   193→        const sessionRes = await fetch("/api/auth/session");
   194→        const sessionData = await sessionRes.json();
   195→        if (sessionData?.user) {
   196→          const u = sessionData.user as any;
   197→          setAuth({ id: u.id || "", email: u.email || "", name: u.name || "", phone: "", role: u.role || "FAMILY", avatarUrl: u.image || null, subscription: "NONE", patientProfiles: [], caregiverProfile: null });
   198→        }
   199→      } catch { /* AuthSync will pick it up */ }
   200→      toast({ title: "Welcome back! \ud83c\udf89", description: "You have successfully logged in." });
   201→      handleClose();
   202→    } catch { setError("Something went wrong. Please try again."); } finally { setLoading(false); }
   203→  };
   204→
   205→  /* ---- Send email OTP ---- */
   206→  const handleSendOtp = async () => {
   207→    if (!regEmail.trim().includes("@")) { setError("Please enter a valid email address."); return; }
   208→    setOtpSending(true); setError("");
   209→    try {
   210→      const res = await api.auth.sendOtp(regEmail.trim(), "REGISTER");
   211→      setRegDevOtp(res.otp || "");
   212→      setRegStep("otp");
   213→      setOtpTimer(120);
   214→      toast({ title: "OTP Sent!", description: `A 6-digit verification code has been sent to ${regEmail.trim()}` });
   215→    } catch (err: any) {
   216→      setError(err?.message || "Failed to send OTP. Please try again.");
   217→    } finally {
   218→      setOtpSending(false);
   219→    }
   220→  };
   221→
   222→  /* ---- Verify OTP and create account ---- */
   223→  const handleVerifyOtpAndRegister = async () => {
   224→    if (regOtp.length !== 6) { setError("Please enter the complete 6-digit OTP."); return; }
   225→    setOtpVerifying(true); setError("");
   226→    try {
   227→      await api.auth.verifyOtp(regEmail.trim(), regOtp, "REGISTER");
   228→      // OTP verified - now create the account
   229→      await api.auth.register({
   230→        name: regName.trim(),
   231→        email: regEmail.trim(),
   232→        phone: regPhone.trim(),
   233→        password: regPassword,
   234→        role: regRole,
   235→      });
   236→      // Auto-login
   237→      const loginResult = await signIn("credentials", { email: regEmail.trim(), password: regPassword, redirect: false });
   238→      if (loginResult?.error) {
   239→        toast({ title: "Account created!", description: "Your email is verified. Please log in with your credentials." });
   240→        setTab("login"); setLoginEmail(regEmail.trim()); return;
   241→      }
   242→      try {
   243→        const sessionRes = await fetch("/api/auth/session");
   244→        const sessionData = await sessionRes.json();
   245→        if (sessionData?.user) {
   246→          const u = sessionData.user as any;
   247→          setAuth({ id: u.id || "", email: u.email || "", name: u.name || "", phone: regPhone.trim(), role: u.role || regRole, avatarUrl: u.image || null, subscription: "NONE", patientProfiles: [], caregiverProfile: null });
   248→        }
   249→      } catch { /* AuthSync will pick it up */ }
   250→      toast({ title: "Welcome to SevaSaathi! \ud83c\udf89", description: "Your email has been verified and account created." });
   251→      handleClose();
   252→    } catch (err: any) {
   253→      setError(err?.message || "Verification failed. Please try again.");
   254→    } finally {
   255→      setOtpVerifying(false);
   256→    }
   257→  };
   258→
   259→  /* ---- Go back to form from OTP ---- */
   260→  const handleBackToForm = () => {
   261→    setRegStep("form");
   262→    setRegOtp("");
   263→    setRegDevOtp("");
   264→    setOtpTimer(0);
   265→    if (timerRef.current) clearInterval(timerRef.current);
   266→  };
   267→
   268→  /* ---- Resend OTP ---- */
   269→  const handleResendOtp = async () => {
   270→    setOtpSending(true); setError("");
   271→    try {
   272→      const res = await api.auth.sendOtp(regEmail.trim(), "REGISTER");
   273→      setRegDevOtp(res.otp || "");
   274→      setRegOtp("");
   275→      setOtpTimer(120);
   276→      toast({ title: "New OTP Sent!", description: "Check your email for the new verification code." });
   277→    } catch (err: any) {
   278→      setError(err?.message || "Failed to resend OTP.");
   279→    } finally {
   280→      setOtpSending(false);
   281→    }
   282→  };
   283→
   284→  const handleReset = async (e: React.FormEvent) => {
   285→    e.preventDefault();
   286→    if (!resetEmail.trim() || !resetNewPassword.trim()) { setResetError("Please enter your email and new password."); return; }
   287→    if (resetNewPassword.length < 6) { setResetError("Password must be at least 6 characters."); return; }
   288→    setResetLoading(true); setResetError("");
   289→    try {
   290→      const res = await fetch("/api/auth/reset-password", { method: "POST", headers: { "Content-Type": "application/json" }, body: JSON.stringify({ email: resetEmail.trim(), newPassword: resetNewPassword }) });
   291→      const data = await res.json();
   292→      if (!res.ok) { setResetError(data.error || "Failed to reset password."); return; }
   293→      setResetDone(true);
   294→      toast({ title: "Password reset successfully!", description: "You can now log in with your new password." });
   295→    } catch { setResetError("Something went wrong. Please try again."); } finally { setResetLoading(false); }
   296→  };
   297→
   298→  /* ================================================================ */
   299→/*  RENDER                                                           */
   300→/* ================================================================ */
   301→  const formatTimer = (seconds: number) => {
   302→    const m = Math.floor(seconds / 60);
   303→    const s = seconds % 60;
   304→    return `${m}:${s.toString().padStart(2, "0")}`;
   305→  };
   306→
   307→  return (
   308→    <Dialog open={isOpen} onOpenChange={(open) => {
   309→      if (!open && !otpSending && !otpVerifying) handleClose();
   310→    }}>
   311→      <DialogContent showCloseButton={false} className="sm:max-w-md p-0 overflow-hidden border-forest-200/40">
   312→        <button onClick={handleClose} className="absolute top-3 right-3 z-50 rounded-full p-1.5 text-gray-400 hover:text-gray-600 hover:bg-gray-100 transition-colors" aria-label="Close">
   313→          <X className="w-5 h-5" />
   314→        </button>
   315→
   316→        <div className="px-6 pt-6 pb-2">
   317→          <DialogHeader className="space-y-1 mb-1">
   318→            <div className="flex items-center gap-2 justify-center">
   319→              <div className="w-9 h-9 rounded-xl bg-forest-900 flex items-center justify-center">
   320→                <Heart className="w-5 h-5 text-lime-400" />
   321→              </div>
   322→              <DialogTitle className="text-xl font-bold text-forest-900">SevaSaathi</DialogTitle>
   323→            </div>
   324→            <DialogDescription className="text-sm text-gray-500 mt-1">
   325→              {tab === "login" ? "Sign in to your account" : regStep === "otp" ? "Verify your email address" : "Create your account to get started"}
   326→            </DialogDescription>
   327→          </DialogHeader>
   328→        </div>
   329→
   330→        <AnimatePresence>
   331→          {error && (
   332→            <motion.div initial={{ height: 0, opacity: 0 }} animate={{ height: "auto", opacity: 1 }} exit={{ height: 0, opacity: 0 }} className="overflow-hidden">
   333→              <div className="mx-6 mt-1 px-3 py-2 rounded-lg bg-red-50 border border-red-200 text-red-700 text-sm flex items-center gap-2">
   334→                <X className="w-4 h-4 flex-shrink-0" />{error}
   335→              </div>
   336→            </motion.div>
   337→          )}
   338→        </AnimatePresence>
   339→
   340→        <div className="px-6 pt-3">
   341→          {regStep === "otp" ? (
   342→            /* ---- OTP VERIFICATION STEP ---- */
   343→            <div className="flex items-center gap-2 mb-3">
   344→              <button onClick={handleBackToForm} className="rounded-full p-1.5 hover:bg-gray-100 transition-colors text-gray-500">
   345→                <ArrowLeft className="w-4 h-4" />
   346→              </button>
   347→              <div className="flex-1 h-1.5 rounded-full bg-gray-100 overflow-hidden">
   348→                <div className="h-full rounded-full bg-forest-500" style={{ width: "100%" }} />
   349→              </div>
   350→              <span className="text-xs font-medium text-forest-600">Step 2/2</span>
   351→            </div>
   352→          ) : (
   353→            <div className="flex bg-gray-100 rounded-lg p-1">
   354→              <button onClick={() => setTab("login")} className={`flex-1 py-2 px-4 rounded-md text-sm font-medium transition-all ${tab === "login" ? "bg-white text-forest-900 shadow-sm" : "text-gray-500 hover:text-gray-700"}`}>Sign In</button>
   355→              <button onClick={() => setTab("register")} className={`flex-1 py-2 px-4 rounded-md text-sm font-medium transition-all ${tab === "register" ? "bg-white text-forest-900 shadow-sm" : "text-gray-500 hover:text-gray-700"}`}>Create Account</button>
   356→            </div>
   357→          )}
   358→        </div>
   359→
   360→        <div className="px-6 py-4 overflow-hidden">
   361→          {/* LOGIN TAB */}
   362→          {tab === "login" && regStep !== "otp" && (
   363→            <div key="login" className="space-y-4">
   364→              <button type="button" onClick={handleGoogleSignIn} className="w-full flex items-center justify-center gap-3 h-11 px-4 rounded-lg border border-gray-300 bg-white hover:bg-gray-50 transition-all text-sm font-medium text-gray-700 shadow-sm">
   365→                <GoogleIcon /> Continue with Google
   366→              </button>
   367→              <div className="flex items-center gap-3">
   368→                <div className="flex-1 h-px bg-gray-200" />
   369→                <span className="text-xs text-gray-400 font-medium uppercase tracking-wider">or continue with email</span>
   370→                <div className="flex-1 h-px bg-gray-200" />
   371→              </div>
   372→              <form onSubmit={handleLogin} className="space-y-3">
   373→                  <div className="space-y-1.5">
   374→                    <Label htmlFor="login-email" className="text-sm font-medium text-gray-700">Email</Label>
   375→                    <div className="relative">
   376→                      <Mail className="absolute left-3 top-1/2 -translate-y-1/2 w-4 h-4 text-gray-400" />
   377→                      <Input id="login-email" type="email" placeholder="you@example.com" value={loginEmail} onChange={(e) => setLoginEmail(e.target.value)} className="pl-10 h-11 border-gray-300" autoComplete="email" disabled={loading} />
   378→                    </div>
   379→                  </div>
   380→                  <div className="space-y-1.5">
   381→                    <div className="flex items-center justify-between">
   382→                      <Label htmlFor="login-password" className="text-sm font-medium text-gray-700">Password</Label>
   383→                      <button type="button" onClick={() => setResetOpen(true)} className="text-xs font-medium text-forest-700 hover:text-forest-900">Forgot password?</button>
   384→                    </div>
   385→                    <div className="relative">
   386→                      <Lock className="absolute left-3 top-1/2 -translate-y-1/2 w-4 h-4 text-gray-400" />
   387→                      <Input id="login-password" type={showPassword ? "text" : "password"} placeholder="Enter your password" value={loginPassword} onChange={(e) => setLoginPassword(e.target.value)} className="pl-10 pr-10 h-11 border-gray-300" autoComplete="current-password" disabled={loading} />
   388→                      <button type="button" onClick={() => setShowPassword(!showPassword)} className="absolute right-3 top-1/2 -translate-y-1/2 text-gray-400 hover:text-gray-600" tabIndex={-1}>
   389→                        {showPassword ? <EyeOff className="w-4 h-4" /> : <Eye className="w-4 h-4" />}
   390→                      </button>
   391→                    </div>
   392→                  </div>
   393→                  <Button type="submit" disabled={loading} className="w-full h-11 bg-forest-900 hover:bg-forest-800 text-white font-semibold rounded-lg shadow-sm hover:shadow-md">
   394→                    {loading ? <><Loader2 className="w-4 h-4 mr-2 animate-spin" />Signing in...</> : <>Sign In <ArrowRight className="w-4 h-4 ml-2" /></>}
   395→                  </Button>
   396→                </form>
   397→                <p className="text-center text-sm text-gray-500 pt-1">Don&apos;t have an account? <button onClick={() => setTab("register")} className="font-semibold text-forest-700 hover:text-forest-900">Create one</button></p>
   398→            </div>
   399→            )}
   400→
   401→            {/* REGISTER TAB - FORM STEP */}
   402→            {tab === "register" && regStep === "form" && (
   403→              <div className="space-y-4">
   404→                {/* Role selector */}
   405→                <div className="space-y-2">
   406→                  <Label className="text-sm font-semibold text-gray-800">How would you like to use SevaSaathi?</Label>
   407→                  <div className="grid grid-cols-1 gap-2">
   408→                    <button type="button" onClick={() => setRegRole("FAMILY")} className={`flex items-start gap-3 p-3 rounded-xl border-2 text-left transition-all ${regRole === "FAMILY" ? "border-forest-500 bg-forest-50" : "border-gray-200 bg-white hover:border-gray-300"}`}>
   409→                      <div className={`w-9 h-9 rounded-lg flex items-center justify-center shrink-0 mt-0.5 ${regRole === "FAMILY" ? "bg-forest-100" : "bg-gray-100"}`}>
   410→                        <Heart className={`w-4.5 h-4.5 ${regRole === "FAMILY" ? "text-forest-700" : "text-gray-400"}`} />
   411→                      </div>
   412→                      <div className="min-w-0">
   413→                        <p className={`text-sm font-semibold ${regRole === "FAMILY" ? "text-forest-900" : "text-gray-700"}`}>I need care for my family</p>
   414→                        <p className={`text-xs mt-0.5 ${regRole === "FAMILY" ? "text-forest-600/80" : "text-gray-400"}`}>Find trusted caregivers for parents, elders, or family members at home</p>
   415→                      </div>
   416→                    </button>
   417→                    <button type="button" onClick={() => setRegRole("CAREGIVER")} className={`flex items-start gap-3 p-3 rounded-xl border-2 text-left transition-all ${regRole === "CAREGIVER" ? "border-forest-500 bg-forest-50" : "border-gray-200 bg-white hover:border-gray-300"}`}>
   418→                      <div className={`w-9 h-9 rounded-lg flex items-center justify-center shrink-0 mt-0.5 ${regRole === "CAREGIVER" ? "bg-forest-100" : "bg-gray-100"}`}>
   419→                        <ShieldCheck className={`w-4.5 h-4.5 ${regRole === "CAREGIVER" ? "text-forest-700" : "text-gray-400"}`} />
   420→                      </div>
   421→                      <div className="min-w-0">
   422→                        <p className={`text-sm font-semibold ${regRole === "CAREGIVER" ? "text-forest-900" : "text-gray-700"}`}>I want to provide care services</p>
