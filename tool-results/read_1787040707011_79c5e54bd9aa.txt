   124→/* ================================================================== */
   125→/*  MAIN COMPONENT                                                     */
   126→/* ================================================================== */
   127→export default function LoginModal({ isOpen, onClose, defaultTab }: LoginModalProps) {
   128→  /* ---- state ---- */
   129→  const [tab, setTab] = useState<Tab>(defaultTab === "register" ? "register" : "login");
   130→  const [showPassword, setShowPassword] = useState(false);
   131→  const [loading, setLoading] = useState(false);
   132→  const [googleLoading, setGoogleLoading] = useState(false);
   133→  const [error, setError] = useState("");
   134→  const [resetOpen, setResetOpen] = useState(false);
   135→
   136→  // Login form
   137→  const [loginEmail, setLoginEmail] = useState("");
   138→  const [loginPassword, setLoginPassword] = useState("");
   139→
   140→  // Register form
   141→  const [regName, setRegName] = useState("");
   142→  const [regEmail, setRegEmail] = useState("");
   143→  const [regPhone, setRegPhone] = useState("");
   144→  const [regPassword, setRegPassword] = useState("");
   145→  const [regRole, setRegRole] = useState<UserRole>("FAMILY");
   146→
   147→  // Reset form (simplified — no OTP)
   148→  const [resetEmail, setResetEmail] = useState("");
   149→  const [resetNewPassword, setResetNewPassword] = useState("");
   150→  const [resetLoading, setResetLoading] = useState(false);
   151→  const [resetError, setResetError] = useState("");
   152→  const [resetDone, setResetDone] = useState(false);
   153→
   154→  const { setAuth } = useAuthStore();
   155→  const { toast } = useToast();
   156→
   157→  // Google popup form
   158→  const [googleOpen, setGoogleOpen] = useState(false);
   159→  const [googleEmail, setGoogleEmail] = useState("");
   160→  const [googleName, setGoogleName] = useState("");
   161→  const [googleError, setGoogleError] = useState("");
   162→  const [googleSubmitting, setGoogleSubmitting] = useState(false);
   163→
   164→  /* ---- reset form fields when switching tabs ---- */
   165→  useEffect(() => {
   166→    setError("");
   167→    setLoginPassword("");
   168→  }, [tab]);
   169→
   170→  /* ---- reset entire state when modal closes ---- */
   171→  const handleClose = useCallback(() => {
   172→    setTab(defaultTab === "register" ? "register" : "login");
   173→    setError("");
   174→    setShowPassword(false);
   175→    setLoading(false);
   176→    setGoogleLoading(false);
   177→    setLoginEmail("");
   178→    setLoginPassword("");
   179→    setRegName("");
   180→    setRegEmail("");
   181→    setRegPhone("");
   182→    setRegPassword("");
   183→    setRegRole("FAMILY");
   184→    setResetOpen(false);
   185→    setResetEmail("");
   186→    setResetNewPassword("");
   187→    setResetLoading(false);
   188→    setResetError("");
   189→    setResetDone(false);
   190→    setGoogleOpen(false);
   191→    setGoogleEmail("");
   192→    setGoogleName("");
   193→    setGoogleError("");
   194→    setGoogleSubmitting(false);
   195→    onClose();
   196→  }, [onClose, defaultTab]);
   197→
   198→  /* ================================================================ */
   199→  /*  HANDLERS                                                         */
   200→  /* ================================================================ */
   201→
   202→  /* ---- Google sign-in — try real OAuth first, fallback to simulated ---- */
   203→  const handleGoogleSignIn = async () => {
   204→    setGoogleLoading(true);
   205→    setGoogleError("");
   206→    try {
   207→      // Check if real Google OAuth is configured
   208→      const checkRes = await fetch("/api/auth/google-configured");
   209→      const { configured } = await checkRes.json();
   210→
   211→      if (configured) {
   212→        // Real Google OAuth — redirects to Google's login page
   213→        signIn("google", {
   214→          callbackUrl: window.location.origin,
   215→        });
   216→        return; // Browser navigates away
   217→      }
   218→    } catch {
   219→      // If check fails, fall through to simulated flow
   220→    }
   221→    setGoogleLoading(false);
   222→    // Google OAuth not configured — open simulated sign-in
   223→    setGoogleOpen(true);
   224→  };
   225→
   226→  /* ---- Google popup submit ---- */
   227→  const handleGoogleSubmit = async (e: React.FormEvent) => {
   228→    e.preventDefault();
   229→    if (!googleEmail.trim() || !googleName.trim()) {
   230→      setGoogleError("Please enter your name and email.");
   231→      return;
   232→    }
   233→    if (!googleEmail.trim().includes("@")) {
   234→      setGoogleError("Please enter a valid email address.");
   235→      return;
   236→    }
   237→    setGoogleSubmitting(true);
   238→    setGoogleError("");
   239→    try {
   240→      // 1. Call the simulate endpoint (sets the session cookie)
   241→      const res = await fetch("/api/auth/google-simulate", {
   242→        method: "POST",
   243→        headers: { "Content-Type": "application/json" },
   244→        body: JSON.stringify({ email: googleEmail.trim(), name: googleName.trim() }),
   245→      });
   246→      const data = await res.json();
   247→      if (!res.ok) {
   248→        setGoogleError(data.error || "Google sign-in failed.");
   249→        return;
   250→      }
   251→
   252→      // 2. Fetch the NextAuth session (cookie is already set by step 1)
   253→      const sessionRes = await fetch("/api/auth/session");
   254→      const sessionData = await sessionRes.json();
   255→
   256→      if (sessionData?.user) {
   257→        // 3. Directly update the Zustand store — no page reload needed
   258→        setAuth({
   259→          id: sessionData.user.id,
   260→          email: sessionData.user.email,
   261→          name: sessionData.user.name,
   262→          phone: "",
   263→          role: sessionData.user.role,
   264→          avatarUrl: sessionData.user.image || null,
   265→          subscription: "NONE" as any,
   266→          patientProfiles: [],
   267→          caregiverProfile: null,
   268→        });
   269→        setGoogleOpen(false);
   270→        toast({
   271→          title: "Welcome" + (data.user.isNewUser ? " to SevaSaathi!" : " back!") + " \ud83c\udf89",
   272→          description: "Signed in as " + data.user.email,
   273→        });
   274→        handleClose();
   275→      } else {
   276→        // Fallback: reload the page so NextAuth picks up the cookie
   277→        window.location.href = window.location.origin;
   278→      }
   279→    } catch {
   280→      setGoogleError("Something went wrong. Please try again.");
   281→    } finally {
   282→      setGoogleSubmitting(false);
   283→    }
   284→  };
   285→
   286→  /* ---- Email / password login ---- */
   287→  const handleLogin = async (e: React.FormEvent) => {
   288→    e.preventDefault();
   289→    if (!loginEmail.trim() || !loginPassword.trim()) {
   290→      setError("Please enter your email and password.");
   291→      return;
   292→    }
   293→    setLoading(true);
   294→    setError("");
   295→    try {
   296→      const result = await signIn("credentials", {
   297→        email: loginEmail.trim(),
   298→        password: loginPassword,
   299→        redirect: false,
   300→      });
   301→
   302→      if (result?.error) {
   303→        setError("Invalid email or password. Please try again.");
   304→        return;
   305→      }
   306→
   307→      // AuthSync in AuthProvider will automatically sync session → Zustand store
   308→      toast({
   309→        title: "Welcome back! 🎉",
   310→        description: "You have successfully logged in.",
   311→      });
   312→      handleClose();
   313→    } catch {
   314→      setError("Something went wrong. Please try again.");
   315→    } finally {
   316→      setLoading(false);
   317→    }
   318→  };
   319→
   320→  /* ---- Registration ---- */
   321→  const handleRegister = async (e: React.FormEvent) => {
   322→    e.preventDefault();
   323→    if (!regName.trim() || !regEmail.trim() || !regPhone.trim() || !regPassword.trim()) {
   324→      setError("Please fill in all fields.");
   325→      return;
   326→    }
   327→    if (regPassword.length < 6) {
   328→      setError("Password must be at least 6 characters.");
   329→      return;
   330→    }
   331→    setLoading(true);
   332→    setError("");
   333→    try {
   334→      // 1. Register
   335→      const regRes = await api.auth.register({
   336→        name: regName.trim(),
   337→        email: regEmail.trim(),
   338→        phone: regPhone.trim(),
   339→        password: regPassword,
   340→        role: regRole,
   341→      });
   342→
   343→      // 2. Auto-login with credentials
   344→      const loginResult = await signIn("credentials", {
   345→        email: regEmail.trim(),
   346→        password: regPassword,
   347→        redirect: false,
   348→      });
   349→
   350→      if (loginResult?.error) {
   351→        // Registration succeeded but auto-login failed — that's ok, just notify
   352→        toast({ title: "Account created!", description: "Please log in with your credentials." });
   353→        setTab("login");
   354→        setLoginEmail(regEmail.trim());
   355→        return;
   356→      }
   357→
   358→      toast({ title: "Welcome to SevaSaathi! 🎉", description: "Your account has been created." });
   359→      handleClose();
   360→    } catch (err: any) {
   361→      setError(err?.message || "Registration failed. Please try again.");
   362→    } finally {
   363→      setLoading(false);
   364→    }
   365→  };
   366→
   367→  /* ---- Reset password (simplified) ---- */
   368→  const handleReset = async (e: React.FormEvent) => {
   369→    e.preventDefault();
   370→    if (!resetEmail.trim() || !resetNewPassword.trim()) {
   371→      setResetError("Please enter your email and new password.");
   372→      return;
   373→    }
   374→    if (resetNewPassword.length < 6) {
   375→      setResetError("Password must be at least 6 characters.");
   376→      return;
   377→    }
   378→    setResetLoading(true);
   379→    setResetError("");
   380→    try {
   381→      const res = await fetch("/api/auth/reset-password", {
   382→        method: "POST",
   383→        headers: { "Content-Type": "application/json" },
   384→        body: JSON.stringify({ email: resetEmail.trim(), newPassword: resetNewPassword }),
   385→      });
   386→      const data = await res.json();
   387→      if (!res.ok) {
   388→        setResetError(data.error || "Failed to reset password.");
   389→        return;
   390→      }
   391→      setResetDone(true);
   392→      toast({ title: "Password reset successfully!", description: "You can now log in with your new password." });
   393→    } catch {
   394→      setResetError("Something went wrong. Please try again.");
   395→    } finally {
   396→      setResetLoading(false);
   397→    }
   398→  };
   399→
   400→  /* ================================================================ */
   401→  /*  RENDER                                                           */
   402→  /* ================================================================ */
   403→  return (
   404→    <Dialog open={isOpen} onOpenChange={(open) => !open && handleClose()}>
   405→      <DialogContent className="sm:max-w-md p-0 overflow-hidden border-forest-200/40">
   406→        {/* Custom X button to override default position */}
   407→        <button
   408→          onClick={handleClose}
   409→          className="absolute top-3 right-3 z-50 rounded-full p-1.5 text-gray-400 hover:text-gray-600 hover:bg-gray-100 transition-colors"
   410→          aria-label="Close"
   411→        >
   412→          <X className="w-5 h-5" />
   413→        </button>
   414→
   415→        <div className="px-6 pt-6 pb-2">
   416→          <DialogHeader className="space-y-1 mb-1">
   417→            <div className="flex items-center gap-2 justify-center">
   418→              <div className="w-9 h-9 rounded-xl bg-forest-900 flex items-center justify-center">
   419→                <Heart className="w-5 h-5 text-lime-400" />
   420→              </div>
   421→              <DialogTitle className="text-xl font-bold text-forest-900">
   422→                SevaSaathi
   423→              </DialogTitle>
   424→            </div>
   425→            <DialogDescription className="text-sm text-gray-500 mt-1">
   426→              {tab === "login"
   427→                ? "Sign in to your account"
   428→                : "Create your account to get started"}
   429→            </DialogDescription>
   430→          </DialogHeader>
   431→        </div>
   432→
   433→        {/* ---- Error banner ---- */}
   434→        <AnimatePresence>
   435→          {error && (
   436→            <motion.div
   437→              initial={{ height: 0, opacity: 0 }}
   438→              animate={{ height: "auto", opacity: 1 }}
   439→              exit={{ height: 0, opacity: 0 }}
   440→              className="overflow-hidden"
   441→            >
   442→              <div className="mx-6 mt-1 px-3 py-2 rounded-lg bg-red-50 border border-red-200 text-red-700 text-sm flex items-center gap-2">
   443→                <X className="w-4 h-4 flex-shrink-0" />
   444→                {error}
   445→              </div>
   446→            </motion.div>
   447→          )}
   448→        </AnimatePresence>
   449→
   450→        {/* ---- Tab switcher ---- */}
   451→        <div className="px-6 pt-3">
   452→          <div className="flex bg-gray-100 rounded-lg p-1">
   453→            <button
   454→              onClick={() => setTab("login")}
   455→              className={`flex-1 py-2 px-4 rounded-md text-sm font-medium transition-all duration-200 ${
   456→                tab === "login"
   457→                  ? "bg-white text-forest-900 shadow-sm"
   458→                  : "text-gray-500 hover:text-gray-700"
   459→              }`}
   460→            >
   461→              Sign In
   462→            </button>
   463→            <button
   464→              onClick={() => setTab("register")}
   465→              className={`flex-1 py-2 px-4 rounded-md text-sm font-medium transition-all duration-200 ${
   466→                tab === "register"
   467→                  ? "bg-white text-forest-900 shadow-sm"
   468→                  : "text-gray-500 hover:text-gray-700"
   469→              }`}
   470→            >
   471→              Create Account
   472→            </button>
   473→          </div>
   474→        </div>
   475→
   476→        {/* ---- Animated content ---- */}
   477→        <div className="px-6 py-4 overflow-hidden">
   478→          <AnimatePresence mode="wait" custom={tab}>
   479→            {/* ========== LOGIN TAB ========== */}
   480→            {tab === "login" && (
   481→              <motion.div
   482→                key="login"
   483→                custom={tab}
   484→                variants={contentVariants}
   485→                initial="enter"
   486→                animate="center"
   487→                exit="exit"
   488→                className="space-y-4"
   489→              >
   490→                {/* Google button */}
   491→                <button
   492→                  type="button"
   493→                  onClick={handleGoogleSignIn}
   494→                  disabled={googleLoading}
   495→                  className="w-full flex items-center justify-center gap-3 h-11 px-4 rounded-lg border border-gray-300 bg-white hover:bg-gray-50 active:bg-gray-100 transition-all duration-150 text-sm font-medium text-gray-700 disabled:opacity-60 disabled:cursor-not-allowed shadow-sm"
   496→                >
   497→                  {googleLoading ? (
   498→                    <Loader2 className="w-5 h-5 animate-spin text-gray-500" />
   499→                  ) : (
   500→                    <GoogleIcon />
   501→                  )}
   502→                  Continue with Google
   503→                </button>
   504→
   505→                {/* Divider */}
   506→                <div className="flex items-center gap-3">
   507→                  <div className="flex-1 h-px bg-gray-200" />
   508→                  <span className="text-xs text-gray-400 font-medium uppercase tracking-wider">
   509→                    or continue with email
   510→                  </span>
   511→                  <div className="flex-1 h-px bg-gray-200" />
   512→                </div>
   513→
   514→                {/* Email / password form */}
   515→                <form onSubmit={handleLogin} className="space-y-3">
   516→                  <div className="space-y-1.5">
   517→                    <Label htmlFor="login-email" className="text-sm font-medium text-gray-700">
   518→                      Email
   519→                    </Label>
   520→                    <div className="relative">
   521→                      <Mail className="absolute left-3 top-1/2 -translate-y-1/2 w-4 h-4 text-gray-400" />
   522→                      <Input
   523→                        id="login-email"
   524→                        type="email"
   525→                        placeholder="you@example.com"
   526→                        value={loginEmail}
   527→                        onChange={(e) => setLoginEmail(e.target.value)}
   528→                        className="pl-10 h-11 border-gray-300 focus:border-forest-500 focus:ring-forest-500"
   529→                        autoComplete="email"
   530→                        disabled={loading}
   531→                      />
   532→                    </div>
   533→                  </div>
   534→
   535→                  <div className="space-y-1.5">
   536→                    <div className="flex items-center justify-between">
   537→                      <Label htmlFor="login-password" className="text-sm font-medium text-gray-700">
   538→                        Password
   539→                      </Label>
   540→                      <button
   541→                        type="button"
   542→                        onClick={() => setResetOpen(true)}
   543→                        className="text-xs font-medium text-forest-700 hover:text-forest-900 transition-colors"
   544→                      >
   545→                        Forgot password?
   546→                      </button>
   547→                    </div>
   548→                    <div className="relative">
   549→                      <Lock className="absolute left-3 top-1/2 -translate-y-1/2 w-4 h-4 text-gray-400" />
   550→                      <Input
   551→                        id="login-password"
   552→                        type={showPassword ? "text" : "password"}
   553→                        placeholder="Enter your password"
   554→                        value={loginPassword}
   555→                        onChange={(e) => setLoginPassword(e.target.value)}
   556→                        className="pl-10 pr-10 h-11 border-gray-300 focus:border-forest-500 focus:ring-forest-500"
   557→                        autoComplete="current-password"
   558→                        disabled={loading}
   559→                      />
   560→                      <button
   561→                        type="button"
   562→                        onClick={() => setShowPassword(!showPassword)}
   563→                        className="absolute right-3 top-1/2 -translate-y-1/2 text-gray-400 hover:text-gray-600 transition-colors"
   564→                        tabIndex={-1}
   565→                      >
   566→                        {showPassword ? <EyeOff className="w-4 h-4" /> : <Eye className="w-4 h-4" />}
   567→                      </button>
   568→                    </div>
   569→                  </div>
   570→
   571→                  <Button
   572→                    type="submit"
   573→                    disabled={loading}
   574→                    className="w-full h-11 bg-forest-900 hover:bg-forest-800 text-white font-semibold rounded-lg transition-all duration-150 shadow-sm hover:shadow-md"
   575→                  >
   576→                    {loading ? (
   577→                      <>
   578→                        <Loader2 className="w-4 h-4 mr-2 animate-spin" />
   579→                        Signing in...
   580→                      </>
   581→                    ) : (
   582→                      <>
   583→                        Sign In
   584→                        <ArrowRight className="w-4 h-4 ml-2" />
   585→                      </>
   586→                    )}
   587→                  </Button>
   588→                </form>
   589→
   590→                {/* Switch to register */}
   591→                <p className="text-center text-sm text-gray-500 pt-1">
   592→                  Don&apos;t have an account?{" "}
