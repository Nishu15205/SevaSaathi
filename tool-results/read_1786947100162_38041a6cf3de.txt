   269→   269→/* ─────────── VERIFICATIONS TAB ─────────── */
   270→   270→function VerificationsTab() {
   271→   271→  const [verifications, setVerifications] = useState<any[]>([]);
   272→   272→  const [loading, setLoading] = useState(true);
   273→   273→  const [rejecting, setRejecting] = useState<string | null>(null);
   274→   274→  const [approving, setApproving] = useState<string | null>(null);
   275→   275→  const [reasons, setReasons] = useState<Record<string, string>>({});
   276→   276→
   277→   277→  const fetchVerifications = useCallback(async () => {
   278→   278→    setLoading(true);
   279→   279→    try {
   280→   280→      const res = await api.admin.verifications.list();
   281→   281→      setVerifications(res.verifications ?? []);
   282→   282→    } catch {
   283→   283→      toast.error('Failed to load verifications');
   284→   284→    } finally {
   285→   285→      setLoading(false);
   286→   286→    }
   287→   287→  }, []);
   288→   288→
   289→   289→  useEffect(() => { fetchVerifications(); }, [fetchVerifications]);
   290→   290→
   291→   291→  const handleApprove = async (id: string) => {
   292→   292→    setApproving(id);
   293→   293→    try {
   294→   294→      await api.admin.verifications.update(id, { status: 'APPROVED' });
   295→   295→      toast.success('Verification approved successfully');
   296→   296→      fetchVerifications();
   297→   297→    } catch {
   298→   298→      toast.error('Failed to approve verification');
   299→   299→    } finally {
   300→   300→      setApproving(null);
   301→   301→    }
   302→   302→  };
   303→   303→
   304→   304→  const handleReject = async (id: string) => {
   305→   305→    const reason = reasons[id]?.trim();
   306→   306→    if (!reason) { toast.error('Please provide a rejection reason'); return; }
   307→   307→    setRejecting(id);
   308→   308→    try {
   309→   309→      await api.admin.verifications.update(id, { status: 'REJECTED', rejectionReason: reason });
   310→   310→      toast.success('Verification rejected');
   311→   311→      setReasons((prev) => { const n = { ...prev }; delete n[id]; return n; });
   312→   312→      fetchVerifications();
   313→   313→    } catch {
   314→   314→      toast.error('Failed to reject verification');
   315→   315→    } finally {
   316→   316→      setRejecting(null);
   317→   317→    }
   318→   318→  };
   319→   319→
   320→   320→  if (loading) {
   321→   321→    return <div className="space-y-3">{Array.from({ length: 3 }).map((_, i) => <Skeleton key={i} className="h-24 rounded-2xl" />)}</div>;
   322→   322→  }
   323→   323→
   324→   324→  if (verifications.length === 0) {
   325→   325→    return <Card className="rounded-2xl border-0 shadow-sm"><CardContent className="py-16 text-center"><ShieldCheck className="w-12 h-12 mx-auto text-[#14532d]/30 mb-3" /><p className="text-muted-foreground">No pending verifications.</p></CardContent></Card>;
   326→   326→  }
   327→   327→
   328→   328→  return (
   329→   329→    <div className="space-y-3">
   330→   330→      {verifications.map((v) => (
   331→   331→        <motion.div key={v.id} variants={fadeVariants} initial="hidden" animate="visible">
   332→   332→          <Card className="rounded-2xl card-hover border-0 shadow-sm">
   333→   333→            <CardContent className="p-4 space-y-3">
   334→   334→              <div className="flex flex-col sm:flex-row sm:items-center justify-between gap-2">
   335→   335→                <div className="flex items-center gap-2 flex-wrap">
   336→   336→                  <span className="font-semibold text-sm">{v.caregiver?.user?.name ?? 'Caregiver'}</span>
   337→   337→                  <Badge className={docTypeBadge[v.docType] ?? 'bg-gray-100 text-gray-700'}>{v.docType ?? 'DOC'}</Badge>
   338→   338→                </div>
   339→   339→                <span className="text-xs text-muted-foreground flex items-center gap-1">
   340→   340→                  <Clock className="w-3 h-3" />{v.createdAt ? new Date(v.createdAt).toLocaleDateString('en-IN', { day: 'numeric', month: 'short', year: 'numeric' }) : '—'}
   341→   341→                </span>
   342→   342→              </div>
   343→   343→              {v.docNumber && <p className="text-xs text-muted-foreground font-mono bg-muted/50 rounded-lg px-3 py-1.5 w-fit">{v.docNumber}</p>}
   344→   344→              <div className="flex items-center gap-2 pt-1">
   345→   345→                <Button
   346→   346→                  size="sm"
   347→   347→                  className="bg-[#14532d] hover:bg-[#14532d]/90 text-white rounded-full"
   348→   348→                  disabled={approving === v.id}
   349→   349→                  onClick={() => handleApprove(v.id)}
   350→   350→                >
   351→   351→                  {approving === v.id ? <Loader2 className="w-4 h-4 animate-spin" /> : <><CheckCircle2 className="w-4 h-4 mr-1" />Approve</>}
   352→   352→                </Button>
   353→   353→                <Button
   354→   354→                  size="sm"
   355→   355→                  variant="outline"
   356→   356→                  className="text-red-600 border-red-200 hover:bg-red-50 rounded-full"
   357→   357→                  disabled={rejecting === v.id}
   358→   358→                  onClick={() => {
   359→   359→                    if (!reasons[v.id]?.trim()) {
   360→   360→                      setReasons((prev) => ({ ...prev, [v.id]: prev[v.id] ?? '' }));
   361→   361→                    } else {
   362→   362→                      handleReject(v.id);
   363→   363→                    }
   364→   364→                  }}
   365→   365→                >
   366→   366→                  {rejecting === v.id ? <Loader2 className="w-4 h-4 animate-spin" /> : <><XCircle className="w-4 h-4 mr-1" />Reject</>}
   367→   367→                </Button>
   368→   368→              </div>
   369→   369→              {reasons[v.id] !== undefined && (
   370→   370→                <motion.div initial={{ height: 0, opacity: 0 }} animate={{ height: 'auto', opacity: 1 }} className="space-y-2">
   371→   371→                  <Textarea
   372→   372→                    placeholder="Enter rejection reason..."
   373→   373→                    value={reasons[v.id]}
   374→   374→                    onChange={(e) => setReasons((prev) => ({ ...prev, [v.id]: e.target.value }))}
   375→   375→                    className="text-sm rounded-xl min-h-[60px] resize-none"
   376→   376→                  />
   377→   377→                  <Button size="sm" variant="destructive" className="rounded-full" disabled={rejecting === v.id} onClick={() => handleReject(v.id)}>
   378→   378→                    {rejecting === v.id ? <Loader2 className="w-4 h-4 animate-spin" /> : <Ban className="w-4 h-4 mr-1" />}Confirm Rejection
   379→   379→                  </Button>
   380→   380→                </motion.div>
   381→   381→              )}
   382→   382→            </CardContent>
   383→   383→          </Card>
