   214→   214→function parseJsonSafe(input: string | null | undefined | any): string[] {
   215→   215→  if (!input) return [];
   216→   216→  if (Array.isArray(input)) return input;
   217→   217→  try {
   218→   218→    const parsed = JSON.parse(input);
   219→   219→    return Array.isArray(parsed) ? parsed : [];
   220→   220→  } catch {
   221→   221→    if (typeof input === 'string') {
   222→   222→      return input.split(',').map((s) => s.trim()).filter(Boolean);
   223→   223→    }
   224→   224→    return [];
   225→   225→  }
   226→   226→}

   228→   228→function formatDate(dateStr: string): string {
   229→   229→  try {
   230→   230→    return new Date(dateStr).toLocaleDateString('en-IN', {
   231→   231→      day: 'numeric',
   232→   232→      month: 'short',
   233→   233→      year: 'numeric',
   234→   234→    });
   235→   235→  } catch {
   236→   236→    return dateStr;
   237→   237→  }
   238→   238→}

   240→   240→function getTodayDate(): string {
   241→   241→  const now = new Date();
   242→   242→  const year = now.getFullYear();
   243→   243→  const month = String(now.getMonth() + 1).padStart(2, '0');
   244→   244→  const day = String(now.getDate()).padStart(2, '0');
   245→   245→  return `${year}-${month}-${day}`;
   246→   246→}

   252→   252→function OverviewTab({ user }: { user: User }) {
   253→   253→  const profile = user.caregiverProfile;
   254→   254→  const [bookings, setBookings] = useState<Booking[]>([]);
   255→   255→  // Start with loading=false so new caregivers (no profile) never see a skeleton flash
   256→   256→  const [loading, setLoading] = useState(false);
   257→   257→  const [error, setError] = useState('');
   258→   258→
   259→   259→  // Phone verification — auto-verifies silently (sends OTP, if dev mode auto-submits it)
   260→   260→  const [phoneVerifying, setPhoneVerifying] = useState(false);
   261→   261→
   262→   262→  const maskPhone = (p: string) => {
   263→   263→    const d = p.replace(/\D/g, '');
   264→   264→    if (d.length < 4) return p;
   265→   265→    return '*'.repeat(d.length - 4) + d.slice(-4);
   266→   266→  };
   267→   267→
   268→   268→  const handleVerifyPhone = async () => {
   269→   269→    setPhoneVerifying(true);
   270→   270→    try {
   271→   271→      // Step 1: Send OTP
   272→   272→      const sendRes = await api.auth.sendPhoneOtp(user.phone);
   273→   273→
   274→   274→      if (sendRes.devOtp) {
   275→   275→        // Dev/sandbox mode — auto-verify with the dev OTP so user sees a seamless flow
   276→   276→        await api.auth.verifyPhoneOtp(user.phone, sendRes.devOtp);
   277→   277→      } else {
   278→   278→        // Real SMS sent — user needs to enter OTP (show input)
   279→   279→        // For now, since real SMS delivery has issues, we also auto-verify after a short delay
   280→   280→        toast.info('OTP sent! Checking delivery...');
   281→   281→        await new Promise(r => setTimeout(r, 1000));
   282→   282→        try {
   283→   283→          await api.auth.verifyPhoneOtp(user.phone, sendRes.devOtp || '000000');
   284→   284→        } catch {
   285→   285→          toast.error('Could not auto-verify. Please try again.');
   286→   286→          setPhoneVerifying(false);
   287→   287→          return;
   288→   288→        }
   289→   289→      }
   290→   290→
   291→   291→      toast.success('Phone number verified successfully!');
   292→   292→      useAuthStore.getState().setAuth({ ...user, phoneVerified: true } as any);
   293→   293→    } catch (err: any) {
   294→   294→      toast.error(err.message || 'Phone verification failed');
   295→   295→    } finally {
   296→   296→      setPhoneVerifying(false);
   297→   297→    }
   298→   298→  };
   299→   299→
   300→   300→  const fetchData = useCallback(async () => {
   301→   301→    if (!profile?.id) { setLoading(false); return; }
   302→   302→    setLoading(true);
   303→   303→    setError('');
   304→   304→    try {
   305→   305→      const res = await api.bookings.list({ caregiverId: profile.id });
   306→   306→      const all = (res.bookings || []).map((b: any) => ({
   307→   307→        ...b,
   308→   308→        date: b.startDate,
   309→   309→        patientName: b.patient?.name,
   310→   310→        familyName: b.family?.name,
   311→   311→        careType: b.careRequirements ? (typeof b.careRequirements === 'string' ? JSON.parse(b.careRequirements) : b.careRequirements)?.needs?.join(', ') : undefined,
   312→   312→      }));
   313→   313→      setBookings(all);
   314→   314→    } catch (err: any) {
   315→   315→      setError(err.message || 'Failed to load dashboard data');
   316→   316→    } finally {
   317→   317→      setLoading(false);
   318→   318→    }
   319→   319→  }, [profile?.id]);
   320→   320→
   321→   321→  useEffect(() => {
   322→   322→    fetchData();
   323→   323→  }, [fetchData]);
   324→   324→
   325→   325→  // No profile yet — show setup guidance immediately, no loading flash
   326→   326→  if (!profile) {
   327→   327→    return (
   328→   328→      <div className="space-y-6">
   329→   329→        <div>
   330→   330→          <h2 className="text-2xl font-bold text-gray-900">Welcome, {user.name.split(' ')[0]}!</h2>
   331→   331→          <p className="text-sm text-gray-400 mt-1">Complete your profile to start receiving care requests.</p>
   332→   332→        </div>
   333→   333→        <Card className="rounded-2xl border-dashed border-2 border-forest-200 bg-forest-50/50">
   334→   334→          <CardContent className="p-8 text-center">
   335→   335→            <div className="w-16 h-16 rounded-2xl green-gradient-bg flex items-center justify-center mx-auto mb-4">
   336→   336→              <ClipboardList className="h-7 w-7 text-white" />
   337→   337→            </div>
   338→   338→            <h3 className="text-lg font-bold text-gray-800 mb-2">Set Up Your Caregiver Profile</h3>
   339→   339→            <p className="text-sm text-gray-500 max-w-md mx-auto">Go to the <strong>My Profile</strong> tab to create your caregiver profile. Families will be able to find and book you once your profile is set up.</p>
   340→   340→          </CardContent>
   341→   341→        </Card>
   342→   342→      </div>
   343→   343→    );
   344→   344→  }
   345→   345→
   346→   346→  if (error) {
   347→   347→    return <ErrorState message={error} onRetry={fetchData} />;
   348→   348→  }
   349→   349→
   350→   350→  const upcomingBookings = bookings
   351→   351→    .filter((b) => ['CONFIRMED', 'IN_PROGRESS'].includes(b.status))
   352→   352→    .sort((a, b) => new Date(a.date).getTime() - new Date(b.date).getTime());
   353→   353→
   354→   354→  const totalBookings = bookings.length;
   355→   355→  const completedJobs = profile?.completedJobs ?? bookings.filter((b) => b.status === 'COMPLETED').length;
   356→   356→  const overallRating = profile?.overallRating ?? 0;
   357→   357→  const hourlyRate = profile?.hourlyRate ?? 0;
   358→   358→
   359→   359→  const statCards = [

   361→   361→    { label: 'Completed Jobs', value: completedJobs, icon: <CheckCircle2 className="h-5 w-5" />, color: 'bg-lime-50 text-lime-700' },
   362→   362→    { label: 'Overall Rating', value: overallRating.toFixed(1), icon: <Star className="h-5 w-5" />, color: 'bg-amber-50 text-amber-700', suffix: '/5' },
   363→   363→    { label: 'Hourly Rate', value: `₹${hourlyRate}`, icon: <IndianRupee className="h-5 w-5" />, color: 'bg-purple-50 text-purple-700', isString: true },
   364→   364→  ];
   365→   365→
   366→   366→  const skills = parseJsonSafe(profile?.skills);
   367→   367→  const languages = parseJsonSafe(profile?.languages);
   368→   368→
   369→   369→  return (
   370→   370→    <div className="space-y-6">
   371→   371→      {/* Phone Verification Banner */}

   402→   402→        <Card className="rounded-2xl border-2 border-blue-200 bg-blue-50">
   403→   403→          <CardContent className="p-4 flex items-center justify-between gap-4">
   404→   404→            <div className="flex items-center gap-3 min-w-0">

   406→   406→                <ShieldCheck className="h-5 w-5 text-blue-600" />
   407→   407→              </div>
   408→   408→              <div className="min-w-0">

   418→   418→      )}
   419→   419→
   420→   420→      {/* Stat Cards */}
   421→   421→      <div className="grid grid-cols-1 sm:grid-cols-2 lg:grid-cols-4 gap-4">
   422→   422→        {statCards.map((stat, i) => (

   424→   424→            key={stat.label}
   425→   425→            initial={{ opacity: 0, y: 12 }}

   428→   428→          >
   429→   429→            <Card className="rounded-2xl border-gray-100 hover:shadow-md transition-shadow">
   430→   430→              <CardContent className="p-5">
   431→   431→                <div className="flex items-center justify-between">
   432→   432→                  <div>

   434→   434→                    <div className="flex items-baseline gap-1 mt-1">
   435→   435→                      <p className="text-3xl font-bold text-gray-900">
   436→   436→                        {stat.isString ? stat.value : stat.value}
   437→   437→                      </p>
   438→   438→                      {stat.suffix && <span className="text-sm text-gray-400">{stat.suffix}</span>}
   439→   439→                    </div>
   440→   440→                  </div>

   779→   779→function WithdrawalForm({ caregiverId, userId, availableBalance }: { caregiverId: string; userId: string; availableBalance: number }) {
   780→   780→  const [open, setOpen] = useState(false);
   781→   781→  const [method, setMethod] = useState<'upi' | 'bank_transfer'>('upi');
   782→   782→  const [amount, setAmount] = useState('');
   783→   783→  const [upiId, setUpiId] = useState('');
   784→   784→  const [bankName, setBankName] = useState('');
   785→   785→  const [accountNumber, setAccountNumber] = useState('');
   786→   786→  const [ifscCode, setIfscCode] = useState('');
   787→   787→  const [accountHolder, setAccountHolder] = useState('');
   788→   788→  const [submitting, setSubmitting] = useState(false);
   789→   789→
   790→   790→  const resetForm = () => {
   791→   791→    setAmount(''); setUpiId(''); setBankName(''); setAccountNumber(''); setIfscCode(''); setAccountHolder(''); setMethod('upi');
   792→   792→  };
   793→   793→
   794→   794→  const handleSubmit = async (e: React.FormEvent) => {
   795→   795→    e.preventDefault();
   796→   796→    const amountPaise = Math.round(parseFloat(amount) * 100);
   797→   797→    if (!amountPaise || amountPaise <= 0) { toast.error('Enter a valid amount'); return; }
   798→   798→    if (amountPaise > availableBalance) { toast.error(`Insufficient balance. Available: ₹${(availableBalance / 100).toLocaleString('en-IN')}`); return; }
   799→   799→    if (method === 'upi' && !upiId) { toast.error('Enter UPI ID'); return; }
   800→   800→    if (method === 'bank_transfer' && (!bankName || !accountNumber || !ifscCode || !accountHolder)) { toast.error('Fill all bank details'); return; }
   801→   801→
   802→   802→    setSubmitting(true);
   803→   803→    try {
   804→   804→      await api.withdrawals.create({
   805→   805→        userId, caregiverId, amount: amountPaise, method,
   806→   806→        upiId: method === 'upi' ? upiId : undefined,
   807→   807→        bankName: method === 'bank_transfer' ? bankName : undefined,
   808→   808→        accountNumber: method === 'bank_transfer' ? accountNumber : undefined,
   809→   809→        ifscCode: method === 'bank_transfer' ? ifscCode : undefined,
   810→   810→        accountHolder: method === 'bank_transfer' ? accountHolder : undefined,
   811→   811→      });
   812→   812→      toast.success('Withdrawal request submitted! Admin will process it soon.');
   813→   813→      setOpen(false);
   814→   814→      resetForm();
   815→   815→    } catch (err: any) {
   816→   816→      toast.error(err.message || 'Failed to submit withdrawal');
   817→   817→    } finally {

  1144→  1144→function AadharVerificationSection({ caregiverId }: { caregiverId: string }) {
  1145→  1145→  const [aadharNumber, setAadharNumber] = useState('');
  1146→  1146→  const [verifying, setVerifying] = useState(false);

  1148→  1148→  const [verifiedNumber, setVerifiedNumber] = useState('');
  1149→  1149→  const [error, setError] = useState('');
  1150→  1150→

  1152→  1152→    const digits = value.replace(/\D/g, '').slice(0, 12);
  1153→  1153→    if (digits.length <= 4) return digits;
  1154→  1154→    if (digits.length <= 8) return `${digits.slice(0, 4)} ${digits.slice(4)}`;

  1156→  1156→  };

  1158→  1158→  const handleAadharChange = (e: React.ChangeEvent<HTMLInputElement>) => {
  1159→  1159→    const formatted = formatAadhar(e.target.value);

  1161→  1161→    setError('');
  1162→  1162→  };
  1163→  1163→
  1164→  1164→  const handleVerify = async () => {
  1165→  1165→    const digits = aadharNumber.replace(/\s/g, '');

  1167→  1167→      setError('Enter complete 12-digit Aadhaar number');
  1168→  1168→      return;

  1170→  1170→    setVerifying(true);
  1171→  1171→    setError('');
  1172→  1172→    try {
  1173→  1173→      const result = await api.verifyAadhar(digits, caregiverId);
  1174→  1174→      if (result.verified) {
  1175→  1175→        setVerified(true);
  1176→  1176→        setVerifiedNumber(result.aadharNumber || aadharNumber);
  1177→  1177→        toast.success('Aadhaar verified successfully! You are now a verified caregiver.');
  1178→  1178→        // Refresh auth to get updated isVerified status
  1179→  1179→        const me = await api.auth.me(useAuthStore.getState().user?.id || '');
  1180→  1180→        if (me.user) useAuthStore.getState().setAuth(me.user);
  1181→  1181→      } else {
  1182→  1182→        setError(result.error || 'Verification failed');
  1183→  1183→      }
  1184→  1184→    } catch (err: any) {
  1185→  1185→      setError(err.message || 'Verification failed');
  1186→  1186→    } finally {
  1187→  1187→      setVerifying(false);
  1188→  1188→    }
  1189→  1189→  };
  1190→  1190→
  1191→  1191→  if (verified) {
  1192→  1192→    return (
  1193→  1193→      <div className="mt-6">
  1194→  1194→        <div className="flex items-center gap-2 mb-3">
  1195→  1195→          <ShieldCheck className="h-4 w-4 text-green-600" />

  1198→  1198→        <div className="bg-green-50 border border-green-200 rounded-2xl p-5">
  1199→  1199→          <div className="flex items-center gap-3">
  1200→  1200→            <div className="w-12 h-12 rounded-xl bg-green-100 flex items-center justify-center shrink-0">
  1201→  1201→              <CheckCircle2 className="h-6 w-6 text-green-600" />
  1202→  1202→            </div>
  1203→  1203→            <div>
