   281→   281→      setGoogleError("Please enter your name and email.");
   282→   282→      return;
   283→   283→    }

   286→   286→      return;
   287→   287→    }

   291→   291→      // 1. Call the simulate endpoint (sets the session cookie)
   292→   292→      const res = await fetch("/api/auth/google-simulate", {
   293→   293→        method: "POST",
   294→   294→        headers: { "Content-Type": "application/json" },
   295→   295→        body: JSON.stringify({ email: googleEmail.trim(), name: googleName.trim(), role: tab === "register" ? regRole : undefined }),
   296→   296→      });

   298→   298→      if (!res.ok) {
   299→   299→        setGoogleError(data.error || "Google sign-in failed.");
   300→   300→        return;
   301→   301→      }
   302→   302→
   303→   303→      // 2. Fetch the NextAuth session (cookie is already set by step 1)
   304→   304→      const sessionRes = await fetch("/api/auth/session");
   305→   305→      const sessionData = await sessionRes.json();

   307→   307→      if (sessionData?.user) {

   310→   310→          id: sessionData.user.id,
   311→   311→          email: sessionData.user.email,
   312→   312→          name: sessionData.user.name,
   313→   313→          phone: "",
   314→   314→          role: sessionData.user.role,
   315→   315→          avatarUrl: sessionData.user.image || null,
   316→   316→          subscription: "NONE" as any,
   317→   317→          phoneVerified: false,
   318→   318→          patientProfiles: [],
   319→   319→          caregiverProfile: null,
   320→   320→        });
   321→   321→        setGoogleOpen(false);
   322→   322→        toast({
   323→   323→          title: "Welcome" + (data.user.isNewUser ? " to SevaSaathi!" : " back!") + " \ud83c\udf89",
   324→   324→          description: "Signed in as " + data.user.email,
   325→   325→        });
   326→   326→        handleClose();
   327→   327→      } else {
   328→   328→        // Fallback: reload the page so NextAuth picks up the cookie
   329→   329→        window.location.href = window.location.origin;
   330→   330→      }
   331→   331→    } catch {
   332→   332→      setGoogleError("Something went wrong. Please try again.");
   333→   333→    } finally {
   334→   334→      setGoogleSubmitting(false);
   335→   335→    }

   382→   382→      toast({
   383→   383→        title: "Welcome back! \ud83c\udf89",
   384→   384→        description: "You have successfully logged in.",

   386→   386→      handleClose();
   387→   387→    } catch {
   388→   388→      setError("Something went wrong. Please try again.");
   389→   389→    } finally {
   390→   390→      setLoading(false);
   391→   391→    }
   392→   392→  };
   393→   393→
   394→   394→  /* ---- Registration ---- */

   410→   410→        name: regName.trim(),
   411→   411→        email: regEmail.trim(),
   412→   412→        phone: regPhone.trim(),
   413→   413→        password: regPassword,

   415→   415→      });
   416→   416→
   417→   417→      const registeredUserId = regRes.user.id;
   418→   418→

   421→   421→        email: regEmail.trim(),
   422→   422→        password: regPassword,
   423→   423→        redirect: false,

   432→   432→      }

   437→   437→        const sessionRes = await fetch("/api/auth/session");
   438→   438→        const sessionData = await sessionRes.json();

   440→   440→          const u = sessionData.user as any;
   441→   441→          authData = {
   442→   442→            id: u.id || registeredUserId,
   443→   443→            email: u.email || regEmail.trim(),
   444→   444→            name: u.name || regName.trim(),
   445→   445→            phone: regPhone.trim(),
   446→   446→            role: u.role || regRole,
   447→   447→            avatarUrl: u.image || null,
   448→   448→            subscription: "NONE",
   449→   449→            phoneVerified: false,
   450→   450→            patientProfiles: [],
   451→   451→            caregiverProfile: null,
   452→   452→          };
   453→   453→          setAuth(authData);
   454→   454→        }
   455→   455→      } catch {
   456→   456→        // AuthSync will pick it up as fallback
   457→   457→      }
   458→   458→
   459→   459→      // 3. After successful registration, show phone verification for CAREGIVER (mandatory)
   460→   460→      //    or show optional banner for FAMILY
   461→   461→      if (regRole === "CAREGIVER") {
   462→   462→        // Mandatory phone verification for caregivers
   463→   463→        setPhoneVerifiedUserId(registeredUserId);
   464→   464→        setPhoneVerifiedPhone(regPhone.trim());
   465→   465→        setModalView("phone-verify");
   466→   466→        setPhoneOtpSent(false);
   467→   467→        setPhoneOtp("");
   468→   468→        setPhoneOtpError("");
   469→   469→        // Auto-send OTP
   470→   470→        handleSendPhoneOtp(regPhone.trim());
   471→   471→      } else {
   472→   472→        // Optional for family members — show a toast and let them in
   473→   473→        toast({
   474→   474→          title: "Welcome to SevaSaathi! \ud83c\udf89",
   475→   475→          description: "Your account has been created. Verify your phone number later for full access.",
   476→   476→        });
   477→   477→        handleClose();
   478→   478→      }
   479→   479→    } catch (err: any) {
   480→   480→      setError(err?.message || "Registration failed. Please try again.");
   481→   481→    } finally {
   482→   482→      setLoading(false);
   483→   483→    }
   484→   484→  };
   485→   485→
   486→   486→  /* ---- Phone OTP Send ---- */
   487→   487→  const handleSendPhoneOtp = async (phone?: string) => {
   488→   488→    const phoneToUse = phone || phoneVerifiedPhone;
   489→   489→    if (!phoneToUse) return;
   490→   490→    setPhoneOtpSending(true);
