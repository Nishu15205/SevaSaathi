   140→/* ================================================================== */
   141→export default function LoginModal({ isOpen, onClose, defaultTab }: LoginModalProps) {
   142→  /* ---- state ---- */
   143→  const [tab, setTab] = useState<Tab>(defaultTab === "register" ? "register" : "login");
   144→  const [showPassword, setShowPassword] = useState(false);
   145→  const [loading, setLoading] = useState(false);
   146→  const [error, setError] = useState("");
   147→  const [resetOpen, setResetOpen] = useState(false);
   148→
   149→  // Login form
   150→  const [loginEmail, setLoginEmail] = useState("");
   151→  const [loginPassword, setLoginPassword] = useState("");
   152→
   153→  // Register form
   154→  const [regName, setRegName] = useState("");
   155→  const [regEmail, setRegEmail] = useState("");
   156→  const [regPhone, setRegPhone] = useState("");
   157→  const [regPassword, setRegPassword] = useState("");
   158→  const [regRole, setRegRole] = useState<UserRole>("FAMILY");
   159→
   160→  // Reset form (simplified — no OTP)
   161→  const [resetEmail, setResetEmail] = useState("");
   162→  const [resetNewPassword, setResetNewPassword] = useState("");
   163→  const [resetLoading, setResetLoading] = useState(false);
   164→  const [resetError, setResetError] = useState("");
   165→  const [resetDone, setResetDone] = useState(false);
   166→
   167→  // Google popup form
   168→  const [googleOpen, setGoogleOpen] = useState(false);
   169→  const [googleEmail, setGoogleEmail] = useState("");
   170→  const [googleName, setGoogleName] = useState("");
   171→  const [googleError, setGoogleError] = useState("");
   172→  const [googleSubmitting, setGoogleSubmitting] = useState(false);
   173→
   174→  // Phone verification state
   175→  const [modalView, setModalView] = useState<ModalView>("auth");
   176→  const [phoneOtp, setPhoneOtp] = useState("");
   177→  const [phoneOtpError, setPhoneOtpError] = useState("");
   178→  const [phoneOtpLoading, setPhoneOtpLoading] = useState(false);
   179→  const [phoneOtpSent, setPhoneOtpSent] = useState(false);
   180→  const [phoneOtpSending, setPhoneOtpSending] = useState(false);
   181→  const [phoneVerifiedUserId, setPhoneVerifiedUserId] = useState("");
   182→  const [phoneVerifiedPhone, setPhoneVerifiedPhone] = useState("");
   183→  const [phoneVerifyCountdown, setPhoneVerifyCountdown] = useState(0);
   184→  const countdownRef = useRef<ReturnType<typeof setInterval> | null>(null);
   185→
   186→  const { setAuth } = useAuthStore();
   187→  const { toast } = useToast();
   188→
   189→  /* ---- reset form fields when switching tabs ---- */
   190→  useEffect(() => {
   191→    setError("");
   192→    setLoginPassword("");
   193→  }, [tab]);
   194→
   195→  /* ---- countdown timer for resend ---- */
   196→  useEffect(() => {
   197→    if (phoneVerifyCountdown > 0) {
   198→      countdownRef.current = setInterval(() => {
   199→        setPhoneVerifyCountdown((prev) => {
   200→          if (prev <= 1) {
   201→            if (countdownRef.current) clearInterval(countdownRef.current);
   202→            return 0;
   203→          }
   204→          return prev - 1;
   205→        });
   206→      }, 1000);
   207→      return () => {
   208→        if (countdownRef.current) clearInterval(countdownRef.current);
   209→      };
   210→    }
   211→  }, [phoneVerifyCountdown]);
   212→
   213→  /* ---- reset entire state when modal closes ---- */
   214→  const handleClose = useCallback(() => {
   215→    setTab(defaultTab === "register" ? "register" : "login");
   216→    setError("");
   217→    setShowPassword(false);
   218→    setLoading(false);
   219→    setLoginEmail("");
   220→    setLoginPassword("");
   221→    setRegName("");
   222→    setRegEmail("");
   223→    setRegPhone("");
   224→    setRegPassword("");
   225→    setRegRole("FAMILY");
   226→    setResetOpen(false);
   227→    setResetEmail("");
   228→    setResetNewPassword("");
   229→    setResetLoading(false);
   230→    setResetError("");
   231→    setResetDone(false);
   232→    setGoogleOpen(false);
   233→    setGoogleEmail("");
   234→    setGoogleName("");
   235→    setGoogleError("");
   236→    setGoogleSubmitting(false);
   237→    // Phone verification reset
   238→    setModalView("auth");
   239→    setPhoneOtp("");
   240→    setPhoneOtpError("");
   241→    setPhoneOtpLoading(false);
   242→    setPhoneOtpSent(false);
   243→    setPhoneOtpSending(false);
   244→    setPhoneVerifiedUserId("");
   245→    setPhoneVerifiedPhone("");
   246→    setPhoneVerifyCountdown(0);
   247→    if (countdownRef.current) clearInterval(countdownRef.current);
   248→    onClose();
   249→  }, [onClose, defaultTab]);
   250→
   251→  /* ================================================================ */
   252→  /*  HANDLERS                                                         */
   253→  /* ================================================================ */
   254→
   255→  /* ---- Google sign-in — redirect flow, fallback to simulated ---- */
   256→  const handleGoogleSignIn = async () => {
   257→    setGoogleError("");
   258→    try {
   259→      // Check if real Google OAuth is configured
   260→      const checkRes = await fetch("/api/auth/google-configured");
   261→      const { configured } = await checkRes.json();
   262→
   263→      if (configured) {
   264→        // Real Google OAuth — full-page redirect to /api/auth/google-go
   265→        const origin = window.location.origin;
   266→        const role = tab === "register" ? regRole : "";
   267→        window.location.href = `/api/auth/google-go?origin=${encodeURIComponent(origin)}&role=${encodeURIComponent(role)}`;
   268→        return;
   269→      }
   270→    } catch {
   271→      // If check fails, fall through to simulated flow
   272→    }
   273→    // Google OAuth not configured — open simulated sign-in
   274→    setGoogleOpen(true);
   275→  };
   276→
   277→  /* ---- Google popup submit ---- */
   278→  const handleGoogleSubmit = async (e: React.FormEvent) => {
   279→    e.preventDefault();
   280→    if (!googleEmail.trim() || !googleName.trim()) {
   281→      setGoogleError("Please enter your name and email.");
   282→      return;
   283→    }
   284→    if (!googleEmail.trim().includes("@")) {
   285→      setGoogleError("Please enter a valid email address.");
   286→      return;
   287→    }
   288→    setGoogleSubmitting(true);
   289→    setGoogleError("");
   290→    try {
   291→      // 1. Call the simulate endpoint (sets the session cookie)
   292→      const res = await fetch("/api/auth/google-simulate", {
   293→        method: "POST",
   294→        headers: { "Content-Type": "application/json" },
   295→        body: JSON.stringify({ email: googleEmail.trim(), name: googleName.trim(), role: tab === "register" ? regRole : undefined }),
   296→      });
   297→      const data = await res.json();
   298→      if (!res.ok) {
   299→        setGoogleError(data.error || "Google sign-in failed.");
   300→        return;
   301→      }
   302→
   303→      // 2. Fetch the NextAuth session (cookie is already set by step 1)
   304→      const sessionRes = await fetch("/api/auth/session");
   305→      const sessionData = await sessionRes.json();
   306→
   307→      if (sessionData?.user) {
   308→        // 3. Directly update the Zustand store — no page reload needed
   309→        setAuth({
   310→          id: sessionData.user.id,
   311→          email: sessionData.user.email,
   312→          name: sessionData.user.name,
   313→          phone: "",
   314→          role: sessionData.user.role,
   315→          avatarUrl: sessionData.user.image || null,
   316→          subscription: "NONE" as any,
   317→          phoneVerified: false,
   318→          patientProfiles: [],
   319→          caregiverProfile: null,
   320→        });
   321→        setGoogleOpen(false);
   322→        toast({
   323→          title: "Welcome" + (data.user.isNewUser ? " to SevaSaathi!" : " back!") + " \ud83c\udf89",
   324→          description: "Signed in as " + data.user.email,
   325→        });
   326→        handleClose();
   327→      } else {
   328→        // Fallback: reload the page so NextAuth picks up the cookie
   329→        window.location.href = window.location.origin;
   330→      }
   331→    } catch {
   332→      setGoogleError("Something went wrong. Please try again.");
   333→    } finally {
   334→      setGoogleSubmitting(false);
   335→    }
   336→  };
   337→
   338→  /* ---- Email / password login ---- */
   339→  const handleLogin = async (e: React.FormEvent) => {
   340→    e.preventDefault();
   341→    if (!loginEmail.trim() || !loginPassword.trim()) {
   342→      setError("Please enter your email and password.");
   343→      return;
   344→    }
   345→    setLoading(true);
   346→    setError("");
   347→    try {
   348→      const result = await signIn("credentials", {
   349→        email: loginEmail.trim(),
   350→        password: loginPassword,
   351→        redirect: false,
   352→      });
   353→
   354→      if (result?.error) {
   355→        setError("Invalid email or password. Please try again.");
   356→        return;
   357→      }
   358→
   359→      // Fetch session to get user data, then set auth explicitly
   360→      try {
   361→        const sessionRes = await fetch("/api/auth/session");
   362→        const sessionData = await sessionRes.json();
   363→        if (sessionData?.user) {
   364→          const u = sessionData.user as any;
   365→          setAuth({
   366→            id: u.id || "",
   367→            email: u.email || "",
   368→            name: u.name || "",
   369→            phone: "",
   370→            role: u.role || "FAMILY",
   371→            avatarUrl: u.image || null,
   372→            subscription: "NONE",
   373→            phoneVerified: u.phoneVerified || false,
   374→            patientProfiles: [],
   375→            caregiverProfile: null,
   376→          });
   377→        }
   378→      } catch {
   379→        // AuthSync will pick it up as fallback
   380→      }
   381→
   382→      toast({
   383→        title: "Welcome back! \ud83c\udf89",
   384→        description: "You have successfully logged in.",
   385→      });
   386→      handleClose();
   387→    } catch {
   388→      setError("Something went wrong. Please try again.");
   389→    } finally {
   390→      setLoading(false);
   391→    }
   392→  };
   393→
   394→  /* ---- Registration ---- */
   395→  const handleRegister = async (e: React.FormEvent) => {
   396→    e.preventDefault();
   397→    if (!regName.trim() || !regEmail.trim() || !regPhone.trim() || !regPassword.trim()) {
   398→      setError("Please fill in all fields.");
   399→      return;
   400→    }
   401→    if (regPassword.length < 6) {
   402→      setError("Password must be at least 6 characters.");
   403→      return;
   404→    }
   405→    setLoading(true);
   406→    setError("");
   407→    try {
   408→      // 1. Register
   409→      const regRes = await api.auth.register({
   410→        name: regName.trim(),
   411→        email: regEmail.trim(),
   412→        phone: regPhone.trim(),
   413→        password: regPassword,
   414→        role: regRole,
   415→      });
   416→
   417→      const registeredUserId = regRes.user.id;
   418→
   419→      // 2. Auto-login with credentials
   420→      const loginResult = await signIn("credentials", {
   421→        email: regEmail.trim(),
   422→        password: regPassword,
   423→        redirect: false,
   424→      });
   425→
   426→      if (loginResult?.error) {
   427→        // Registration succeeded but auto-login failed — that's ok, just notify
   428→        toast({ title: "Account created!", description: "Please log in with your credentials." });
   429→        setTab("login");
   430→        setLoginEmail(regEmail.trim());
   431→        return;
   432→      }
   433→
   434→      // Fetch session to get user data, then set auth explicitly
   435→      let authData: any = null;
   436→      try {
   437→        const sessionRes = await fetch("/api/auth/session");
   438→        const sessionData = await sessionRes.json();
   439→        if (sessionData?.user) {
   440→          const u = sessionData.user as any;
   441→          authData = {
   442→            id: u.id || registeredUserId,
   443→            email: u.email || regEmail.trim(),
   444→            name: u.name || regName.trim(),
   445→            phone: regPhone.trim(),
   446→            role: u.role || regRole,
   447→            avatarUrl: u.image || null,
   448→            subscription: "NONE",
   449→            phoneVerified: false,
   450→            patientProfiles: [],
   451→            caregiverProfile: null,
   452→          };
   453→          setAuth(authData);
   454→        }
   455→      } catch {
   456→        // AuthSync will pick it up as fallback
   457→      }
   458→
   459→      // 3. After successful registration, show phone verification for CAREGIVER (mandatory)
   460→      //    or show optional banner for FAMILY
   461→      if (regRole === "CAREGIVER") {
   462→        // Mandatory phone verification for caregivers
   463→        setPhoneVerifiedUserId(registeredUserId);
   464→        setPhoneVerifiedPhone(regPhone.trim());
   465→        setModalView("phone-verify");
   466→        setPhoneOtpSent(false);
   467→        setPhoneOtp("");
   468→        setPhoneOtpError("");
   469→        // Auto-send OTP
   470→        handleSendPhoneOtp(regPhone.trim());
   471→      } else {
   472→        // Optional for family members — show a toast and let them in
   473→        toast({
   474→          title: "Welcome to SevaSaathi! \ud83c\udf89",
   475→          description: "Your account has been created. Verify your phone number later for full access.",
   476→        });
   477→        handleClose();
   478→      }
   479→    } catch (err: any) {
   480→      setError(err?.message || "Registration failed. Please try again.");
   481→    } finally {
   482→      setLoading(false);
   483→    }
   484→  };
   485→
   486→  /* ---- Phone OTP Send ---- */
   487→  const handleSendPhoneOtp = async (phone?: string) => {
   488→    const phoneToUse = phone || phoneVerifiedPhone;
   489→    if (!phoneToUse) return;
   490→    setPhoneOtpSending(true);
   491→    setPhoneOtpError("");
   492→    try {
   493→      await api.auth.sendPhoneOtp(phoneToUse);
   494→      setPhoneOtpSent(true);
   495→      setPhoneOtp("");
   496→      setPhoneVerifyCountdown(60);
   497→      toast({ title: "OTP Sent", description: `A 6-digit OTP has been sent to ${phoneToUse}` });
   498→    } catch (err: any) {
   499→      setPhoneOtpError(err?.message || "Failed to send OTP. Please try again.");
   500→    } finally {
   501→      setPhoneOtpSending(false);
   502→    }
   503→  };
   504→
   505→  /* ---- Phone OTP Verify ---- */
   506→  const handleVerifyPhoneOtp = async () => {
   507→    if (phoneOtp.length !== 6) {
   508→      setPhoneOtpError("Please enter the complete 6-digit OTP.");
   509→      return;
   510→    }
   511→    setPhoneOtpLoading(true);
   512→    setPhoneOtpError("");
   513→    try {
   514→      await api.auth.verifyPhoneOtp(phoneVerifiedPhone, phoneOtp);
   515→
   516→      // Update auth store with phoneVerified = true
   517→      const currentUser = useAuthStore.getState().user;
   518→      if (currentUser) {
   519→        setAuth({ ...currentUser, phoneVerified: true });
   520→      }
   521→
   522→      toast({
   523→        title: "Phone Verified! \u2705",
   524→        description: "Your phone number has been successfully verified.",
   525→      });
   526→      handleClose();
   527→    } catch (err: any) {
   528→      setPhoneOtpError(err?.message || "Verification failed. Please try again.");
   529→    } finally {
   530→      setPhoneOtpLoading(false);
   531→    }
   532→  };
   533→
   534→  /* ---- Reset password (simplified) ---- */
   535→  const handleReset = async (e: React.FormEvent) => {
   536→    e.preventDefault();
   537→    if (!resetEmail.trim() || !resetNewPassword.trim()) {
   538→      setResetError("Please enter your email and new password.");
   539→      return;
   540→    }
   541→    if (resetNewPassword.length < 6) {
   542→      setResetError("Password must be at least 6 characters.");
   543→      return;
   544→    }
   545→    setResetLoading(true);
   546→    setResetError("");
   547→    try {
   548→      const res = await fetch("/api/auth/reset-password", {
   549→        method: "POST",
   550→        headers: { "Content-Type": "application/json" },
   551→        body: JSON.stringify({ email: resetEmail.trim(), newPassword: resetNewPassword }),
   552→      });
   553→      const data = await res.json();
   554→      if (!res.ok) {
   555→        setResetError(data.error || "Failed to reset password.");
   556→        return;
   557→      }
   558→      setResetDone(true);
   559→      toast({ title: "Password reset successfully!", description: "You can now log in with your new password." });
   560→    } catch {
   561→      setResetError("Something went wrong. Please try again.");
   562→    } finally {
   563→      setResetLoading(false);
   564→    }
   565→  };
   566→
   567→  /* ================================================================ */
   568→  /*  RENDER                                                           */
   569→  /* ================================================================ */
   570→  return (
   571→    <Dialog open={isOpen} onOpenChange={(open) => !open && handleClose()}>
   572→      <DialogContent showCloseButton={false} className="sm:max-w-md p-0 overflow-hidden border-forest-200/40">
   573→        {/* Single custom close button */
   574→        <button
   575→          onClick={handleClose}
   576→          className="absolute top-3 right-3 z-50 rounded-full p-1.5 text-gray-400 hover:text-gray-600 hover:bg-gray-100 transition-colors"
   577→          aria-label="Close"
   578→        >
   579→          <X className="w-5 h-5" />
   580→        </button>
   581→
   582→        {/* ============================================================ */}
   583→        {/*  PHONE VERIFICATION VIEW (CAREGIVER - mandatory)             */}
   584→        {/* ============================================================ */}
   585→        {modalView === "phone-verify" && (
   586→          <div className="px-6 pt-6 pb-2">
   587→            <DialogHeader className="space-y-1 mb-4">
   588→              <div className="flex items-center gap-2 justify-center">
   589→                <div className="w-10 h-10 rounded-xl bg-amber-100 flex items-center justify-center">
   590→                  <Smartphone className="w-5 h-5 text-amber-700" />
   591→                </div>
   592→              </div>
